```
@singletonclass Submiter(object):    RETRY = 3    RETRY_DURATION = 3
    def __init__(self, submit_flag_fn, delay=0):        self._flag_lst = queue.Queue()        self._exit = False        self._thread = None        self._submit_flag_fn = submit_flag_fn        self._delay = delay        # self._max_workers = max_workers
    def add_flag(self, flag: dict):        """flag should be a dict such as `{'flag': '', 'ip': ''}`"""        if not isinstance(flag, dict):            flag = {'flag': flag}        for k in flag.keys():            flag[k] = flag[k].strip()        self._flag_lst.put(flag)
    def submit_flags(self):        while not self._exit:            try:                flag = self._flag_lst.get()                self._submit_flag_fn(**flag)                time.sleep(self._delay)            except Exception as e:                logger.error(e.__str__())
    def submit_flag_with_retry(self):        while True:            retry = self.RETRY + 2            flag = ''
            try:                flag = self._flag_lst.get_nowait()            except queue.Empty:                pass
            while retry and not flag:                time.sleep(self.RETRY_DURATION)                try:                    flag = self._flag_lst.get_nowait()                except queue.Empty:                    pass                retry -= 1
            if flag:                success = self._submit_flag_fn(**flag)                _retry = self.RETRY - 1
                while not success and _retry:                    time.sleep(self.RETRY_DURATION)                    _retry -= 1                    success = self._submit_flag_fn(**flag)
            if not retry:                break
    def _start(self):        _thread.start_new_thread(self.submit_flags, ())        while not self._exit:            time.sleep(1)
    def start(self):        if self._thread is not None and self._thread.is_alive():            self._exit = True            time.sleep(2)            self._exit = False
        self._thread = threading.Thread(target=self._start)        self._thread.start()
    def exit(self):        self._exit = True
```

可以看到，`Submiter.start`中封装了一层，为什么要这样做呢。

因为当我们使用`threading`库启动新线程时，是没有简单直观的方法控制退出的，必须等到线程函数执行完成（所以一般需要在线程函数中做控制）。而这里使用更底层的`_thread`库的好处是，由`_thread.start_new_thread`启动的线程会随函数返回而静默退出，这样就完全不用侵入式的修改线程函数逻辑，控制退出的逻辑完全由外部实现。（当然，这种情况仅限于线程函数不需要做资源释放）

`exec`方法中，为了读取TCP流时获取准确的结果，我在命令执行过程中进行了分包操作，也就是echo一个随机的token，例如`cat /etc/passwd ; echo ABCDEF`（Windows的话则是添加`& echo ABCDEF`，但AWD基本不会遇见Windows），然后通过`recv_until`读取准确结果。这种操作对于有Socket编程经验的读者来说应该是基本功了。
