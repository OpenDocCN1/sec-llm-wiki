def gauss(a, n):
    r = 0
    for c in range(n):
        t = r
        for i in range(r, n):
            if a[i][c] == 1: 
                t = i
                break
        if a[t][c] == 0: 
            continue
        t_list = a[t]
        a[t] = a[r]
        a[r] = t_list
        for i in range(r + 1, n):
            if a[i][c] == 1:
                for j in range(c, n + 1):
                    a[i][j] ^= a[r][j]
        r += 1
    for i in range(n - 1, -1, -1):
        for j in range(i + 1, n):
            a[i][n] ^= a[i][j] & a[j][n]
    mask = 0
    for i in range(n):
        mask <<= 1
        mask |= a[i][n]
    return mask
