01

0x1 从helloworld开始


       我们从最简单的hello world开始。严格来讲这个模块都不算是字符型驱动模块，只能算是可加载内核模块。这里先不实现设备的申请注册，先从最基本框架开始，可以对可加载内核模块如何实现以及如何工作有更好的理解。

编写模块代码

//hello.c
#include <linux/init.h>
#include <linux/kernel.h> 
#include <linux/module.h>
static int hello_init(void)
{
    printk("Hello，kernel World！\n");
    return 0;
}

static void hello_exit(void)
{
    printk("Goodbye，good luck！\n");
}
module_init(hello_init);
module_exit(hello_exit);
MODULE_LICENSE("GPL");

上述代码在模块基本框架的基础上分别在装在函数和卸载函数进行了输出，因为内核的代码不能使用libc，只能使用内核的函数，这里printk就是内核的打印函数。

编写Makefile

obj-m := hello.o
KDIR:= /lib/modules/$(shell uname -r)/build
all:
    make -C $(KDIR) M=$(PWD) modules cc=gcc-5

clean:
    make -C $(KDIR) M=$(PWD) modules clean

Makefile这里只需要关注KDIR，代表编译的内核路径，这里我们以当前系统的内核进行编译。

加载模块

sudo insmod hello.ko

查看运行结果

cat /var/log/syslog | grep Hello

卸载模块

sudo rmmod hello.ko

查看结果

cat /var/log/syslog | grep Goodbye


可以看到，当模块加载到内核时，执行了chardev_init()打印了相应内容，而当模块释放时，执行了chardev_exit()，打印了相应内容。 



02

0x2 设备申请



       上述程序只具备在内核缓冲区进行输出的能力，而我们希望我们的模块起码可以和用户进行交互，以提供后续利用的可能，因此首先我们要实现设备申请注册，一般情况下，注册设备需要三步：

  1. 申请设备号

  2. 创建cdev

  3. 将设备添加到系统

这里直接给出代码

//device.c
#include <linux/init.h>
#include <linux/module.h>
#include <linux/types.h>
#include <linux/kernel.h>
#include <linux/fs.h>
#include <linux/cdev.h>
#include <linux/sched.h>
#include <linux/device.h>
#include <linux/slab.h>
#include <asm/current.h>
#include <linux/uaccess.h>

#define DRIVER_NAME "chardev"
#define BUFFER_SIZE 64


static const unsigned int MINOR_BASE = 0;
static const unsigned int MINOR_NUM = 1;
static unsigned int chardev_major;
static struct cdev chardev_cdev;
static struct class *chardev_class = NULL;

struct file_operations chardev_fops = {                    //定义file_options，用于后续添加函数
};

static int chardev_init(void)
{
    int alloc_ret = 0;
    int cdev_err = 0;
    dev_t dev;

    printk("[dev_test:] The chardev_init() function has been called.\n");
    printk("[dev_test:] Trying to get major number value.\n");

    alloc_ret = alloc_chrdev_region(&dev, MINOR_BASE, MINOR_NUM, DRIVER_NAME); //注册设备号
    if (alloc_ret != 0) {
        printk("alloc_chrdev_region = %d\n", alloc_ret);
        return -1;
    }
    //Get the major number value in dev.
    chardev_major = MAJOR(dev);
    printk("[device_test:] Get major number done.The major is %d\n", chardev_major);
    dev = MKDEV(chardev_major, MINOR_BASE); //获取设备在设备表中的位置

    //initialize a cdev structure
    cdev_init(&chardev_cdev, &chardev_fops); //静态初始化cdev结构
    chardev_cdev.owner = THIS_MODULE;

    //add a char device to the system
    cdev_err = cdev_add(&chardev_cdev, dev, MINOR_NUM); //注册cdev，成功返回0
    if (cdev_err != 0) {
        printk("cdev_add = %d\n", alloc_ret);
        unregister_chrdev_region(dev, MINOR_NUM);
        return -1;
    }

    chardev_class = class_create(THIS_MODULE, "chardev"); //为该设备创建设备类
    if (IS_ERR(chardev_class)) {
        printk("class_create\n");
        cdev_del(&chardev_cdev);
        unregister_chrdev_region(dev, MINOR_NUM);
        return -1;
    } 
    device_create(chardev_class, NULL, MKDEV(chardev_major, MINOR_BASE), NULL, "chardev%d", MINOR_BASE); 
    printk("[device_test:] Register complete.\n");
    return 0; //在/dev下创建对应的设备节点。生成设备文件
}

static void chardev_exit(void)
{
    int minor = 0;
    dev_t dev = MKDEV(chardev_major, MINOR_BASE); 
    printk("[dev_test:] The chardev_exit() function has been called."); 
    device_destroy(chardev_class, MKDEV(chardev_major, minor));
    class_destroy(chardev_class);
    cdev_del(&chardev_cdev);
    unregister_chrdev_region(dev, MINOR_NUM);
}

module_init(chardev_init);
module_exit(chardev_exit);
MODULE_LICENSE("GPL");

03

0x3 IO接口实现



       完成设备注册之后，再进一步就是实现I/O接口，以供后续与程序或用户进行交互。

      在上一个文章中提过，其实对于设备注册和销毁来说，流程相对固定，核心代码差异不大，一般不需要过多关注，拿来直接用即可，重点在于模块功能代码，而I/O接口就是在模块功能代码中实现。后续将按上述示例代码内容固定模块加载和模块卸载函数，着重对模块功能函数进行编写。首先实现I/O功能。

    switch (cmd) {

04

0x4 提权功能



       这里先简单介绍一下内核里面最常见的提权方式，后续内容将以这块为基础。Linux内核每个进程中都有一个 cred 结构，这个结构保存了该进程的权限等信息（uid，gid 等），类似于一种凭证。有兴趣可以简单看一下下面的cred结构体，该代码来自5.9.9版本内核。

struct cred {
    atomic_t    usage;
#ifdef CONFIG_DEBUG_CREDENTIALS
    atomic_t    subscribers; /* number of processes subscribed 使用这个 cred 的进程数*/
    void        *put_addr;
    unsigned    magic;
#define CRED_MAGIC 0x43736564
#define CRED_MAGIC_DEAD 0x44656144
#endif
    kuid_t        uid; /* real UID of the task 创建进程的用户的 id ，不是创建可执行程序的用户 id*/
    kgid_t        gid; /* real GID of the task */
    kuid_t        suid; /* saved UID of the task 保存的 euid 切换之前的 id，用于 euid 切换回来*/
    kgid_t        sgid; /* saved GID of the task */
    kuid_t        euid; /* effective UID of the task euid 是进程运行过程中实时的 id*/
    kgid_t        egid; /* effective GID of the task */
    kuid_t        fsuid; /* UID for VFS ops */
    kgid_t        fsgid; /* GID for VFS ops */
    unsigned    securebits; /* SUID-less security management */
    kernel_cap_t    cap_inheritable; /* caps our children can inherit */
    kernel_cap_t    cap_permitted; /* caps we're permitted */
    kernel_cap_t    cap_effective; /* caps we can actually use */
    kernel_cap_t    cap_bset; /* capability bounding set */
    kernel_cap_t    cap_ambient; /* Ambient capability set */
#ifdef CONFIG_KEYS
    unsigned char    jit_keyring; /* default keyring to attach requested
                     * keys to */
    struct key    *session_keyring; /* keyring inherited over fork */
    struct key    *process_keyring; /* keyring private to this process */
    struct key    *thread_keyring; /* keyring private to this thread */
    struct key    *request_key_auth; /* assumed request_key authority */
#endif
#ifdef CONFIG_SECURITY
    void        *security; /* subjective LSM security */
#endif
    struct user_struct *user;    /* real user ID subscription 创建进程的用户的 id 描述符*/
    struct user_namespace *user_ns; /* user_ns the caps and keyrings are relative to. */
    struct group_info *group_info;    /* supplementary groups for euid/fsgid */
    /* RCU deletion */
    union {
        int non_rcu; /* Can we skip RCU deletion? */
        struct rcu_head    rcu;        /* RCU deletion hook */
    };
} __randomize_layout;

如果去查看该结构体，会发现其大小为168字节。

目前为止还不需要关注这个点，这里如果能修改某个进程的 cred，将uid和gid修改为0，那么也就修改了这个进程的权限，实现了提权。一般情况下，在内核中实现提权会使用commit_creds(prepare_kernel_cred(0))，即首先使用prepare_kernel_cred(0)创建一个root用户的cred，之后调用commit_creds用新创建的cred更新当前进程的cred实现提权。
