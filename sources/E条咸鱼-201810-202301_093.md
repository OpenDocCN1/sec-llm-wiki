# x86架构与x64架构  

1、x86架构 x86架构是intel开发的一种32位的指令集。8个32位通用寄存器 `eax,ebx,ecx,edx,ebp,esp,esi,edi`。

2、x64架构 x64架构是64位的，有16个通用寄存器 `rax,rbx,rcx,rdx,rsi,rdi,rsp,rbp,r8,r9,r10,r11,r12,r13,r14,r15`。

# x86架构下函数于栈中调用过程

```
1.  `0804840b <Add>:`
    
2.  ` 804840b:    55                      push   ebp          //将ebp压入栈`
    
3.  ` 804840c:    89 e5                   mov    ebp,esp      //将esp压入栈`
    
4.  ` 804840e:    83 ec 08                sub    esp,0x8      //抬高栈帧`
    
5.  ` 8048411:    8b 55 08                mov    edx,DWORD PTR [ebp+0x8]  //将[ebp+0x8]压入edx`
    
6.  ` 8048414:    8b 45 0c                mov    eax,DWORD PTR [ebp+0xc]  //将[ebp+0xc]压入eax`
    
7.  ` 8048417:    01 d0                   add    eax,edx  //将eax与edx中的内容相加赋给eax`
    
8.  ` 8048419:    83 ec 08                sub    esp,0x8  //抬高栈帧`
    
9.  ` 804841c:    50                      push   eax  //将eax压入栈`
    
10.  ` 804841d:    68 e0 84 04 08          push   0x80484e0    //压入x+y=`
     
11.  ` 8048422:    e8 b9 fe ff ff          call   80482e0 <printf@plt> //调用printf函数`
     
12.  ` 8048427:    83 c4 10                add    esp,0x10 //调整栈帧`
     
13.  ` 804842a:    90                      nop`
     
14.  ` 804842b:    c9                      leave`
     
15.  ` 804842c:    c3                      ret`
     
16.  
     
17.  `0804842d <main>:`
     
18.  ` 804842d:    8d 4c 24 04             lea    ecx,[esp+0x4]`
     
19.  ` 8048431:    83 e4 f0                and    esp,0xfffffff0`
     
20.  ` 8048434:    ff 71 fc                push   DWORD PTR [ecx-0x4]`
     
21.  
     
22.  ` 8048437:    55                      push   ebp      //将ebp压入栈`
     
23.  ` 8048438:    89 e5                   mov    ebp,esp  //将esp压入栈`
     
24.  ` 804843a:    51                      push   ecx      //将ecx压入栈`
     
25.  ` 804843b:    83 ec 04                sub    esp,0x4  //抬高栈帧,esp=esp+0x4`
     
26.  ` 804843e:    83 ec 08                sub    esp,0x8  //抬高栈帧`
     
27.  ` 8048441:    6a 14                   push   0x14     //压入参数20`
     
28.  ` 8048443:    6a 0a                   push   0xa      //压入参数10`
     
29.  ` 8048445:    e8 c1 ff ff ff          call   804840b <Add>    //调用Add函数`
     
30.  ` 804844a:    83 c4 10                add    esp,0x10 //调整栈帧`
     
31.  ` 804844d:    b8 00 00 00 00          mov    eax,0x0  //清空eax`
     
32.  ` 8048452:    8b 4d fc                mov    ecx,DWORD PTR [ebp-0x4]  //将[ebp-0x4]内的数据赋给ecx`
     
33.  ` 8048455:    c9                      leave           //mov esp,ebp;pop ebp`
     
34.  ` 8048456:    8d 61 fc                lea    esp,[ecx-0x4]    //取[ecx-0x4]中的数据赋给esp`
     
35.  ` 8048459:    c3                      ret             //pop eip`
     
36.  
     
37.  ` 804845a:    66 90                   xchg   ax,ax`
     
38.  ` 804845c:    66 90                   xchg   ax,ax`
     
39.  ` 804845e:    66 90                   xchg   ax,ax`
     
```

```
21.  `  400554:    48 83 ec 10             sub    rsp,0x10     //抬高栈帧`

```

# 总结

x86架构与x64架构在函数于栈中调用过程的不同之处在于：x86架构下，函数的参数是直接存放到栈帧中的；而x64架构下，函数的参数是通过寄存器传参进入栈帧的。（当寄存器不够用，才会将参数直接存入栈帧）
