#include <stdio.h>
#include <stdlib.h>
int main(void)
{
    int i, j;
    char name[10], s[10];

    printf("please input five chars : ");
    scanf("%s", name);

    for (i = 0; i < 5; i++)
    {
        s[i] = (name[i] % 10) ^ i;
        s[i] = (s[i] + 2) % 10;
    }

    for (i = 0; i < 5; i++)
    {
        j = s[i];
        while (j < 'A')
        {
            j += 10;
        }
        printf("%c", j);
    }
    printf("\n");

    return 0;
}
