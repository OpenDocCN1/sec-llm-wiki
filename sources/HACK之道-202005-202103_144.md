for (i \= 0; i < 5; i++)
{
s\[i\] \= (name\[i\] % 10) ^ i;
s\[i\] \= (s\[i\] + 2) % 10;
}

for (i \= 0; i < 5; i++)
{
j \= s\[i\];

{
