### meta refresh

```
1.  `class Arachni::Parser::Extractors::MetaRefresh < Arachni::Parser::Extractors::Base`
    
2.  
    
3.  `    def run`
    
4.  `        return [] if !check_for?( 'http-equiv' )`
    
5.  
    
6.  `        document.nodes_by_attribute_name_and_value( 'http-equiv', 'refresh' ).`
    
7.  `            map do |url|`
    
8.  `                begin`
    
9.  `                    _, url = url['content'].split( ';', 2 )`
    
10.  `                    next if !url`
     
11.  `                    unquote( url.split( '=', 2 ).last.strip )`
     
12.  `                rescue`
     
13.  `                    next`
     
14.  `                end`
     
15.  `            end`
     
16.  `    end`
     
17.  
     
18.  `    def unquote( str )`
     
19.  `        [ '\'', '"' ].each do |q|`
     
20.  `            return str[1...-1] if str.start_with?( q ) && str.end_with?( q )`
     
21.  `        end`
     
22.  `        str`
     
23.  `    end`
     
24.  
     
25.  `end`
     
```

### script 中提取url

```
1.  `class Arachni::Parser::Extractors::Scripts < Arachni::Parser::Extractors::Base`
    
2.  
    
3.  `    def run`
    
4.  `        return [] if !check_for?( 'script' )`
    
5.  
    
6.  `        document.nodes_by_name( 'script' ).map do |s|`
    
7.  `            [s['src']].flatten.compact | from_text( s.text.to_s )`
    
8.  `        end`
    
9.  `    end`
    
10.  
     
11.  `    def from_text( text )`
     
12.  `        text.scan( /[\/a-zA-Z0-9%._-]+/ ).`
     
13.  `            select do |s|`
     
14.  `            # String looks like a path, but don't get fooled by comments.`
     
15.  `            s.include?( '.' ) && s.include?( '/' )  &&`
     
16.  `                !s.include?( '*' ) && !s.start_with?( '//' ) &&`
     
17.  
     
18.  `                # Require absolute paths, otherwise we may get caught in`
     
19.  `                # a loop, this context isn't the most reliable for extracting`
     
20.  `                # real paths.`
     
21.  `                s.start_with?( '/' )`
     
22.  `        end`
     
23.  `    end`
     
24.  
     
25.  `end`
     
```
