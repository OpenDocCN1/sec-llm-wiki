本来按理说先放个通防waf的，但是听说十分容易被check down，最好不要乱动，有时候拿flag分还不如down的分多，这就很难受了。

<?php  
  
error\_reporting(0);

{

function getallheaders() {  
foreach ($\_SERVER as $name => $value) {  
if (substr($name, 0, 5) == 'HTTP\_')  
$headers\[str\_replace(' ', '-', ucwords(strtolower(str\_replace('\_', ' ', substr($name, 5)))))\] = $value;

}

}  
  
function logging($var){  
$filename = $\_SERVER\['REMOTE\_ADDR'\];  
$LOG\_FILENAME = LOG\_FILEDIR."/".$filename;  
$time = date("Y-m-d G:i:s");  
file\_put\_contents($LOG\_FILENAME, "\\r\\n".$time."\\r\\n".print\_r($var, true), FILE\_APPEND);  
file\_put\_contents($LOG\_FILENAME,"\\r\\n".'http://'.$\_SERVER\['HTTP\_HOST'\].$\_SERVER\['PHP\_SELF'\].'?'.$\_SERVER\['QUERY\_STRING'\], FILE\_APPEND);  
file\_put\_contents($LOG\_FILENAME,"\\r\\n\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*",FILE\_APPEND);  
}  
  
waf();
