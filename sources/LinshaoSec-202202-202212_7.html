说明：

远线程注入，是通过调用Windows API，在其他应用程序进程中创建线程执行代码的技术，

使用的关键函数是CreateRemoteThread()，

用处：

进程迁移，文件自删除，杀软对抗，外挂编写等等

常见的注入方式：

远程线程注入，APC注入，消息钩子注入，注册表注入，导入表注入，输入法注入等

在此记录一下测试过程，抛砖引玉，望不吝赐教。

一、测试步骤：

    分为两个方面，构建dll和注入dll

1.编写测试Dll：

  • 编写dll

  • 测试dll

2.远线程注入Dll：

  • 打开目标进程

  • 通过句柄在目标进程中申请空间

  • 在目标进程内写入要注入的目标dll路径

  • 获取LoadLibraryA()函数地址

  • 创建远程线程调用LoadLibraryA()加载dll

  • 等待线程结束，清理环境

三、远程线程注入Dll：

1.通过pid打开目标进程，返回句柄

2.通过句柄在目标进程中申请空间，返回空内存地址

3.在目标进程内存地址写入要注入的目标dll路径

4.获取LoadLibraryA()函数地址，此函数存在于kernel32.dll模块中，同一机器其内存地址相同

5.创建远程线程调用LoadLibraryA()加载dll

6.等待线程结束，清理环境

(1)windows api函数

在此之前我们先了解几个函数

OpenProcess()打开目标进程，返回句柄

VirtualAllocEx()申请空间，返回空内存地址

WriteProcessMemory()写入要注入的目标dll路径

LoadLibraryA()加载dll

CreateRemoteThread()创建远程线程调用LoadLibraryA()函数

windows函数原形：    HANDLE OpenProcess(      DWORD dwDesiredAccess,               //渴望得到的访问权限（标志）            //PROCESS_ALL_ACCESS(所有权限)在c++的定义中值为2097151      BOOL bInheritHandle,                 // 是否继承句柄，true 或者false随意      DWORD dwProcessId               // 进程标示符，置NULLj就行    );
  LPVOID VirtualAllocEx(    HANDLE hProcess,    LPVOID lpAddress,    SIZE_T dwSize,    DWORD flAllocationType,    DWORD flProtect  );
  BOOL WriteProcessMemory(    HANDLE hProcess,    LPVOID lpBaseAddress,    LPVOID lpBuffer,    DWORD nSize,    LPDWORD lpNumberOfBytesWritten  );
  HMODULE LoadLibrary(_In_ LPCTSTR lpFileName);
  HANDLE CreateRemoteThread(    __in HANDLE hProcess,    __in LPSECURITY_ATTRIBUTES lpThreadAttributes,    __in SIZE_T dwStackSize,    __in LPTHREAD_START_ROUTINE lpStartAddress,    __in LPVOID lpParameter,    __in DWORD dwCreationFlags,    __out LPDWORD lpThreadId  );

2.修改Dlltest.cpp代码如下：

#include <windows.h>#include <iostream>
int main() {   //通过pid号拿到进程句柄   HANDLE h = OpenProcess(2097151, true, 19300);   if (h) { printf("打开句柄成功\n"); }else { printf("打开句柄失败"); }
   //定义要注入dll的路径，下一步将他传入注入目标的进程内存里, dll里面的代码注意x86和x64位shellcode   const char* s = "F:\\TEMP\\dll1.dll";   printf(s); 
   //在目标进程里面申请一块虚拟内存存放dll路径   LPVOID pbuf=VirtualAllocEx(h, 0, strlen(s), MEM_RESERVE | MEM_COMMIT, PAGE_EXECUTE_READWRITE);    if (!pbuf) {printf("申请内存失败\n");}else {printf("申请内存成功\n");}
   //写入内存区域   BOOL aaa=WriteProcessMemory(h,pbuf,s, strlen(s),0);          if (aaa){printf("写入成功\n");}else{printf("写入失败\n");}
   //创建远程线程，传入LoadLibrary函数，参数为上面返回的目标进程里的dll路径   HANDLE hand1 = CreateRemoteThread(h,NULL,0,(LPTHREAD_START_ROUTINE)LoadLibraryA,pbuf,0,NULL);   if (hand1 != NULL) {  printf("注入成功\n");  }else { printf("注入失败\n"); }   WaitForSingleObject(h, -1);   CloseHandle(h);   return 0;}

四、使用go重写dll注入

编译时需要注意对应系统位数，否则两种架构程序的loadLibraryA函数的内存地址是不一样的，会造成宿主进程异常结束，eventvwr日志会有事件号1000。通过以下命令来设置64或者32位目标。


DllLoad.go


package main
import (  "fmt"  "syscall"  "unsafe")var(  kernel32=syscall.MustLoadDLL("kernel32.dll")  OpenProcess=kernel32.MustFindProc("OpenProcess")  VirtualAllocEx=kernel32.MustFindProc("VirtualAllocEx")  //分配内存  WriteProcessMemory=kernel32.MustFindProc("WriteProcessMemory")  //写入内存  CreateRemoteThread=kernel32.MustFindProc("CreateRemoteThread")  pid int  dllpath string  lenth int  ss []byte)
func main(){  myinit()
  hand,_, err :=OpenProcess.Call(2097151, uintptr(0), uintptr(pid))  fmt.Println("打开目标进程成功",hand)  checkErr(err)
  addr,_,err:=VirtualAllocEx.Call(hand,0,uintptr(lenth),0x00002000|0x00001000,0x40)  checkErr(err)  fmt.Printf("申请内存地址成功  %x\n",addr)
  isok,_,c:=WriteProcessMemory.Call(hand,addr,uintptr(unsafe.Pointer(&ss[0])),uintptr(lenth),0)  fmt.Println("写入内存成功",isok)
  LoadLibraryA:=getFuncAddr("kernel32.dll","LoadLibraryA")  a,_,c:=CreateRemoteThread.Call(hand,uintptr(unsafe.Pointer(nil)),0, uintptr(LoadLibraryA),uintptr(unsafe.Pointer(addr)),0,uintptr(unsafe.Pointer(nil)))  checkErr(c)  fmt.Println("注入成功",a)}//初始化输入func myinit() {  fmt.Printf("请输入pid:\n->")  fmt.Scanln(&pid)  fmt.Printf("请输入dll路径:\n->")  fmt.Scanln(&dllpath)  ss= []byte(dllpath)  lenth=len(ss)}
//获取函数地址,返回函数指针func getFuncAddr(module string,proc string)unsafe.Pointer{  handle,err:=syscall.LoadLibrary(module)  checkErr(err)  s,err:=syscall.GetProcAddress(handle,proc)  checkErr(err)  sss:=unsafe.Pointer(s)  //fmt.Println(sss)  return sss}func checkErr(err error){  if err!=nil {    if err.Error() != "The operation completed successfully." {      fmt.Println("报错：",err.Error())      syscall.Exit(1);      return    }  }}
