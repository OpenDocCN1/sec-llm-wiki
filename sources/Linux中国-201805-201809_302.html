我们继续 无密码验证[1] 的文章。上一篇文章中，我们用 Go 写了一个 HTTP 服务，用这个服务来做无密码验证 API。今天，我们为它再写一个 JavaScript 客户端。

我们将使用 这里的[2] 这个单页面应用程序（SPA）来展示使用的技术。如果你还没有读过它，请先读它。

记住流程：

◈ 用户输入其 email。
◈ 用户收到一个带有魔法链接的邮件。
◈ 用户点击该链接、
◈ 用户验证成功。

对于根 URL（/），我们将根据验证的状态分别使用两个不同的页面：一个是带有访问表单的页面，或者是已验证通过的用户的欢迎页面。另一个页面是验证回调的重定向页面。

伺服

我们将使用相同的 Go 服务器来为客户端提供服务，因此，在我们前面的 main.go 中添加一些路由：

  1. router.Handle("GET", "/...", http.FileServer(SPAFileSystem{http.Dir("static")}))
  1. type SPAFileSystem struct {
  2.     fs http.FileSystem
  3. }
  5. func (spa SPAFileSystem) Open(name string) (http.File, error) {
  6.     f, err := spa.fs.Open(name)
  7.     if err != nil {
  8.         return spa.fs.Open("index.html")
  9.     }
  10.     return f, nil
  11. }

这个伺服文件放在 static 下，配合 static/index.html 作为回调。

你可以使用你自己的服务器，但是你得在服务器上启用 CORS[3]。

单页面应用程序的所有渲染由 JavaScript 来完成，因此，我们使用了一个空的 body 部分和一个 main.js 文件。

我们将使用 上篇文章[2] 中的 Router。

渲染

现在，我们使用下面的内容来创建一个 static/js/main.js 文件：

  1. import Router from 'https://unpkg.com/@nicolasparada/router'
  2. import { isAuthenticated } from './auth.js'
  4. const router = new Router()
  6. router.handle('/', guard(view('home')))
  7. router.handle('/callback', view('callback'))
  8. router.handle(/^\//, view('not-found'))
  10. router.install(async resultPromise => {
  11.     document.body.innerHTML = ''
  12.     document.body.appendChild(await resultPromise)
  13. })
  15. function view(name) {
  16.     return (...args) => import(`/js/pages/${name}-page.js`)
  17.         .then(m => m.default(...args))
  18. }
  20. function guard(fn1, fn2 = view('welcome')) {
  21.     return (...args) => isAuthenticated()
  22.         ? fn1(...args)
  23.         : fn2(...args)
  24. }

与上篇文章不同的是，我们实现了一个 isAuthenticated() 函数和一个 guard() 函数，使用它去渲染两种验证状态的页面。因此，当用户访问 / 时，它将根据用户是否通过了验证来展示主页或者是欢迎页面。

验证

现在，我们来编写 isAuthenticated() 函数。使用下面的内容来创建一个 static/js/auth.js 文件：

  1. export function getAuthUser() {
  2.     const authUserItem = localStorage.getItem('auth_user')
  3.     const expiresAtItem = localStorage.getItem('expires_at')
  5.     if (authUserItem !== null && expiresAtItem !== null) {
  6.         const expiresAt = new Date(expiresAtItem)
  8.         if (!isNaN(expiresAt.valueOf()) && expiresAt > new Date()) {
  9.             try {
  10.                 return JSON.parse(authUserItem)
  11.             } catch (_) { }
  12.         }
  13.     }
  15.     return null
  16. }
  18. export function isAuthenticated() {
  19.     return localStorage.getItem('jwt') !== null && getAuthUser() !== null
  20. }

当有人登入时，我们将保存 JSON 格式的 web 令牌、它的过期日期，以及在 localStorage上的当前已验证用户。这个模块就是这个用处。

◈ getAuthUser() 用于从 localStorage 获取已认证的用户，以确认 JSON 格式的 Web 令牌没有过期。
◈ isAuthenticated() 在前面的函数中用于去检查它是否没有返回 null。

获取

在继续这个页面之前，我将写一些与服务器 API 一起使用的 HTTP 工具。

我们使用以下的内容去创建一个 static/js/http.js 文件：

  1. import { isAuthenticated } from './auth.js'
  3. function get(url, headers) {
  4.     return fetch(url, {
  5.         headers: Object.assign(getAuthHeader(), headers),
  6.     }).then(handleResponse)
  7. }
  9. function post(url, body, headers) {
  10.     return fetch(url, {
  11.         method: 'POST',
  12.         headers: Object.assign(getAuthHeader(), { 'content-type': 'application/json' }, headers),
  13.         body: JSON.stringify(body),
  14.     }).then(handleResponse)
  15. }
  17. function getAuthHeader() {
  18.     return isAuthenticated()
  19.         ? { authorization: `Bearer ${localStorage.getItem('jwt')}` }
  20.         : {}
  21. }
  23. export async function handleResponse(res) {
  24.     const body = await res.clone().json().catch(() => res.text())
  25.     const response = {
  26.         statusCode: res.status,
  27.         statusText: res.statusText,
  28.         headers: res.headers,
  29.         body,
  30.     }
  31.     if (!res.ok) {
  32.         const message = typeof body === 'object' && body !== null && 'message' in body
  33.             ? body.message
  34.             : typeof body === 'string' && body !== ''
  35.                 ? body
  36.                 : res.statusText
  37.         const err = new Error(message)
  38.         throw Object.assign(err, response)
  39.     }
  40.     return response
  41. }
  43. export default {
  44.     get,
  45.     post,
  46. }

这个模块导出了 get() 和 post() 函数。它们是 fetch API 的封装。当用户是已验证的，这二个函数注入一个 Authorization: Bearer <token_here> 头到请求中；这样服务器就能对我们进行身份验证。

这个页面使用一个 HTMLTemplateElement 作为视图。这只是一个输入用户 email 的简单表单。

为了避免干扰，我将跳过错误处理部分，只是将它们输出到控制台上。

现在，我们来写 onAccessFormSubmit() 函数。

  1. import http from '../http.js'
  3. function onAccessFormSubmit(ev) {
  4.     ev.preventDefault()
  6.     const form = ev.currentTarget
  7.     const input = form.querySelector('input')
  8.     const email = input.value
  10.     sendMagicLink(email).catch(err => {
  11.         console.error(err)
  12.         if (err.statusCode === 404 && wantToCreateAccount()) {
  13.             runCreateUserProgram(email)
  14.         }
  15.     })
  16. }
  18. function sendMagicLink(email) {
  19.     return http.post('/api/passwordless/start', {
  20.         email,
  21.         redirectUri: location.origin + '/callback',
  22.     }).then(() => {
  23.         alert('Magic link sent. Go check your email inbox.')
  24.     })
  25. }
  27. function wantToCreateAccount() {
  28.     return prompt('No user found. Do you want to create an account?')
  29. }

它对 /api/passwordless/start 发起了 POST 请求，请求体中包含 email 和 redirectUri。在本例中它返回 404 Not Found 状态码时，我们将创建一个用户。

  1. function runCreateUserProgram(email) {
  2.     const username = prompt("Enter username")
  3.     if (username === null) return
  5.     http.post('/api/users', { email, username })
  6.         .then(res => res.body)
  7.         .then(user => sendMagicLink(user.email))
  8.         .catch(console.error)
  9. }

这个用户创建程序，首先询问用户名，然后使用 email 和用户名做一个 POST 请求到 /api/users。成功之后，给创建的用户发送一个魔法链接。

回调页

这是访问表单的全部功能，现在我们来做回调页面。使用如下的内容来创建一个 static/js/pages/callback-page.js 文件：

  1. import http from '../http.js'
  3. const template = document.createElement('template')
  4. template.innerHTML = `
  5.     <h1>Authenticating you</h1>
  6. `
  8. export default function callbackPage() {
  9.     const page = template.content.cloneNode(true)
  11.     const hash = location.hash.substr(1)
  12.     const fragment = new URLSearchParams(hash)
  13.     for (const [k, v] of fragment.entries()) {
  14.         fragment.set(decodeURIComponent(k), decodeURIComponent(v))
  15.     }
  16.     const jwt = fragment.get('jwt')
  17.     const expiresAt = fragment.get('expires_at')
  19.     http.get('/api/auth_user', { authorization: `Bearer ${jwt}` })
  20.         .then(res => res.body)
  21.         .then(authUser => {
  22.             localStorage.setItem('jwt', jwt)
  23.             localStorage.setItem('auth_user', JSON.stringify(authUser))
  24.             localStorage.setItem('expires_at', expiresAt)
  26.             location.replace('/')
  27.         })
  28.         .catch(console.error)
  30.     return page
  31. }

请记住……当点击魔法链接时，我们会来到 /api/passwordless/verify_redirect，它将把我们重定向到重定向 URI，我们将放在哈希中的 JWT 和过期日期传递给 /callback。

回调页面解码 URL 中的哈希，提取这些参数去做一个 GET 请求到 /api/auth_user，用 JWT 保存所有数据到 localStorage 中。最后，重定向到主页面。
