对开发人员来说，定时某些事件是一项常见任务。定时器的常见场景是看门狗、任务的循环执行，或在特定时间安排事件。在这篇文章中，我将演示如何使用 timer\_create(...)🔗 linux.die.net 创建一个 POSIX 兼容的间隔定时器。

你可以从 GitHub🔗 github.com 下载下面样例的源代码。

![](../Images/b592b203e0a2a8675a50463d903d1d67.png)

线程定时器

让我们看看 `simple_threading_timer.c` 样例。这是最简单的一个。它展示了一个调用了超时函数 `expired` 的间隔定时器是如何被创建的。在每次过期时，都会创建一个新的线程，在其中调用函数 `expired`：

```
1.  `#include <stdio.h>`
2.  `#include <stdlib.h>`
3.  `#include <time.h>`
4.  `#include <signal.h>`
5.  `#include <unistd.h>`
6.  `#include <string.h>`
7.  `#include <errno.h>`
8.  
9.  `void expired(union sigval timer_data);`
10.  
11.  `pid_t gettid(void);`
12.  
13.  `struct t_eventData{`
14.  `    int myData;`
15.  `};`
16.  
17.  `int main()`
18.  `{`
19.  `    int res = 0;`
20.  `    timer_t timerId = 0;`
21.  
22.  `    struct t_eventData eventData = { .myData = 0 };`
23.  
24.  `    /*  sigevent 指定了过期时要执行的操作  */`
25.  `    struct sigevent sev = { 0 };`
26.  
27.  `    /* 指定启动延时时间和间隔时间 `
28.  `    * it_value和it_interval 不能为零 */`
29.  
30.  `    struct itimerspec its = {   .it_value.tv_sec  = 1,`
31.  `                                .it_value.tv_nsec = 0,`
32.  `                                .it_interval.tv_sec  = 1,`
33.  `                                .it_interval.tv_nsec = 0`
34.  `                            };`
35.  
36.  `    printf("Simple Threading Timer - thread-id: %d\n", gettid());`
37.  
38.  `    sev.sigev_notify = SIGEV_THREAD;`
39.  `    sev.sigev_notify_function = &amp;expired;`
40.  `    sev.sigev_value.sival_ptr = &amp;eventData;`
41.  
42.  `    /* 创建定时器 */`
43.  `    res = timer_create(CLOCK_REALTIME, &amp;sev, &amp;timerId);`
44.  
45.  `    if (res != 0){`
46.  `        fprintf(stderr, "Error timer_create: %s\n", strerror(errno));`
47.  `        exit(-1);`
48.  `    }`
49.  
50.  `    /* 启动定时器 */`
51.  `    res = timer_settime(timerId, 0, &amp;its, NULL);`
52.  
53.  `    if (res != 0){`
54.  `        fprintf(stderr, "Error timer_settime: %s\n", strerror(errno));`
55.  `        exit(-1);`
56.  `    }`
57.  
58.  `    printf("Press ETNER Key to Exit\n");`
59.  `    while(getchar()!='\n'){}`
60.  `    return 0;`
61.  `}`
62.  
63.  `void expired(union sigval timer_data){`
64.  `    struct t_eventData *data = timer_data.sival_ptr;`
65.  `    printf("Timer fired %d - thread-id: %d\n", ++data->myData, gettid());`
66.  `}`
```

这种方法的优点是在代码和简单调试方面用量小。缺点是由于到期时创建新线程而增加额外的开销，因此行为不太确定。

中断信号定时器

超时定时器通知的另一种可能性是基于 内核信号🔗 man7.org。内核不是在每次定时器过期时创建一个新线程，而是向进程发送一个信号，进程被中断，并调用相应的信号处理程序。

由于接收信号时的默认操作是终止进程（参考 signal🔗 linux.die.net 手册页），我们必须要提前设置好 Qt Creator，以便进行正确的调试。

当被调试对象接收到一个信号时，Qt Creator 的默认行为是：

◈ 中断执行并切换到调试器上下文。◈ 显示一个弹出窗口，通知用户接收到信号。

这两种操作都不需要，因为信号的接收是我们应用程序的一部分。

Qt Creator 在后台使用 GDB。为了防止 GDB 在进程接收到信号时停止执行，进入 “工具(Tools) -> 选项(Options)” 菜单，选择 “调试器(Debugger)”，并导航到 “本地变量和表达式(Locals & Expressions)”。添加下面的表达式到 “定制调试助手(Debugging Helper Customization)”：

```
1.  `handle SIG34 nostop pass`
```

![](../Images/92feaa2c85653a1d51665888c32b8e5b.png "Sig 34 时不停止")

*Sig 34 时不停止*

你可以在 GDB 文档🔗 sourceware.org 中找到更多关于 GDB 信号处理的信息。

接下来，当我们在信号处理程序中停止时，我们要抑制每次接收到信号时通知我们的弹出窗口:

![](../Images/762e9e24cd50b9b520f2bd4636633fc9.png "Signal 34 弹出窗口")

*Signal 34 弹出窗口*

为此，导航到 “GDB” 标签并取消勾选标记的复选框:

![](../Images/61e998b5ce6364878f87e4e0ac311c8f.png "定时器信号窗口")

*定时器信号窗口*

现在你可以正确的调试 `signal_interrupt_timer`。真正的信号定时器的实施会更复杂一些：

```
1.  `#include <stdio.h>`
2.  `#include <stdlib.h>`
3.  `#include <signal.h>`
4.  `#include <unistd.h>`
5.  `#include <signal.h>`
6.  `#include <time.h>`
7.  `#include <unistd.h>`
8.  `#include <errno.h>`
9.  `#include <string.h>`
10.  
11.  `#define UNUSED(x) (void)(x)`
12.  
13.  `static void handler(int sig, siginfo_t *si, void *uc);`
14.  `pid_t gettid(void);`
15.  
16.  `struct t_eventData{`
17.  `    int myData;`
18.  `};`
19.  
20.  `int main()`
21.  `{`
22.  `    int res = 0;`
23.  `    timer_t timerId = 0;`
24.  
25.  `    struct sigevent sev = { 0 };`
26.  `    struct t_eventData eventData = { .myData = 0 };`
27.  
28.  `    /* 指定收到信号时的操作 */`
29.  `    struct sigaction sa = { 0 };`
30.  
31.  `    /* 指定启动延时的时间和间隔时间 */`
32.  `    struct itimerspec its = {   .it_value.tv_sec  = 1,`
33.  `                                .it_value.tv_nsec = 0,`
34.  `                                .it_interval.tv_sec  = 1,`
35.  `                                .it_interval.tv_nsec = 0`
36.  `                            };`
37.  
38.  `    printf("Signal Interrupt Timer - thread-id: %d\n", gettid());`
39.  
40.  `    sev.sigev_notify = SIGEV_SIGNAL; // Linux-specific`
41.  `    sev.sigev_signo = SIGRTMIN;`
42.  `    sev.sigev_value.sival_ptr = &amp;eventData;`
43.  
44.  `    /* 创建定时器 */`
45.  `    res = timer_create(CLOCK_REALTIME, &amp;sev, &amp;timerId);`
46.  
47.  `    if ( res != 0){`
48.  `        fprintf(stderr, "Error timer_create: %s\n", strerror(errno));`
49.  `        exit(-1);`
50.  `    }`
51.  
52.  `    /* 指定信号和处理程序 */`
53.  `    sa.sa_flags = SA_SIGINFO;`
54.  `    sa.sa_sigaction = handler;`
55.  
56.  `    /* 初始化信号 */`
57.  `    sigemptyset(&amp;sa.sa_mask);`
58.  
59.  `    printf("Establishing handler for signal %d\n", SIGRTMIN);`
60.  
61.  `    /* 注册信号处理程序 */`
62.  `    if (sigaction(SIGRTMIN, &amp;sa, NULL) == -1){`
63.  `        fprintf(stderr, "Error sigaction: %s\n", strerror(errno));`
64.  `        exit(-1);`
65.  `    }`
66.  
67.  `    /* 启动定时器 */`
68.  `    res = timer_settime(timerId, 0, &amp;its, NULL);`
69.  
70.  `    if ( res != 0){`
71.  `        fprintf(stderr, "Error timer_settime: %s\n", strerror(errno));`
72.  `        exit(-1);`
73.  `    }`
74.  
75.  `    printf("Press ENTER to Exit\n");`
76.  `    while(getchar()!='\n'){}`
77.  `    return 0;`
78.  `}`
79.  
80.  `static void`
81.  `handler(int sig, siginfo_t *si, void *uc)`
82.  `{`
83.  `    UNUSED(sig);`
84.  `    UNUSED(uc);`
85.  `    struct t_eventData *data = (struct t_eventData *) si->_sifields._rt.si_sigval.sival_ptr;`
86.  `    printf("Timer fired %d - thread-id: %d\n", ++data->myData, gettid());`
87.  `}`
```

与线程定时器相比，我们必须初始化信号并注册一个信号处理程序。这种方法性能更好，因为它不会导致创建额外的线程。因此，信号处理程序的执行也更加确定。缺点显然是正确调试需要额外的配置工作。

![](../Images/b592b203e0a2a8675a50463d903d1d67.png)

总结

本文中描述的两种方法都是接近内核的定时器的实现。不过，即使 timer\_create(...)🔗 linux.die.net 函数是 POSIX 规范的一部分，由于数据结构的细微差别，也不可能在 FreeBSD 系统上编译样例代码。除了这个缺点之外，这种实现还为通用计时应用程序提供了细粒度控制。
