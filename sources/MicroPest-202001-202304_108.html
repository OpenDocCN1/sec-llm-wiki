_start:
    jmp main
    
    ; Constants
win32_library_hashes:
    call win32_library_hashes_return
    ; LoadLibraryA
    dd 0xEC0E4E8E
    
; Function: find_kernel32
find_kernel32:
    push esi
    xor eax, eax
    mov eax, [fs:eax+0x30]
    mov eax, [eax+0x0C]
    mov esi, [eax+0x1C]
    lodsd
    mov eax, [eax+0x08]
    pop esi
    ret
        
; Function: find_function
find_function:
    pushad
    mov ebp, [esp+0x24]
    mov eax, [ebp+0x3C]
    mov edx, [ebp+eax+0x78]
    add edx, ebp
    mov ecx, [edx+0x18]
    mov ebx, [edx+0x20]
    add ebx, ebp
find_function_loop:
    jecxz find_function_finished
    dec ecx
    mov esi, [ebx+ecx*4]
    add esi, ebp
        
compute_hash:
    xor edi, edi
    xor eax, eax
    cld
compute_hash_again:
    lodsb
    test al, al
    jz compute_hash_finished
    ror edi, 0x0D
    add edi, eax
    jmp compute_hash_again
compute_hash_finished:
find_function_compare:
    cmp edi, [esp+0x28]
    jnz find_function_loop
    mov ebx, [edx+0x24]
    add ebx, ebp
    mov cx, [ebx+2*ecx]
    mov ebx, [edx+0x1C]
    add ebx, ebp
    mov eax, [ebx+4*ecx]
    add eax, ebp
    mov [esp+0x1C], eax
find_function_finished:
    popad
    ret
        
; Function: resolve_symbols_for_dll
resolve_symbols_for_dll:
    lodsd
    push eax
    push edx
    call find_function
    mov [edi], eax
    add esp, 0x08
    add edi, 0x04
    cmp esi, ecx
    jne resolve_symbols_for_dll
resolve_symbols_for_dll_finished:
    ret
        
main:
    sub esp, 0x88                       
; Allocate space on stack for function addresses
    mov ebp, esp                        
; Set ebp as frame ptr for relative offset on stack
    call find_kernel32                  
; Find base address of kernel32.dll
    mov edx, eax                       
 ; Store base address of kernel32.dll in EDX
    jmp win32_library_hashes
win32_library_hashes_return:
    pop esi
    lea edi, [ebp+0x04]                
 ; This is where we store our function addresses
    mov ecx, esi
    add ecx, 0x04                      
 ; Length of kernel32 hash list
    call resolve_symbols_for_dll


代码清单 1：完整的 32 位函数定位程序集清单


主要功能

设置堆栈和存储

我们从第6行开始，跳转到位于第83行的main函数。前几行在堆栈上留出一些空间并设置一个帧指针EBP，它将在整个汇编代码中用于引用存储的数据。这很重要，因为一旦我们找到某物的值或地址，我们就需要一种稍后引用它的方法。

找到 Kernel32 的基地址

第88行，我们调用find_kernel32函数在内存中查找kernel32.dll的基地址。找到基地址后，它会存储在第90行的EDX寄存器中以进行安全保存。

获取编码函数名称的位置

第92行的win32_library_hashes又反过来call win32_library_hashes_return。然后将返回eip弹出到ESI 中。

第95-98行正在设置我们的存储位置。ESI和EDI将用于指向我们需要引用的位置。ESI将用于指向散列函数名称的位置，EDI将用于指向我们将存储解析函数地址的位置。

解析符号

在第100行，调用了resolve_symbols_for_dll函数。重要的是要记住kernel32的基地址当前存储在EDX寄存器中。这将在以后很重要。

目前，我们将只关注这个函数。我们将参考“代码清单2”以避免来回滚动太多。

代码清单 2：32 位解析符号函数

1.在第3行，lodsd指令将存储在ESI 中的值加载到EAX寄存器中，并递增ESI以指向下一个地址。如果我们的代码正在寻找多个函数，ESI将准备就绪并指向下一个编码的函数名称。

2.EAX现在包含我们的第一个函数名称的编码值，而EDX包含kernel32的地址，然后在第4-5行被压入堆栈。

3.现在我们将跳过对find_function的调用。重要的是要知道EAX现在包含我们在返回后要查找的函数的地址。

4.在第7行，解析函数的地址存储在EDI当前指向的地址处。

5.第8行将堆栈恢复到EAX和EDX被推送到它之前的原始状态。

6.在第9行，EDI递增以指向可以存储函数地址的下一个位置。

7.然后比较ESI和ECX以查看是否已到达散列函数列表的末尾。如果没有，函数将循环直到到达列表的末尾。如果已到达结尾，则函数返回。


查找函数地址

现在将在本部分中介绍汇编代码的两个最重要的部分。汇编代码第一部分通过PE Structure来定位_IMAGE_EXPORT_DIRECTORY。一旦找到_IMAGE_EXPORT_DIRECTORY结构，代码清单4中汇编代码第二部分将遍历函数名称列表，以找到与正在搜索的函数匹配的函数。最后，定位的地址将保存到一个位置，以后可以引用它来进行函数调用。


查找导出的函数名称

代码清单3中的第一部分代码将定位_IMAGE_EXPORT_DIRECTORY结构。找到结构后，将收集存储在NumberOfFunctions变量中的导出函数的数量和存储在AddressOfNames变量中的导出函数名称列表的 RVA 。需要这两个值来遍历导出的函数名称以找到与正在搜索的内容匹配的函数。一旦找到匹配的函数，AddressOfNameOrdinals将用于获取导出的函数地址。

以下分步分析代码清单 3 ：

1.第3行，PUSHAD指令用于将所有当前寄存器存储在堆栈上。PUSHAD按以下顺序（自顶向下）入在栈中：EAX，ECX，EDX，EBX，原始ESP，EBP，ESI和EDI

2.kernel32.dll的基地址存储在堆栈中的ESP + 0x24（36 字节）然后被移动到第4行的EBP寄存器。

3.第5行的 move 指令将位于kernel32.dll基址偏移0x3C处的值加载到EAX寄存器中。根据PE Structure图，PE Header的基地址位于PE文件基地址的0x3C偏移处。EAX现在包含从kernel32.dll基址到PE Header地址的偏移量。

4.第6行的移动指令将EBP（kernel32.dll 的基地址）、EAX（PE 头偏移）和0x78相加，并将结果存储在EDX寄存器中。回头看PE Structure图，ExportTable结构的偏移值就在那个位置。EDX现在包含导出表相对于kernel32.dll基址的偏移值。

5.EBP（kernel32.dll 的基地址）被添加到第7行的EDX（_IMAGE_EXPORT_DIRECTORY 的偏移量）。EDX现在指向_IMAGE_EXPORT_DIRECTORY结构。结构布局见下图。

6.0x18被添加到EDX并存储在第8行的ECX 中。根据图1中的_IMAGE_EXPORT_DIRECTORY结构，NuberOfNames变量位于偏移量0x18 处。ECX现在保存导出函数的数量。

7.0x20被添加到EDX并存储在第9行的EBX 中。根据图1中的_IMAGE_EXPORT_DIRECTORY结构，AddressOfNames变量位于偏移量0x20 处。此变量包含从kernel32.dll的基址到导出的函数名称列表的相对偏移量。

8.在第10行，将EBP添加到EDX 中。EDX现在指向导出的函数名称列表。

代码清单 4：32 位迭代函数列表
  1. 第2行检查ECX是否为零。如果ECX已达到零，该函数将通过恢复寄存器并返回来完成。
  2. 在第3行，ECX-1，这表明代码将反向迭代列表。
  3. 在第4行，将EBX（函数名称列表）指向的值添加到ECX（导出名称数量）乘以4的总和并存储在ESI中。ESI现在包含从kernel32的基址到list中最后一个字符串的偏移量。
  4. EBP（kernel32.dll 的基地址）在第5行添加到ESI。ESI现在指向最后导出的函数名称，一个以 NULL 结尾的字符串。
  5. 在第7-18行，函数名称的哈希被计算并存储在EDI中。
  6. 计算出的散列存储在EDI寄存器中，在第20行与存储在 [ESP+0x28] 中的值进行比较。哈希函数名称存储在该位置。它在调用find_function函数之前被压入堆栈。
  7. 如果值不匹配，则循环移动到下一个函数名称。如果找到匹配项，则继续。
  8. 接下来的几个步骤（第22-28行）可能有点难以理解，请参阅图2以帮助您继续学习。0x24被添加到EDX（_IMAGE_EXPORT_DIRECTORY 结构）并存储在第22行的EBX 中。根据图1中的_IMAGE_EXPORT_DIRECTORY 结构，AddressOfOrdinals位于偏移量0x24处。此变量包含与导出函数对应的有序值列表的相对偏移量。
图 2：导出表可视化

  1. 在第23行，将EBP（kernel32.dll 的基地址）添加到EBX 中。EBX现在指向由AddressOfOrdinals变量引用的序数列表。
  2. EBX（序数列表地址）被添加到ECX（函数编号）的总和*2 并存储在ECX 中。ECX现在包含AddressOfFunctions列表中的偏移量，该列表将包含正在搜索的函数地址。
  3. 0x1C被添加到EDX（_IMAGE_EXPORT_DIRECTORY 结构）并存储在第25行的EBX中。EBX现在在_IMAGE_EXPORT_DIRECTORY结构中包含AddressOfFunctions变量的偏移量。
  4. 在第26行，将EBP（kernel32.dll 的基地址）添加到EBX 中，使EBX指向函数地址列表。
  5. 在第27行，ECX（函数编号）乘以 4 并添加到EBX（函数地址列表）并存储在EAX 中。EAX现在包含从kernel32.dll的基址到正在搜索的函数地址的偏移量。
  6. 在第28行，EAX（函数地址偏移）被添加到EBP（kernel32.dll 的基地址）。EAX现在包含正在搜索的函数的地址。
  7. 该值存储在 [ ESP + 0x1C ] 中，其中包含在函数开始时由PUSHAD命令压入堆栈的EDI值。这存储了函数地址，我们以后可以根据需要引用它。
  8. POPAD函数从堆栈寄存器恢复到其以前的状态，并执行返回到resolve_symbols_for_dll功能。

_start:
    jmp main
    
    ; Constants
 win32_library_hashes:
    call win32_library_hashes_return
    ; LoadLibraryA      R13
    dd 0xEC0E4E8E
    ; CreateProcessA    R13 + 0x08
    dd 0x16B3FE72
    
 ; Function: find_kernel32
 find_kernel32:
    push rsi
    mov rax, [gs:0x60]
    mov rax, [rax+0x18]
    mov rax, [rax+0x20]
    mov rax, [rax]
    mov rax, [rax]
    mov r11, [rax+0x20]   ; Kernel32 Base Stored in R11
    pop rsi
    ret
        
    ; Function: find_function
 find_function:
    mov eax, [r11+0x3C]
    mov edx, [r11+rax+0x88]
    add rdx, r11            ; RDX now points to the IMAGE_DATA_DIRECTORY structure
    mov ecx, [rdx+0x18]     ; ECX = Number of named exported functions
    mov ebx, [rdx+0x20]
    add rbx, r11            ; RBX = List of exported named functions
 find_function_loop:
    jecxz find_function_finished
    dec ecx                 ; Going backwards
    lea rsi, [rbx+rcx*4]    ; Point RSI at offset value of the next function name
    mov esi, [rsi]          ; Put the offset value into ESI
    add rsi, r11            ; RSI now points to the exported function name
        
 compute_hash:
    xor edi, edi            ; Zero EDI
    xor eax, eax            ; Zero EAX
    cld                     ; Reset direction flag
 compute_hash_again:
    mov al, [rsi]           ; Place the first character from the function name into AL
    inc rsi                 ; Point RSI to the next character of the function name
    test al, al             ; Test to see if the NULL terminator has been reached
    jz compute_hash_finished
    ror edi, 0x0D           ; Rotate the bits of EDI right 13 bits
    add edi, eax            ; Add EAX to EDI
    jmp compute_hash_again
 compute_hash_finished:
 find_function_compare:
    cmp edi, r12d           ; Compare the calculated hash to the stored hash
    jnz find_function_loop
    mov ebx, [rdx+0x24]     ; EBX contains the offset to the AddressNameOrdinals list
    add rbx, r11            ; RBX points to the AddressNameOrdinals list
    mov cx, [rbx+2*rcx]     ; CX contains the function number matching the current function 
    mov ebx, [rdx+0x1C]     ; EBX contains the offset to the AddressOfNames list
    add rbx, r11            ; RBX points tot he AddressOfNames List
    mov eax, [rbx+4*rcx]    ; EAX contains the offset of the desired function address
    add rax, r11            ; RAX contains the address of the desired function
 find_function_finished:
    ret
        
 ;  Function: resolve_symbols_for_dll
 resolve_symbols_for_dll:
    mov r12d, [r8d]           ; Move the next function hash into R12
    add r8, 0x04              ; Point R8 to the next function hash
    call find_function
    mov [r15], rax            ; Store the resolved function address
    add r15, 0x08             ; Point to the next free space
    cmp r9, r8                ; Check to see if the end of the hash list was reached
    jne resolve_symbols_for_dll
 resolve_symbols_for_dll_finished:
    ret
        
 main:
    sub rsp, 0x110             ; Allocate space on stack for function addresses
    mov rbp, rsp               ; Set ebp as frame ptr for relative offset on stack
    call find_kernel32         ; Find base address of kernel32.dll
    jmp  win32_library_hashes
 win32_library_hashes_return:
    pop r8                     ; R8 is the hash list location
    mov r9, r8
    add r9, 0x08               ; R9 marks the end of the hash list
    lea r15, [rbp+0x10]        ; This will be a working address used to store our function addresses
    mov r13, r15               ; R13 will be used to reference the stored function addresses
    call resolve_symbols_for_dll
    int3
缺少命令？
当使用 Assembly 在 64 位上运行时，有一些命令不再存在或不再有用。PUSHAD和POPAD不会在64位寄存器工作。在这种情况下，它们没有用，我们需要找到替代方案。该LODSB命令也不会在64位寄存器工作。有必要将其替换为MOV AL、[RSI]和INC RSI序列。它做完全相同的事情，只是有更多的字节和指令。
寄存器差异
64位程序集有更多可用的寄存器。添加了寄存器R8-R15。这为我们提供了8个新的地方来存放我们需要的东西。有了这个额外的存储空间是相当容易的补偿缺少PUSHAD和POPAD命令。不过也有一些考虑。为了对我们现在找到的函数地址做任何事情，我们需要调用这些函数。根据文档，64位stdcall函数调用有点不同。根据参数传递部分，R8和R9寄存器用于将第三个和第四个参数传递给函数。我们应该避免在其中长期存放任何东西。这同样适用于在RCX和RDX参数。
文件偏移量差异
PE头偏移值仍处于KERNEL32.DLL的0x3c偏移量上。在这里，查找_IMAGE_EXPORT_DIRECTORY的偏移量有点不同。检查下图3中WinDbg中的_IMAGE_DOS_HEADER和_IMAGE_NT_HEADERS64的表转储，我们可以看到布局只是略有不同。_IMAGE_DATA_DIRECTORY位于OptionalHeader偏移量0x18的0x70处。这意味着总偏移量需要是0x88，而不是32位模式下的0x78。好消息是IMAGE_DATA_DIRECTORY的布局仍然与图1相同。
