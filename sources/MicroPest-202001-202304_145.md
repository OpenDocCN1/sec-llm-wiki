    下一篇将介绍个PE的插入技术，这里先作个铺垫，熟悉并回忆一下以前的基础知识。

  

    我们来看看最简单的PE Header，以一个16位的 MS-DOS 可执行文件Stub为例。

  

![](../Images/c83a50d5dfb58a99c4cdee5140ea6a8e.png)

这是详细的 MS-DOS 头。  

  

#define IMAGE\_DOS\_SIGNATURE               0x5A4D          // MZ

#define IMAGE\_NT\_SIGNATURE                  0x00004550  // PE00

typedef struct \_IMAGE\_DOS\_HEADER {      // DOS .EXE header

    WORD   e\_magic;              // Magic number

    WORD   e\_cblp;                 // Bytes on last page of file

    WORD   e\_cp;                    // Pages in file

    WORD   e\_crlc;                  // Relocations

    WORD   e\_cparhdr;           // Size of header in paragraphs

    WORD   e\_minalloc;          // Minimum extra paragraphs needed

    WORD   e\_maxalloc;          // Maximum extra paragraphs needed

    WORD   e\_ss;                     // Initial (relative) SS value

    WORD   e\_sp;                     // Initial SP value

    WORD   e\_csum;                // Checksum

    WORD   e\_ip;                      // Initial IP value

    WORD   e\_cs;                      // Initial (relative) CS value

    WORD   e\_lfarlc;                  // File address of relocation table

    WORD   e\_ovno;                  // Overlay number

    WORD   e\_res\[4\];                 // Reserved words

    WORD   e\_oemid;                // OEM identifier (for e\_oeminfo)

    WORD   e\_oeminfo;             // OEM information; e\_oemid specific

    WORD   e\_res2\[10\];              // Reserved words

    LONG   e\_lfanew;                 // File address of new exe header

  } IMAGE\_DOS\_HEADER, \*PIMAGE\_DOS\_HEADER;

  

    每个 PE 文件都以一个 MS-DOS 可执行文件开始，该文件以 IMAGE\_dos\_signature 开始。ASCII 表示为0x5a4d，即 MZ。字母“ MZ”代表 Mark Zbikowski，他是 MS-DOS 的原始架构师之一，也是 MS-DOS 可执行格式的设计者。

  

![](../Images/a8c84cbf3c94f640205214e10498f6b6.png)

在上面的图片中，0x3c是IMAGE\_dos\_header的e\_lfanew成员。此地址指向新的 EXE 标头，即 \_image\_nt\_headers。在本例中，e\_lfanew 包含0x00000120，它指向结构 IMAGE\_nt\_headers 的开头,包含IMAGE\_nt\_Signature的签名。经过实验，Windows加载程序只关心 e\_magic和e\_lfanew，成员来自 IMAGE\_dos\_header，因为 DOS 头的其余成员被 MS-DOS 用来执行Stub。  

  

![](../Images/a8ccf10d2e1a8cb5e12ffdc3643bd558.png)

  

在任何32位/64位 PE 文件中，都会看到一个很小的 MS-DOS Stub。从偏移量0x40到0x7f 是这个64字节的Stub。让我们移除带有Stub的 MS-DOS 头并反汇编代码。

![](../Images/b50dfff4f86410a10138509a5b5b3288.png)

在 DOS下运行此程序，会报提示；

![](../Images/076a737a2ceac159e99041a85362d9eb.png)

如果我们查看 DOS 头，可以在调试器中看到这些值，这证明只需要 MS-DOS 头就可以执行它的Stub,但是正如我之前提到的，e\_magic 和 e\_lfanew 对于 Windows 加载器来说非常重要。

![](../Images/a516b9c80fde2be0e7d94fdc1b849370.png)

从上图中可以看到初始堆栈指针0x00B8的值 e\_sp 和初始指令指针0x0000的值 e\_ip。我们解码的16位汇编代码可以看到将执行和打印该文本。  

![](../Images/c8fad98a42d6d1b90e32c757f0402908.png)

我们来写一段简单的程序，C语言如下：  

  

| #include <stdio.h>int main() {  size\_t i \= 0;  while (i++<5) puts("@OsandaMalith");  return 0;} |
| --- |

下面的代码是16位 MASM。这将是我们的 MS-DOS Stub程序。

dosseg 

 

ifndef ??Version 

 option m510 

endif

 

.model small

.8086

.stack 0400h 

 

.data

    message db '@OsandaMalith',0Dh,0dh,0Ah

    len \= $ \- message

 

.code

Main proc 

    mov ax, @data

    mov ds, ax              

    mov es, ax            

 

    lea bp, message 

 

    mov cx, len

    mov bl, 2 

    

    xor dx, dx

    push dx

 

.repeat

    push cx                 

; Save CX (needed for Int 10h/AH=13h below)

    mov ah, 03h            

 ; VIDEO - GET CURSOR POSITION AND SIZE

    xor bh, bh              

; Page 0

    int 10h                

 ; Call Video-BIOS => DX is current cursor position

    pop cx    

    

    mov ah, 13h            

 ; VIDEO - WRITE STRING (AT and later,EGA)

    mov al, 1              

 ; Mode 1: move cursor

    xor bh, bh             

 ; Page 0

    int 10h      

 

    pop dx

    inc dx

    push dx

.until dx \== 5

 

    mov  ax, 4C01h          

    int  21h

 

Main endp 

end Main
