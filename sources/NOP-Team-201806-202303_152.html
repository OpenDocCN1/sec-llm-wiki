这个脚本很简单，也是以后扩展的一个基础，在此基础上再开发其他功能。

下面是源码

import requests

import sys

import time

def backup_scan():

    if  len(sys.argv) == 1 or sys.argv[1] == '-h':

        # print("Usage:(使用方法：)\npython3 backup_scan.py domain dict.txt result.txt")

        print('Usage :')

        print("eg: python3 backup_scan.py http://www.freebuf.com /root/dict.txt /root/result.txt")

        exit()

    domain = sys.argv[1]

    dict_rar_path = sys.argv[2]

    output_file = sys.argv[3]

    if domain.endswith('/'):       # 为了解决 http://www.example.com/ 中的最后一个 /  

        domain = domain[:-1]

    elif not domain.startswith('http') and not domain.startswith('https'):  # 不晓得Python如何自行判断http 和 https，所以还是让用户自行填写上

        print('别忘了输入协议头！')

        exit()

    try:

        dict_file = open(dict_rar_path, 'r+')

        headers = {

                   'User-Agent': 'Mozilla/5.0 (X11; Ubuntu; Linux x86_64; rv:57.0) Gecko/20100101 Firefox/57.0',

                   'Accept': 'text/html,application/xhtml+xml,application/xml;q=0.9,*/*;q=0.8',

                   'Accept-Language': 'zh-CN,zh;q=0.8,zh-TW;q=0.7,zh-HK;q=0.5,en-US;q=0.3,en;q=0.2',

                   'Accept-Encoding': 'gzip, deflate',

                   'Connection': 'close',

                   'Upgrade-Insecure-Requests': '1'

        }

        result_file = open(output_file, 'a+')

        result_file.write('='*50+'\n')

        result_file.write(time.asctime( time.localtime(time.time()) )+'\n')

        for i in dict_file:

            if i.startswith('/'):

                url = domain + i.strip('\n') # 字符串从字典中获取出来时候是带有 \n 的，直接拼接会出现问题

            else:

                url = domain+'/' + i.strip('\n')

            resp = requests.get(url=url, headers=headers)

            if resp.status_code == 200:

                result_file.write(url+'\n')

            else:

                continue

    except Exception as e:

        print(e)

    result_file.close() # 保证文件关闭

    print("="*(len(output_file)+14))

    print('The Result is %s'%output_file)

    print("="*(len(output_file)+14))

def main():  # 方便以后扩展

    backup_scan() 

if __name__ == '__main__':

    main()


此脚本使用Python3 编写，其中request 库是第三方依赖，大家需要自行安装 pip3 install  requests

使用方法异常简单，大家直接执行就会出现使用方法。
