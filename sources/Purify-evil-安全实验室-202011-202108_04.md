# 目标：实现for循环的完整语法 # for循环的完整语法：for....else语句# 1.作用：遍历数据时，使用的语法格式，满足条件时执行break语句提前终止循环# 2.格式：#     for 临时变量 in 容器:#         对临时变量的处理#         满足条件时，执行break语句#     else:#         在上面的for循环中没有执行的break语句，容器遍历完成，一定会执行else部分 # 定义列表info_list = [10, 20, 30, 40, 50, 60, 70] # 使用for循环完整语法格式遍历列表# 查找数据30# 如果找到了30，执行break语句，停止遍历# 如果容器遍历完成，没有找到数据弹出提示信息 # 1.找到数据info_list = [10, 20, 30, 40, 50, 60, 70] #定义列表find_data = 30 #定义变量存储需要查找的数据  for a in info_list:    print("找到数据 %d" % a) #for循环遍历临时容器数据    if find_data == a: #添加if判断从for循环遍历的临时容器中找到数据退出循环        breakelse:    print("没有找到数据 %d" % find_data) print("循环结束") print("-"*50) # 2.没有找到数据info_list = [10, 20, 30, 40, 50, 60, 70] #定义列表find_data = 90 #定义变量存储需要查找的数据  for a in info_list:    print("找到数据 %d" % a) #for循环遍历临时容器数据    if find_data == a: #添加if判断从for循环遍历的临时容器中找到数据退出循环        breakelse:    print("没有找到数据 %d" % find_data) #容器遍历完成没有找到数据 print("循环结束")
