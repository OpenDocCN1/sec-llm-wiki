# for循环# 作用：可以让指定代码重复执行，可以循环遍历容器中的数据(遍历：从容器中把数据一个一个取出)# 格式: for() 临时变量 in 容器变量：#             对临时变量的处理# 特点：临时变量从容器中把数据一个一个获取到# for循环中可以添加条件判断# for循环可以只能遍历 像列表，元组，字符串，字典这样可以遍历的容器#             不能遍历数字，浮点数，布尔类型，这样不可遍历的数字
# 定义列表：info_list = [100, "hello", True, 12.5, (10, 20)]for value in info_list:    print(value, end=" ")    # print(type(value))print()
# 定义元组info_tuple = (10, 20, 30, 40, 50, 60)for i in info_tuple:    print(i, end=" ")  # end = “指定字符串" ，以”指定字符串结尾“后同一行输出    # print(type(i), end="")  # 打印类型    print()
# 定义字符串info_str = "hello python"for s in info_str:    print(s, end=" ")    # print(type(s),end="")print()# 定义字典info_dict = {"name": "王权富贵", "age": 22, "gender": "男", "stature": 178.8}for d in info_dict:    print(d,info_dict[d]) #d获取键，info_dict获取键的值    print()for d in info_dict:    print("%s 是 %s"%(d,info_dict[d]))
# 列表例子，找到列表里的数据：info_list = [100, "hello", True, 12.5, (10, 20)]for value in info_list:    if value == 12.5:        print("已找到数据：%s" % value)    elif value == "hello":        print("已找到数据：%s" % value)    else:        print(value)    # print(value, end=" ")
