# 目标：打印星星"""        * * * * *        * * * * *        * * * * *        * * * * *        * * * * *"""
# 1.打印一颗星星# 2.打印一行星星 5颗# 3.连续打印五行星星
a = 1  # 控制星星的行数while a <= 5:
    b = 1  # 控制星星的列数    while b <= 5:        print("*", end=" ")  # 星星没打印完不换行        b += 1    print()  # 打印完之后，必须换行
    a += 1
print("循环结束", "-" * 50, "分隔符")
# 目标打印三角形："""* * * * * * * * * * * * * * * """a = 1  # 控制星星的行数while a <= 5:
    b = 1  # 控制星星的列数    while b <= a:  # 以列数制约行数        print("*", end=" ")  # 星星没打印完不换行        b += 1    print()  # 打印完之后，必须换行
    a += 1
print("循环结束", "-" * 50, "分隔符")# 目标打印九九乘法表a = 1  # 控制行数while a <= 9:
    b = 1  # 控制列数    while b <= a:  # 以列数制约行数,一行一列，两行两列 递增        #\t 表示tab键，文本输出显示时，在竖直方向保持对其        print("%d * %d = %d\t" % (b, a, b * a),end=" ")  # 没打印完不换行        b += 1    print()  # 打印完之后，必须换行
    a += 1
