# print()函数拓展# print()函数输出# print()函数打印输出自带换行，因为有end = "\n" (\n是换行符)print("---------")print("=========")# print()函数打印完输出后不换行 end = "指定字符串"，print()函数打印完后，以”指定字符串“结尾#格式：end = “指定字符串" ，以”指定字符串结尾“后同一行输出print("----------", end="%")print("==========")# print()函数打印完输出完不换行以空字符串为结尾print("----------", end="")print("==========")
# 拓展：print()函数一次性输出单个或多个数据,变量时，默认以空格隔开#格式：sep = "指定字符串"，以“指定字符串”隔开print("小明", "男", 18, 178.5)print("小李", "男", 19, 178.5, sep=" ")print("小王", "男", 20, 178.5, sep="##")
