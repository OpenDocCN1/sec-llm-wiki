# 目标：实现whlie循环嵌套# 项目：道歉三次，连续三天# 定义变量保存天数，定义变量保存道歉次数
i = 1 #定义变量保存天数 -- 计数器while i<= 3:
    j = 1 #定义变量保存道歉次数 -- 计数器    while j <= 3:        print("对不起 %d 次，一套流程" % j)        j += 1 #修改循环变量+1次 --修改计数器
    print("-----第%d天-----,道歉" %i)    i += 1 #修改循环变量+1天 -- 修改计数器
print("循环结束")
# 2.while循环分析i = 1 #保存天数while i<= 3:    print("-----第%d天-----,道歉" %i)
    j = 1 #道歉次数    while j <= 3:        print("对不起 %d 次，一套流程" % j)        j += 1
    i += 1
print("循环结束")
# ps：whlie循环嵌套# 1.概念：wulie循环中嵌套whlie循环# 2.作用：用来处理排序，一些特殊场景（平面坐标）# 3.特点：外循环执行一次，内循环执行一圈# 4.在坐标系中，外层循环变量一边控制行号/内层循环变量一般控制列号# 5.从内到外读循环容易理解
