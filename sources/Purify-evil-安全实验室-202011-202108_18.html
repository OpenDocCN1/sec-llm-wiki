# 导入模块随机数模块import random # 格式：random.randint(a,b)  a是随机数的下限（最小值），b是随机数上线（最大值）# random.变量(下限数，上限数)print(random.randint(12, 20)) #生成随机数n：12 <=n <=20 上限是20，下线是12print(random.randint(20, 20)) #结果永远是20# print(random.randint(20, 12)) ValueError: empty rangse for randrange() (20,13, -7) 返回错误，下限大于上限
