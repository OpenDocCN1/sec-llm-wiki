# 需求：把字符串"hello,python,python,java"把第二个"python"换成"go" # 1.定义一个字符串info = "hello,python,python,java" # 使用split切分print(info.split(",")) #返回切分['hello', 'python', 'python', 'java']#print(type(info)) 返回类型：<class 'str'>字符串 #定义一个新的变量保存到列表类型切分，然后索引替换new_list = info.split(",")new_list[2] = "go"print(new_list) #返回替换结果['hello', 'python', 'go', 'java']print(type(new_list)) #返回列表类型：<class 'list'>#使用join拼接列表到一起print(",".join(new_list)) #返回替换结果 hello,python,go,java
