"""if语句中嵌套if语句作用：外层if条件判断成立的前提条件下，进行内层if条件判断#1.数字类型：int（整数） float（浮点数/小数） bool（布尔）=> 参与运算时，True=1/False=0#2.非数字类型：str "字符串" #list[列表] tuple（元组） dict{字典}语法格式：if 条件1：    条件1满足执行的代码    if条件1基础上的条件2:    条件2满足时，执行的代码 #条件2不满足的处理 else:    条件2不满足时，执行的代码#条件1不满足时的处理else:    条件1不满足时，执行的代码    """# 目标：使用if嵌套上火车# 1.定义bool类型变量 has_ticket 表示是否有车票has_ticket = True #True 有车票 False没有车票# 2.定义int类型knif_length 表示刀的长度，单位：cm/厘米knif_length = 22# 3.首先检查是否有车票，如果有才允许进行安检# 4.安检时，需要检查刀的长度，大于20cm不允许上车/小于20cm，安检通过if has_ticket:    print("有车票允许上车")    if knif_length > 20:        print("刀具长度危险不允许上车")    else:        print("安检刀具不危险允许上车")else:    print("没有车票请补票")
