# elif多次判断# 语法格式：# if 条件:#     条件1满足，执行的代码# elif 条件2:#     条件2满足时，执行的代码# elif 条件3:#     条件3满足时，执行的代码# else:#     上述条件都不成立时，执行else部分代码#根据考试成绩，进行分级 优秀 良好 中等 差劲 不及格 # 定义一个变量保存分数score = 81# 使用if...elif...else进行判断if score>= 90:    print("优秀")elif score>=80 and score<90:    print("良好")elif score >= 70 and score <80:    print("中等")elif score >=60 and score <70:    print("差劲")else:    print("不及格") print("程序已结束") print("-"*50)# 优化：if score>= 90:    print("优秀")elif score>=80:    print("良好")elif score >= 70:    print("中等")elif score >=60:    print("差劲")else:    print("不及格") print("程序已结束")
