# 数据类型转换：# 1.数字类型：int（整数） float（浮点数/小数） bool（布尔）=> 参与运算时，True=1/False=0# 2.非数字类型：str "字符串" #list[列表] tuple（元组） dict{字典} # 作用：把一种数据类型转换为另一种数据类型# int(x) 把x转换为整数，x如果是字符串，必须是数字类型的# 浮点数转整数x = 123.6print(type(x))  # 浮点数：<class 'float'>print(type(int(x)))  # 整数:<class 'int'>print("-" * 50) # 字符串转整数x = "123"print(type(x))  # 字符串：<class 'str'>print(type(int(x)))  # 整数：‘<class 'int'>print("-" * 50)# 字符串转整数/因为 "hello python" 是非数字类型所以报错x = "hello python"print(type(x))  # 字符串：<class 'str'># print(type(int(x))) #ValueError: invalid literal for int() with base 10: 'hello python' 报错print("-" * 50) # 数字类型：int（整数） float（浮点数/小数） bool（布尔）=> 参与运算时，True=1/False=0# int(x) x如果是字符串，必须是数字类型的# float(x) x如果是字符串，必须是数字类型的# str(x) 把x转换为任何类型，任何数据都可以转换为字符串类型 # 序列类型（包括：字符串，列表，元组）# list(x) 把序列数x转换为列表# tuple(x) 把序列数x转换为列表 x = 123.6  # 浮点类型print(int(x))print(type(int(x)))  # 转换为<class 'int'>整数类型，x如果是字符串，必须是数字类型的print("-" * 50) x = 123  # 整数类型print(float(x))print(type(float(x)))  # 转换为<class 'float'>浮点数类型，x如果是字符串，必须是数字类型的print("-" * 50) x = 123  # 整数类型print(str(x))print(type(str(x)))  # 转换为<class 'str'>字符串类型，任何数据都可以转换为字符串类型print("-" * 50) x = "123"  # 字符串类型print(list(x))print(type(list(x)))  # 转换为<class 'list'>列表类型，把序列数x转化为列表  序列类型（包括：字符串，列表，元组）print("-" * 50)x = (10,20,30) #元组类型print(type(x))print(type(list(x)))print("-" * 50) x = "123"  # 字符串类型print(tuple(x))print(type(tuple(x))) #转换为<class 'tuple'>元组类型，把序列数x转化为列表  序列类型（包括：字符串，列表，元组）print("-" * 50) # 逆转-拼接字符串x = "python"  # 字符串类型print(tuple(x))print(type(tuple(x)))print("-"*50)#Python join() 方法用于将序列中的元素以指定的字符连接生成一个新的字符串。# 格式：字符.join(类型(变量名))print("+".join(tuple(x)))print("#".join(tuple(x)))print(" ".join(tuple(x)))print("".join(tuple(x)))
