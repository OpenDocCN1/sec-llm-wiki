# 目标：实现字典的常见操作方法：增 => 删 => 改 =>查
# 字典组成：字典名{key:value,key1:value1....}# 定义字典 保存 慕容铁汁 的详细信息info_dict = {"name":"慕容铁汁",             "age":20,             "high":165.5,             "gender":"男"    }print("字典操作前：", info_dict)print("-"*50)# 查# 1.字典名[key] key存在,取对应的value值，key不存在,返回错误，后面的代码不会执行info_dict = {"name":"慕容铁汁",             "age":20,             "high":165.5,             "gender":"男"    }print(info_dict["name"])print(info_dict["age"])# print(info_dict["agee"]) #不存在的key# print("这一行是否正常执行") KeyError: 'agee' key不存在报错不执行print("-"*50)
# 2.字典名.get(key:value) key存在,取对应的value值，key不存在,不会报错，返回None,后面的代码正常执行info_dict = {"name":"慕容铁汁",             "age":20,             "high":165.5,             "gender":"男"    }
print(info_dict.get("name"))print(info_dict.get("age"))print(info_dict.get("agee","航头镇")) #key不存在不会返回错误，如果键不在字典中返回默认值 None 或者设置的默认值。print("这一行是否正常执行")print("-"*50)
#增/改# 1.字典名[key] = 新值 key存在修改，修改对应的vlaue值，key不存在，新建 键值对info_dict = {"name":"慕容铁汁",             "age":20,             "high":165.5,             "gender":"男"    }print("字典操作前：", info_dict)info_dict["name"] = "上官翠花"print(info_dict) #key存在修改info_dict["addr"] = "航头镇"print(info_dict) #key不存在，新建 键值对print("-"*50)
# 2.字典名.setdefault(key,value) key存在，不修改对应value值，使用默认值 #key不存在,新建键值对info_dict = {"name":"慕容铁汁",             "age":20,             "high":165.5,             "gender":"男"    }info_dict.setdefault("name","西门二狗")print(info_dict) #key存在，不修改对应value值，使用默认值info_dict.setdefault("addr","航头镇")print(info_dict) #key不存在,新建键值对print("-"*50)
# 3.字典1.update(字典2) 把字典2合并到字典1，key存在更新对应的value值，key不存在，创建新的键值对info_dict = {"name":"慕容铁汁",             "age":20,             "high":165.5,             "gender":"男"    }print("字典操作前：", info_dict)#存在键值对info_dict.update({"name":"欧阳狗蛋","age":23,"high":156.2,"gender":"男"})print(info_dict) #key存在更新对应的value值#不存在键值对info_dict.update({"name":"欧阳铁蛋","weight":600,"high":156.2,"gender":"男"})print(info_dict) #key不存在，创建新的键值对print("-"*50)
#删# 1.del 字典名[key] key存在，正常删除键值对，key不存在，报错info_dict = {"name":"慕容铁汁",             "age":20,             "high":165.5,             "gender":"男"    }print("字典操作前：", info_dict)#存在keydel info_dict["name"]print(info_dict)#不存在key# del info_dict["name"]# print(info_dict) #刚刚name已经删除所以不存在了，KeyError: 'name'不存在报错print("-"*50)
# 2.字典名.pop(key)  key存在，正常删除键值对，key不存在，报错info_dict = {"name":"慕容铁汁",             "age":20,             "high":165.5,             "gender":"男"    }print("字典操作前：", info_dict)#存在keyinfo_dict.pop("name")print(info_dict)#不存在key# info_dict.pop("name")# print(info_dict) #刚刚name已经删除所以不存在了，KeyError: 'name'不存在报错print("-"*50)
# 3.字典名.clear() 清空字典info_dict = {"name":"慕容铁汁",             "age":20,             "high":165.5,             "gender":"男"    }print("清空字典前：", info_dict)info_dict.clear()print("清空字典后:",info_dict)
# 统计字典中键值对的个数info_dict = {"name":"慕容铁汁",             "age":20,             "high":165.5,             "gender":"男"    }print("清空字典前：", info_dict)print(len(info_dict),"对,键值对")
