# 字典中的注意事项：# 1.字典中的键key是唯一的# 字典中有相同的键key对应不同的值，后面键对应的值会替换前面的值my_dict = {"name":"常威","age":30,"high":180.5,"age":40,"age":50}print(my_dict)print("-"*50)#{'name': '常威', 'age': 50, 'high': 180.5} # 2.字典中的key键的数据类型，可以是除了列表和字典以外的所有数据类型# 字典中的键key不能是列表或字典本身info_dict = {"name":"来福",100:"一百",12.3:"小数",True:"是真的",(10,20,30):"元组"}# info_dict = {"name":"来福",100:"一百",12.3:"小数",True:"是真的",[10,20,30]:"列表"} TypeError: unhashable type: 'list'# 注意：列表不能作为字典的键# info_dict = {"name":"来福",100:"一百",12.3:"小数",True:"是真的",{"one":1,"two":2}:"字典"}  #TypeError: unhashable type: 'dict'# 注意：字典本身也不能做字典的键print(info_dict)print(type(info_dict))print("-"*50)# 字典名{key:value}# 字典的value值是可以是任意数据类型的 # 3.字典是无序的数据容器my_dict_1 = {"name":"常威","age":30,"high":180.5}my_dict_2 = {"high":180.5,"name":"常威","age":30}print(my_dict_1 == my_dict_2) #返回True，只需要字典里的键值成对匹配print("-"*50)my_dict_1 = {"name":"常威","age":30,"high":180.5}my_dict_2 = {"high":180.5,"name":"常威","age":30,"gender":"男"}print(my_dict_1 == my_dict_2) #返回False，字典内的成对键值不匹配
