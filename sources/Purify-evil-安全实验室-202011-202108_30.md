# 元组基础 # 定义：# 元组：也是用来存储多个数据的高级容器（可以存储任意数据类型）#   注意：元组中的数据修改是收到限制的# 格式：元组名 = (数据1,数据2,数据3 .......)# 元组名 = () 或 元组名 = tuple() => 空元组 my_tuple = (100, 12.3, False, "python", 20, "中文")# my_tuple =() 空元组# my_tuple = tuple() 空元组print(my_tuple)print(type(my_tuple)) #类型：<class 'tuple'>print("-"*50) # 定义只有一个数据的元组，需要加逗号my_tuple =(30,) #加一个逗号<class 'tuple'>是元组my_tuple_01 =(30) #不加逗号<class 'int'>是整数print(my_tuple,my_tuple_01)print(type(my_tuple),type(my_tuple_01))print("-"*50) #通过索引访问数据：元组名[索引值]# 索引：0    1      2       3       4    5my_tuple = (100, 12.3, False, "python", 20, "中文")print(my_tuple[0])# print(my_tuple[6]) #IndexError: tuple index out of range 元组索引越界错误print("-"*50)
