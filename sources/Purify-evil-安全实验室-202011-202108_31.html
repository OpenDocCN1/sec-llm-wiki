# 目标：实现对列表的常见操作# 增=>删=>改=>查 # 定义列表# 列表的快速格式化 ctrl+alt+l (在列表逗号后面添加空格符合格式)# 索引：      0  1  2  3  4  5  6  7  8info_list = [1, 2, 3, 4, 5, 6, 7, 8, 9]print("原来的列表是：", info_list)  # 原来的列表是： [1, 2, 3, 4, 5, 6, 7, 8, 9]print("-" * 50) # 增加：# 列表.insert(索引,数据) 在指定位置插入数据info_list = [1, 2, 3, 4, 5, 6, 7, 8, 9]info_list.insert(2, 100)  # 增加数据[100, 1, 2, 3, 4, 5, 6, 7, 8, 9] 插入到索引值后面一位print(info_list)print("-" * 50)# 列表.insert(索引,[数据]) 插入小列表info_list = [1, 2, 3, 4, 5, 6, 7, 8, 9]info_list.insert(2, [100, 200])print(info_list)print("-" * 50) # 列表.append(数据) 在末尾追加数据 =>着重记忆info_list = [1, 2, 3, 4, 5, 6, 7, 8, 9]info_list.append(200)print(info_list)print("-" * 50)# 列表.append(索引,[数据]) 插入小列表info_list = [1, 2, 3, 4, 5, 6, 7, 8, 9]info_list.append([200, 300])print(info_list)print("-" * 50) # 列表合并：列表1.extend(列表2) 将列表2的数据追加到列表1(把列表2合并到列表1去)# info_list = [1, 2, 3, 4, 5, 6, 7, 8, 9]# info_list.extend([200,300])# print(info_list) info_list_01 = [1, 2, 3]info_list_02 = [4, 5, 6]info_list_01.extend(info_list_02)print(info_list_01)print("-" * 50) # 删除：# del 列表[索引] 删除指定的数据# 索引：      0  1  2  3  4  5  6  7  8info_list = [1, 2, 3, 4, 5, 6, 7, 8, 9]del info_list[0]print(info_list)print("-" * 50)# 拓展：把列表从内存清除# del info_list# print(info_list) #NameError: name 'info_list' is not defined 列表已删除  # 列表.remove(数据) 删除第一个出现的指定的数据 => 常用info_list = [1, 2, 3, 4, 5, 6, 7, 8, 9]info_list.remove(4)print(info_list)print("-" * 50) # 列表.pop()删除末尾数据(返回删除的数据)info_list = [1, 2, 3, 4, 5, 6, 7, 8, 9]num = info_list.pop()print(info_list)print(num)print("-" * 50)# 列表.pop(索引)删除末尾数据(返回删除的数据)# 索引：      0  1  2  3  4  5  6  7  8info_list = [1, 2, 3, 4, 5, 6, 7, 8, 9]num = info_list.pop(0)print(info_list)print(num)print("-" * 50) # 列表.clear()清空列表info_list = [1, 2, 3, 4, 5, 6, 7, 8, 9]info_list.clear()print(info_list)print("-" * 50) # 修改：# 列表[索引] = 数据 修改指定索引的数据# 列表中的数据可以修改# 索引：      0  1  2  3  4  5  6  7  8info_list = [1, 2, 3, 4, 5, 6, 7, 8, 9]info_list[0] = 111print(info_list)print("-" * 50)# 查询：# 通过索引获取数据# 列表名[索引]# 索引：      0  1  2  3  4  5  6  7  8info_list = [1, 2, 3, 4, 5, 6, 7, 8, 9]print(info_list[0])print(info_list[1])print("-" * 50) # 通过数据获取索引# 列表名.index(数据) 获取数据在列表中第一次出现的索引值# 索引：      0  1  2  3  4  5  6  7  8info_list = [1, 2, 3, 4, 5, 6, 7, 8, 9]print(info_list.index(2))# print(info_list.index(222)) ValueError: 222 is not in list 列表越界错误print("-" * 50) # 列表的高级操作 # 定义列表my_list = [1, 2, 3, 4, 5, 55, 6.6, 7, -8, 9, 0.2] # 统计# len(列表) 列表长度 (列表中元素的个数)my_list = [1, 2, 3, 4, 5, 55, 6.6, 7, -8, 9, 0.2]print(len(my_list))print("-" * 50) # 列表.count(数据) 数据在列表中出现的次数my_list = [1, 2, 3, 4, 4, 4, 55, 55, 6.6, 7, -8, 9, 0.2]print(my_list.count(4))print(my_list.count(55))print(my_list.count(666)) #没有这个数据返回0print("-" * 50) #排序# 列表.sort() 升序(从小到大排序)my_list = [1, 2, 3, 4, 4, 4, 55, 55, 6.6, 7, -8, 9, 0.2]my_list.sort()print(my_list)print("-" * 50) # 列表.sort(reverse=True) 升序(从大到小排序)my_list = [1, 2, 3, 4, 4, 4, 55, 55, 6.6, 7, -8, 9, 0.2]my_list.sort(reverse=True)# my_list.sort(reverse=False) 从小到大排序print(my_list)print("-" * 50) #列表.reverse() 逆转，反转,倒序my_list = [1, 2, 3, 4, 4, 4, 55, 55, 6.6, 7, -8, 9, 0.2]my_list.reverse()print(my_list)print("-" * 50) #列表拷贝#列表名.copy()my_list = [1, 2, 3, 4, 4, 4, 55, 55, 6.6, 7, -8, 9, 0.2]new_list = my_list.copy()print(new_list)# id()函数可以查看变量保存数据的内存地址 => 拓展print(my_list, id(my_list)) #内存地址：2778205491016print(new_list, id(new_list)) #内存地址：2778205512392# my_list.copy()# new_list = my_list# print(new_list) ##dir(查看变量或数据或对象 所有的方法和属性)print(dir(list))#['__add__', '__class__', '__contains__', '__delattr__', '__delitem__', '__dir__', '__doc__', '__eq__', '__format__', '__ge__', '__getattribute__', '__getitem__', '__gt__', '__hash__', '__iadd__', '__imul__', '__init__', '__init_subclass__', '__iter__', '__le__', '__len__',  '__lt__', '__mul__', '__ne__', '__new__', '__reduce__', '__reduce_ex__', '__repr__', '__reversed__', '__rmul__', '__setattr__', '__setitem__', '__sizeof__', '__str__', '__subclasshook__', #
