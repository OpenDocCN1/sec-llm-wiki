##字符串的查找与替换： #查找：#定义字符串：#索引值：012345............18#my_str = "hello,python,python" # string.startswith(str)#判断：变量名.startswith(字符串) 检查字符串是否以xxx开头，是则返回True,否则返回False#可以查找单个字母，也可以查找多个my_str = "hello,python,python"print(my_str.startswith("h"))print(my_str.startswith("he"))print(my_str.startswith("hello")) print(my_str.startswith("e"))#print(my_str.startswith("h"),my_str.startswith("o")) # string.endswith(str)#判断：变量名.endswith(字符串) 检查字符串是否以xxx结束，是则返回True,否则返回False#可以查找单个字母，也可以查找多个my_str = "hello,python,python"print(my_str.endswith("n"))print(my_str.endswith("on"))print(my_str.endswith("python")) print(my_str.endswith("java"))#print(my_str.endswith("o"),my_str.endswith("n")) #string.find(str, start=0,end=len(string))#索引值：012345............18 在指定范围内查找到则返回索引值，否则返回-1#my_str = "hello,python,python"my_str = "hello,python,python"print(my_str.find("h")) #已查到到，返回索引值print(my_str.find("python")) #返回最左边第一个字符串的索引值print(my_str.find("j")) #未查找到，返回-1 #startswith和end指定范围内查找检查是否包含在内#print(my_str.find("查找内容", 从第几个索引值开始查找，len（查找范围）))#要查找字符串，要从查找内容后一个索引值开始查找my_str = "hello,python,python"print(my_str.find("python", 7)) #指定查找范围，从8索引值往后查找最左边第一个字符串的索引值print(my_str.find("python", 7, len(my_str))) #指定范围内查找print(my_str.find("java", 1, len(my_str))) #指定范围内字符串查找不到返回 -1print("上行代码报错是否会继续执行？") #这行代码依旧执行#find与index 两个查找函数对比代码是否运行my_str = "hello,python,python"print(my_str.index('python' ,7)) #要查找字符串，要从查找内容后一个索引值开始查找#print(my_str.index("java"))#index报错后后面的代码无法正常执行print("上行代码报错是否会继续执行？")  #替换：#string.replace(old_str,new_str,num=string.count(old))# 把string中的old_str替换成new_str/变量.replace("替换字符串", "被替换字符串"，替换次数)my_str = "hello,python,python,java"print(my_str.replace("python", "go"))  #num值没有指定，默认全部替换print(my_str.replace("python", "go",1)) #num值指定为1，替换1次
