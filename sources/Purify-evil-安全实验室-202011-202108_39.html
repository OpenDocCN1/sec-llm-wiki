#数字类型分类：#1.数字类型：int（整数） float（浮点数/小数） bool（布尔）=> 参与运算时，True=1/False=0#2.非数字类型：str "字符串" #list[列表] tuple（元组） dict{字典}
#通过type（）函数可以查看当前数据所属类型 => 使用：type（变量名）/type（数据）
a = 123print(type(a)) #int（整数）
a = 178.5print(type(a)) #float（浮点数/小数）
a = Trueb = Falseprint(type(a),type(b)) #bool（布尔 ）print(a+1,b+1) #参与运算时，True=1/False=0
a = "你好"print(type(a)) #str"字符串"
a = [10,20,30]print(type(a)) #list[列表]
a = (10,20,30)print(type(a)) #tuple（元组）
a = {'one':1 , "two":2}print(type(a)) #dict{字典}
