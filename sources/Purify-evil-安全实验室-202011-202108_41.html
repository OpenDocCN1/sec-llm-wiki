#比较运算符: ==,!=,>,<,>=,<=#作用：用作比较两个操作数的大小，通常和if判断用于一起#False 假0----- True 真1 所有比较运算符返回1表示真，返回0表示假。这分别与特殊的变量True和False等价。注意，这些变量名的大小写a=10b=20print(a==b) #等于 - 比较对象是否相等print(a!=b) #不等于 - 比较两个对象是否不相等print(a>b) #大于 - 返回a否大于bprint(a<b) #小于 - 返回a否小于bprint(a>=b) #大于或等于 - 返回a是否大于bprint(a<=b) #小于或等于 - 返回a是否小于等于b
#比较运算符if实例演示：a = 21b = 10
if (a == b):    print("1：a 等于 b")else:    print("1：a 不等于 b")
if (a != b):    print("2：a 不等于 b")else:    print("2：a 等于 b")
if (a < b):    print("3：a 小于 b")else:    print("3：a 大于等于 b")
if (a > b):    print("4：a 大于 b")else:    print("4：a 小于等于 b")
# 修改变量 a 和 b 的值a = 0b = 1if (a <= b):    print("5：a 小于等于 b")else:    print("5：a 大于  b")
if (b >= a):    print("6：b 大于等于 a")else:    print("6：b 小于 a")
#逻辑运算符：and，or，not#作用：判断真假a=0 #false（假）b=1 #true （真）
print(a and b)print(b and b) #布尔"与" - 两边都为真，结果为真
print(a or a)print(a or b) #布尔"或" - 两边只要有一个为真，结果就为真
print(not a) #布尔"非" - 取反，取否定
#逻辑运算符if实例演示：a = 0b = 1
if (a and b):    print("1：变量 a 和 b 都为 true")else:    print("1：变量 a 和 b 有一个不为 true")
if (a or b):    print("2：变量 a 和 b 都为 true，或其中一个变量为 false")else:    print("2：变量 a 和 b 都不为 true")
