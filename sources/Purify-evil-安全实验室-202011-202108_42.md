#print（）拓展
#定义变量 保存姓名name = "张全蛋"#定义变量 保存年龄age = 22#定义变量 保存性别gender = "男"#定义变量 保存身高high = 178.5
#使用print（）函数打印变量中的内容,print()函数自带换行print(name)print(age)print(gender)print(high)
#使用print（）函数一次性打印多个函数
print(name, age, gender, high)
#占位符：%d（数字） %s（字符串） %f（浮点数/小数/默认保留六位，可以使用%.xf指定x保留几位） %%（两个百分号表示一个百分号）#使用格式：print（"字符串信息 占位符" % 变量或数据）name = "小张"print("姓名：%s " % name)
age = 22print("年龄：%d 岁" % age)
gender = "男"print("性别：%s" % gender)
high = 178.5print("身高：%f" % high) #默认保留六位print("身高：%.1f" % high) #指定保留几位num = 89print("合格率：%d%%" % 89)
#print（）一次把多个变量全部输出#格式化输出，按照一定的格式输出
print("姓名：%s，年龄：%d，性别：%s，身高：%.1f" % (name,age,gender,high))
#任何数据都可以转化为字符串，占位符都可以替换为%sprint("姓名：%s，年龄：%s，性别：%s，身高：%s" % (name,age,gender,high))
