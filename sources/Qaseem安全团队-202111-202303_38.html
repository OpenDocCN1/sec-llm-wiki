/app/admin/controller/Friendshiplinks.php
/app/admin/controller/Payment.php
漏洞触发点在后台添加滴滴友链处，token参数是我们可控的， 在didi函数中，请求值不经过过滤就file_put_contents()写入PHP文件，导致代码执行注入漏洞 ,访问前台任意界面即可getshell
  public function didi()
    {
        if (request()->isPost()) {
            $site_id = input('siteid');
            $token = input('token');//post获取参数token
            $code = <<<INFO
<?php
return [
    'siteid' => '{$site_id}',
    'token' => '{$token}'
 ];
INFO;
            $file = App::getRootPath() . 'config/didi.php';
            if (!file_exists($file)) {
                return json(['err' => 1, 'msg' => '配置文件不存在']);
            }
            file_put_contents($file, $token);//未经检验直接写入导致getshell
            return json(['err' => 0, 'msg' => '修改成功']);
        }
        $site_id = config('didi.siteid');
        $token = config('didi.token');
        View::assign([
            'siteid' => $site_id,
            'token' => $token
        ]);
        return \view();
    }
}
