```
from concurrent.futures.thread import ThreadPoolExecutorimport requestsimport urllib3
# 取消报错urllib3.disable_warnings(urllib3.exceptions.InsecureRequestWarning)# 常见的网站源码备份文件名list1 = ['web', 'website', 'backup', 'back', 'www', 'wwwroot', 'temp']# 常见的网站源码备份文件后缀list2 = ['tar', 'tar.gz', 'zip', 'rar', '7-zip', '7z']# 请求头headers = {    'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; WOW64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/57.0.2987.110 Safari/537.36'}
def scan(url):    # 检测URL显示    print(url)    # 错误排除    r = requests.get(url + "/1223.zip", headers=headers, verify=False, timeout=3)    if r.status_code == 200:        return    for i in list1:        for j in list2:            back = str(i) + '.' + str(j)            scan_url = url + '/' + back            r = requests.get(scan_url, headers=headers, verify=False, timeout=3)            # 200认为存在            if r.status_code == 200:                # 写入result.txt                with open('result.txt', 'a')as f:                    f.write(scan_url + "\t" + str(len(r.text)) + '\n')                    f.close()                # 输出成功信息                print("[++++++] " + scan_url + "-->" + str(r.status_code))                # 找到了直接返回                return
if __name__ == '__main__':    # 清空result.txt    file = open("result.txt", 'w').close()    # 进程池数10    with ThreadPoolExecutor(max_workers=10) as executor:        # 输入文件, 同目录下的url.txt        for i in open("url.txt", errors="ignore").readlines():            executor.submit(scan, i.strip())
```
