linux  /etc                                         /*一般存放配置文件*/

       /dev                                        /*一般存放硬件文件*/

       /bin or /sbin or /usr/bin or /usr/sbin     /*一般存放命令文件*/ 

       /boot                                     /*一般存放内核和grub等文件*/               <!--sbin和bin的区别在于sbin即是superbin也就是说只有超级管理员才能执行修改的-->   

        

                                         注意了linux的单用户可以大致理解为是win的安全模式也就是说最小系统程序启动模式

       /lib                                    /*系统调用的函数库保存位置*/

       /media                                 /*挂载目录挂载媒体设备的目录*/

       /mnt                                  /*系统建议是用来挂载额外的设备的目录例如u盘或者移动硬盘*/

       /opt   or   /usr/local               /*第三方安装的软件保存位置*/

       /var                                /*动态数据保存位置主要保存缓存和日志以及软件的运行所产生的文件*/

      /tmp                                /*临时目录就是我们自己实验或者玩的目录建议*/

文件第一个字符 -表示文件

               d表示目录

               l表示软连接（类似快捷方式）



文件权限分类一般是 u g o

硬链接的特点：

  相当于拷贝cp -p和同步更新

chmod 权限管理命令   //改变文件或目录权限  u所有者 g所属组 o其他人 a所有人 +表示增加权限 -表示减少权限 =表示直接赋值权限

      例如chmod u+x 文件名 //指给所有者增加一个执行权限

      例如chmod g+w，o-r 文件名 //指给所属组加一个写的权限同时给其他人去掉读的权限 （注意多个操作要用“，”（逗号）隔开）

      例如chmod g=rwx 文件名    //指给所属组直接赋予读写执行的权限

      

然而Linux一般是用数字授予权限位的

r----4

w----2

x----1

例如

rwxrw-r--  //就是764

532        //r-x-wxr--


例如chmod 640 文件名 //就是rw-r-----


chmod -R  文件目录（名）  //递归修改（改变目录的同时也更改目录下的子目录权限）

例如chmod -R 777 /tmp/love       //就会更改love下的BuDing以及BuDing的子目录orochi以及love目录的权限


useradd orochi //创建账户的命令

 pwd查看当前所在目录

rwx权限具体是指

 ----------------------------------------------------------------

 | r  | 读权限 可以查看文件    | 可以列出目录中的内容            |

 | w  |写权限 可以修改文件内容 |   可以在目录中创建、删除文件    |

 | x  | 执行权限 可以执行文件  |  可以进入目录                   |

 ----------------------------------------------------------------

    -S 以rwx形式显示新建文件缺省权限


修改默认的权限形式

find命令

find [搜索范围] [匹配条件]


文件搜索

例如：

find /etc -name init  //在目录/etc中查找文件init

-iname 不区分大小写

例如：

find /etc -iname init  //在目录/etc中查找文件名为init（或者大写的init的）


find size 命令 

+n  //大于

-n  //小于

n   //等于

例如：

find / -size +204800 //在根目录下查找大于100mb的文件   注：单位是数据块 也就是512字节 也就是 0.5k


find -group命令

根据所属组查找

例如：

find /home -user orochi  //在根目录查找所有者为orochi的文件

-admin访问时间 access

-cmin 文件属性change

-mmin 文件内容modify

例如：

find /etc -cmin -5  //在/etc下查找5分钟内呗修改过属性的文件和目录


find -a两个条件同时满足的命令

     -o两个条件满足任意一个即可

     -type根据文件类型查找 

           f 文件

           d 目录

           l 软链接文件

     -inum根据i节点查找   例如： find . -inum 31531（一个文件的i节点可以通过ls -i可以看到的）//就是在当前目录找到31531节点的文件

例如：

find /love -name BuDing -a -type d  //就是指我在love目录下查找名为BuDing的目录


-exec/-ok 命令 {} \; 对搜索结果执行操作

例如：

find /love -name orochi -exec ls -l {} \;     //就是指我在love目录下查找名叫orochi的文件并且对搜索到的执行详细信息操作


-ok的话就可以进行一次询问操作也就是谨慎起见


例如：

find /home -user admin -ok rm {} \; //就是指在home下找到所有者为admin的用户的文件并进行询问的删除操作


find . -inum 31531 -exec rm {} \;   //就是在当前目录找到31531 i节点的文件然后删除掉

grep命令

查询某一个文件里的文件内容的一行并将那一行列出来

-i 不区分大小写查询

-v 排除指定字串



！！！以#开头的行都是注释行！！！


例如：

grep -v # /etc/inittab  //就是指把所有这个文件里的有#的行过滤掉然后显示其他行

grep -v ^# /etc/inittab  //就是指只去掉这个文件以#开头的文件行去掉显示其他行

crontab linux添加计划任务的命令



crontab 命令常见于 Unix 和类 Unix 的操作系统之中（Linux 就属于类 Unix 操作系统），用于设置周期性被执行的指令。该命令从输入设备读取指令，并将其存放于 crontab 文件中，以供之后读取和执行。通常，crontab 储存的指令被守护进程激活，crond 为其守护进程，crond 常常在后台运行，每一分钟会检查一次是否有预定的作业需要执行。


通过 crontab 命令，我们可以在固定的间隔时间执行指定的系统指令或 shell　script 脚本。时间间隔的单位可以是分钟、小时、日、月、周的任意组合。



# Example of job definition:

# .---------------- minute (0 - 59)

# |  .------------- hour (0 - 23)

# |  |  .---------- day of month (1 - 31)

# |  |  |  .------- month (1 - 12) OR jan,feb,mar,apr ...

# |  |  |  |  .---- day of week (0 - 6) (Sunday=0 or 7) OR sun,mon,tue,wed,thu,fri,sat

# |  |  |  |  |

# *  *  *  *  * user-name command to be executed


crontab -e 使用这个命令的时候添加计划任务都会在/var/spool/cron/crontabs中添加一个该用户自己的任务文档，这样的目的是为了隔离

如果是系统级别的定时任务，只需要sudo权限编辑/etc/crontab文件即可


e.g: */1 * * * * touch /home/shiyanlou/$(date +\%Y\%m\%d\%H\%M\%S)



cron服务监测时间最小单位是分钟 所以cron会每分钟读取一次/etc/crontab和/var/spool/cron/crontab里的内容



每个目录的作用：


/etc/cron.daily，目录下的脚本会每天执行一次，在每天的6点25分时运行；

/etc/cron.hourly，目录下的脚本会每个小时执行一次，在每小时的17分钟时运行；

/etc/cron.mouthly，目录下的脚本会每月执行一次，在每月1号的6点52分时运行；

/etc/cron.weekly，目录下的脚本会每周执行一次，在每周第七天的6点47分时运行；


注意！！！

如果crontab环境没开启时可能需要手动配置


sudo service rsyslog start 


sudo cron -f &


crontab -e
