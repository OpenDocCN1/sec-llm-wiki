三、数据库相关查询命令

1、show databases;默认有4个数据库

2、create database aiyou; 创建数据库

3、use aiyou; 选择要操作的数据库

4、select database();查看当前数据库

![](../Images/5aabcde97e08cb167632f88c6fe83545.png)

5、drop database aiyou;删除数据库

6、exit 退出

  

四、在数据库中添加表

1、新建表

create table 专辑(序号 int(4) auto\_increment,

                            专辑名称 varchar(100) not null,

                            主打歌 varchar(100) not null,

                            发行时间 date not null,

                            primary key(序号)

);

2、desc 专辑; 查看表的结构

3、show tables 显示目前存在的表名

![](../Images/c3ff679d3ec8af403d143fc1ed93d398.png)

4、表的相关操作

alter table 专辑 rename 专属;                            修改表名

alter table 专辑 modify 主打歌  varchar(55);   修改字段数据类型

alter table 专辑 change 序号 id int(2);              修改字段名字和属性

alter table 专辑 add 作词 varchar(100) not null after 主打歌;
