二、数据库连接

<?php 

$dbhost = 'localhost'; // mysql服务器主机地址 

$dbuser = 'root'; // mysql用户名 

$dbpass = 'root'; // mysql用户名密码 

$conn = mysqli\_connect($dbhost, $dbuser, $dbpass); 

if(! $conn ) 

{ die('Could not connect: ' . mysqli\_error());

} 

echo '数据库连接成功！';

mysqli\_close($conn);

?>
