```
     //基础命令     python hawkscan.py -u https://www.exemple.com/
    //使用字典     python hawkscan.py -u https://www.exemple.com/ -w dico_extra.txt
    //使用多线程     python hawkscan.py -u https://www.exemple.com/ -t 30
    //指定文件     python hawkscan.py -u https://www.exemple.com/ -b
    //排除页面     python hawkscan.py -u https://www.exemple.com/ --exclude profile.php
    //排除响应代码     python hawkscan.py -u https://www.exemple.com/ --exclude 403
    //排除大小     python hawkscan.py -u https://www.exemple.com/ --exclude 1337b 
    //排除类型     python hawkscan.py -u https://www.exemple.com/ --exclude 1337b,403
```
