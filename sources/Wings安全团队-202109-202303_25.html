找dll基址


先列过程：


  1. 找到TEB

  2. 找到PEB

  1. 找到_PEB_LDR_DATA

  2. 跳两次到kernel32的位置

  1. 找到kernel的entry


和x86不同的一点是，TEB存放的位置不在fs寄存器中，而是在gs寄存器中，TEB的地址存放于 gs:[30h]处。


PEB也变了，PEB在TEB+0x60处。其余过程和x86的区别不大，代码如下图所示。


	mov rbx, gs:[30h]               ;TEB
	mov rbx,[rbx+60h]			        	;PEB
	mov rbx,[rbx+18h]		        		;LDR
	mov rbx,[rbx+20h]			        	;LIST
	mov rbx,[rbx]                   ;ntdll
	mov rbx,[rbx]                   ;kernel32
	mov rbx,[rbx+20h]               ;entry
	mov [rbp-10h],rbx               ;var8


找函数的绝对地址


继续列过程：


  1. 找到E_lfanew的RVA地址（DllBase+0x3c）

  2. 找到PE文件签名地址（DllBase+1中保存的RVA）

  1. 找到导出表的RVA(2中得到的地址+0x88)

  2. 找到导出表的绝对地址（DllBase+3中的RVA）

  1. 找到导出名称表的RVA(2中地址+0x20)

  2. 找出导出名称表的绝对地址（DllBase+5中的RVA）

  1. 找到导出序号表的RVA（2中地址+0x24）

  2. 找到导出序号表的绝对地址（DllBase+7中的RVA）

  1. 找到导出地址表的RVA(2中地址+0x1c)

  2. 找到导出地址表的绝对地址（2中地址+9中RVA）

  1. 找到导出函数的数量（2中地址+0x14）

  2. 遍历名称表，找到WinExec并记住遍历过的数量。


代码如下所示：


	; Find WinExec address
	xor rax,rax
	xor rcx,rcx
	mov eax, [rbx + 3Ch]		; RVA of PE signature E8
	add rax, rbx       	  	; Address of PE signature = base address + RVA of PE signature
	mov ecx, [rax + 88h]		; RVA of Export Table
	add rcx, rbx 		      	; Address of Export Table
	mov rax,rcx

	xor rdi,rdi
	mov edi, [rax + 20h] 		; RVA of Name Pointer Table
	add rdi, rbx 		      	; Address of Name Pointer Table
	mov [rbp-18h], rdi 	   	; var12 = Address of Name Pointer Table

	xor rcx,rcx
	mov ecx, [rax + 24h]		; RVA of Ordinal Table
    add rcx, rbx 			    ; Address of Ordinal Table
	mov [rbp-20h], rcx 	  	; var16 = Address of Ordinal Table

	xor rdx,rdx
	mov edx, [rax + 1Ch] 		; RVA of Address Table
	add rdx, rbx 			      ; Address of Address Table
	mov [rbp-28h], rdx 	  	; var20 = Address of Address Table

	xor rdx,rdx
	mov edx, [rax + 14h] 		; Number of exported functions

	xor rax, rax 		      	; counter = 0


	s: mov rdi,[rbp-18h]  	; rdi = var16 = Address of Name Pointer Table
       mov rsi, [rbp-8h] 	; rsi = var4 = "WinExec\x00"
       xor rcx, rcx

       cld  		            	; set DF=0 => process strings from left to right
       mov edi, [rdi + rax*4]	; Entries in Name Pointer Table are 4 bytes long
        		              	; edi = RVA Nth entry = Address of Name Table * 4
       add rdi, rbx       	; rdi = address of string = base address + RVA Nth entry
       add cx, 8 	        	; Length of strings to compare (len('WinExec') = 8)
       repe cmpsb         	; Compare the first 8 bytes of strings in 
        			              ; esi and edi registers. ZF=1 if equal, ZF=0 if not
       jz funfound

       inc eax 		      ; counter++
       cmp eax, edx    	; check if last function is reached
       jb short s 		  ; if not the last -> loop

       add rsp, 26h      		
       jmp functionend 		; if function is not found, jump to end

过程


  1. 在序号表在对应位置处找到WinExec并得到其序号（序号表的绝对地址+（12中的值*2））

  2. 在地址表里找到WinExec的RVA（地址表的绝对地址+（13中得到的序号*4））

  1. 找到WinExec的绝对地址（RVA+DllBase）

  2. 将参数放入寄存器中

  1. 调用


    ; the counter (eax) now holds the position of WinExec
