```
#学习代码库函数的使用方法import openpyxl 
table = openpyxl.Workbook()#创建一个excel对象
#创建时，会自动生成一个sheet，获取使用activesh = table.active
#修改当前，sheet标题为成绩表sh.title = '成绩表'
sh1 = table.create_sheet('工资表',0)#增加一个sheet为工资表，索引为0
sh3 = table['成绩表'] #选择使用哪个sheet#print(sh3)
sh3['A1'] = '姓名' #修改A1单元格内容sh3['B1'] = '成绩'print(sh3['A1'].value) #输出A1单元格的值sh3.cell(1,3,value = '年龄') #读取或者写入单元格数据print(sh3.cell(1,3).value)
table.save('成绩表.xlsx') #保存文件名

```

  
记住修改完成后一定要进行保存处理 table.save() 放在最后执行
