git

docker-compose 1.23.2 及以上

1.4 快速安装docker

环境信息

  1. 操作系统：CentOS Linux release 7.7.1908 (Core，

  2. 操作账号：root

  3. Docker版本：当前最新版本：19.03.2

  4. docker-compose版本：当前最新版本：1.24.1

安装

执行以下命令(若非root账号，请自行添加sudo)：

yum install -y yum-utils device-mapper-persistent-data lvm2 \
&& yum-config-manager \
    --add-repo \
    https://download.docker.com/linux/centos/docker-ce.repo \
&& yum install  -y docker-ce docker-ce-cli containerd.io \
&& systemctl start docker \
&& yum -y install epel-release \
&& yum -y install python-pip \
&& pip install --upgrade pip \
&& pip install docker-compose

等待几分钟，即完成docker和docker-compose的安装

2. linux/Mac下docker的食用方式

2.1 下载git源码

git clone https://github.com/testerSunshine/12306.git
cd 12306                        //进入项目目录
cp TickerConfig.py TickerConfig.py.bak              //备份配置文件

2.2 编辑配置文件

vim TickerConfig.py
# 出发日期(list) "2018-01-06", "2018-01-07"
STATION_DATES = [
    "2020-01-24"                    //填写日期
]
# 填入需要购买的车次(list)，"G6087"
# 修改车次填入规则，注：(以前设置的车次逻辑不变)，如果车次填入为空，那么就是当日乘车所有车次都纳入筛选返回
# 不填车次是整个list为空才算，如果不是为空，依然会判断车次的，这种是错误的写法 [""], 正确的写法 []
STATION_TRAINS = []                             //留空则视为全部车次，如果指定车次，则为["车次号"]
# 出发城市，比如深圳北，就填深圳就搜得到
FROM_STATION = "北京"

# 到达城市 比如深圳北，就填深圳就搜得到
TO_STATION = "上海"
# 座位(list) 多个座位ex:
# "商务座",
# "一等座",
# "二等座",
# "特等座",
# "软卧",
# "硬卧",
# "硬座",
# "无座",
# "动卧",
SET_TYPE = ["二等座"]                  //设置座位信息
# 乘车人(list) 多个乘车人ex:
# "张三",
# "李四"
TICKET_PEOPLES = ["王麻子"]			//输入乘车人姓名
# 12306登录账号
USER = "账号"
PWD = "密码"
# 此处设置云打码服务器地址，如果有自建的服务器，可以自行更改
HOST = "120.77.154.140:8000"					//默认使用云服务器
REQ_URL = "/verify/base64/"
HTTP_TYPE = "http"
#  邮箱配置，如果抢票成功，将通过邮件配置通知给您

邮箱密码为授权码

server酱推送

# 是否开启 server酱 微信提醒， 使用前需要前往 http://sc.ftqq.com/3.version 扫码绑定获取 SECRET 并关注获得抢票结果通知的公众号

SERVER_CHAN_CONF = {
    "is_server_chan": True,
    "secret": "xxxxxx"
}



3. 运行

docker-compose up --build -d			//12306目录下执行，开始抢票
docker logs --follow ticket					//查看抢票
docker-compose down									//停止抢票
