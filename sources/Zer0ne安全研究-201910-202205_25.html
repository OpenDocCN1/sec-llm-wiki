在做安全测试的过程中，遇到一个请求报文是全加密的项目，经分析后发现采用AES/ECB/PKCS7Padding进行加密。为了方便使用burp进行测试，需要找到一款AES加解密的burp插件，然而找到的都是只支持到AES/ECB/PKCS5Padding的插件。若要实现AES/ECB/PKCS7Padding则需要导入bcprov-jdk15这个jar包，原因如下：

JAVA实现“AES/ECB/PKCS7Padding”对称加解密，尤其是 AES256的加解密需要注意两点： 

· 技术出口限制，国内的JDK 默认不支持； 

· JDK默认不支持PKCS7Padding填充模式，需借助第三方提供者。

https://blog.csdn.net/lijun169/article/details/82736103

burp加载使用bcprov-jdk15实现的AES/ECB/PKCS7Padding加解密插件，会显示找不到AES/ECB/PKCS7Padding。

根据https://blog.csdn.net/fyq201749/article/details/81504251找到不使用BC实现PKCS7Padding的方法，经修改后，代码如下：（写的burp插件目前不具备通用性，先放出加解密算法）


package burp;
import javax.crypto.Cipher;import javax.crypto.spec.IvParameterSpec;import javax.crypto.spec.SecretKeySpec;
import org.apache.commons.codec.binary.Base64;


public class AESUtil {    private static String ALGO = "AES";    private static String ALGO_MODE = "AES/ECB/NoPadding";    private static String akey = "123456789abcdef0";    private static String aiv = "22222222222222222222222222222222";    public static final String ENCRYPT_MODE_CBC = "CBC";    public static final String ENCRYPT_MODE_ECB = "ECB";    /**     * 默认加密类型     */    private static final String _ENCRYPT_MODE = "ECB";
    public static void main(String[] args) throws Exception {        String data = "原文";
        System.out.println("原始数据:"+data);        String rstData = pkcs7padding(data);//进行PKCS7Padding填充        String passwordEnc = AESUtil.encrypt(rstData,akey);//进行java的AES/CBC/NoPadding加密        String passwordDec = AESUtil.decrypt(passwordEnc,akey);//解密        System.out.println("加密之后的字符串:"+passwordEnc);        System.out.println("解密后的数据:"+passwordDec);    }    public static String encrypt(String Data,String key) throws Exception {        Data = pkcs7padding(Data);        return encrypt(Data,key,_ENCRYPT_MODE);    }    public static String decrypt(String Data,String key) {        Data = pkcs7padding(Data);        try {            return decrypt(Data,key,_ENCRYPT_MODE);        } catch (Exception e) {            e.printStackTrace();        }        return null;    }    /**     * AESUtil加密     *     * @param Data 明文     * @param key 密钥     * @param encryptMode AESUtil加密模式，CBC或ECB     * @return 该字符串的AESUtil密文值     */    public static String encrypt(String Data,String key,String encryptMode) throws Exception {        try {            byte[] iv = toByteArray(aiv);//因为要求IV为16byte，而此处aiv串为32位字符串，所以将32位字符串转为16byte            Cipher cipher = Cipher.getInstance(ALGO_MODE);            int blockSize = cipher.getBlockSize();            byte[] dataBytes = Data.getBytes();            int plaintextLength = dataBytes.length;            if (plaintextLength % blockSize != 0) {                plaintextLength = plaintextLength + (blockSize - (plaintextLength % blockSize));            }            byte[] plaintext = new byte[plaintextLength];            System.arraycopy(dataBytes, 0, plaintext, 0, dataBytes.length);            SecretKeySpec keyspec = new SecretKeySpec(key.getBytes("utf-8"), ALGO);            if(encryptMode== AESUtil.ENCRYPT_MODE_ECB) {                cipher.init(Cipher.ENCRYPT_MODE,keyspec);            }else{                IvParameterSpec ivspec = new IvParameterSpec(iv);                cipher.init(Cipher.ENCRYPT_MODE, keyspec, ivspec);            }            byte[] encrypted = cipher.doFinal(plaintext);            String EncStr = new String((new Base64()).encode(encrypted));//将cipher加密后的byte数组用base64加密生成字符串            return EncStr ;        } catch (Exception e) {            e.printStackTrace();            return null;        }    }    /**     * AESUtil解密     *     * @param encryptedData 密文     * @param key 密钥     * @param encryptMode AESUtil加密模式，CBC或ECB     * @return 该密文的明文     */    public static String decrypt(String encryptedData,String key,String encryptMode) throws Exception {        try {            byte[] encrypted1 = (new Base64()).decode(encryptedData);            byte[] iv = toByteArray(aiv);            Cipher cipher = Cipher.getInstance(ALGO_MODE);            SecretKeySpec keyspec = new SecretKeySpec(key.getBytes("utf-8"), ALGO);            // 根据加密模式是CBC还是ECB进行选择            if(encryptMode== AESUtil.ENCRYPT_MODE_ECB) {                cipher.init(Cipher.DECRYPT_MODE,keyspec);            }else{                IvParameterSpec ivspec = new IvParameterSpec(iv);                cipher.init(Cipher.DECRYPT_MODE, keyspec, ivspec);            }            byte[] original = cipher.doFinal(encrypted1);            String originalString = new String(original);            return originalString.trim();//此处添加trim（）是为了去除多余的填充字符，就不用去填充了，具体有什么问题我还没有遇到，有强迫症的同学可以自己写一个PKCS7UnPadding函数        } catch (Exception e) {            e.printStackTrace();            return null;        }    }    //此函数是将字符串每两个字符合并生成byte数组    public static byte[] toByteArray(String hexString) {        hexString = hexString.toLowerCase();        final byte[] byteArray = new byte[hexString.length() >> 1];        int index = 0;        for (int i = 0; i < hexString.length(); i++) {            if (index  > hexString.length() - 1)                return byteArray;            byte highDit = (byte) (Character.digit(hexString.charAt(index), 16) & 0xFF);            byte lowDit = (byte) (Character.digit(hexString.charAt(index + 1), 16) & 0xFF);            byteArray[i] = (byte) (highDit << 4 | lowDit);            index += 2;        }        System.out.println(byteArray.length);        return byteArray;    }    //此函数是pkcs7padding填充函数    public static String pkcs7padding(String data) {        int bs = 16;        int padding = bs - (data.length() % bs);        String padding_text = "";        for (int i = 0; i < padding; i++) {            padding_text += (char)padding;        }        return data+padding_text;    }}
