```
for /f "delims=" %%A in ('dir /s /b %WINDIR%\system32\*htable.xsl') do set "var=%%A"
wmic process get CSName,Description,ExecutablePath,ProcessId /format:"%var%" >> wmic_info.htmlwmic service get Caption,Name,PathName,ServiceType,Started,StartMode,StartName /format:"%var%" >> wmic_info.htmlwmic USERACCOUNT list full /format:"%var%" >> wmic_info.htmlwmic group list full /format:"%var%" >> wmic_info.htmlwmic nicconfig where IPEnabled='true' get Caption,DefaultIPGateway,Description,DHCPEnabled,DHCPServer,IPAddress,IPSubnet,MACAddress /format:"%var%" >> wmic_info.htmlwmic volume get Label,DeviceID,DriveLetter,FileSystem,Capacity,FreeSpace /format:"%var%" >> wmic_info.htmlwmic netuse list full /format:"%var%" >> wmic_info.htmlwmic qfe get Caption,Description,HotFixID,InstalledOn /format:"%var%" >> wmic_info.htmlwmic startup get Caption,Command,Location,User /format:"%var%" >> wmic_info.htmlwmic PRODUCT get Description,InstallDate,InstallLocation,PackageCache,Vendor,Version /format:"%var%" >> wmic_info.htmlwmic os get name,version,InstallDate,LastBootUpTime,LocalDateTime,Manufacturer,RegisteredUser,ServicePackMajorVersion,SystemDirectory /format:"%var%" >> wmic_info.htmlwmic Timezone get DaylightName,Description,StandardName /format:"%var%" >> wmic_info.htmlwmic environment /format:"%var%" >> wmic_info.htmlwmic share get name,path,status /format:"%var%" >> wmic_info.htmlwmic onboarddevice get Description,DeviceType,Enabled,Status /format:"%var%" >> wmic_info.htmlwmic /namespace:\\root\securitycenter2 path antivirusproduct GET displayName,productState,pathToSignedProductExe /format:"%var%" >> wmic_info.htmlwmic nteventlog where filename='system' call cleareventlog /format:"%var%" >> wmic_info.html
echo ----------systeminfo---------- >> cmd_info.txtsysteminfo >> cmd_info.txtecho ----------netstat -anto---------- >> cmd_info.txtnetstat -anto >> cmd_info.txtecho ----------route print---------- >> cmd_info.txtroute print >> cmd_info.txtecho ----------logicaldisk3---------- >> cmd_info.txtwmic logicaldisk where drivetype=3 get deviceid >> cmd_info.txtecho ----------logicaldisk2---------- >> cmd_info.txtwmic logicaldisk where drivetype=2 get deviceid >> cmd_info.txtecho ----------net user---------- >> cmd_info.txtnet user >> cmd_info.txtecho ----------net start---------- >> cmd_info.txtnet start >> cmd_info.txtecho ----------set---------- >> cmd_info.txtset >> cmd_info.txtecho ----------host---------- >> cmd_info.txttype c:\windows\system32\drivers\etc\hosts >> cmd_info.txtecho ----------ifconfig---------- >> cmd_info.txtifconfig /all >> cmd_info.txtecho ----------session---------- >> cmd_info.txtnet session >> cmd_info.txtecho ----------ipc---------- >> cmd_info.txtnet use >> cmd_info.txtecho ----------product---------- >> cmd_info.txtwmic /namespace:"\\root\cimv2" path win32_product get name /format:table >> cmd_info.txt
:: dir /s /a c:\ >filec.txt
```

  

执行后可根据已有的盘符再去列文件。
