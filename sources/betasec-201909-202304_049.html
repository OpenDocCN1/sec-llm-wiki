面对企业众多的服务器，互联网暴露面的探测成为了一项长期的、重要的工作内容。每天人为的进行扫描探测，浪费相当大的人工成本。如果购买相应暴露面探测的设备呢，还会无形增加了企业的资金成本问题。由此小编写了一个基于钉钉自动化推送的互联网暴露面探测工具。代码如下所示：

# 2022/6/28 by betaimport jsonimport hashlibimport base64import hmacimport osimport timeimport nmapimport requestsimport refrom urllib.parse import quote_plus
class Messenger:    def __init__(self, token=os.getenv("DD_ACCESS_TOKEN"), secret=os.getenv("DD_SECRET")):        self.timestamp = str(round(time.time() * 1000))        self.URL = "https://oapi.dingtalk.com/robot/send"        self.headers = {'Content-Type': 'application/json','Connection': 'close'}        secret = secret        secret_enc = secret.encode('utf-8')        string_to_sign = '{}\n{}'.format(self.timestamp, secret)        string_to_sign_enc = string_to_sign.encode('utf-8')        hmac_code = hmac.new(secret_enc, string_to_sign_enc, digestmod=hashlib.sha256).digest()        self.sign = quote_plus(base64.b64encode(hmac_code))        self.params = {'access_token': token, "sign": self.sign}
    def send_text(self, content):        """        发送文本        @param content: str, 文本内容        """        data = {"msgtype": "text", "text": {"content": content}}        self.params["timestamp"] = self.timestamp        return requests.post(            url=self.URL,            data=json.dumps(data),            params=self.params,            headers=self.headers        )
def nmap_result(hosts):    nm = nmap.PortScanner()    scan_result = ""    hosts = hosts.split('\n')    for host in hosts:        try:            info = "\n\n**"+host+"**"            IP = re.findall(r"\b(?:(?:25[0-5]|2[0-4][0-9]|[01]?[0-9][0-9]?)\.){3}(?:25[0-5]|2[0-4][0-9]|[01]?[0-9][0-9]?)\b", host)            result = nm.scan(IP[0], '1-65535', '-sS -Pn -n --open  --min-parallelism 1024  -T4')            scan = result['scan'][IP[0]]['tcp']            for line in scan.keys():                port_info = "\n" + str(line)+'  '+scan[line]['state'] +'  '+ scan[line]['name']                info = info + port_info            if len(info)>31:                scan_result = scan_result + info        except Exception as e:            pass
