#!/bin/bash
echo " Get Subdomain Script"echo "                     "echo "            by:crhua "echo "Github => https://github.com/crhua"echo "                     "
if [[ $# -eq 0 ]] ;then  echo "Usage: bash Get_Subdomain.sh huasec.com"  exit 1else  curl -s 'https://crt.sh/?q=%.'$1'&output=json' | jq '.[] | {name_value}' | sed 's/\"//g' | sed 's/\*\.//g' | grep "name_value"|cut -d ' ' -f4 > $1.txt.bak  echo "crt.sh over!!!"    curl -s -x 127.0.0.1:1080 "http://web.archive.org/cdx/search/cdx?url=*."$1"/*&output=text&fl=original&collapse=urlkey" |sort| sed -e 's_https*://__' -e "s/\/.*//" -e 's/:.*//' -e 's/^www\.//' | uniq >$1.txt.bak  echo "web.archive.org over!!!"    curl -s "https://certspotter.com/api/v0/certs?domain="$1 | jq '.[].dns_names[]' | sed 's/\"//g' | sed 's/\*\.//g' | sort | grep $1 >>$1.txt.bak  echo "certspotter.com over!!!"
  curl -s -x 127.0.0.1:1080 "http://dns.bufferover.run/dns?q=.$1" | jq -r '.FDNS_A[]' | awk -F "," '{print $2}' >>$1.txt.bak  python2 ../Sublist3r/sublist3r.py -d $1  >> $1_sub.txt  sed -i '1,23d' $1_sub.txt  cat $1.txt.bak $1_sub.txt > $1_tmp.txt
  #To heavy  sort $1_tmp.txt | uniq > $1.txt  rm -rf $1.txt.bak $1_sub.txt $1_tmp.txtfiecho "Detect Subdomain $(wc -l $1.txt|awk '{ print $1 }' )" "=> ${1}"echo "File Location : "$(pwd)/"$1.txt"
