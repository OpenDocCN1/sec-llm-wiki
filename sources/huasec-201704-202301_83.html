0x02  自动化脚本

#!/usr/bin/python

import nmap
import optparse
import sys
def scan(thost,tport):
    try:
        nm = nmap.PortScanner()
        nm.scan(thost,tport)
        status = nm[thost]['tcp'][int(tport)]['state']
        print "[*] " + thost + "   tcp/  " + tport +"  " +status
    except:
        print "[*] " + thost + " is invalid ip"


def Main():
    parse = optparse.OptionParser("usage:nmap_scan.py -h <target host> -p <target port> "+"\n" + "usage:nmap_scan.py -l <hosts_file> -p <target port>")
    parse.add_option("-t",dest="thost",type="string",help="specify target host")
    parse.add_option("-p",dest="tport",type="string",help="specify target ports")
    parse.add_option("-l",dest="tlist",type="string",help="specify target hosts_list")
    (options,args) = parse.parse_args()
    if (options.tport == None):
        print parse.usage
        exit()
    else:
        tp = str(options.tport).split(',')
    if '-l' in sys.argv:
        thost = open(options.tlist,'r')
        for i in thost.readlines():
            i = i.strip('\n')
            for p in tp:
                scan(i,p)
    else:

        print thost
        for p in tp:
            scan(thost,p)

if __name__ == '__main__':
    Main()
