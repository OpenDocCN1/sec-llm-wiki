nm = nmap.PortScanner()

  

#扫描192.168.1.1 端口1-100

nm.scan ('192.168.1.1','1-100')

  

#打印扫描结果的简单信息

nm.scaninfo()

  

#查看有多少hosts

nm.all\_hosts()

  

#查看192.168.1.1的详细信息

nm\['192.168.1.1'\]

  

#查看该host包含的所有协议

nm.all\_ptotocols()

  

#查看哪些端口提供了tcp协议

nm\['192.168.1.1'\]\['tcp'\]

nm\['192.168.1.1'\]\['tcp'\].keys()

#!/usr/bin/python  
  
import nmap  
import optparse  
import sys  
def scan(thost,tport):  
    try:  
        nm = nmap.PortScanner()  
        nm.scan(thost,tport)  
        status = nm\[thost\]\['tcp'\]\[int(tport)\]\['state'\]  
        print "\[\*\] " + thost + "   tcp/  " + tport +"  " +status  
    except:  
        print "\[\*\] " + thost + " is invalid ip"  
  
  
def Main():  
    parse = optparse.OptionParser("usage:nmap\_scan.py -h <target host> -p <target port> "+"\\n" + "usage:nmap\_scan.py -l <hosts\_file> -p <target port>")  
    parse.add\_option("-t",dest="thost",type="string",help="specify target host")  
    parse.add\_option("-p",dest="tport",type="string",help="specify target ports")  
    parse.add\_option("-l",dest="tlist",type="string",help="specify target hosts\_list")  
    (options,args) = parse.parse\_args()  
    if (options.tport == None):  
        print parse.usage  
        exit()  
    else:  
        tp = str(options.tport).split(',')  
    if '-l' in sys.argv:  
        thost = open(options.tlist,'r')  
        for i in thost.readlines():  
            i = i.strip('\\n')

                scan(i,p)
