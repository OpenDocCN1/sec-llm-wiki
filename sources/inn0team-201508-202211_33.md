14.  
     
15.  `import cv2`
     
16.  `from PIL import Image, ImageDraw`
     
17.  
     
18.  
     
19.  `class Opencv(object):`
     
20.  `    # 初始化`
     
21.  `    def __init__(self):`
     
22.  `        # opencv 官方识别库地址 根据自己电脑路径更改`
     
23.  `        self.faces_xml = '/Users/sanjin/work/gittools/opencv/data/haarcascades/haarcascade_frontalface_default.xml'`
     
24.  
     
25.  `    # 人脸识别`
     
26.  `    def faces(self, img_path):`
     
27.  `        img = cv2.imread(img_path)`
     
28.  `        # 加载 官方的 识别数据`
     
29.  `        face_data = cv2.CascadeClassifier(self.faces_xml)`
     
30.  `        print img`
     
31.  `        # 判断图片维度 如果为3 证明不是灰度图 然后转化成灰度图`
     
32.  `        # 不是3 就是 2  那就是 灰度图`
     
33.  `        if img.ndim == 3:`
     
34.  `            # 转化 灰度`
     
35.  `            gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)`
     
36.  `        else:`
     
37.  `            gray = img`
     
38.  
     
39.  `        # 1.2和5是特征的最小、最大检测窗口，它改变检测结果也会改变`
     
40.  `        faces = face_data.detectMultiScale(gray, 1.2, 5)`
     
41.  
     
42.  `        # 创建返回list`
     
43.  `        result = []`
     
44.  
     
45.  `        # 进行 for 循环 把 人脸坐标 和 宽高 插入到 list`
     
46.  `        for (x, y, width, height) in faces:`
     
47.  `            result.append((x, y, x + width, y + height))`
     
48.  
     
49.  `        return result`
     
50.  
     
51.  `    # 画出人脸`
     
52.  `    def h_faces(self, img_path, save_path):`
     
53.  `        faces = self.faces(img_path)`
     
54.  `        # 判断 人脸数据 是否存在`
     
55.  `        if faces:`
     
56.  `            # 打开图片`
     
57.  `            img = Image.open(img_path)`
     
58.  `            # 声明 Draw 方法 进行 画边框`
     
59.  `            draw_instance = ImageDraw.Draw(img)`
     
60.  `            # for 循环 根据 坐标 和 宽 高 画出人脸位置   outline 颜色值`
     
61.  `            for (x1, y1, x2, y2) in faces:`
     
62.  `                draw_instance.rectangle((x1, y1, x2, y2), outline=(255, 0, 0))`
     
63.  
     
64.  `            # 保存图片`
     
65.  `            img.save(save_path)`
     
66.  
     
67.  
     
68.  `if __name__ == '__main__':`
     
69.  `    # 图片地址 , 输出图片地址`
     
70.  `    Opencv().h_faces('/Users/sanjin/Desktop/1.jpg', '/Users/sanjin/Desktop/2.jpg')`
