"""根据分析得到的码表和用于比较的字符串，base32解码，得到L@h\_Xa@J\_o@f332\_@Aq\_e0g13,之后对数据进行rot13解码，得到flag:Y@u\_Kn@W\_b@s332\_@Nd\_r0t13"""

import base64

import string

 

tr \= string.maketrans("NoPqRsTuVwXyZaBcDeFgHiJkLm567234","ABCDEFGHIJKLMNOPQRSTUVWXYZ234567")

 

EnBase32 \= "weNTDk5LZsNRHk6cVogqTZmFy2NRP7X4ZHLTBZwg".translate(tr)

 

rot\_13\= base64.b32decode(EnBase32)

 

print rot\_13

 

Flag \= ''

n \= 13

for letter in rot\_13:

 if letter.isupper():

 c \= ord(letter) \- ord('A')

 i \= (c + n) % 26 + ord('A')

 Flag+= chr(i)

 elif letter.islower():

 c \= ord(letter) \- ord('a')

 i \= (c + n) % 26 + ord('a')

 Flag+= chr(i)

 else:

 Flag+= letter

 

print Flag
