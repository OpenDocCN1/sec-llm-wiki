1.  `     $(".js-game").bind("touchstart", function(t) {`

3.  `                var s = 0;`
    
4.  `                s = e.landscape ? t.touches[0].pageX < parseInt(e.canvas.style.width) / 2 ? 0 : 1 : t.touches[0].pageY < parseInt(e.canvas.style.width) / 2 ? 0 : 1,`
    
5.  `                e.gameData.answer === s ? (e.score += e.increment,`
    
6.  `                $(".js-game-score p").text(e.score),`
    
7.  `                e.generateNext(),`
    
8.  `                e.nextStep(),`
    
9.  `                e.playAudioEffect("choose")) : (e.showFailChoose(0),`

修改后代码如下：

  

1.  `     $(".js-game").bind("touchstart", function(t) {`
    
2.  `            if (e.enableGameClick) {`
    
3.  `                var s = 0;`
    
4.  `                s = e.landscape ? t.touches[0].pageX < parseInt(e.canvas.style.width) / 2 ? 0 : 1 : t.touches[0].pageY < parseInt(e.canvas.style.width) / 2 ? 0 : 1,`
    
5.  `                1 === 1 ? (e.score += 1000,`
    
6.  `                $(".js-game-score p").text(e.score),`
    
7.  `                e.generateNext(),`
    
8.  `                e.nextStep(),`
    
9.  `                e.playAudioEffect("choose")) : (e.showFailChoose(0),`
    
10.  `                e.playAudioEffect("fail"))`
     
11.  `            }`
     

  

**前后的功能区别在于，不管你点对，还是点错，都加1000分。**

  

**直接在浏览器中改是不会生效的，需要在流量中改**。先从浏览器中将game.js 下载下来，并将js代码修改，为接下来使用准备。

1.  `from baseproxy.proxy import ReqIntercept, RspIntercept, AsyncMitmProxy`
    
2.  `class DebugInterceptor(ReqIntercept, RspIntercept):`
    
3.  `    def deal_request(self, request):`
    
4.  `        # print(request.hostname,request.path)`
    
5.    
    
6.  `        return request`
    
7.    
    
8.  `    def deal_response(self, response):`
    
9.  `        request = response.request`
    
10.  `        if request.hostname == 'act.security.tencent.com' and '201812/js/game.js' in request.path:`
     
11.  `            with open("game.js","rb") as f:`
     
12.  `                data = f.read()`
     
13.  `            response.set_body_data(data)`
     
14.    
     
15.  `        return response`
     
16.    
     
17.  `if __name__=="__main__":`
     
18.    
     
19.  `    baseproxy = AsyncMitmProxy(https=True)`
     
20.  `    baseproxy.register(DebugInterceptor)`
     
21.  `    baseproxy.serve_forever()`
