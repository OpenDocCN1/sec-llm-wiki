高性能端口扫描

前言


不要以为我是标题党，真的是so fast。。。最近有个小项目的需要，使用golang写了个端口扫描工具，不得不说golang的效率确实比python快的太多了。在使用一段时间golang之后，感觉有三个方面是优于python的：

  1. 一个方面是性能优越

  2. 第二方面是兼容性好

  3. 第三方面是可以跨平台编译成本地二进制文件，发布项目很方便。


接下来我把这个工具的源代码，以及使用方式给大家给大家分享一下。


第一节

PortScan工具

工具名称：PortScan

  1. 采用Go语言开发，支持从config.txt文件中读取目的ip和端口，对指定的目的服务器进行端口扫描

  2. config.txt支持配置端口列表，默认为22、36000、56000、3306

  3. 在服务器上连接目的服务器端口,仅做一次TCP三次握手



PortScan参数如下：



参数说明:

  •  -c:用来指定config文件路径

  •  -limit:用来指定扫描端口的并发数，默认为1000


第二节

使用效果

首先生成一个config.txt文件，内容如下：


  1. [ip]
  2. 10.27.3.1
  3. 10.27.3.2
  4. 10.27.3.3
  5. 10.27.3.4
  6. 10.27.3.5
  7. 10.27.3.6
  8. ......
  9. 10.27.3.254
  10. 127.0.0.1
  11. [port]
  12. 4300
  13. 4301
  14. 4302
  15. 4303
  16. 4304
  17. 4305
  18. 4306
  19. 4307
  20. 4308
  21. 4309


ip共有255个，端口有10个，也就是说总共要进行2550次TCP连接。


接着使用go build PortScan.go命令，生成一个本地二进制文件PortScan.exe,然后执行命令：


PortScan.exe -c config.txt


最后的结果存储到result.txt文件中，大致用时是30s。

端口扫描核心函数：

  2. import (

  5.     "io/ioutil"
  6.     "bufio"
  7.     "log"
  8.     "regexp"
  9.     "net"
  10.     "runtime"
  11.     "time"
  12.     "bytes"
  13.     "flag"
  14.     "os"
  15.     "strings"
  17. )
  19. var(
  22.     P = fmt.Println
  24. )
  26. //首先从命令行中读取线程数和配置文件路径
  27. //从配置文件中解析出ip和port
  28. //配置文件格式为
  29. // [ip]
  30. // 127.0.0.1
  31. // [port]
  32. // 22
  33. // 36000
  34. // 56000
  35. // 3306
  36. //根据开启的线程数对指定ip和端口进行tcp连接
  37. //如果端口开启，把ip:port按照格式返回
  39. func Scanner(configFile string, limit int){
  41.     runtime.GOMAXPROCS(runtime.NumCPU())
  42.     data, err := ioutil.ReadFile(configFile)
  43.     portlist := make([]string,0,10)
  45.     if err != nil{
  46.         log.Fatal(err)
  47.         panic(err)
  48.     }
  50.     ip_index := bytes.Index(data,[]byte("[ip]"))
  52.     port_index := bytes.Index(data,[]byte("[port]"))
  54.     if ip_index < 0{
  56.         P("文件格式有误: missing [ip]")
  57.         return
  58.     }
  60.     if port_index <0{
  62.         portlist = append(portlist,"22")
  63.         portlist = append(portlist,"36000")
  64.         portlist = append(portlist,"56000")
  65.         portlist = append(portlist,"3306")
  67.     }else{
  68.         reg_port := regexp.MustCompile(`\d+`)
  69.         var ports [][]byte
  71.         if ip_index>port_index{
  72.             ports = reg_port.FindAll(data[:ip_index],-1)
  74.         }else{
  76.             ports = reg_port.FindAll(data[port_index:],-1)
  77.         }
  80.         for _,v := range(ports){
  82.             portlist = append(portlist,string(v))
  84.         } 
  85.     }
  88.     reg_ip := regexp.MustCompile(`((25[0-5]|2[0-4]\d|((1\d{2})|([1-9]?\d)))\.){3}(25[0-5]|2[0-4]\d|((1\d{2})|([1-9]?\d)))`)
  90.     ips := reg_ip.FindAll(data,-1)
  92.     input := make(chan []byte, len(ips))
  94.     result := make(chan string, len(ips))
  96.     defer close(input)
  97.     defer close(result)
  99.     for _, v := range(ips){
  100.         input <- v
  101.     }
  102.     //控制多少并发
  103.     for i:=0; i<limit;i++{
  104.         //这个时候可以启动扫描函数
  105.         go ScanPort(portlist,input,result)
  107.     }
  109.     result_str := ""
  111.     for i :=0; i< len(ips);i++{
  113.         //将扫描的结果输出
  114.         ip_result,ok:= <-result
  116.         if !strings.Contains(ip_result,"ok"){
  117.             P(ip_result)
  118.             result_str += ip_result+"\n"
  121.         }
  123.         if !ok {
  126.             break;
  127.         }
  129.     }
  131.     f,err := os.Create("result.txt")
  132.     if err !=nil{
  134.         P(err)
  135.         return
  136.     }
  137.     defer f.Close()
  138.     w := bufio.NewWriter(f)
  140.     fmt.Fprintln(w, result_str)
  142.     w.Flush()
  144. }
  146. func ScanPort(portlist []string,intput chan []byte,result chan string ) {
  148.     for{
  150.         task,ok := <-intput
  152.         if !ok{
  154.             return
  155.         }
  156.         ip := string(task)
  157.         P("scaning ",ip,"port",portlist)
  158.         port_str :=""
  159.         for i:=0; i<len(portlist); i++{
  160.             _, err  := net.DialTimeout("tcp", ip+":"+portlist[i], time.Second*3)
  162.             if err != nil{
  163.                 continue
  164.             }
  165.             port_str+=portlist[i]+" "
  166.         }
  168.         if len(port_str) >0{
  169.             //说明有打开的端口
  170.             // P(ip+":"+port_str+"open")
  171.             result <- ip+"----"+"1"+"----"+"open port:"+port_str
  173.         }else{
  174.             result <- ip+"----"+"0"+"----"+"ok"
  175.         }
  177.     }
  179. }
