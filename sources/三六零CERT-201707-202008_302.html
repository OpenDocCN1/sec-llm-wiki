ar = [1, 2, 3, 4, 5, 8, 9, 14, …, 1866653692288920481394023416971]

 

def egcd(a, b):

         if  a == 0:

                   return  (b, 0, 1)

         else:

                   g,  y, x = egcd(b % a, a)

                   return  (g, x - (b // a) * y, y)

def modinv(a, m):

         g,  x, y = egcd(a, m)

         if  g != 1:

                   return  0

         else:

                   return  x % m

def solver(ar=[]):

         n  = len(ar)

         ar.reverse()

         u  = ar[0]

 

         cr  = []

         dr  = []

         for  i in range(0,n):

                   a  = ar[i]

                   if  a%2 == 1 or a%4 == 0:

                            continue

 

                   flag  = 1

                   a  /= 2

                   for  c in cr:

                            g,  x, y = egcd(c,a)

                            if  g != 1:

                                     flag  = 0

                                     break

 

                   if  flag:

                            cr.append(a)

                            t  = ar[i-1]-ar[i]

                            if  t > a:

                                     t  -= a

                            dr.append([t,a])


         ar  = dr[:]

         n  = len(ar)

         m  = 1

         Mi  = [0]*n

         Mj  = [0]*n

 

         for  i in range(0,n):

                   m  *= ar[i][1]


         for  i in range(0,n):

                   Mi[i]  = m/ar[i][1]

                   Mj[i]  = modinv(Mi[i],ar[i][1])

 

         x  = 0

         for  i in range(0,n):

                   x  += Mi[i]*Mj[i]*ar[i][0]

 

         print(u+x%m%u)

 

solver(ar)
