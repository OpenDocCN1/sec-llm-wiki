> ```
> #coding = utf-8
> import requests
> import re
> import urllib
> page = range(1,5)
> github\_username = raw\_input('github\_username:')
> for pg in page:
> 	url = 'https://github.com/%s?page=%s&tab=repositories' %(github\_username,pg)
> 	get = requests.get(url)
> 	regex = '\\<a href=\\"/(.\*?)\\" itemprop\\=\\"name codeRepository\\"\\>'
> 	key = re.findall(regex,get.content)
> 	for k in key:
> 		#print k
> 		url2 = 'https://github.com/'
> 		xiangmu\_name = url2 + k
> 		print xiangmu\_name
> 		url3 = '/archive/master.zip'
> 		download = xiangmu\_name + url3
> 		print download
> 		filename = urllib.urlopen(download)
> 		info = filename.info()
> 		filename\_regex = 'Content-Disposition: attachment; filename=(.\*)'
> 		filename\_down = re.findall(filename\_regex,str(info))
> 		#print filename\_down
> 		for filename1 in filename\_down:
> 			print filename1
> 			filename2 = filename1.strip()
> 			urllib.urlretrieve(download,filename2)
> ```
