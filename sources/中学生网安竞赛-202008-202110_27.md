### 安全评估

如果没有路由器的话，这显然就是一个单源最短路裸题。直接上堆优化的Dijkstra算法即可，复杂度为O(NlogN)（M和N同规模）。

现在加入路由器的设定，就必须知道一个常见的建图技巧，及引入虚拟结点。这里需要对于每个路由器自定义一个leader结点，路由器内的每个结点到leader需要花费b的代价，而从leader离开到其他结点不需要花费代价。这样就把问题转化为了一个单源最短路问题，然后还是用堆优化的Dijkstra解。

#include<iostream> #include<cstdio> #define LL long long using namespace std; const long long oo=(long long)1000000007*1000000007; struct edg{int nxt,y,v;}e[810000];  int lk[210000],ltp=0; void ist(int x,int y,int z){   e[++ltp]=(edg){lk[x],y,z};  lk[x]=ltp; } struct nds{   int x;  LL y;   bool operator<(nds z){  return y<z.y;} }hp[810000];  int sz=0; void psh(nds z){   hp[++sz]=z;   for(int i=sz;i!=1 && hp[i]<hp[i>>1];i>>=1)  swap(hp[i],hp[i>>1]); } void pp(){   hp[1]=hp[sz--];   int x=1,mn=1;   for(;;){     mn=x;     if((x<<1)<=sz && hp[x<<1]<hp[mn])  mn=(x<<1);     if((x<<1|1)<=sz && hp[x<<1|1]<hp[mn])  mn=(x<<1|1);     if(mn==x)  break;     swap(hp[mn],hp[x]);     x=mn;   } } int n,m,o,s; LL dst[210000],lst[210000]; void dij(int x){   sz=0;   for(int i=1;i<=n+o;++i)  dst[i]=oo;   dst[x]=0;   psh((nds){x,0});   while(sz){     while(sz && hp[1].y>dst[hp[1].x])  pp();     if(!sz)  break;     x=hp[1].x;  dst[x]=hp[1].y;     pp();     for(int i=lk[x];i;i=e[i].nxt)if(dst[x]+e[i].v<dst[e[i].y]){       dst[e[i].y]=dst[x]+e[i].v;       psh((nds){e[i].y,dst[e[i].y]});       lst[e[i].y]=x;     }   } } void prvs(){   sz=0;   ltp=0;   for(int i=1;i<=n+o;++i)  lk[i]=0; } int main(){   int T;  cin>>T;   while(T --> 0){     prvs();     scanf("%d%d%d%d",&n,&m,&o,&s);     int l,r,mk;     for(int i=1;i<=m;++i){       scanf("%d%d%d",&l,&r,&mk);       ist(l,r,mk);       ist(r,l,mk);     }     for(int i=1;i<=o;++i){       scanf("%d%d",&l,&mk);       for(int j=1;j<=l;++j){         scanf("%d",&r);         ist(r,n+i,mk);         ist(n+i,r,0);       }     }     dij(s);     for(int i=1;i<=n;++i){       printf("%lld",dst[i]);       if(i!=n)  printf(" ");     }     printf("\n");   }   return 0; }
