栅栏密码

•有一串字符串作为明文，每n个为一组，然后每组序号相同提取处理重新组成一组，拼接起来就为密文。给出密文要求解题得出明文。解题思路：本次题目，其实主要是代码基本能力，在手上模拟一下加密过程以及解密过程能找出规律。

 

#include<iostream>#include<cstdio>#include<cstring>using namespace std;int n,m;char s[110000];char t[110000];int main(){  int T;  cin>>T;  while(T --> 0){    scanf("%d",&m);    scanf("%s",s+1);    n=strlen(s+1);    int o=n%m;    if(!o)  o=m;    int tp=0;    for(int i=1;i<=(n+m-1)/m;++i){      int st=(n+m-1)/m;      for(int j=i,k=1;j<=n;j+=st,k++){        if(k==o+1)  --st;        t[++tp]=s[j];      }    }    for(int i=1;i<=n;++i)      printf("%c",t[i]);    printf("\n");  }  return 0;}

量子波动速读

给你一串整数数列做乘法，判断最后结果是否越界越界输出-1，没越界输出结果

本题主要思路在于两个

 1、对于c++选手要采用除法判别是否越界，以及特判为0的两种情况。

2、对于python以及java可以直接采用自身大数模板来实现，一个循环就可以解决。


#include<iostream>#include<cstdio>#define LL long longusing namespace std;const LL m=(LL)1000000000*1000000000;int n;int main(){  freopen("ddd.in","r",stdin);  freopen("ddd.out","w",stdout);  int T;  cin>>T;  while(T --> 0){    scanf("%d",&n);    LL x;    LL bwl=1;    while(n --> 0){      scanf("%lld",&x);      if(x==0){        bwl=0;        //break;  Attention!      }      else if(bwl!=-1 && bwl!=0){        if(m/bwl<x){          bwl=-1;        }        else{          bwl=bwl*x;        }      }    }    printf("%lld\n",bwl);  }  return 0;}


 

RSA加密

•题目大意：

• 有一个n，它是由两个质数组成。

•‘找出一个整数e，满足1<e<φ(n)且e和φ(n)互质，作为公钥

 找出一个整数d，使得e*d除以φ(n)的余数为1，作为私钥

•

•解题思路：本题题意看上去很复杂，实际上本质就是求e对于φ(n)的乘法逆元。因此解题思路分为两步：

•1、找出φ(n)

2、求e对φ(n)的逆元

这个主要需要一定的数学知识，或者对于加密过程的了解，实际上这道题在网上其实有很多类似板子可以做，基本知道了这个加密知识点就可以做。

#include<iostream>#include<cstdio>#include<cmath>#define LL long longusing namespace std;void exgcd(LL a,LL b,LL &x,LL &y){  if(!b){    x=1,y=0;    return ;  }  exgcd(b,a%b,x,y);  int z=x;  x=y,y=(z-a/b*y);}LL n,e;LL p,q,phi;int main(){  int T;  cin>>T;  while(T --> 0){    scanf("%lld%lld",&n,&e);    LL m=(LL)sqrt((double)n);    for(int i=2;i<=m;++i)if(n%i==0){      p=i;      q=n/i;      break;    }    phi=(p-1)*(q-1);    LL x,y;    exgcd(e,phi,x,y);    x=(x%phi+phi)%phi;    printf("%lld\n",x);  }  return 0;}


 

肝能力

•题意：1、有五个位置，可以理解为每个人在不同位置上产生不同贡献，同时每个人还有一个场外固定提升队伍阵容的贡献。


•这道题第一个想到思想是贪心，由于每个卡牌在场外贡献是固定的，对于一个阵容来说，能作为羁绊提供场外贡献前5+k个卡牌。

•因此首要思想就是对以卡牌作为场外关键字进行排序。


•你们可以考虑，如果排序卡牌必定是在前面连续的选人区间里面，如果不是，那我只要和交换最后一个卡牌交换必然更优。

排序以后我们就来到这道题的解法：动态规划

 

我们可以看出其实由于只有五个位置，我们可以用二进制状态来表示当前位置是否被选（其实就是状压） j就是代表二进制数

 

$F\[i][j]$代表当前i个已经进行选择，位置占有情况为j的最优策略。

然后，分两种情况

 

i<=m+k 那么就需要考虑两种情况

 1、当前i作为观众

 dp[i][j] =max(dp[i - 1][j] + q[i].c,dp[i-1][j]);

2、考虑i作为三个职业五个位置的情况的人

dp[i][j] = max(dp[i][j], dp[i - 1][j^(1 <<(i))] + V计算得到战斗力);

 V战斗力其实就是按公式计算。

i>m+k时只需要考虑上面第二种情况（即当前位置为上阵卡牌的情况）。

#include <bits/stdc++.h>using namespace std;#define ll long longconst int maxn = 1e5 + 10;ll attack[maxn],defense[maxn],c[maxn];struct node{    ll attack;    ll defence;    ll c;}   q[maxn];ll dp[maxn][128];bool cmp(struct node a,struct node b){    return a.c>b.c;}signed main() {    int n,m;    freopen("1.in","r",stdin);    freopen("1.out","w",stdout);
    scanf("%d%d",&n,&m);    for (int i=1;i<=n;i++)    {        scanf("%lld%lld%lld",&q[i].attack,&q[i].defence,&q[i].c);    }    memset(dp, 0xc0, sizeof(dp));    dp[0][0]=0;    sort(q+1,q+n+1,cmp);    for (int i=1; i<= n; i++)    for (int j = 0; j < (1 << 5); j++) {      if (i <=m+5 && i - __builtin_popcount(j) <= m )               {                   dp[i][j] =max(dp[i - 1][j] + q[i].c,dp[i-1][j]);
               }      else dp[i][j] = dp[i - 1][j];
        if (j & (1)) dp[i][j] = max(dp[i][j], dp[i - 1][j^(1)] + q[i].defence*q[i].defence);        if (j & (1 <<(1))) dp[i][j] = max(dp[i][j], dp[i - 1][j^(1 <<(1))] + q[i].defence*q[i].defence);                if (j & (1 <<(2))) dp[i][j] = max(dp[i][j], dp[i - 1][j^(1 <<(2))] + q[i].attack*q[i].defence);        if (j & (1 <<(3))) dp[i][j] = max(dp[i][j], dp[i - 1][j^(1 <<(3))] + q[i].attack*q[i].defence);                if (j & (1 <<(4))) dp[i][j] = max(dp[i][j], dp[i - 1][j^(1 <<(4))] + q[i].attack*q[i].attack);    }
    printf("%lld\n", dp[n][(1 << 5) - 1]);
    fclose(stdin);   fclose(stdout);  return 0;}
