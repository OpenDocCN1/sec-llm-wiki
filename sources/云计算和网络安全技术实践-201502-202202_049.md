本期是配合kubevirt实践虚机对multus多网卡的使用，

环境的搭建之前都实践过，就不再重复了，

直接上模板，

apiVersion: kubevirt.io/v1alpha3  
kind: VirtualMachine  
metadata:  
  labels:  
    kubevirt.io/vm: vm-ubuntu  
  name: vm-ubuntu  
spec:  
  running: false  
  template:  
    metadata:  
      labels:  
        kubevirt.io/vm: vm-ubuntu  
    spec:  
      domain:  
        devices:  
          disks:  
          - disk:  
              bus: virtio  
            name: containerdisk  
          - disk:  
              bus: virtio  
            name: cloudinitdisk  
          interfaces:  
          - bridge: {}  
            name: macvlan-conf  
        machine:  
          type: ""  
        resources:  
          requests:  
            memory: 2048M  
      terminationGracePeriodSeconds: 0  
      networks:  
      - multus:  
          networkName: default/bridge-conf  
        name: macvlan-conf  
      volumes:  
      - containerDisk:  
          image: ubuntu:16.04  
          imagePullPolicy: IfNotPresent  
        name: containerdisk  
      - cloudInitNoCloud:  
          userData: |  
            #cloud-config  
            ssh\_pwauth: True  
            chpasswd:  
             list: |  
               ubuntu:ubuntu  
             expire: False  
        name: cloudinitdisk

pod裹着的虚机正常启动，但是发现两个问题，一个好，一个坏，

好的是发现虚机不需要用容器集群网络的网卡，仅使用multus的网络，

反正虚机用容器集群网络的网卡也没啥用，因此这个是好事二，

坏的是地址正常拿到了，但是网络不通，于是就想到把macvlan改成bridge试试，给宿主机加了块物理网卡，然后创建一个bridge绑上，

sudo brctl addbr br0，  
sudo brctl addif br0 ens35，  
sudo ifconfig ens35 promisc up，

再创建一个bridge类型的网络，sudo vim bridge-conf.yaml，

apiVersion: "k8s.cni.cncf.io/v1"  
kind: NetworkAttachmentDefinition  
metadata:  
  name: bridge-conf  
spec:  
  config: '{  
      "cniVersion": "0.3.0",  
      "type": "bridge",  
      "bridge": "br0",  
      "ipam": {  
        "type": "host-local",  
        "subnet": "192.168.128.0/24",  
        "rangeStart": "192.168.128.200",  
        "rangeEnd": "192.168.128.216",  
        "routes": \[  
          { "dst": "0.0.0.0/0" }  
        \],  
        "gateway": "192.168.128.1"  
      }  
    }'

最后再用这个bridge类型的网络创建虚机，

apiVersion: kubevirt.io/v1alpha3  
kind: VirtualMachine  
metadata:  
  labels:  
    kubevirt.io/vm: vm-ubuntu  
  name: vm-ubuntu  
spec:  
  running: false  
  template:  
    metadata:  
      labels:  
        kubevirt.io/vm: vm-ubuntu  
    spec:  
      domain:  
        devices:  
          disks:  
          - disk:  
              bus: virtio  
            name: containerdisk  
          - disk:  
              bus: virtio  
            name: cloudinitdisk  
          interfaces:  
          - bridge: {}  
            name: bridge-conf  
        machine:  
          type: ""  
        resources:  
          requests:  
            memory: 2048M  
      terminationGracePeriodSeconds: 0  
      networks:  
      - multus:  
          networkName: default/bridge-conf  
        name: bridge-conf  
      volumes:  
      - containerDisk:  
          image: ubuntu:16.04  
          imagePullPolicy: IfNotPresent  
        name: containerdisk  
      - cloudInitNoCloud:  
          userData: |  
            #cloud-config  
            ssh\_pwauth: True  
            chpasswd:  
             list: |  
               ubuntu:ubuntu  
             expire: False  
        name: cloudinitdisk
