写一段Python程序进行测试，

#!/usr/bin/python  
import smtplib  
count = 0  
while (count < 10):  
        server = smtplib.SMTP('192.168.137.44', 25)  
        server.login("test2@test.com", "123456")  
        msg = "Hello!"  
        server.sendmail("test2@test.com", "test1@test.com", msg)  
        server.quit()  
        count = count + 1  
print "Finish!"
