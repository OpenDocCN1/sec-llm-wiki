1、端口扫描 vim port_scan.py

import socket

target_host = "192.168.1.202"

client = socket.socket(socket.AF_INET,socket.SOCK_STREAM)

for target_port in range(20,500):
try:
client.connect((target_host,target_port))
print('Port %d is open\r\n' % (target_port))
client.close()
client = socket.socket(socket.AF_INET,socket.SOCK_STREAM)
except Exception:
pass
client.close()

import socket

target_host = "192.168.1.202"
target_port = 80

client = socket.socket(socket.AF_INET,socket.SOCK_STREAM)

client.connect((target_host,target_port))

client.send("GET /sqli/example4.php?id=1 HTTP/1.1\r\nHost: 192.168.1.202\r\n\r\n")
