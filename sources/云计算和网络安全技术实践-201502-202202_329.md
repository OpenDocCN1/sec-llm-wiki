1、端口扫描 vim port\_scan.py

import socket

target\_host = "192.168.1.202"

client = socket.socket(socket.AF\_INET,socket.SOCK\_STREAM)

for target\_port in range(20,500):  
try:  
client.connect((target\_host,target\_port))  
print('Port %d is open\\r\\n' % (target\_port))  
client.close()  
client = socket.socket(socket.AF\_INET,socket.SOCK\_STREAM)  
except Exception:  
pass  
client.close()
