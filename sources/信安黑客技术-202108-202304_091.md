#!/bin/ bash
"[0-9]+"

)for pid in ${ALL_PROCESS[*]}do
test -f /proc/$pid/status && state=$(egrep "state"/proc/$pid/status / awk'{print $2}'
)case ""$state

" in$((stoped_count+1)

)$((sleeping_count+1)

);$( (zombie_count+1)) echo "$pid
" >>zombie.txt$pid

"
total:$running\_count\\nstoped: $stoped\_count\\nsleeping: $sleeping\_count\\nzombie:$zombie\_count"
