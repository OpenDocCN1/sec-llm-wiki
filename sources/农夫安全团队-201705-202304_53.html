同一套试卷，题目的顺序会随机打乱，如果想每次都看到一模一样的试卷咋办？


继续看

http://quanguo.jingsai.ciwong.com/javascript/dowork.js?v=20161010的第67行，

其随机打乱题目顺序的算法都在这里

将第73行的

var iRand = parseInt(num.length *Math.random()); 

替换成

var iRand =1; 

即可得到每次顺序一致的卷面。

function randomq(num) {

   var inputdata=[]; 

   for (var i = 0; i < num.length; i++) { 

       inputdata[i] = num[i];

   } 

   for (var i = 0; i < num.length; i++) { 

      // var iRand = parseInt(num.length * Math.random()); 

   var iRand =1; 

        var temp = inputdata[i]; 

       inputdata[i] = inputdata[iRand]; 

       inputdata[iRand] = temp; 

    }

   for (var i = 0; i < num.length; i++) {

       var c = inputdata[i];

       var gs = $(c).find(".seq")[0].outerHTML;

       c = c.replace(gs, '<span>' + parseInt(i+ 1)+ '</span>');

       inputdata[i] = c;

    }

   return inputdata; 

}
