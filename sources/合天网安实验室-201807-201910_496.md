function get\_real\_ip(){

$ip=false;

if(!empty($\_SERVER\["HTTP\_CLIENT\_IP"\])){

$ip = $\_SERVER\["HTTP\_CLIENT\_IP"\];

}

if (!empty($\_SERVER\['HTTP\_X\_FORWARDED\_FOR'\])) {

$ips = explode (", ", $\_SERVER\['HTTP\_X\_FORWARDED\_FOR'\]);

if ($ip) { array\_unshift($ips, $ip); $ip = FALSE; }

for ($i = 0; $i < count($ips); $i++) {

if (!eregi ("^(10|172\\.16|192\\.168)\\.", $ips\[$i\])) {

$ip = $ips\[$i\];

break;

}

}

}

return ($ip ? $ip : $\_SERVER\['REMOTE\_ADDR'\]);

}
