    def gcd(a,b):
        if a<b:
            a,b = b,a
        while(b!=0):
            temp = a%b
            a = b
            b = temp

    def CRT(a,n):

        N = reduce(lambda x,y:x*y,n)   # ni 的乘积,N=n1*n2*n3

            sum += a_i*N_i*invert(N_i,n_i)   #sum=C1M1y1+C2M2y2+C3M3y3
        return sum % N 

    x = CRT(c,n)

    m = iroot(x,e)[0]           #开e次方根
    print(n2s(m))               #数值转字符串
