       crc32 = binascii.crc32(data) & 0xffffffff       if crc32 == 0x9E916964: # 0x9E916964是这个 png文件头的 CRC校验码，在 21~25byte处           print('\n\n',i,j,crc32)           print(type(data))           exit(0)   print(i, end=' ')
