def sieve(n):
i = 0
count = 0
L=[]
for k in range(2,n):
L.append(k)
while L ** 2 < L[-1]:
for j in range(i + 1,len(L)):
if L[j] % L == 0:
L[j] = 0
count += 1
L.sort()
L = L[count:]
count = 0
i += 1
return L
