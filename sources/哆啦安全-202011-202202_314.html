配置linageOs源码中的交叉编译环境

sudo vim ~/.bashrc

export ARCH=arm &&export PATH=/usr/local/Android7.1.2/prebuilts/gcc/linux-x86/arm/arm-eabi-4.8/bin:$PATH &&export CROSS_COMPILE=arm-eabi- &&
export ARCH=arm &&export PATH=/usr/local/Android7.1.2/prebuilts/gcc/linux-x86/arm/arm-linux-androideabi-4.9/bin:$PATH &&export CROSS_COMPILE=arm-linux-androideabi- &&
export ARCH=arm64 &&export PATH=/usr/local/Android7.1.2/prebuilts/gcc/linux-x86/aarch64/aarch64-linux-android-4.9/bin:$PATH &&export CROSS_COMPILE=aarch64-linux-android- &&
export ARCH=x86 &&export PATH=/usr/local/Android7.1.2/prebuilts/gcc/linux-x86/x86/x86_64-linux-android-4.9/bin:$PATH &&export CROSS_COMPILE=x86_64-linux-android- &&

source ~/.bashrc
