**1、手动测试**

 BOOL APIENTRY DllMain( HANDLE hModule,                        DWORD  ul\_reason\_for\_call,                        LPVOID lpReserved                     ){    switch (ul\_reason\_for\_call)    {        case DLL\_PROCESS\_ATTACH:            MessageBox(NULL,"hello world，I'm 3kb","title",MB\_OK);        case DLL\_THREAD\_ATTACH:        case DLL\_THREAD\_DETACH:        case DLL\_PROCESS\_DETACH:            break;    }return TRUE;}

· Build：release

· 工程设置->Win32 Release->C/C++->Code Generation->Use run-time library:->Multithreaded DLL

 C:\\Users\\a\\AppData\\Roaming\\MicrosoftWordStartup

对字符串截取，截取出版本号14，代码如下：
