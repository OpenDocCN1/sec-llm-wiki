**Request与Selenium**

  

  

**Request**对网站的爬取非常简单快捷，占用的资源相对较少，且速度更快，但是适用于静态网页的爬取，没有下拉窗口，不需要对界面进行操作，对没有反爬机制的网站更加方便。  

**Selenium**则是一种自动化的web测试工具，本身并不是专门用来做爬虫的，但是其功能强大，模拟人操作网页，可见即可爬取，并且可以有效针对反爬机制。

import csvimport jsonimport randomimport reimport time
from selenium.webdriver.chrome.options import Options
from selenium import webdriver
# chrome.exe --remote-debugging-port=9222 --user-data-dir="C:\selenum\AutomationProfile"
def login():    driver.get(        "https://i.taobao.com/my_taobao.htm")
    while True:        flag = input("是否登陆/Y or /N")        if flag == "Y":            driver.get(                "https://i.taobao.com/my_taobao.htm")            break
def get_infomathino(item_url):    driver.get(item_url)    if "亲，请拖动下方滑块完成验证" in driver.page_source:        while True:            flag = input("是否解锁/Y or /N")            if flag == "Y":                driver.get(                    item_url)                break
    bbb = re.findall(r'<li title=.*?</li>', driver.page_source)    a = ''    print(bbb)    for i in bbb:        ccc = re.findall(r'>.*?<', i)[0]        ccc = ccc.replace('>', '')        ccc = ccc.replace('<', '')        ccc = ccc.replace("&nbsp;", '')        # title = re.findall(r"(.*?):", ccc)[0]        # text = re.findall(r":(.*)", ccc)[0]        # title = title.replace(':', '')        # text = text.replace(':', '')        a = a + "|**|" + ccc        a = a + "|**|"    return a
options = Options()options.add_experimental_option("debuggerAddress", "127.0.0.1:9527")driver = webdriver.Chrome(executable_path=r'chromedriver.exe',options=options)# 获取目标urllogin()page = 0target_good = input("请输入需要查询的商品")with open(r"taobao3.csv", 'a', newline='', encoding='ANSI') as f:    writer = csv.writer(f)    for i in range(1, 100):        search_url = "https://s.taobao.com/search?q=" + target_good + "&suggest=history_1&commend=all&ssid=s5-e&search_type=item" \                                                                      "&sourceId=tb.index&spm=a21bo.jianhua.201856-taobao-item.2" \                                                                      "&ie=utf8&initiative_id=tbindexz_20170306&_input_charset" \                                                                      "=utf-8&wq=&suggest_query=&source=suggest&bcoffset=4" \                                                                      "&p4ppushleft=2%2C48&s=" + str(            page) + "&ntoffset=4"
        driver.get(search_url)        if "亲，请拖动下方滑块完成验证" in driver.page_source:            while True:                flag = input("是否解锁/Y or /N")                if flag == "Y":                    driver.get(                        search_url)                    break
        html_data = re.findall('g_page_config = (.*);', driver.page_source)[0]        json_data = json.loads(html_data)        data = json_data["mods"]['itemlist']['data']['auctions']        print(json_data["mods"]['itemlist']['data']['auctions'])        for index in data:            dicts = {                '标题': index['raw_title'],                '店铺': index['nick'],                '价格': index['view_price'],                '网址': index['detail_url']            }            infoma = 'NULL'            if "click" not in index['detail_url'] and "tmall" not in index['detail_url']:                if "http:" not in index['detail_url']:                    index['detail_url'] = "https:" + index['detail_url']                    infoma = get_infomathino(index['detail_url'])                else:                    infoma = get_infomathino(index['detail_url'])
            try:                writer.writerow([index['raw_title'], index['nick'], index['view_price'], index['detail_url'],infoma])            except:                print(dicts)                print(infoma)        page = page + 44        time.sleep(random.randint(0, 9))
