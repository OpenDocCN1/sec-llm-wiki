## **神经元**

**  

  

在说神经网络之前，我们讨论一下神经元（Neurons），它是神经网络的基本单元。神经元先获得输入，然后执行某些数学运算后，再产生一个输出。

数学表达式为

f(x,w) = w\*x + b

其中x为我们的特征输入，w为该特征输入的权重而，b则是对整体结果进行的一个微调。

我们可以设想这样一个场景在一战的堑壕战中我们作为防守方的指挥官要坚守我们的阵地，敌人会配合炮火对我阵地发起冲锋，但是多数的时候敌人在我们各个阵地的位置发起的是佯攻，以骗取我们的火力掩护真正的进攻，我们需要根据敌方战力值和我方地形优势，来进行分析找出敌人在哪里是真正的进攻。那么首先我们让下属把近几次的进攻详情为我们做了一个汇报：

![](../Images/041f716112139b22f28c52fa64f13e74.png)

我们需要根据汇报的情况作为一个模型进而制作神经网络这其中的数值都是我们的x

激活函数的作用是将无限制的输入转换为可预测形式的输出。  

这里我们使用sigmoid函数其输出介于0和1，我们可以理解为它把 (−∞,+∞) 范围内的数压缩到 (0, 1)以内。正值越大输出越接近1，负向数值越大输出越接近0。

![](../Images/bdb31d70ba16a1e7a87b2c04082ae0b8.png)

这里我们是这样定义的

y =1 / (1 \+ np.exp(-x))

import numpy as np
def sigmoid(x):    return 1 / (1 + np.exp(-x))
class Neuron():    def __init__(self, weights, bias):        self.weights = weights        self.bias = bias
    def feedforward(self, inputs):        total = np.dot(self.weights, inputs) + self.bias        return sigmoid(total)
class OurNeuralNetworks():
    def __init__(self):        weights = np.array([0, 0, 0])        bias = 0
        # The Neuron class here is from the previous section        self.h1 = Neuron(weights, bias)        self.h2 = Neuron(weights, bias)        self.h3 = Neuron(weights, bias)        self.o1 = Neuron(weights, bias)
    def feedforward(self, x):        out_h1 = self.h1.feedforward(x)        out_h2 = self.h2.feedforward(x)        out_h3 = self.h3.feedforward(x)
        # The inputs for o1 are the outputs from h1 and h2        out_o1 = self.o1.feedforward(np.array([out_h1, out_h2,out_h3]))        return out_o1
network = OurNeuralNetworks()x = np.array([0, 0, 0])print(network.feedforward(x))

**随机梯度下降**

使用一种称为**随机梯度下降**（**SGD**）的优化算法，来训练网络。

![](../Images/417892af66f0e4c7a77e6cc464262977.png)

η称为学习率，它决定了我们训练网络速率的快慢。

  

**训练流程**

1.  选择一个样本；
    
2.  计算损失函数对所有权重和偏置(b)的偏导数；
    
3.  使用更新公式更新每个权重和偏置(b)；
    
4.  回到第1步。

import numpy as np
def sigmoid(x):    return 1 / (1 + np.exp(-x))
def deriv_sigmoid(x):    fx = sigmoid(x)    return fx * (1 - fx)
# 计算损失def mse_loss(y_true, y_pred):    return ((y_true - y_pred) ** 2).mean()
class OurNeuralNetwork():    # 初始化    def __init__(self):        # 权重        self.w1 = 0        self.w2 = 0        self.w3 = 0        self.w4 = 0        self.w5 = 0        self.w6 = 0        self.w7 = 0        self.w8 = 0        self.w9 = 0        self.w10 = 0        self.w11 = 0        self.w12 = 0        # 偏置        self.b1 = np.random.normal()        self.b2 = np.random.normal()        self.b3 = np.random.normal()        self.b4 = np.random.normal()
    def feedforward(self, x):        h1 = sigmoid(self.w1 * x[0] + self.w2 * x[1] + self.w3 * x[2] + self.b1)        h2 = sigmoid(self.w4 * x[0] + self.w5 * x[1] + self.w6 * x[2] + self.b2)        h3 = sigmoid(self.w7 * x[0] + self.w8 * x[1] + self.w9 * x[2] + self.b3)        o1 = sigmoid(self.w10 * h1 + self.w11 * h2 + self.w12 * h3 + self.b4)        return o1
    def train(self, data, all_y_trues):        learn_rate = 0.1        epochs = 10000  # 训练次数
        for epoch in range(epochs):            for x, y_true in zip(data, all_y_trues):                # 计算输出                sum_h1 = self.w1 * x[0] + self.w2 * x[1] + self.w3 * x[2] + self.b1                h1 = sigmoid(sum_h1)
                sum_h2 = self.w4 * x[0] + self.w5 * x[1] + self.w6 * x[2] + self.b2                h2 = sigmoid(sum_h2)
                sum_h3 = self.w7 * x[0] + self.w8 * x[1] + self.w9 * x[2] + self.b3                h3 = sigmoid(sum_h3)
                sum_o1 = self.w10 * h1 + self.w11 * h2 + self.w12 * h3 + self.b4                o1 = sigmoid(sum_o1)                y_pred = o1
                d_L_d_ypred = -2 * (y_true - y_pred)
                # 输出                d_ypred_d_w10 = h1 * deriv_sigmoid(sum_o1)                d_ypred_d_w11 = h2 * deriv_sigmoid(sum_o1)                d_ypred_d_w12 = h3 * deriv_sigmoid(sum_o1)                d_ypred_d_b4 = deriv_sigmoid(sum_o1)
                d_ypred_d_h1 = self.w10 * deriv_sigmoid(sum_o1)                d_ypred_d_h2 = self.w11 * deriv_sigmoid(sum_o1)                d_ypred_d_h3 = self.w12 * deriv_sigmoid(sum_o1)
                # 隐藏层的计算                d_h1_d_w1 = x[0] * deriv_sigmoid(sum_h1)                d_h1_d_w2 = x[1] * deriv_sigmoid(sum_h1)                d_h1_d_w3 = x[2] * deriv_sigmoid(sum_h1)                d_h1_d_b1 = deriv_sigmoid(sum_h1)
                d_h2_d_w4 = x[0] * deriv_sigmoid(sum_h2)                d_h2_d_w5 = x[1] * deriv_sigmoid(sum_h2)                d_h2_d_w6 = x[2] * deriv_sigmoid(sum_h2)                d_h2_d_b2 = deriv_sigmoid(sum_h2)
                d_h3_d_w7 = x[0] * deriv_sigmoid(sum_h3)                d_h3_d_w8 = x[1] * deriv_sigmoid(sum_h3)                d_h3_d_w9 = x[2] * deriv_sigmoid(sum_h3)                d_h3_d_b3 = deriv_sigmoid(sum_h3)
                # 更新权重和偏置                self.w10 -= learn_rate * d_L_d_ypred * d_ypred_d_w10                self.w11 -= learn_rate * d_L_d_ypred * d_ypred_d_w11                self.w12 -= learn_rate * d_L_d_ypred * d_ypred_d_w12                self.b4 -= learn_rate * d_L_d_ypred * d_ypred_d_b4
                self.w1 -= learn_rate * d_L_d_ypred * d_ypred_d_h1 * d_h1_d_w1                self.w2 -= learn_rate * d_L_d_ypred * d_ypred_d_h1 * d_h1_d_w2                self.w3 -= learn_rate * d_L_d_ypred * d_ypred_d_h1 * d_h1_d_w3                self.b1 -= learn_rate * d_L_d_ypred * d_ypred_d_h1 * d_h1_d_b1
                self.w4 -= learn_rate * d_L_d_ypred * d_ypred_d_h2 * d_h2_d_w4                self.w5 -= learn_rate * d_L_d_ypred * d_ypred_d_h2 * d_h2_d_w5                self.w6 -= learn_rate * d_L_d_ypred * d_ypred_d_h2 * d_h2_d_w6                self.b2 -= learn_rate * d_L_d_ypred * d_ypred_d_h2 * d_h2_d_b2
                self.w7 -= learn_rate * d_L_d_ypred * d_ypred_d_h3 * d_h3_d_w7                self.w8 -= learn_rate * d_L_d_ypred * d_ypred_d_h3 * d_h3_d_w8                self.w9 -= learn_rate * d_L_d_ypred * d_ypred_d_h3 * d_h3_d_w9                self.b3 -= learn_rate * d_L_d_ypred * d_ypred_d_h3 * d_h3_d_b3
            if epoch % 10 == 0:                y_preds = np.apply_along_axis(self.feedforward, 1, data)                # 计算损失                loss = mse_loss(all_y_trues, y_preds)                print("Epoch %d loss: %.3f" % (epoch, loss))
data = np.array([    [-3, -20, 53],  # 123号阵地    [-23, 10, -46],  # 423号阵地    [26, 10, -6],  # 583号阵地])all_y_trues = np.array([    0,  # 123号阵地    1,  # 423号阵地    0,  # 583号阵地])
network = OurNeuralNetwork()network.train(data, all_y_trues)
# 测试number5 = np.array([60 - 53, 230 - 220, 220 - 206])  # 5号阵地number947 = np.array([70 - 53, 170 - 220, 160 - 206])  # 947号阵地print("5号阵地: %.3f" % network.feedforward(number5))print("947号阵地: %.3f" % network.feedforward(number947))
