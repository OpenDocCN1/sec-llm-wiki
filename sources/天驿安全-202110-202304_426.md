*   """)
    
*   for i in range(begin\_num, end\_num + 1):
    
*   if i < 10:
    
*   i = "0000" + str(i)
    
*   elif i < 100:
    
*   i = "000" + str(i)
    
*   elif i < 1000:
    
*   i = "00" + str(i)
    
*   elif i < 10000:
    
*   i = "0"+str(i)
    
*   with open("5.txt", 'a') as f:
