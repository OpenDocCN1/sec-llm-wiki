服务器数据

上面的NewsList.vue的数据都是硬编码，如果需要从服务器上获取数据呢？由于NewsList.vue直接从服务器获取数据会减低它的可重用性和增加复杂度。所以，可以通过一个服务从服务器上获取数据。

新建一个目录src/services，在目录下创建newsService.js

const newsService = {
    getArticlesByType (articleType) {
        return fetch( '/data/articles.json')
        .then((response) => {
            return response.json()
        })
        .then( (serverArticles) => {
            const newsArticles = serverArticles
            .filter(s => s.articleType === articleType)
            .map( s => {
                return {
                    id: s.id,
                    title: s.title,
                    content: s.content,
                    dateString: s.dateString,
                    baseImageName: s.baseImageName,
                    articleType: s.articleType,
                    isFavourite: s.isFavourite
                }
            })
            return newsArticles
        })
    },
    getFavorites() {
        return fetch( '/data/articles.json')
        .then((response) => {
            return response.json()
        })
        .then( (serverArticles) => {
            const newsArticles = serverArticles
            .filter(s => s.isFavourite === true )
            .map( s => {
                return {
                    id: s.id,
                    title: s.title,
                    content: s.content,
                    dateString: s.dateString,
                    baseImageName: s.baseImageName,
                    articleType: s.articleType,
                    isFavourite: s.isFavourite
                }
            })
            return newsArticles
        })
    }
}

export default newsService

方法getArticlesByType从静态文件/data/articles.json获取所有文章，并根据ArticleType筛选，映射成页面上需要的数据。在实际的应用中，是需要服务器来做这种动作，但在本文中，不涉及到这些，只由前端完成。

在TopStories.vue上使用这个服务

<template>

import newsService from '../services/newsService'
import NewsList from '../components/NewsList.vue'

export default {
  components: {
    NewsList,
  },
  data() {

      newsArticles : [],

  mounted() {

    })

</script>

顶部工具栏标题

如果用户每次点击底部工具栏，顶部工具栏的标题会随着不同按钮而改变，友好度会增加不少。

为了达到这种效果，需要：

  • 对每条路由添加元数据，使得它包含工具栏标题
  • 当路由切换时，更新Vuex存储的工具栏标题
  • 在顶部工具栏组件中，用Vuex存储的数据来渲染工具栏标题

Vuex支持将存储分隔到不同的模块中，这有助于保持代码的组织，并允许将相关状态分组在一起。

在目录src/store/modules创建一个模块topToolbar.js

const state = {
    title: 'Top Stories',
}

// getters
const getters = {
    getTitle: (state) => {
        return state.title
    },
}

const actions = {
    changeTitle(context, title) {
        context.commit('SET_TITLE', title)
    },
}

const mutations = {
    SET_TITLE(state, data) {
        state.title = data
    },
}

export default {
    namespace: true,
    state,
    getters,
    actions,
    mutations,
}

接着在src/store的index.js调用它

import Vue from 'vue'
import Vuex from 'vuex'
import topToolbar from './modules/topToolbar'

Vue.use(Vuex)

const store = new Vuex.Store({
  modules: {
    topToolbar,
  }
})

export default store

然后在路由里调用changeTitle

src/route/index.js

import Vue from 'vue'
import Router from 'vue-router'
import store from '../store'

import TopStories from '../views/TopStories.vue'
import CodeExamples from '../views/CodeExamples.vue'
import MyFavorites from '../views/MyFavorites.vue'

Vue.use(Router)

const router = new Router({
    mode: 'history',
    base: process.env.BASE_URL,
    routes: [
        {
            path: '/',
            name: 'top-stories',
            component: TopStories,
            meta: {
                title: "Top Stories",
            },
        },
        {
            path: '/code-examples',
            name: 'code-examples',
            component: CodeExamples,
            meta: {
                title: "Code Examples",
            },
        },
        {
            path: '/my-favorites',
            name: 'my-favorites',
            component: MyFavorites,
            meta: {
                title: "Favourites",
            },
        },
    ]
})

router.beforeEach((to, from,next) => {
    store.dispatch('changeTitle', to.meta.title)
    next()
})

export default router
