我们在上一篇探讨了mcheck+eBPF+uprobes 的方案来解决堆内存损坏，但由于本人水平问题，没办法解决得了eBPF的问题，所以并没有解决这个问题。

那，有没有办法通过调试器的方式来做呢，就是使用GDB+python的方式来解决 。毕竟，在调试器下，可以直接读取和写入被调试程序的内存。那么，通过python脚本就可以对相应的地址进行检测了。

按照之前的思路，类似的方案应该是

主要是两个阶段：

  • 获取申请调用栈

  1. 每次申请前，把之前已经申请过的地址全部进行检测，一旦发现有破坏，立马打印相应的栈信息，并中断到调试器，来检查相应的代码
  2. 每次申请时，记录获得的内存地址和调用栈。
  3. 每次释放时，把之前已经申请过的地址全部进行检测，如果是没遇到任何损坏的，从记录的地址列表删除该记录，如果有破坏的，立马打印相应的栈信息，并中断到调试器来检查相应代码

  • 定位破坏时的调用栈

    1. 在第一阶段获得的申请调用栈上对调用malloc处的代码或指令地址打断点
    2. 断点命中，对申请到的内存块尾部打数据断点
    3. 运行程序直至命中数据断点，获得破坏时调用栈

    由于程序是用了mcheck机制，那么，在调试器里里可以直接通过call命令调用mprobe，不过，不知道为啥，在我机器上的GDB每次做这种操作就crash掉，所以，只能用python脚本来实现类似mprobe的功能。

    幸好mprobe的功能不复杂，根据malloc/mcheck.c的代码，只是在读取内存里的二进制数据需要用struct模块。

    首先，要获取调试实例，才能够读取它的内存空间

    inferior = gdb.selected_inferior()

    再获取每块内存的头部

    def gethdr(ptr):
        buf = inferior.read_memory(ptr-MCHECK_HDR_SIZE,MCHECK_HDR_SIZE)
        fmt = "QQQQQQ"
        hdr = {}
        hdr["size"],hdr["magic"],hdr["prev"],hdr["next"],hdr["block"],hdr["magic2"] = struct.unpack(fmt,buf)
        return hdr

    然后对申请过的每场内存都检查头部和尾部，一旦有问题，就立马打印出来有问题堆块的地址和申请调用栈

            if hdr["size"] == 0:

            ret = hdr["magic"]^(hdr["prev"]+hdr["next"])
            if ret == MAGICWORD:
                buf = inferior.read_memory(m+hdr["size"],1)
                tail, = struct.unpack("B",buf)
                if tail != MAGICBYTE:
                    alert("memory block %d tail damage:", m)
                    damage = True
                elif hdr["magic2"]^hdr["block"] != MAGICWORD:
                    alert("memory block %d head damage:", m)
                    damage = True
                else:
                    continue
            elif ret == MAGICFREE:
                alert("memory block %d double free:", m)
                damage = True
            else:
                alert("memory block %d head damage:", m)
                damage = True
        return damage

    import sys
    import gdb
    import struct

    entries = {}
    MAGICWORD = 0xfedabeeb
    MAGICFREE = 0xd8675309
    MAGICBYTE = 0xd7
    MALLCFLOOD = 0x93
    FREEFLOOD = 0x95

    MCHECK_HDR_SIZE = 0x30

    inferior = gdb.selected_inferior()

    def gethdr(ptr):
        buf = inferior.read_memory(ptr-MCHECK_HDR_SIZE,MCHECK_HDR_SIZE)
        fmt = "QQQQQQ"
        hdr = {}
        hdr["size"],hdr["magic"],hdr["prev"],hdr["next"],hdr["block"],hdr["magic2"] = struct.unpack(fmt,buf)
        return hdr

    def alert(fmt,m):
        print(fmt%m)
        print("stack:")
        print(entries[m])
        
    def checkMem():
        damage = False
        mems = list(entries.keys())
        mems.sort()
        for m in mems:
            hdr = gethdr(m)
            if hdr["size"] == 0:
                continue
            ret = hdr["magic"]^(hdr["prev"]+hdr["next"])
            if ret == MAGICWORD:
                buf = inferior.read_memory(m+hdr["size"],1)
                tail, = struct.unpack("B",buf)
                if tail != MAGICBYTE:
                    alert("memory block %d tail damage:", m)
                    damage = True
                elif hdr["magic2"]^hdr["block"] != MAGICWORD:
                    alert("memory block %d head damage:", m)
                    damage = True
                else:
                    continue
            elif ret == MAGICFREE:
                alert("memory block %d double free:", m)
                damage = True
            else:
                alert("memory block %d head damage:", m)
                damage = True
        return damage

    class MallocAfterBreakpoint(gdb.FinishBreakpoint):
        def stop(self):
            frs = []
            fr = gdb.newest_frame()
            rval = int(fr.read_register("rax"))
            if rval == 0:
                return False
            hdr = gethdr(rval)
            if hdr["size"] == 0:
                return False
            while fr is not None and fr.is_valid():
                pc = fr.pc()
                info = {
                        "pc": "0x%x"%int(pc),
                        "address": gdb.execute("info symbol %d"%pc, to_string=True)
                        }
                frs.append(info)
                fr = fr.older()
            entries[rval] = frs 
            return False 

    class MallocBeforeBreakpoint(gdb.Breakpoint):
        def stop(self):
            if checkMem():
                return True
            fr = gdb.newest_frame()
            fin = MallocAfterBreakpoint(fr, internal=True)
            return False 

    class FreeBreakpoint(gdb.Breakpoint):
        def stop(self):
            if checkMem():
                return True 
            fr = gdb.newest_frame()
            v = int(fr.read_register("rdi"))
            if v == 0:
                return False
            return False 

    arch = gdb.selected_inferior().architecture()
    if "x86-64" not in arch.name():
        sys.exit(1)
    mallocBp = MallocBeforeBreakpoint("malloc")
    freeBp = FreeBreakpoint("free")
    gdb.execute("r")

    测试程序corrupt.c

    #include <unistd.h>
    #include <stdlib.h>
    #include <stdio.h>
    #include <string.h>
    #include <mcheck.h>
    #include <execinfo.h>
    #include <signal.h>

    #define BT_BUF_SIZE 100

    char* mem[BT_BUF_SIZE];
    int idx = 0;

    void a(size_t n)
    {
        char* p = malloc(n);
        if( p != NULL )
        {
            memset(p, 0, n);
            mem[idx++] = p;
        }
    }

    void b(size_t n)
    {
        char* p = malloc(n);
        if( p != NULL )
        {
            memset(p, 0, n + 16);
            mem[idx++] = p;
        }
    }

    void c(size_t n)
    {
        char* p = malloc(n);
        if( p != NULL )
        {
            memset(p, 0, n + 24);
            mem[idx++] = p;
        }
    }

    int main(int argc, char* argv[])
    {
        if (mcheck(NULL) != 0) 
        {
            fprintf(stderr, "mcheck() failed\n");

            exit(EXIT_FAILURE);
        }

        size_t i = 8;
        unsigned seed = atoi(argv[1]);
        memset(mem,0,BT_BUF_SIZE);
        srand(seed);
        for ( ; i < 100; i++ )
        {
            int r = rand()%seed;
            switch( r % 3 )
            {
                case 0:
                    a(r);
                    break;
                case 1:
                    b(r);
                    break;
                case 2:
                    c(r);
                    break;
            }
        }

        int j = 0;
        while(mem[j] != NULL)
        {
            free(mem[j]);
            mem[j] = NULL;
            j++;
        }
        return 0;
