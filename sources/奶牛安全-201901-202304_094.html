我们在上一篇Linux堆内存损坏2：valgrind篇探索了一下valgrind的定位手段，很完美。基本上，不想深究的朋友，可以停下来。但有兴趣的朋友可以继续看一下。

根据glibc的手册3.2.3.8 Heap Consistency Checkinghttps://www.gnu.org/software/libc/manual/html_node/Heap-Consistency-Checking.html，提到一种mcheck的手段。这里试一下看一下它的效果。

继续上一篇的代码corrupt.c，但会加上一些东西，比如

  1. 引入mcheck, 在main函数开头调用mcheck和随后调用mcheck_check_all
  2. 引入backtrace, 在myfunc里调用，而myfunc由mcheck注册
  3. 引入一个新的数组存储新申请下来的内存，结束时统一释放
#include <unistd.h>
#include <stdlib.h>
#include <stdio.h>
#include <string.h>
#include <mcheck.h>
#include <execinfo.h>

#define BT_BUF_SIZE 100

char* mem[BT_BUF_SIZE];
int idx = 0;


void myfunc(enum mcheck_status mstatus)
{
    int nptrs;
    void *buffer[BT_BUF_SIZE];
    char **strings;

    printf("check status:%d\n", mstatus );
    nptrs = backtrace(buffer, BT_BUF_SIZE);
    printf("backtrace() returned %d addresses\n", nptrs);

    /* The call backtrace_symbols_fd(buffer, nptrs, STDOUT_FILENO)
       would produce similar output to the following: */

    strings = backtrace_symbols(buffer, nptrs);
    if (strings == NULL) {
        perror("backtrace_symbols");
        exit(EXIT_FAILURE);
    }

    for (int j = 0; j < nptrs; j++)
        printf("%s\n", strings[j]);

    free(strings);

}

void a(size_t n)
{
    char* p = malloc(n);
    if( p != NULL )
    {
        memset(p, 0, n);
        mem[idx++] = p;
    }
}

void b(size_t n)
{
    char* p = malloc(n);
    if( p != NULL )
    {
        memset(p, 0, n + 16);
        mem[idx++] = p;
    }
}

void c(size_t n)
{
    char* p = malloc(n);
    if( p != NULL )
    {
        memset(p, 0, n + 24);
        mem[idx++] = p;
    }
}

int main(int argc, char* argv[])
{
    if (mcheck(myfunc) != 0) 
    {
        fprintf(stderr, "mcheck() failed\n");

        exit(EXIT_FAILURE);
    }

    size_t i = 8;
    unsigned seed = atoi(argv[1]);
    memset(mem,0,BT_BUF_SIZE);
    mcheck_check_all();
    srand(seed);
    for ( ; i < 100; i++ )
    {
        int r = rand()%seed;
        printf("count:%u,random:%d, type:%u\n",i, r, r%3);
        switch( r % 3 )
        {
            case 0:
                a(r);
                break;
            case 1:
                b(r);
                break;
            case 2:
                c(r);
                break;
        }
    }

    int j = 0;
    while(mem[j] != NULL)
    {
        free(mem[j]);
        mem[j] = NULL;
        j++;
    }
    return 0;
}

这个结果，只让人说一句“浦你阿母，死扑街“。一堆乱七八糟的东西，是什么鬼。

根据mcheck.h的定义：

enum mcheck_status
{
  MCHECK_DISABLED = -1,         /* Consistency checking is not turned on.  */
  MCHECK_OK,                    /* Block is fine.  */
  MCHECK_FREE,                  /* Block freed twice.  */
  MCHECK_HEAD,                  /* Memory before the block was clobbered.  */
  MCHECK_TAIL                   /* Memory after the block was clobbered.  */
};

可见，这里是检测出来块的尾部被损坏了。3就是MCHECK_TAIL
