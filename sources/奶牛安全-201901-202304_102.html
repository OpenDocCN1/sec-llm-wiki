本文介绍sytstemd成为Linux一号进程的原理，同时从反入侵角度来说明它的安全风险

原理

systemd是在内核文件init/main.c里的rest_init启动的。

noinline void __ref rest_init(void)
{
 struct task_struct *tsk;
 int pid;

 rcu_scheduler_starting();
 /*
  * We need to spawn init first so that it obtains pid 1, however
  * the init task will end up wanting to create kthreads, which, if
  * we schedule it before we create kthreadd, will OOPS.
  */
 pid = kernel_thread(kernel_init, NULL, CLONE_FS);
 /*
  * Pin init on the boot CPU. Task migration is not properly working
  * until sched_init_smp() has been run. It will set the allowed
  * CPUs for init to the non isolated CPUs.
  */
 rcu_read_lock();
 tsk = find_task_by_pid_ns(pid, &init_pid_ns);
 tsk->flags |= PF_NO_SETAFFINITY;
 set_cpus_allowed_ptr(tsk, cpumask_of(smp_processor_id()));
 rcu_read_unlock();

 numa_default_policy();
 pid = kernel_thread(kthreadd, NULL, CLONE_FS | CLONE_FILES);
 rcu_read_lock();
 kthreadd_task = find_task_by_pid_ns(pid, &init_pid_ns);
 rcu_read_unlock();

 /*
  * Enable might_sleep() and smp_processor_id() checks.
  * They cannot be enabled earlier because with CONFIG_PREEMPTION=y
  * kernel_thread() would trigger might_sleep() splats. With
  * CONFIG_PREEMPT_VOLUNTARY=y the init task might have scheduled
  * already, but it's stuck on the kthreadd_done completion.
  */
 system_state = SYSTEM_SCHEDULING;

 complete(&kthreadd_done);

 /*
  * The boot idle thread must execute schedule()
  * at least once to get things moving:
  */
 schedule_preempt_disabled();
 /* Call into cpu_idle with preempt disabled */
 cpu_startup_entry(CPUHP_ONLINE);
}

从这一段

 /*
  * We need to spawn init first so that it obtains pid 1, however
  * the init task will end up wanting to create kthreads, which, if
  * we schedule it before we create kthreadd, will OOPS.
  */
 pid = kernel_thread(kernel_init, NULL, CLONE_FS);

就可以看出来了，说明它是调用kernel_init启动一个内核线程执行。

再看kernel_init函数

static int __ref kernel_init(void *unused)
{
 int ret;

 /*
  * Wait until kthreadd is all set-up.
  */
 wait_for_completion(&kthreadd_done);

 kernel_init_freeable();
 /* need to finish all async __init code before freeing the memory */
 async_synchronize_full();

 system_state = SYSTEM_FREEING_INITMEM;
 kprobe_free_init_mem();
 ftrace_free_init_mem();
 kgdb_free_init_mem();
 exit_boot_config();
 free_initmem();
 mark_readonly();

 /*
  * Kernel mappings are now finalized - update the userspace page-table
  * to finalize PTI.
  */
 pti_finalize();

 system_state = SYSTEM_RUNNING;
 numa_default_policy();

 rcu_end_inkernel_boot();

 do_sysctl_args();

 if (ramdisk_execute_command) {
  ret = run_init_process(ramdisk_execute_command);
  if (!ret)
   return 0;
  pr_err("Failed to execute %s (error %d)\n",
         ramdisk_execute_command, ret);
 }

 /*
  * We try each of these until one succeeds.
  *
  * The Bourne shell can be used instead of init if we are
  * trying to recover a really broken machine.
  */
 if (execute_command) {
  ret = run_init_process(execute_command);
  if (!ret)
   return 0;
  panic("Requested init %s failed (error %d).",
        execute_command, ret);
 }

 if (CONFIG_DEFAULT_INIT[0] != '\0') {
  ret = run_init_process(CONFIG_DEFAULT_INIT);
  if (ret)
   pr_err("Default init %s failed (error %d)\n",
          CONFIG_DEFAULT_INIT, ret);
  else
   return 0;
 }

 if (!try_to_run_init_process("/sbin/init") ||
     !try_to_run_init_process("/etc/init") ||
     !try_to_run_init_process("/bin/init") ||
     !try_to_run_init_process("/bin/sh"))
  return 0;

 panic("No working init found.  Try passing init= option to kernel. "
       "See Linux Documentation/admin-guide/init.rst for guidance.");
}

  • execute_command是在这个函数设置的
static int __init init_setup(char *str)
{
 unsigned int i;

 execute_command = str;
 /*
  * In case LILO is going to boot us with default command line,
  * it prepends "auto" before the whole cmdline which makes
  * the shell think it should execute a script with such name.
  * So we ignore all arguments entered _before_ init=... [MJ]
  */
 for (i = 1; i < MAX_INIT_ARGS; i++)
  argv_init[i] = NULL;
 return 1;
}
__setup("init=", init_setup);

**可见在这几种情况下，systemd才会成为一号进程: **

  1. /init不存在，或/init是systemd的软链
  2. grub启动参数不设置init=，或者init=指向systemd路径
  3. 编译内核时不设置CONFIG_DEFAULT_INIT，或指向systemd路径
  4. /sbin/init, /etc/init， /bin/init是systemd的软链

安全风险

根据上面原理，要设置某个程序成为一号进程是很简单的事情。所以，不少路由器或防火墙厂商都是通过软链的方式让系统启动到它们指定的文本操作界面，从而限制用户的操作。相信不少运维的兄弟都有过操作体验。

但如果系统被入侵了，黑客把/init, /sbin/init, /etc/init, /bin/init或grub启动配置指向到恶意程序呢？一旦系统重启，整个系统就被恶意软件掌管了。

针对这种情况，应该怎么做呢？

  • HIDS对上面的路径和配置进行监控，一旦发现有可疑变更，立马处理，恢复到正常的指向。
  • 如果指向被更改且系统已经被接管了，立马使用liveCD启动，把系统所在的盘挂载上去，从而进行定位和恢复工作
