讲述如何在命名空间里建立隔离的网络。

译自: https://medium.com/@teddyking/namespaces-in-go-network-fdcf63e76100

在上一篇文章用Go探究命名空间：Mount命名空间篇，已经使用PivotRoot和Mount命名空间把ns-process切换到一个新的根文件系统。虽然ns-process只是运行一个/bin/sh进程，但越来越像其它容器了，已经具有一些非常酷的特性：

  1. 由于User命名空间缘故，能够以非root用户运行
  2. 由于Mount命名空间缘故，能够有自己的根文件系统
  3. 由于PID命名空间缘故，已经和宿主机有了进程隔离功能

但它还缺少一个很关键的功能：网络。

出现这种情况是因为ns-process创建了一个新的Network命名空间，它隔离所有网络相关的资源（IP, 端口，网络接口之类）。

本文将探讨怎样设置新的Network命名空间，使得ns-process有一个网络接口和可路由的IP地址。

网络知识速览

如果想把网络连接加入到ns-process，对Network命名空间的坚实理解是必需的。在这里，推荐阅读https://blog.scottlowe.org/2013/09/04/introducing-linux-network-namespaces/，这个链接里的内容会让人形成如何在ns-process对Network命名空间配置的基础。链接里的内容汇总如下：

  1. 在宿主机的Network命名空间创建一个桥接设备
  2. 创建虚拟以太对
  3. 把虚拟以太对一头对接到桥接设备
  4. 把虚拟以太对另一头对接到ns-process的Network命名空间
  5. 确保命名空间里进程发起的网络流量都通过虚拟以太对路由。

大致方案是在ns-process的Network命名空间和宿主机之间建立连接。如图所示

看起来相当大的工作量，特别当创建和配置需要涉及两个不同的Network命名空间，事情变得更加复杂。且创建网络连接是需要root权限，这和前面几篇Linux命名空间，用Go探究命名空间：基础篇，用Go探究命名空间：User命名空间篇，用Go探究命名空间：reexec篇，用Go探究命名空间：Mount命名空间篇可以使用普通用户运行ns-process的初衷相悖。

幸好创建网络连接需要root权限可以通过setuid的方式解决。可以把创建网络连接的代码独立出来，确保这个代码属于root用户且设置了setuid标志。然后，再通过ns-process调用它。

基于这个思路，引入netsetgo。

netsetgo

netsetgo是一个用于对容器创建Network命名空间的小程序。它实现上面列举的网络配置功能。它的代码在https://github.com/teddyking/netsetgo，大致逻辑如下：

  1. 在bridge.go里调用netlink.LinkAdd创建桥接设备
  2. 在veth.go调用netlink.LinkAdd创建虚拟以太对
  3. 在bridge.go里调用netlink.LinkSetMaster把虚拟以太对一头对接到桥接设备
  4. 在veth.go调用netlink.LinkSetNsPid把虚拟以太对一头对接到新的Network命名空间
  5. 在container.go里调用netlink.RouteAdd在新的network命名空间里添加一条默认路由

为了ns-process通过调用netsetgo， 需要下载它并给它设置合适的权限。

ns-process

netsetgo已经部署完了，开始修改ns-process调用它来配置网络。

乍一看，只要创建一个执行netsetgo的*exec.Cmd对象，然后在合适时候运行它就行。

然而，事情并没有这么简单。先看一下之前的代码

// Git repo: https://github.com/teddyking/ns-process
// Git tag: 4.1
// Filename: ns_process.go
func main() {
 cmd := reexec.Command("nsInitialisation", rootfsPath)

 cmd.SysProcAttr = &syscall.SysProcAttr{
  Cloneflags: syscall.CLONE_NEWNS |
   syscall.CLONE_NEWUTS |
   syscall.CLONE_NEWIPC |
   syscall.CLONE_NEWPID |
   syscall.CLONE_NEWNET |
   syscall.CLONE_NEWUSER,
 }

 if err := cmd.Run(); err != nil {
  fmt.Printf("Error running Command - %s\n", err)
  os.Exit(1)
 }
}

在这里，使用cmd.Run()运行一个带有各种CLONE_NEW*的reexec命令，而cmd.Run()在所运行进程没退出之前是不会返回的。在这里是没有问题，因为所有配置操作都是在新创建命名空间里进行。

然而，netsetgo还需要在宿主机的命名空间里配置，这意味着不能够使用类似cmd.Run()这种阻塞的调用。

幸好，可以通过cmd.Start()（立刻返回）和cmd.Wait()（在启动命令退出时返回）来替代cmd.Run()。这可以保证既在宿主机命名空间执行netsetgo，也可以在新的命名空间里执行它。

// Git repo: https://github.com/teddyking/ns-process
// Git tag: 5.0
// Filename: ns_process.go
if err := cmd.Start(); err != nil {
 fmt.Printf("Error starting the reexec.Command - %s\n", err)
 os.Exit(1)
}

pid := fmt.Sprintf("%d", cmd.Process.Pid)
netsetgoCmd := exec.Command(netsetgoPath, "-pid", pid)
if err := netsetgoCmd.Run(); err != nil {
 fmt.Printf("Error running netsetgo - %s\n", err)
 os.Exit(1)
}

if err := cmd.Wait(); err != nil {
 fmt.Printf("Error waiting for reexec.Command - %s\n", err)
 os.Exit(1)
}

下面要确保在网络没配置之前不会执行/bin/sh，而判断网络配置OK的依据是虚拟以太对出现在新的Network命名空间，这可以通过一个简单的循环来做

// Git repo: https://github.com/teddyking/ns-process
// Git tag: 5.0

func waitForNetwork() error {
 maxWait := time.Second * 3
 checkInterval := time.Second
 timeStarted := time.Now()

 for {
  interfaces, err := net.Interfaces()
  if err != nil {
   return err
  }

  // pretty basic check ...
  // > 1 as a lo device will already exist
  if len(interfaces) > 1 {
   return nil
  }

  if time.Since(timeStarted) > maxWait {
   return fmt.Errorf("Timeout after %s waiting for network", maxWait)
  }

  time.Sleep(checkInterval)
 }
}

上面的逻辑是在3秒内检测是否有多于一个网络接口存在（包括环回接口）。

然后，在nsInitialisation调用它

func nsInitialisation() {
 newrootPath := os.Args[1]

 if err := mountProc(newrootPath); err != nil {
  fmt.Printf("Error mounting /proc - %s\n", err)

 }

}

整体的ns-process.go如下

// Git repo: https://github.com/teddyking/ns-process
// Git tag: 5.0
// Filename: ns_process.go
package main

import (
    "flag"
    "fmt"
    "os"
    "os/exec"
    "time"
    "net"
    "syscall"
    "github.com/docker/docker/pkg/reexec"
    "path/filepath"
)
func mountProc( newroot string ) error {
    source := "proc"
    target := filepath.Join( newroot, "/proc" )
    fstype := "proc"
    flags := 0
    data := ""

    os.MkdirAll( target, 0755 )
    if err := syscall.Mount(
        source,
        target,
        fstype,
        uintptr( flags ),
        data,
    ); err != nil {
        return err
    }

    return nil
}

func pivotRoot( newroot string ) error {
    putold := filepath.Join( newroot, "/.pivot_root" )
    
    // bind mount newroot to itself - this is a slight hack
    // needed to work around a pivot_root requirement
    if err := syscall.Mount(
        newroot,
        newroot,
        "",
        syscall.MS_BIND|syscall.MS_REC,
        "",
    ); err != nil {
        return err
    }

    // create putold directory
    if  err := os.MkdirAll( putold, 0700 ); err != nil {
        return err
    }

    // call pivot_root
    if err := syscall.PivotRoot( newroot, putold ); err != nil {
        return err
    }

    // ensure current working directory is set to new root
    if err := os.Chdir( "/" ); err != nil {
        return err
    }

    // unmount putold, which now lives at /.pivot_root
    putold = "/.pivot_root"
    if err := syscall.Unmount( 
        putold,
        syscall.MNT_DETACH,
    ); err != nil {
        return err
    }

    // remove putold
    if err := os.RemoveAll( putold ); err != nil {
        return err
    }
    
    return nil
}

func waitForNetwork() error {
    maxWait := time.Second*3
    checkInterval := time.Second
    timeStarted := time.Now()

    for {
        interfaces, err := net.Interfaces()
        if err != nil  {
            return err
        }

        if  len( interfaces ) >1 {
            return nil
        }

        if time.Since( timeStarted) > maxWait {
            return fmt.Errorf( "Timeout after %s waiting for network", maxWait )
        }
        time.Sleep( checkInterval )
    }
}

func init() {
    fmt.Printf( "program %s pid %d init\n", os.Args[0], os.Getpid() )
    reexec.Register( "nsInitialisation", nsInitialisation )
    if reexec.Init() {
        os.Exit(0)
    }
}

func nsInitialisation() {
    fmt.Printf( "\n>> namespace setup code goes here << \n\n" )
    newrootPath := os.Args[1]

    if err := mountProc(newrootPath ); err != nil {
        fmt.Print( "Error mounting /proc - %s\n", err )
        os.Exit(1)
    }

    if err := pivotRoot( newrootPath ); err != nil {
        fmt.Printf( "Error running pivot_root - %s\n", err )
        os.Exit(1)
    }

    if err := waitForNetwork(); err != nil {
        fmt.Printf( "Error waiting for network - %s \n", err )
        os.Exit(1)
    }
    nsRun()
}

func nsRun() {
    cmd := exec.Command( "/bin/sh" )
    cmd.Stdin = os.Stdin
    cmd.Stdout = os.Stdout
    cmd.Stderr = os.Stderr

    cmd.Env = []string{"PS1=-[ns-process]- # "}

    if err := cmd.Run(); err != nil {
        fmt.Printf("Error running the /bin/sh command - %s\n", err)
        os.Exit(1)
    }
}


func main() {
    fmt.Printf( "program %s pid %d main\n", os.Args[0], os.Getpid() )
    var rootfsPath string
    var netsetgoPath string
    flag.StringVar( &rootfsPath, "rootfs", "/tmp/ns-process/rootfs", "Path to the root filesystem to use" )
    flag.StringVar( &netsetgoPath, "netsetgo", "/usr/local/bin/netsetgo", "Path to netsetgo binary" )
    flag.Parse() 


    cmd := reexec.Command("nsInitialisation", rootfsPath)

    cmd.Stdin = os.Stdin
    cmd.Stdout = os.Stdout
    cmd.Stderr = os.Stderr

    cmd.Env = []string{"PS1=-[ns-process]- # "}

    cmd.SysProcAttr = &syscall.SysProcAttr{
        Cloneflags: syscall.CLONE_NEWNS |
        syscall.CLONE_NEWUTS |
        syscall.CLONE_NEWIPC |
        syscall.CLONE_NEWPID |
        syscall.CLONE_NEWNET |
        syscall.CLONE_NEWUSER,
        UidMappings: []syscall.SysProcIDMap{
            {
                ContainerID: 0,
                HostID:      os.Getuid(),
                Size:        1,
            },
        },
        GidMappings: []syscall.SysProcIDMap{
            {
                ContainerID: 0,
                HostID:      os.Getgid(),
                Size:        1,
            },
        },
    }

    if err := cmd.Start(); err != nil {
        fmt.Printf( "Error starting the reexec.Command - %s\n", err )
        os.Exit(1)
    }

    pid := fmt.Sprintf( "%d", cmd.Process.Pid )
    fmt.Printf( "%s\n", pid )
    netsetgoCmd := exec.Command(netsetgoPath, "-pid", pid )
    if  err := netsetgoCmd.Run(); err != nil {
        fmt.Printf( "Error running netsetgo - %s\n", err )
        os.Exit(1)
    }

    if err := cmd.Wait(); err != nil {
        fmt.Printf( "Error waiting for reexec.Command - %s\n", err )
        os.Exit(1)
    }
}

首先在宿主机上设置iptables做转发

$ sudo iptables -tnat -N netsetgo
$ sudo iptables -tnat -A PREROUTING -m addrtype --dst-type LOCAL -j netsetgo
$ sudo iptables -tnat -A OUTPUT ! -d 127.0.0.0/8 -m addrtype --dst-type LOCAL -j netsetgo
$ sudo iptables -tnat -A POSTROUTING -s 10.10.10.0/24 ! -o brg0 -j MASQUERADE
$ sudo iptables -tnat -A netsetgo -i brg0 -j RETURN

在新的命名空间里执行

-[ns-process]- # echo "nameserver 8.8.8.8" >> /etc/resolv.conf
-[ns-process]- # ping google.com
^C
