按照容器设计的理念，在命名空间里的进程应该尽量少知道它所在的宿主机，不知道宿主机运行的进程，也不知道宿主机上挂载的分区。为什么会这样呢？

按照man 7 mount_namespaces的结果

“When a process creates a new mount namespace using clone(2) or unshare(2) with the CLONE_NEWNS flag, the mount point list for the new namespace is a copy of the caller’s mount point list.”

根据上面所述，新的Mount命名空间是直接复制调用者的挂载点列表，这也是/proc/mounts为什么会显示宿主机的挂载列表。按照容器设计理念，需要清除宿主机的挂载信息，达到隔离效果，需要用到pivot_root系统调用。

pivot_root系统调用

pivot_root允许进程设置新的根文件系统，就是可以改变/的位置。它允许在其它位置挂载当前根文件系统，同时把新的根文件系统挂载到/。一旦原先的根文件系统移动后，它可以卸载。由此，就有了一种在新的Mount命名空间卸载宿主机挂载点的机制：把它们旋走，然后卸载。

这也是可以在Ubuntu上跑类似CentOS之类的其它发行版的原因。只要Ubuntu宿主机磁盘有CentOS文件系统的影像，就可以创建一个新的Mount命名空间，调用pivot_root把根文件系统指向CentOS文件系统，然后在这个命名空间运行的任意进程都会认为它们在CentOS上运行。

按照上面描述，pivot_root应该是在命名空间创建之后，运行任何进程之前进行。这也是上一篇用Go探究命名空间：reexec篇里reexec发挥的时候。

在Go里，pivot_root是由syscall包的PivotRoot实现的。

func PivotRoot(newroot string, putold string) (err error)

newroot是指定新根文件系统的目录，putold是要把当前根目录移动到的目标目录。

newroot和putold有如下限制：

  1. 它们必须是目录
  2. 它们和当前根目录不在同一个文件系统
  3. putold应该在newroot里面
  4. putold上没有其它文件系统挂载

准备newroot文件系统的过程可能非常详细和复杂。以Docker的分层文件系统方法为例，许多文件系统层融合在一起以表示单个一致的根。这里只做一些简单得多的事情，即假设一个合适的根文件系统已经准备好可以使用。

// Git repo: https://github.com/teddyking/ns-process
// Git tag: 4.0

func pivotRoot(newroot string) error {
 putold := filepath.Join(newroot, "/.pivot_root")

 // bind mount newroot to itself - this is a slight hack
 // needed to work around a pivot_root requirement
 if err := syscall.Mount(
  newroot,
  newroot,
  "",
  syscall.MS_BIND|syscall.MS_REC,
  "",
 ); err != nil {
  return err
 }

 // create putold directory
 if err := os.MkdirAll(putold, 0700); err != nil {
  return err
 }

 // call pivot_root
 if err := syscall.PivotRoot(newroot, putold); err != nil {
  return err
 }

 // ensure current working directory is set to new root
 if err := os.Chdir("/"); err != nil {
  return err
 }

 // umount putold, which now lives at /.pivot_root
 putold = "/.pivot_root"
 if err := syscall.Unmount(
  putold,
  syscall.MNT_DETACH,
 ); err != nil {
  return err
 }

 // remove putold
 if err := os.RemoveAll(putold); err != nil {
  return err
 }

 return nil
}

// Git repo: https://github.com/teddyking/ns-process

func nsInitialisation() {
 newrootPath := os.Args[1]

 if err := pivotRoot(newrootPath); err != nil {

package main

import (
    "flag"

func mountProc( newroot string ) error {
    source := "proc"
    target := filepath.Join( newroot, "/proc" )
    fstype := "proc"
    flags := 0
    data := ""

    os.MkdirAll( target, 0755 )
    if err := syscall.Mount(
        source,
        target,
        fstype,
        uintptr( flags ),
        data,
    ); err != nil {
        return err
    }

    return nil
}

func pivotRoot( newroot string ) error {
    putold := filepath.Join( newroot, "/.pivot_root" )
    
    // bind mount newroot to itself - this is a slight hack
    // needed to work around a pivot_root requirement
    if err := syscall.Mount(
        newroot,
        newroot,
        "",
        syscall.MS_BIND|syscall.MS_REC,
        "",
    ); err != nil {
        return err
    }

    // create putold directory
    if  err := os.MkdirAll( putold, 0700 ); err != nil {
        return err
    }

    // call pivot_root
    if err := syscall.PivotRoot( newroot, putold ); err != nil {
        return err
    }

    // ensure current working directory is set to new root
    if err := os.Chdir( "/" ); err != nil {
        return err
    }

    // unmount putold, which now lives at /.pivot_root
    putold = "/.pivot_root"
    if err := syscall.Unmount( 
        putold,
        syscall.MNT_DETACH,
    ); err != nil {
        return err
    }

    // remove putold
    if err := os.RemoveAll( putold ); err != nil {
        return err
    }
    
    return nil
}



func init() {
    fmt.Printf( "program %s pid %d init\n", os.Args[0], os.Getpid() )
    reexec.Register( "nsInitialisation", nsInitialisation )
    if reexec.Init() {
        os.Exit(0)
    }
}

func nsInitialisation() {
    fmt.Printf( "\n>> namespace setup code goes here << \n\n" )
    newrootPath := os.Args[1]

    if err := mountProc(newrootPath ); err != nil {
        fmt.Print( "Error mounting /proc - %s\n", err )
        os.Exit(1)
    }

    if err := pivotRoot( newrootPath ); err != nil {
        fmt.Printf( "Error running pivot_root - %s\n", err )
        os.Exit(1)
    }
    nsRun()
}

func nsRun() {
    cmd := exec.Command( "/bin/sh" )
    cmd.Stdin = os.Stdin
    cmd.Stdout = os.Stdout
    cmd.Stderr = os.Stderr

    cmd.Env = []string{"PS1=-[ns-process]- # "}

    if err := cmd.Run(); err != nil {
        fmt.Printf("Error running the /bin/sh command - %s\n", err)
        os.Exit(1)
    }
}


func main() {
    fmt.Printf( "program %s pid %d main\n", os.Args[0], os.Getpid() )
    var rootfsPath string
    flag.StringVar( &rootfsPath, "rootfs", "/tmp/ns-process/rootfs", "Path to the root filesystem to use" )
    flag.Parse() 


    cmd := reexec.Command("nsInitialisation", rootfsPath)

    cmd.Stdin = os.Stdin
    cmd.Stdout = os.Stdout
    cmd.Stderr = os.Stderr

    cmd.Env = []string{"PS1=-[ns-process]- # "}

    cmd.SysProcAttr = &syscall.SysProcAttr{
        Cloneflags: syscall.CLONE_NEWNS |
        syscall.CLONE_NEWUTS |
        syscall.CLONE_NEWIPC |
        syscall.CLONE_NEWPID |
        syscall.CLONE_NEWNET |
        syscall.CLONE_NEWUSER,
        UidMappings: []syscall.SysProcIDMap{
            {
                ContainerID: 0,
                HostID:      os.Getuid(),
                Size:        1,
            },
        },

            {
                ContainerID: 0,

                Size:        1,
            },
        },
    }

    if err := cmd.Run(); err != nil {
        fmt.Printf("Error running the /bin/sh command - %s\n", err)
        os.Exit(1)
    }
}
