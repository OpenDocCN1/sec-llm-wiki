寄存器:

esp

栈寄存器,用来指示栈顶,对整个系统栈而言

ebp

桢指针寄存器,用来表明函数的栈底.只是对于函数本身而言

eax

经常用来寄放函数的返回值

eip

指令寄存器,存放着下一条要执行的指令地址

指令与堆栈变化:

push    ebp

l  esp  = esp -4

l  *esp  = ebp   //C风格赋值

l  表示函数的开始,即函数入口

 



mov     ebp,esp

 

ebp = esp

 

call    dump!FuncB (00401010)

l  esp  = esp-4

l  *esp  = 预期eip  //即下一个指令地址

l  eip  = FuncB  // FuncB的地址赋给eip

 

xor     eax,eax

eax = 0

 

pop     ebp

l  ebp  = *esp

l  esp  = esp + 4

 

ret

l  eip  = *esp

l  esp  = esp + 4
