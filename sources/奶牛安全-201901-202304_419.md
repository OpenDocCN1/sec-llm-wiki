#include <set>
void init( std::set<int>& set ){    for ( int i = 0; i < 0x10; i++ )    {        set.insert( i );    }}
int getSum( std::set<int>& set ){    std::set<int>::iterator iter;    int result = 0;
    for ( iter = set.begin(); iter != set.end(); iter++ )    {        result += *iter;    }
    return result;}
int main(){    std::set<int> set;    init( set );
    return getSum( set );}
