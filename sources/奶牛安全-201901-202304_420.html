这一节，看一下list的iterator对象在内存的布局

#include <list> void init( std::list<int>& lst ){    for ( int i = 0; i < 0x10; i++ )    {        lst.push_back( i );    }}
int getSum( std::list<int>& lst ){    std::list<int>::iterator iter;    int result = 0;
    for ( iter = lst.begin(); iter != lst.end(); iter++ )    {        result += *iter;    }
    return result;}
int main(){    std::list<int> lst;    init( lst );
    return getSum( lst );}
