[A,B] = size(test_label);classmlr = zeros(A,B);for i = 1:length(test_data)        tmp=test_data(i,:)*ww'+b;        z0=abs(tmp-tmp0);        z1=abs(tmp-tmp1);        if z0 < z1                classmlr(i,:) = 0;        else                classmlr(i,:) = 1;        endend%训练%多元线性回归————————————————————————
