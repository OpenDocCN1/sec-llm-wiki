import itertools as iteimport sys,os words = sys.argv[1]num = int(sys.argv[2])path = str(os.path)r = ite.product(words,repeat=num) with open('pass.txt','a') as f:    for i in r:        f.write(''.join(i))        f.write(''.join('\n'))print('已生成由'+words+'随机组成的'+str(num)+'位的密码字典！')print('密码字典名称：pass.txt')print('字典路径：',path)
