# coding=utf8# @time:2022/4/21 20:14# Author 浩宇
# 一、时间模块# import time,datetime# 1.python和时间有关的模块有两个：time、datetime# 2.时间戳'''''''''时间戳指的是当前时间到1970年1月1日0时0分0秒(指的是格林威治时间)的时间差（单位是秒）好处:1)使用时间戳保存时间比使用字符串保存时间所占用的内容要少得多2)使用时间戳对时间进行加密更简单1650544031.5202246  ->   包含了年\月\日\时\分\秒\毫秒
'''''''''# 一.时间戳# 1.time()用法  -   获取当前时间# t1 = time.time()# print(t1,type(t1))    # 1650543825.3692148 <class 'float'>  (时间戳)
# 2.locatime用法# 1)localtime()   -   获取当地本地时间# t2 = time.localtime()# print(t2,type(t2))   # time.struct_time(tm_year=2022, tm_mon=4, tm_mday=21, tm_hour=20, tm_min=40, tm_sec=15, tm_wday=3, tm_yday=111, tm_isdst=0)
# 2)localtime(时间戳)    -   将时间戳转换成本地的结构体时间# t3 = time.localtime(0)# print(t3)
# 3.将结构体时间转换成字符串时间'''''''''时间格式字符串 -   包含时间占位字符串%Y  -   年%m  -   月%d  -   日%H  -   时%M  -   分%S  -   秒%a  -   星期（英文单词简写）%A  -   星期（英文单词全拼）%b  -   月份（英文简写）%B  -   月份（英文全拼）'''''''''# strftime(时间格式字符串,结构体时间)# t5 = time.localtime()# s_t5 = time.strftime('%Y年-%m月-%d日',t5)# print(s_t5)# -----------------------------------------------
# 4.sleep（时间）   -   程序暂停时间（单位秒）# time.sleep(2)# print('================')
# 4.datetime'''''''''from datetime import time,date,datetime,timedelta# time - 时分秒# date - 年月日# datetime - 年月日时分秒t1 = date.today()print(t1,t1.year,t1.month,t1.day)   # 2022-04-21 2022 4 21
t2 = datetime.now()print(t2)   # 2022-04-21 22:50:40.328680'''''''''
# 二.hashlib模块import hashlib
# hashlib是python自带的一个专门提供hash加密的模块# 1.hash加密的特点'''''''''1）同一个数据通过同一个加密算法得到的结果是一样的（加密后的结果叫密文或者摘要）2）加密后的结果不可逆3）不通大小的数据通过相同的算法生成的摘要的长度是一样的'''''''''
# 2.应用场景'''''''''1）创建数据不可逆的密文（加密）2）验证数据的完整性和是否被修改'''''''''
# 3.怎么生成摘要'''''''''# 1）根据加密算法创建hash对象hash = hashlib.md5()    # 常见hash算法：md5、sha相关# 2）添加加密对象# hash对象.update（数据）# 数据    -   必须是bytes对象hash.update(bytes('老子天下第一',encoding = 'utf--8'))  # 添加的数据需要转换成二进制# 3)生成摘要（生成密文）# hash对象.hexdigest（）result = hash.hexdigest()print(result)   # 1bcbf481b5a6b208c8e3d8b2765f1146'''''''''
# 生成图片摘要'''''''''hash = hashlib.md5()with open('美女.jpg','rt') as f:    hash.update(f.read())    print(hash.hexdigest())'''''''''
# 生成文本文件的摘要'''''''''hash = hashlib.md5()with open('01-异常捕获.py','rb') as f:    hash.update(f.read())    print(hash.hexdigest())'''''''''

# 方法二：字符串.encode()b2=str1.encode()print(b1,type(b1))
# 2)二进制转字符串# 方法一：a.str(二进制,encoding='utf-8')s1=str(b1.encoding='utf-8')print(s1,type(s1))
