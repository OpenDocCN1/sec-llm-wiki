一、部署loki日志中心

下载二进制包cd /home/starunion/softwget https://github.com/grafana/loki/releases/download/v2.2.0/loki-linux-amd64.zipunzip loki-linux-amd64.zip -d loki

mkdir -p /home/starunion/apps/lokicp -afr /home/starunion/soft/loki/loki-linux-amd64 /home/starunion/apps/loki/loki-linux
修改配置文件vim config.yaml

auth_enabled: false
server:  http_listen_port: 3100  grpc_listen_port: 39095 #grpc监听端口，默认为9095  grpc_server_max_recv_msg_size: 15728640  #grpc最大接收消息值，默认4m  grpc_server_max_send_msg_size: 15728640  #grpc最大发送消息值，默认4m
ingester:  lifecycler:    address: 172.16.10.45 #IP地址    ring:      kvstore:        store: inmemory      replication_factor: 1    final_sleep: 0s  chunk_idle_period: 5m  chunk_retain_period: 30s  max_transfer_retries: 0  max_chunk_age: 20m  #一个timeseries块在内存中的最大持续时间。如果timeseries运行的时间超过此时间，则当前块将刷新到存储并创建一个新块
schema_config:  configs:    - from: 2018-04-15      store: boltdb      object_store: filesystem      schema: v11      index:        prefix: index_        period: 168h
storage_config:  boltdb:    directory: /tmp/loki_data/index
  filesystem:    directory: /tmp/loki_data/chunks
limits_config:  enforce_metric_name: false  reject_old_samples: true  reject_old_samples_max_age: 168h  ingestion_rate_mb: 30  #修改每用户摄入速率限制，即每秒样本量，默认值为4M  ingestion_burst_size_mb: 15  #修改每用户摄入速率限制，即每秒样本量，默认值为6M
chunk_store_config:  max_look_back_period: 168h   #回看日志行的最大时间，只适用于即时日志
table_manager:  retention_deletes_enabled: true #日志保留周期开关，默认为false  retention_period: 168h  #日志保留周期
创建systemd服务启动lokivim /etc/systemd/system/loki.service
[Unit]Description=lokiDocumentation=https://grafana.com/oss/loki/After=network.target
[Service]User=starunionGroup=starunionType=simpleExecStart=/home/starunion/apps/loki/loki-linux --config.file=/home/starunion/apps/loki/config.yaml Restart=on-failure
[Install]WantedBy=multi-user.target
启动systemctl start lokisystemctl status lokisystemctl enable loki

二、部署promtail日志采集组件

下载二进制包cd /home/starunion/softwget https://github.com/grafana/loki/releases/download/v2.2.0/promtail-linux-amd64.zipunzip promtail-linux-amd64.zip -d loki-promtail
mkdir -p /home/starunion/apps/loki-promtailcp -afr /home/starunion/soft/loki-promtail/loki-promtail-linux-amd64 /home/starunion/apps/loki-promtail/loki-promtail-linux
修改配置文件vim config.yaml

server:  http_listen_port: 9080  grpc_listen_port: 0  grpc_server_max_recv_msg_size: 15728640  grpc_server_max_send_msg_size: 15728640
positions:  filename: /tmp/positions.yaml # This location needs to be writeable by promtail.
client:  url: http://loki-server-ip:3100/loki/api/v1/push
scrape_configs: - job_name: gos-game-local-release   pipeline_stages:   - multiline:       firstline: '^\d{4}-\d{2}-\d{2}[a-zA-Z]\d{1,2}:\d{2}:\d{2}\.\d{3}[a-zA-Z]'       max_wait_time: 3s   static_configs:   - targets:      - 172.16.10.46     labels:       logs_data_source: p102       job: gos-game-local-release       __path__: /home/starunion/apps/gos-game-local-release/bin/logs/*/*.log

 - job_name: gos-game-local-cn   pipeline_stages:   - multiline:       firstline: '^\d{4}-\d{2}-\d{2}[a-zA-Z]\d{1,2}:\d{2}:\d{2}\.\d{3}[a-zA-Z]'       max_wait_time: 3s   static_configs:   - targets:      - 172.16.10.46     labels:       logs_data_source: p102       job: gos-game-local-release       __path__: /home/starunion/apps/gos-game-local-cn/bin/logs/*.log
创建systemd服务启动loki-pormtailvim /etc/systemd/system/promtail.service
[Unit]Description=lokiDocumentation=https://grafana.com/oss/loki/After=network.target
[Service]User=starunionGroup=starunionType=simpleExecStart=/home/starunion/apps/loki-promtail/loki-promtail-linux --config.file=/home/starunion/apps/loki-promtail/config.yaml Restart=on-failure
[Install]WantedBy=multi-user.target
启动systemctl start promtailsystemctl status promtailsystemctl enable promtail
