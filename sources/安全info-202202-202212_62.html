tar zxvf kibana-7.7.0-linux-x86_64.tar.gzmv kibana-7.7.0-linux-x86_64 kibanacd kibanacp config/kibana.yml config/kibana.yml.bak.exampleecho '' > config/kibana.ymlcat >> config/kibana.yml << eofserver.port: 5601server.host: "0.0.0.0"server.name: "starunion"elasticsearch.hosts: ["http://127.0.0.1:9200", "http://127.0.0.1:9201","http://127.0.0.1:9202"]kibana.index: ".kibana"kibana.defaultAppId: "home"i18n.locale: "zh-CN"eof
