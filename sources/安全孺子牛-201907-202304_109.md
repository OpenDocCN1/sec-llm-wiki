```

```
#!/bin/bashtime=`more files.log | zeek-cut ts`thost=`more files.log | zeek-cut tx_hosts`rhost=`more files.log | zeek-cut rx_hosts`protocol=`more files.log | zeek-cut source`filetype=`more files.log | zeek-cut mime_type`filename=`more files.log | zeek-cut extracted`echo "时间：$time"echo "服务器：$thost"echo "客户端：$rhost"echo "协议：$protocol"echo "文件类型：$filetype"echo "文件名称：$filename"
