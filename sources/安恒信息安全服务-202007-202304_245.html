#Skytrain Inc Ticket Validation System 0.1#Do not distribute this file.

def load_file(loc):\\判断文件是否为md后缀    if loc.endswith(".md"):        return open(loc, 'r')    else:        print("Wrong file type.")        exit()

def evaluate(ticketFile):    #Evaluates a ticket to check for ireggularities.    code_line = None    for i,x in enumerate(ticketFile.readlines()):        if i == 0:\\判断文件第一行是否为# Skytrain Inc            if not x.startswith("# Skytrain Inc"):                return False            continue        if i == 1:\\判断文件第二行是否为## Ticket to            if not x.startswith("## Ticket to "):                return False            print(f"Destination: {' '.join(x.strip().split(' ')[3:])}")            continue

        if x.startswith("__Ticket Code:__"):\\判断_Ticket Code:_在第几行            code_line = i+1            continue

        if code_line and i == code_line:            if not x.startswith("**"):\\判断第四行开头是否有**                return False            ticketCode = x.replace("**", "").split("+")[0]\\取这一行**和+中间的数字            if int(ticketCode) % 7 == 4:\\ticketCode和7取余等于4                validationNumber = eval(x.replace("**", ""))\\用eval()函数执行**后面的命令                if validationNumber > 100:                    return True                else:                    return False    return False

def main():    fileName = input("Please enter the path to the ticket file.\n")    ticket = load_file(fileName)    #DEBUG print(ticket)    result = evaluate(ticket)    if (result):\\判断result是否为ture        print("Valid ticket.")    else:        print("Invalid ticket.")ticket.close

四 加固建议


4.1 XML外部实体注入


（1）使用开发语言提供的禁用外部实体的方法

PHP：

libxml_disable_entity_loader(true);

（左右滑动查看更多）

JAVA:

DocumentBuilderFactory dbf =DocumentBuilderFactory.newInstance();dbf.setExpandEntityReferences(false);

（左右滑动查看更多）

Python：

from lxml import etreexmlData = etree.parse(xmlSource,etree.XMLParser(resolve_entities=False))

（左右滑动查看更多）

（2）过滤用户提交的XML数据

关键词：<!DOCTYPE和<!ENTITY，或者，SYSTEM和PUBLIC。


4.2 代码执行


（1）严格检查程序参数，特别是 “&”, “&&”, “|”, “||”，“eval”, “execute” 这类参数。

（2）禁止把通过外部传入的参数传入到该类可执行函数的参数中。

（3）如果使用的是第三方建站程序，务必升级到最新版本。


总结


针对XXE漏洞是XML在允许引用外部实体时，通过构造恶意内容，可导致读取任意文件、执行系统命令、探测内网端口、攻击内网网站等危害，所以开发者在编写代码时禁用外部实体是非常有效的防止XML外部实体注入的方式。


同时代码执行漏洞的本质是危险函数eval、asset等函数中的参数为用户可控导致。如果开发者在编写代码时将外部参数传入到这类可执行函数中，那么攻击者就有机可乘，所以大家在编写代码时候一定要主要这类情况哦！


当然各种协议的暴力破解也是攻击者常用的入侵手段，平时可以将一些默认端口改为其他不常用的端口；设置安全策略，登陆失败达到一定次数就锁定当前IP一段时间；设置登陆安全组，避免普通用户以root权限登陆。最后提高密码复杂度也能有效避免系统被攻击者入侵哦！
