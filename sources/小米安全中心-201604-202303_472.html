[ -f /scripts/utils.sh ] && . /scripts/utils.sh

urldecode() {
    local url_encoded="${1//+/ }"
    printf '%b' "${url_encoded//%/\\x}"
}

oIFS=$IFS
IFS=$'\n';
for i in `env`;do
    eval "${i%%=*}=\`urldecode \$${i%%=*}\`";
done

正确答案




bash变量里的内容在使用时不会被解析诸如`或"等控制符号。但是会以空格分隔为独立的调用参数。利用curl的-o参数可以任意写文件，同时利用--resolve参数更改解析可以控制写入内容，这样可以利用一些写权限变可执行的方法来实现命令执行。poc中是通过覆盖脚本中导入的脚本来执行命令。
