static main()
{
    auto addr = 0x400AA6;
    auto addr_key = 0x41E1B0;
    auto a,i = 0;
    auto str = "F1@gChe";
    for(i=0;i<=0x104;i++)
    {   
        a = i%7;
        PatchByte(addr+i,Byte(addr+i)^ord(str[a:a+1]));

    }
}
