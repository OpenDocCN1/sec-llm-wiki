  $this->method = $_SERVER['REQUEST_METHOD'] == 'GET' ? 'get' : 'post';

 private function get_dv()
 {

 {
  $sbin = "";
  $len = strlen($str);
  for ($i = 0; $i < $len; $i += 2) {
   $sbin .= pack("H*", substr($str, $i, 2));

 }
