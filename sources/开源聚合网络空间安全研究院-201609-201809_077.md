#!/usr/bin/env python

import sys

import json

from Crypto.Cipher import AES

from Crypto import Random

  

def get\_padding(rawstr):

    remainder = len(rawstr) % 16

    if remainder != 0:

        return '\\x00' \* (16 - remainder)

    return ''

  

def aes\_encrypt(key, plaintext):

    plaintext += get\_padding(plaintext)

    aes = AES.new(key, AES.MODE\_ECB)

    cipher\_text = aes.encrypt(plaintext).encode('hex')

    return cipher\_text

  

def generate\_hello(key, name, flag):

    message = "Connection for mission: {}, your mission's flag is: {}".format(name, flag)

    return aes\_encrypt(key, message)

    return raw\_input()

    sys.stdout.flush()

print\_output("Please enter your Agent ID to secure communications:")

**分析：**  

分析代码发现这是个AES ECB的加密通信，mission key是防作弊用的，而agent id是我们可控的，有可能切入点就在这里。

先了解下**AES ECB加密原理**（别喷哈，我也是先学现卖）

电码本模式（Electronic Codebook Book (ECB)

这种模式是将整个明文分成若干段相同的小段，然后对每一小段进行加密，每组16位。具体的可以百度或者谷歌。

这段代码是利用key（key是随机的）对message进行加密，而message中的name（也就是agent id）是我们可控的，每16位一组，不够16位用'\\x00'补齐，我们可以通过输入agent id来控制flag的长度（根据上边的flag可以猜想这道题的flag很有可能是32位的16进制数）。传入agent id之后，系统会返回一串对message加密后的密文，我们要做的是通过控制agent id长度来控制包含flag的那组的长度（前15位为我们已知的，最后一位我们可以爆破），具体操作如下：

原来的message：

message = “Connection for mission: {}, your mission’s flag is: {}”.format(name, flag)
