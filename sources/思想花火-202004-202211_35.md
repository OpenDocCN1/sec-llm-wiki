**1、功能需求**

需求很简单：  

1\. 将需要检测的URL写入一个文本文件，每个URL一行

2\. 对每一个URL进行探测

3\. 如果探测到的站点包含“美团”或者“meituan"关键字，则打印“发现钓鱼网站：URL”

```
package main
/*1. 将需要检测的URL写入一个文本文件，每个URL一行2. 对每一个URL进行探测3. 如果探测到的URL包含“美团”或者“meituan"关键字，则打印”发现钓鱼网站：URL“*/
import (  "bufio"  "fmt"  "io/ioutil"  "net/http"  "os"  "regexp"  "strings")//读取网站bodyfunc detect(url string) string {  response, err := http.Get(url)  if err != nil {    fmt.Println(err)  }  body, _ := ioutil.ReadAll(response.Body)  defer response.Body.Close()  return string(body)}//读取含有URL文件，并输出func Readfile(filename string) []string {  file, err := os.Open(filename)  if err != nil {    fmt.Println("File is broken")  }  defer file.Close()  data := make([]string, 0)  reader := bufio.NewReader(file)  for {    linestr, err := reader.ReadString('\n')    if err != nil {      break    }    linestr = strings.TrimSpace(linestr)    if linestr == "" {      continue    }    data = append(data, linestr)  }  return data}//包含关键词1func IsKeyword1(word string) (b bool) {  pattern := "meituan"  result, _ := regexp.MatchString(pattern, word)  return result}//包含关键词2func IsKeyword2(word string) (b bool) {  pattern := "美团"  result, _ := regexp.MatchString(pattern, word)  return result}
func main() {  filename := os.Args[1]  url := Readfile(filename)  for _, k := range url { //string数组的值    site := detect(k)    if IsKeyword1(site) || IsKeyword2(site) {      fmt.Printf("发现钓鱼网站：%s \n", k)      continue    }  }}
```
