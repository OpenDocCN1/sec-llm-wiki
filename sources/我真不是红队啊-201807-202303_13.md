```
9.  `                config.dialect.forModelSave(table, attrs, sql, paras);`
    
10.  `                // if (paras.size() == 0)        return false;        // The sql "insert into tableName() values()" works fine, so delete this line`
     
11.  
     
12.  `                // --------`
     
13.  `                Connection conn = null;`
     
14.  `                PreparedStatement pst = null;`
     
15.  `                int result = 0;`
     
16.  `                try {`
     
17.  `                        conn = config.getConnection();`
     
18.  `                        if (config.dialect.isOracle()) {`
     
19.  `                                pst = conn.prepareStatement(sql.toString(), table.getPrimaryKey());`
     
20.  `                        } else {`
     
21.  `                                pst = conn.prepareStatement(sql.toString(), Statement.RETURN_GENERATED_KEYS);`
     
22.  `                        }`
     
23.  `                        config.dialect.fillStatement(pst, paras);`
     
24.  `                        result = pst.executeUpdate();`
     
25.  `                        config.dialect.getModelGeneratedKey(this, pst, table);`
     
26.  `                        _getModifyFlag().clear();`
     
27.  `                        return result >= 1;`
     
28.  `                } catch (Exception e) {`
     
29.  `                        throw new ActiveRecordException(e);`
     
30.  `                } finally {`

32.  `                }`

```
