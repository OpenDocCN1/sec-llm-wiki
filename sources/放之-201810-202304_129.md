golang提供了 `plugin`包，首先通过将代码编译为共享库( `go build-buildmode=plugin code.go-o lib.so`)的形式，然后由 `plugin`包进行调用，以实现插件模式。  

plugin的使用方式：

1.    
    
    在插件中设置 `Symbol`(A Symbol is a pointer to a variable or function)
    
      
    
2.    
    
    然后通过 `plugin.Open("xxx.so")` 打开这个库，最后采用 `plugin.Lookup("Symbolname")`即可。
    
      
    

示例:

*   greeter.go
    

```
1.  `package main`
    
2.  
    
3.  `import (`
    
4.  `    "fmt"`
    
5.  `    "os"`
    
6.  `    "plugin"`
    
7.  `)`
    
8.  
    
9.  `type Greeter interface {`
    
10.  `    Greet()`
     
11.  `}`
     
12.  
     
13.  `func main() {`
     
14.  `    lang := "english"`
     
15.  `    if len(os.Args) == 2 {`
     
16.  `        lang = os.Args[1]`
     
17.  `    }`
     
18.  
     
19.  `    var mod string`
     
20.  `    switch lang {`
     
21.  `    case "english":`
     
22.  `        mod = "./eng/eng.so"`
     
23.  `    case "chinese":`
     
24.  `        mod = "./chi/chi.so"`
     
25.  `    default:`
     
26.  `        fmt.Println("Don't speak that language")`
     
27.  `    }`
     
28.  
     
29.  `    plug, err := plugin.Open(mod)`
     
30.  `    if err != nil {`
     
31.  `        fmt.Println(err)`
     
32.  `        os.Exit(1)`
     
33.  `    }`
     
34.  
     
35.  `    symGreeter, err := plug.Lookup("Greeter")`
     
36.  `    if err != nil {`
     
37.  `        fmt.Println(err)`
     
38.  `        os.Exit(1)`
     
39.  `    }`
     
40.  
     
41.  `    var greeter Greeter`
     
42.  `    greeter, ok := symGreeter.(Greeter)`
     
43.  
     
44.  `    if !ok {`
     
45.  `        fmt.Println("UnExpected type from module symbol")`
     
46.  `        os.Exit(1)`
     
47.  
     
48.  `    }`
     
49.  
     
50.  `    greeter.Greet()`
     
51.  
     
52.  `}`
     
```

*   chi/greeter.go
    

```
1.  `package main`
    
2.  
    
3.  `import "fmt"`
    
4.  
    
5.  `type greeting string`
    
6.  
    
7.  `func (g greeting) Greet() {`
    
8.  `    fmt.Println("你真帅")`
    
9.  `}`
    
10.  
     
11.  `var Greeter greeting`
     
```
