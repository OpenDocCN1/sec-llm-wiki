# 制作微信好友头像拼图

利用 `ItChat`库，通过获得微信好友头像并将其做成拼图。 同时将其发送到文件传输助手。

```
1.  `import os`
    
2.  `import math`
    
3.  `import itchat`
    
4.  `import PIL.Image as Image`
    
5.  
    
6.  `x = 0`
    
7.  `y = 0`
    
8.  `num = 0`
    
9.  
    
10.  `itchat.auto_login(enableCmdQR=-1, hotReload=True)`
     
11.  
     
12.  `friends = itchat.get_friends(update=True)[0:]`
     
13.  `user = friends[0]["UserName"]`
     
14.  
     
15.  `if not os.path.isdir(user):`
     
16.  `    os.mkdir(user)`
     
17.  
     
18.  `for friend in friends:`
     
19.  `    img = itchat.get_head_img(userName=friend["UserName"])`
     
20.  `    with open("{}/{}.jpg".format(user,friend["UserName"]), 'wb') as icon:`
     
21.  `        icon.write(img)`
     
22.  
     
23.  `pics = os.listdir(user)`
     
24.  
     
25.  `eachsize = int(math.sqrt(float(640 * 640) / len(pics)))`
     
26.  `numline = int(640 / eachsize)`
     
27.  
     
28.  `toImage = Image.new('RGBA', (640, 640))`
     
29.  
     
30.  `for pic in pics:`
     
31.  `    try:`
     
32.  `        img = Image.open(os.path.join(user,pic))`
     
33.  `    except Exception as e:`
     
34.  `        continue`
     
35.  
     
36.  `    img = img.resize((eachsize, eachsize), Image.ANTIALIAS)`
     
37.  `    toImage.paste(img, (x * eachsize, y * eachsize))`
     
38.  
     
39.  `    x += 1`
     
40.  `    if x == numline:`
     
41.  `        x = 0`
     
42.  `        y += 1`
     
43.  
     
44.  `fimage = "{}.jpg".format(user)`
     
45.  `toImage = toImage.convert("RGB")`
     
46.  `toImage.save(fimage)`
     
47.  
     
48.  `itchat.send_image(fimage, 'filehelper')`
     
```
