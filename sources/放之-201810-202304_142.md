```
1.  `import uuid`
    
2.  `from telethon import TelegramClient, events, sync`
    
3.  `from telethon.tl.types import PeerUser, PeerChat, PeerChannel`
    
4.  
    
5.  `api_id=1234567`
    
6.  `api_hash='thisi is fake hash'`
    
7.  
    
8.  `client = TelegramClient('session_name', api_id, api_hash)`
    
9.  `client.start()`
    
10.  
     
11.  `titswiki = client.get_entity('titswiki')`
     
12.  
     
13.  `c = client.get_entity(PeerChannel(titswiki.id))`
     
14.  `for m in client.iter_messages(c):`
     
15.  `        print(m.download_media("image/"+str(uuid.uuid1())))`
     
```
