unit uD4GoHttp;


interface


uses

  System.SysUtils;


function Get(url: PAnsiChar; parameter: PAnsiChar): PAnsiChar; cdecl; external 'd4gohttp.dll';


function Post(url: PAnsiChar; parameter: PAnsiChar): PAnsiChar; cdecl; external 'd4gohttp.dll';


type

  TD4GoHttp = class

  public

    /// <param name="url">网址,要加http://或者https://</param>

    /// <param name="parameter">参数不需要"?"号</param>

    /// <returns>网站或服务器接口返回值,字符串</returns>

    function GGet(url: string; parameter: string): string;


    /// <param name="url">网址,要加http://或者https://</param>

    /// <param name="parameter">如果是json请用json字符串</param>

    /// <returns>网站或服务器接口返回值,字符串</returns>

    function GPost(url: string; parameter: string): string;

  end;


implementation


uses

  HTTPApp, system.classes, EncdDecd, System.NetEncoding;


{ TD4GoHttp }


function TD4GoHttp.GGet(url, parameter: string): string;

var

  res: PAnsiChar;

begin

  res := Get(PAnsiChar(AnsiString(EncodeString(url))), PAnsiChar(AnsiString(EncodeString(parameter))));

  Result := DecodeString(string(res));

  Exit;

end;


function TD4GoHttp.GPost(url, parameter: string): string;

var

  res: PAnsiChar;

begin

  res := Post(PAnsiChar(AnsiString(EncodeString(url))), PAnsiChar(AnsiString(EncodeString(parameter))));

  Result := DecodeString(string(res));

  Exit;

end;


end.

package main


import "C"

func main() {

}


//export Get

func Get(url, parameter *C.char) *C.char {

var (

gUrl,gParameter []byte

res string

cRes *C.char

)

gUrl,_=gbase64.DecodeString(C.GoString(url))

gParameter, _ = gbase64.DecodeString(C.GoString(parameter))

v, e := http.HttpGet(string(gUrl), string(gParameter))

if e != nil {

res = e.Error()

} else {

res = v

}

cRes = C.CString(gbase64.EncodeString([]byte(res)))

}

package http


import "github.com/gogf/gf/net/ghttp"


// HttpPost post请求

func HttpPost(url,parameter string) (string,error) {

var (

urls string

e error

res string

)

if parameter == "" {

urls = url

} else  {

urls = url + parameter

}

if r, err := ghttp.Post(urls); e != nil {

e = err

} else {

defer r.Close()

res = r.ReadAllString()

}

return res,e

}

package http


import (

"github.com/gogf/gf/net/ghttp"

)


// HttpGet get请求

func HttpGet(url,parameter string) (string,error) {

var (

urls string

e error

res string

)

if parameter == "" {

urls = url

} else  {

urls = url + "?" + parameter

}

if r, err := ghttp.Get(urls); e != nil {

e = err

} else {

defer r.Close()

res = r.ReadAllString()

}

return res,e

}
