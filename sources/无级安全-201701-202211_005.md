#### 获取key

使用工具 获取微信key  
![](../Images/7b0a892efedcc08a65c6f4275f8e8953.png "null")  
获取key后通过脚本进行解密

  
**Python3**

```
1.  `from Crypto.Cipher import AES`
    
2.  `import hashlib, hmac, ctypes, sys, getopt`
    
3.  
    
4.  `SQLITE_FILE_HEADER = bytes('SQLite format 3', encoding='ASCII') + bytes(1)`
    
5.  `IV_SIZE = 16`
    
6.  `HMAC_SHA1_SIZE = 20`
    
7.  `KEY_SIZE = 32`
    
8.  `DEFAULT_PAGESIZE = 4096`
    
9.  `DEFAULT_ITER = 64000`
    
10.  `opts, args = getopt.getopt(sys.argv[1:], 'hk:d:')`
     
11.  `input_pass = ''`
     
12.  `input_dir = ''`
     
13.  
     
14.  `for op, value in opts:`
     
15.  `    if op == '-k':`
     
16.  `        input_pass = value`
     
17.  `    else:`
     
18.  `        if op == '-d':`
     
19.  `            input_dir = value`
     
20.  
     
21.  `password = bytes.fromhex(input_pass.replace(' ', ''))`
     
22.  
     
23.  `with open(input_dir, 'rb') as (f):`
     
24.  `    blist = f.read()`
     
25.  `print(len(blist))`
     
26.  `salt = blist[:16]`
     
27.  `key = hashlib.pbkdf2_hmac('sha1', password, salt, DEFAULT_ITER, KEY_SIZE)`
     
28.  `first = blist[16:DEFAULT_PAGESIZE]`
     
29.  `mac_salt = bytes([x ^ 58 for x in salt])`
     
30.  `mac_key = hashlib.pbkdf2_hmac('sha1', key, mac_salt, 2, KEY_SIZE)`
     
31.  `hash_mac = hmac.new(mac_key, digestmod='sha1')`
     
32.  `hash_mac.update(first[:-32])`
     
33.  `hash_mac.update(bytes(ctypes.c_int(1)))`
     
34.  
     
35.  `if hash_mac.digest() == first[-32:-12]:`
     
36.  `    print('Decryption Success')`
     
37.  `else:`
     
38.  `    print('Password Error')`
     
39.  `blist = [blist[i:i + DEFAULT_PAGESIZE] for i in range(DEFAULT_PAGESIZE, len(blist), DEFAULT_PAGESIZE)]`
     
40.  
     
41.  `with open(input_dir, 'wb') as (f):`
     
42.  `    f.write(SQLITE_FILE_HEADER)`
     
43.  `    t = AES.new(key, AES.MODE_CBC, first[-48:-32])`
     
44.  `    f.write(t.decrypt(first[:-48]))`
     
45.  `    f.write(first[-48:])`
     
46.  `    for i in blist:`
     
47.  `        t = AES.new(key, AES.MODE_CBC, i[-48:-32])`
     
48.  `        f.write(t.decrypt(i[:-48]))`
     
49.  `        f.write(i[-48:])`
     
```

运行脚本安装pip包时，一直安装失败。出现了一些问题，一番探查 发现原因是pip包冲突了  
解决方法:  
![](../Images/59c86b50fcaba5cc9d77d50e69a16454.png "null")  
卸载crypto

```
1.  `pip3 uninstall crypto`
    
```

卸载

```
1.  `pip3 uninstall pycrypto`
    
```

重新安装

```
1.  `pip3 install crypto`
    
```

此时就可以运行起来
