def rc4(data,key):

        j=0

        s=range(256)

        for i in range(256):

                j=(j+s[i]+ord(key[i%len(key)]))%256

                s[i],s[j]=s[j],s[i]

        i=0

        j=0

        out=[]

        for char in data:

                i=(i+1)%256

                j=(j+s[i])%256

                s[i],s[j]=s[j],s[i]

                out.append(chr(ord(char)^s[(s[i]+s[j])%256]))

        return ''.join(out)
