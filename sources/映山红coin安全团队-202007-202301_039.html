IDA脱壳脚本




static main(void){auto fp, dexAddress, end, size;dexAddress = 0xA644C008;   //起始地址  size = 0x0086CAB0;               //结束地址=起始地址+文件大小  end = dexAddress + size;      //打开文件  fp = fopen("D:\\classes.dex", "wb");         //循环写入  for ( ; dexAddress < end; dexAddress++ )fputc(Byte(dexAddress), fp);}
