  1. 下载附件，发现是一个vmdk硬盘，那么使用vmware导入该硬盘（注意版本，低版本vmware可能无法正确导入）。

  2. 开启win10虚拟机，观察发现多了一块盘符。有一个锁的形状，确认为bitlocker加密盘。


3. 使用计算机管理查看磁盘管理，发现存在多余的空间没有分配，猜测可能隐藏有解密盘的关键信息。


4. 查找资料或者亲自实践，我们可以得知bitlocker在加密过程中可能会生成一个密钥txt文件，猜测可能会存放于未分配的空间中，由于生成密钥是一个55位的数字加下划线的组合，所以我们可以查找一下该硬盘的内容。

5. 找到解密密钥之后，就可以看到硬盘的内容了，存在两个文件，一个exe，一个excel。先运行exe，发现有虚拟机检测，那应该是vmp加壳了。


6. 使用ida查看，发现确实是vmp壳，所以就放弃了调试器分析程序的方法。



7. 既然是chat，那肯定有网络通讯请求，所以可以尝试使用wireshark分析程序网络行为。简单捕获后，发现有一个明显的dns请求。



8. 我们可以猜想这个域名可能是一个聊天服务，所以尝试将该域名重定向到本地虚拟机，测试一下。写一个简单的flask

from flask import Flask
app = Flask(__name__)
app.run(host='0.0.0.0',port=80)


9. 发现并没有请求这个flask服务，于是继续抓包（或者在虚拟机中使用tcpdump抓包，推荐此方法），发现该程序走的端口是5000。


10. 修改端口，继续观察，发现有请求了，也有错误代码。


11. 查找错误代码excel发现，不存在这个代码，所以继续抓包，观察程序到底发送了什么数据。


12. 发现是直接做了一个http的post请求，请求的位置是根目录，数据为json格式，内容是 {"username": "admin", "password": "******"}。我们尝试修改一下返回的数据，比如随便从错误代码的excel中选择一个试试。发现错误代码变成了199。

     * Create a connection to the FastCGI application

    private function connect()

            $res = socket_connect($this->_sock,$this->_host, $this->_port);

    /**
     * Build a FastCGI packet
     *
     * @param Integer $type Type of the packet
     * @param String $content Content of the packet
     * @param Integer $requestId RequestId
     */
    private function buildPacket($type, $content, $requestId = 1)
    {
        $clen = strlen($content);
        return chr(self::VERSION_1)         /* version */
            . chr($type)                    /* type */
            . chr(($requestId >> 8) & 0xFF) /* requestIdB1 */
            . chr($requestId & 0xFF)        /* requestIdB0 */

            . chr(0)                        /* paddingLength */
            . chr(0)                        /* reserved */

    /**
     * Build an FastCGI Name value pair

        if ($vlen < 128) {
            /* valueLengthB0 */
            $nvpair .= chr($vlen);
        } else {
            /* valueLengthB3 & valueLengthB2 & valueLengthB1 & valueLengthB0 */
            $nvpair .= chr(($vlen >> 24) | 0x80) . chr(($vlen >> 16) & 0xFF) . chr(($vlen >> 8) & 0xFF) . chr($vlen & 0xFF);
        }
        /* nameData & valueData */
        return $nvpair . $name . $value;

    /**
     * Read a set of FastCGI Name value pairs

     * @param String $data Data containing the set of FastCGI NVPair
     * @return array of NVPair

    private function readNvpair($data, $length = null)

        $array = array();

            $length = strlen($data);

        while ($p != $length) {

            if ($nlen >= 128) {
                $nlen = ($nlen & 0x7F << 24);
                $nlen |= (ord($data{$p++}) << 16);
                $nlen |= (ord($data{$p++}) << 8);
                $nlen |= (ord($data{$p++}));

            $vlen = ord($data{$p++});

    private function decodePacketHeader($data)

        }
        if ($paramsRequest) {
            $request .= $this->buildPacket(self::PARAMS, $paramsRequest);
        }
        $request .= $this->buildPacket(self::PARAMS, '');

        if ($stdin) {
            $request .= $this->buildPacket(self::STDIN, $stdin);
        }

        socket_write($this->_sock, $request);

        do {
            $resp = $this->readPacket();
            if ($resp['type'] == self::STDOUT || $resp['type'] == self::STDERR) {
                $response .= $resp['content'];
            }
        } while ($resp && $resp['type'] != self::END_REQUEST);

        if (!is_array($resp)) {
            throw new Exception('Bad request');
        }

        switch (ord($resp['content']{4})) {
            case self::CANT_MPX_CONN:
                throw new Exception('This app can\'t multiplex [CANT_MPX_CONN]');

            case self::OVERLOADED:

                break;
            case self::UNKNOWN_ROLE:
                throw new Exception('Role value not known [UNKNOWN_ROLE]');
                break;
            case self::REQUEST_COMPLETE:
                return $response;
        }
    }
}
