#include<stdio.h>#define N 20
struct pcbs{  int pno;//定义编号  int arrtime;//到达时间  int runtime;//运行时间  int starttime;//开始时间  int roundtime;//周转时间  int endtime;//结束时间    };

int inputp(pcbs p[],int n) //子函数调用，通过进程数的数量初始化要输入的内容 {  for(int i=0;i<n;i++)  {    printf("请输入作业的编号，到达时间和执行时间：");    scanf("%d%d%d",&p[i].pno,&p[i].arrtime,&p[i].runtime);  }  return 0;}

//按照到达时间使用冒泡排序，从小到大排序 int sort_pop(pcbs p[],int n){  pcbs m;  for(int i=0;i<n-1;i++)//外循环控制趟数，n个数为n-1趟    for(int k=0;k<n-1-i;k++)//内循环为每趟比较的次数，第i趟比较n-i次      if(p[k].arrtime>p[k+1].arrtime)//若本次到达时间大于下次到达时间则进行交换（升序排列左大于右）         {          m=p[k];          p[k]=p[k+1];          p[k+1]=m;         }          return 0;} 

float f=0;//fcfs算法 int fcfs(pcbs p[],int n){
  sort_pop(p,n);//先执行排序  p[0].starttime=p[0].arrtime;//第一个作业的开始时间等于到达时间  p[0].endtime=p[0].starttime+p[0].runtime;//第一个作业的结束时间等于第一个作业的开始时间+运行时间   p[0].roundtime=p[0].endtime-p[0].starttime;//第一个作业的周转时间等于第一个作业的结束时间减去到达时间  f+=p[0].roundtime;//周转时间相加   for(int i=1;i<n;i++)  {    if(p[i].arrtime>p[i-1].endtime)//假如本次到达时间大于上一个结束时间， 本次开始时间等于本次到达时间，否则本次开始时间等于上一个结束时间       p[i].starttime=p[i].arrtime;    else      p[i].starttime=p[i-1].endtime;          p[i].endtime=p[i].starttime+p[i].runtime;//结束时间等于开始时间+运行时间     p[i].roundtime=p[i].endtime-p[i].arrtime;//周转时间等于结束时间减去到达时间    f+=p[i].roundtime;  }
  return 0; } 

int printpcb(pcbs p[],int n){    printf("执行结果如下：\n");  printf("编号\t到达\t运行\t开始\t结束\t周转\n");  for(int i=0;i<n;i++)  {      printf("%d\t%d\t%d\t%d\t%d\t%d\n",p[i].pno,p[i].arrtime,p[i].runtime,p[i].starttime,p[i].endtime,p[i].roundtime);  }  printf("平均周转时间为:%f\n",f/n);  return 0;}



int main(){  int n;  pcbs p[N]={0};  printf("请输入进程数量：");  scanf("%d",&n);  inputp(p,n);//输入内容进行初始化，函数调用   fcfs(p,n); //执行先来先服务算法   printpcb(p,n);//输出结果   return 0; }
