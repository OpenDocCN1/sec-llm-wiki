配置步骤

(1) 配置 MSR36-20_1

# 配置各接口的 IP 地址，具体略。

# 配置一个 ACL，定义要保护由子网 10.1.1.0/24 去往子网 10.1.2.0/24的数据流。

<H3C>system-view

[H3C] aclnumber 3101

[H3C-acl-adv-3101]rule permit ip source 10.1.1.0 0.0.0.255 destination 10.1.2.0 0.0.0.255

[H3C-acl-adv-3101] quit

# 创建 IPsec 安全提议 lzc。

[H3C] ipsec transform-set lzc

# 配置安全协议对 IP 报文的封装形式为隧道模式。

[H3C-ipsec-transform-set-lzc]encapsulation-mode tunnel

# 配置采用的安全协议为 ESP。

[H3C-ipsec-transform-set-lzc] protocolesp

# 配置 ESP 协议采用的加密算法为采用 128比特的 AES，认证算法为 HMAC-SHA1。

[H3C-ipsec-transform-set-lzc] esp encryption-algorithm aes-cbc-128

[H3C-ipsec-transform-set-lzc] esp authentication-algorithm sha1

[H3C-ipsec-transform-set-lzc] quit

# 创建一条手工方式的 IPsec安全策略，名称为 LZC，序列号为 10。

[H3C] ipsec policy lzc110 manual

# 指定引用 ACL 3101。

[H3C-ipsec-policy-manual- lzc1-10]security acl 3101

# 指定引用的 IPsec 安全提议为 lzc1。

[H3C-ipsec-policy-manual- lzc1-10]transform-set lzc

# 指定 IPsec 隧道对端 IP 地址为 2.2.2.1。

[H3C-ipsec-policy-manual- lzc1-10]remote-address 2.2.2.1

# 配置 ESP 协议的出方向 SPI 为 12345，入方向 SPI 为 54321。

[H3C-ipsec-policy-manual-lzc1-10] sa spi outbound esp 12345

[H3C-ipsec-policy-manual- lzc1-10] sa spiinbound esp 54321

# 配置 ESP 协议的出方向 SA 的密钥为明文字符串123456，入方向 SA 的密钥为明文字符串 654321。

[H3C-ipsec-policy-manual-lzc1-10] sa string-key outbound esp simple 123456

[H3C-ipsec-policy-manual-lzc1-10] sa string-key inbound esp simple 654321

[H3C-ipsec-policy-manual- lzc1-10] quit

 # 在接口 GigabitEthernet2/1/2上应用 IPsec安全策略 lzc1。

[H3C] interfacegigabitethernet 2/1/2

[H3C-GigabitEthernet2/1/2]ipsec apply policy lzc1

[H3C-GigabitEthernet2/1/2] quit

(2) 配置 MSR36-20_3

# 配置各接口的 IP 地址，具体略。

# 配置一个 ACL，定义要保护由子网 10.1.2.0/24 去往子网 10.1.1.0/24的数据流。

<H3C>system-view

[H3C] acl number 3101

[H3C-acl-adv-3101] rulepermit ip source 10.1.2.0 0.0.0.255 destination10.1.1.0 0.0.0.255

[H3C-acl-adv-3101] quit

# 创建 IPsec 安全提议lzc。

[H3C] ipsec transform-set lzc

# 配置安全协议对 IP 报文的封装形式为隧道模式。

[H3C-ipsec-transform-set- lzc]encapsulation-mode tunnel

# 配置采用的安全协议为 ESP。

[H3C-ipsec-transform-set- lzc] protocolesp

# 配置 ESP 协议采用的加密算法为 128比特的 AES，认证算法为 HMAC-SHA1。

[H3C-ipsec-transform-set-lzc] esp encryption-algorithm aes-cbc-128 [H3C-ipsec-transform-set- lzc] espauthentication-algorithm sha1

[H3C-ipsec-transform-set- lzc] quit

# 创建一条手工方式的 IPsec安全策略，名称为 LZC1，序列号为 10。

[H3C] ipsec policy LZC1 10 manual

# 指定引用 ACL 3101。

[H3C-ipsec-policy-manual-LZC1-10] security acl 3101

# 指定引用的 IPsec 安全提议为 lzc。

[H3C-ipsec-policy-manual- LZC1-10]transform-set lzc

# 指定 IPsec 隧道对端 IP 地址为 1.1.1.1。

[H3C-ipsec-policy-manual- LZC1-10] remote-address1.1.1.1

# 配置 ESP 协议的出方向 SPI 为 54321，入方向 SPI 为 12345。

[H3C-ipsec-policy-manual-LZC1-10] sa spi outbound esp 54321

[H3C-ipsec-policy-manual- LZC1-10] sa spiinbound esp 12345

# 配置 ESP 协议的出方向 SA 的密钥为明文字符串654321，入方向 SA 的密钥为明文字符串 123456。

[H3C-ipsec-policy-manual-LZC1-10] sa string-key outbound esp simple gfedcba

[H3C-ipsec-policy-manual-LZC1-10] sa string-key inbound esp simple abcdefg

[H3C-ipsec-policy-manual- LZC1-10] quit

# 在接口 GigabitEthernet2/1/2 上应用 IPsec 安全策略 LZC1。

[H3C] interfacegigabitethernet 2/1/2

[H3C-GigabitEthernet2/1/2]ipsec apply policy LZC1

[H3C-GigabitEthernet2/1/2] quit
