qmake: QT += core gui axcontainer
#include 

void Widget::on_pushButton_clicked()
{
    QAxObject axObject("SAPI.SpVoice");
    axObject.dynamicCall("Speak(QString, SpeechVoiceSpeakFlags)", ui -> textEdit -> toPlainText());
}
