### 1.获取字符串

*   
*   
*   
*   
*   
*   
*   
*   
*   
*   
*   

```
#include 
// 输入对话框void MyWidget::on_pushButton_4_clicked(){    bool ok;
    // 获取字符串    QString string = QInputDialog::getText(this, tr("输入字符串对话框"), tr("请输入用户名："), QLineEdit::Normal,tr("admin"), &ok);    if(ok) qDebug() << "string:" << string;}
```

### 2.获取整数

*   
*   
*   
*   
*   
*   
*   
*   
*   
*   
*   

```
#include 
// 输入对话框void MyWidget::on_pushButton_4_clicked(){    bool ok;
    // 获取整数    int value1 = QInputDialog::getInt(this, tr("输入整数对话框"), tr("请输入-1000到1000之间的数值"), 100, -1000, 1000, 10, &ok);    if(ok) qDebug() << "value1:" << value1;}
```

### 4.获取条目

*   
*   
*   
*   
*   
*   
*   
*   
*   
*   
*   
*   
*   

```
#include 
// 输入对话框void MyWidget::on_pushButton_4_clicked(){    bool ok;
    QStringList items;    items << tr("条目1") << tr("条目2");    // 获取条目    QString item = QInputDialog::getItem(this, tr("输入条目对话框"), tr("请选择或输入一个条目"), items, 0, true, &ok);    if(ok) qDebug() << "item:" << item;}
```
