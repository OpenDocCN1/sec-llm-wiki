void QCustomDelegate::paint(QPainter *painter, const QStyleOptionViewItem &option, const QModelIndex &index) const{    QRectF rect = option.rect;    if(option.state & QStyle::State_Selected)        painter->fillRect(rect,QBrush(QColor("#1296bd")));    else        painter->fillRect(rect,QBrush(QColor("#FFFFFF")));    painter->drawText(rect, Qt::AlignCenter, index.model()->data(index, Qt::EditRole).toString());}
