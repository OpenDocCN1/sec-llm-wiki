```

  
复制

# 4.下标访问

重载下标访问运算符

```

&at(int n)const;
```

  
复制

# 5.String类提供的方法

```
int size()const;
       //返回当前字符串的大小，string内部不会在我们每一次的字符串连接的时候都会分配空间，
//它会多分配一些内存，下次连接时的字符串小于剩下的空间它就不用再为这个字符串分配另外的空间了。
//它分配的所有空间就是size，而字符串的实际长度length
length()const;       //返回当前字符串的长度
empty()const;        //当前字符串是否为空
```

  
复制

```

  
复制

# 7.连接两个字符串

重载+运算符

```

compare(const string &s) const;//比较当前字符串和s的大小
```

  
复制这个实现的string类是比较简单的，C++标准模板库里面的string类的方法是非常多的，而且非常复杂。 在这里仅仅给大家演示string内部大概的实现方法

## String.h文件：类的声明

```
#ifndef STRING\_H//头文件卫士，避免头文件被多次编译
STRING\_H
<iostream\>  //输入输出头文件
; //命名空间
class String
{
public://
公有的声明String(const char\*); //构造函数
String(int n,char c); //传入大小的构造函数
~String(); //析构函数

public:
String(String &str);  //拷贝构造函数
const String& operator\=(const String& str);//赋值运算符重载，我们在不需要修改参数的值的地方都应该声明为const

\[\](int pos)const;//下标访问运算符重载
at(int pos)const;//检查越界的下标访问

size()const; //字符串的空间大小
length()const; //字符串的字符大小
empty()const; //判断为空的函数

const String& operator+=(const String& str); //字符串的连接
\==(const String& str);//字符串的相等判断
compare(const String& str);//字符串的相等判断

& operator\>>(istream& input, String& str); //输入操作符的重载
& operator<<(ostream& output, String& str); //输出操作符的重载
private:
\* m\_pBuff;//保存字符串的首地址
;  //字符串的字符大小
;  //字符串的空间大小

};

## String.cpp：类的实现

```
#include"String.h" //包含类的声明头文件

EXT\_LEN 50  //定义一个宏，用户申请的内存大小，我们实际上为字符串分配的空间要多EXT\_LEN，以便字符串的连接
String::String(const char\* pStr)
{
\= strlen(pStr); //获得字符串的实际大小
\= new char\[m\_nRealLen + EXT\_LEN\]; //实际分配的内存比字符串多了EXT\_LEN
//strcpy(m\_pBuff, pStr);  //C提供的一个字符串函数，将后面参数的字符复制到第一个字符串后面，并添加一个结束字符\\0，C中字符串的指针指向的是字符串的首地址，这个函数遇到\\0以为字符串结束
memcpy(m\_pBuff, pStr, m\_nRealLen);  //内存拷贝，遇到\\0不会结束，提供了需要拷贝的数据的长度
\= m\_nRealLen + EXT\_LEN;  //计算字符串的空间大小
}

String::String(int n, char c)  //构造一个由n个一种字符串构成的字符串
{
\= new char\[n + EXT\_LEN\];  //实际分配的内存比字符串多了EXT\_LEN
for (int i \= 0; i < n; i++)
{
\[i\] \= c;  //利用下标运算符和循环赋值
}
\= n;  //赋值字符串大小
\= n + EXT\_LEN;  //分配的空间大小
}

String::~String()
{
if (m\_pBuff)   //判断字符串指针是否为空
{
delete\[\] m\_pBuff;
\= NULL;  //避免野指针的产生
}

\= 0;  //对字符串的长度清零
\= 0; //对分配的空间清零 
}

String::String(String& str)
{
//这里不用判断 左操作数的字符串首地址是否指向内存，就销毁再指向空
//因为左操作数m\_pBuff开始没有赋值，默认会指向一个无法访问的地址，在这里销毁无法访问的地址，就会出现异常
//if (m\_pBuff) //如果左操作数的字符串首地址指向内存，就销毁再指向空
//{
//  delete\[\]m\_pBuff;
//  m\_pBuff = NULL;
//}
\= new char\[str.length() + EXT\_LEN\];//重新分配一段空间
//strcpy(m\_pBuff, str.m\_pBuff);
memcpy(m\_pBuff, str.m\_pBuff, str.length());  //字符串的拷贝
\= str.length();   //实际长度等于字符串的字符长度
\= m\_nRealLen + EXT\_LEN;
}

const String& String::operator\=(const String& str)
{
if (this \== &str)  //避免自赋值
{
return \*this;
}

if (m\_pBuff) //如果左操作数的字符串首地址指向内存，就销毁再指向空
{
delete\[\]m\_pBuff;
\= NULL;
}

\= new char\[str.length() + EXT\_LEN\];  //分配空间
//strcpy(m\_pBuff, str.m\_pBuff);
memcpy(m\_pBuff, str.m\_pBuff, str.length());
\= str.length();
\= m\_nRealLen + EXT\_LEN;

return \*this;  //返回左操作数的引用
}

::operator\[\](int nPos)const
{

return m\_pBuff\[nPos\];  //使用了C里面字符串的下标访问
}

::at(int nPos)const
{
if (nPos \>= m\_nRealLen)  //如果给出的下标超出了字符串的实际长度，就抛出一个异常
{
//throw exception
}
 return m\_pBuff\[nPos\];
}

::size()const
{
return m\_nBuffSize;  //返回字符串的空间大小
}

::length()const
{
return m\_nRealLen;  //返回字符串的字符大小
}

::empty()const
{
return !m\_nRealLen;  //根据字符串的实际长度返回是否为空
}

const String& String::operator+=(const String& str)
{
if (m\_nBuffSize \- m\_nRealLen \>= str.length())
  //计算 空间大小减去字符大小 剩余的分配了的但是没有使用的空间 是否可以连接后面的字符串，
//如果可以不用再次分配空间直接连接，就不用再分配空间了。
{
//strcat(m\_pBuff, str.m\_pBuff);
memcpy(m\_pBuff + m\_nRealLen, str.m\_pBuff, str.length());
\= m\_nRealLen + str.length();

}
   else  //需要重新分配空间情况
{
\= m\_nRealLen + str.length();
\*p \= new char\[nLen + EXT\_LEN\];
//strcpy(p, m\_pBuff);
memcpy(p, m\_pBuff, m\_nRealLen); //将前面的字符串拷贝到新申请的空间中
//strcpy(p + m\_nRealLen, str.m\_pBuff);
memcpy(p + m\_nRealLen, str.m\_pBuff, str.length());
  //因为新空间已经有了一个字符串，我们不能再从首地址拷贝开始了，我们将首地址加上已经拷贝过去的字符串的长度，
//从这个位置开始后面没有拷贝字符串的位置拼接好第二个字符串
\= nLen;
\= nLen + EXT\_LEN;

if (m\_pBuff)  //如果前面的字符串不是空的，我们就删除前面拼接之前的第一个字符串
{
delete\[\]m\_pBuff;
}

\= p;  //将拼接好的字符串赋值给对象指向字符串首地址的指针
}

return \*this;
}
& operator\>>(istream& input, String& str)
{
std::cin.get(str.m\_pBuff, str.size(), '\\n');
  //不直接使用cin是怕cin的输入没有结束或者字符串的长度导致越界，这个是c的一个读入指定长度字符串的函数，
//该函数将str.size()长度的字符串读取到str.m\_pBuff中，第三个字符是结束字符，即使没有达到指定长度，遇到这个字符也会结束的。
return input;
}
& operator<<(ostream& output, String& str)
{
for (int i \= 0; i < str.length(); i++)
  //因为cout对字符串的输出是以\\0来结束输出，我们使用的memcpy函数是不会在字符串的结尾自动加入结束符号\\0，
//所以我们需要利用循环和它的实际长度来实现遍历输出
{
std::cout.put(str\[i\]);  //每次向屏幕输出一个字符
}
return output;
}
