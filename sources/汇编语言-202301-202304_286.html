大部分程序中90%的运行时间都在执行10%的代码 – Hennessy & Patterson. Computer Architecture, A quantitative Approch

这意味着，可能优化了90%的代码，但程序运行时间只提升了10%。更好地关注10%的运行代码是优化性能的基础；

题目要求：设计一个英文名计数任务，给定一个500500行人名的txt文件，统计不同名字出现的频率和对应的排名情况。第一版设计中考虑使用map作为数据结构，每次读取一行字符时需要判断名字是否存在，存在则频率加1，不存在则名字对应的值为1。

实例代码如下：



// namedb.h#pragma once#include<string>#include<map>#include<fstream>

using namespace std;

class NameDB{    public:        NameDB(const std::string& nameFile);        int getNameRank(const std::string& name) const;        int getAbsoluteNumber(const std::string& name) const;    private:        std::map<std::string, int> mNames;        bool nameExistAndIncrement(const std::string& name);        void addNewName(const std::string& name);};



NameDB::NameDB(const std::string& nameFile){    ifstream inFile(nameFile);    if(!inFile){        throw invalid_argument("file name does not exist. ");    }    string name;    while(inFile >> name){        if(!nameExistAndIncrement(name)){            addNewName(name);        }    }    inFile.close();}

int NameDB::getNameRank(const std::string& name) const{    int num = getAbsoluteNumber(name);    if(num == -1){        return -1;    }    int rank = 1;    for(auto& item : mNames){        if(item.second > num){            rank++;        }    }    return rank;}

int NameDB::getAbsoluteNumber(const std::string& name) const{    auto res = mNames.find(name);    if(res != end(mNames)){        return res -> second;    }    return -1;}

bool NameDB::nameExistAndIncrement(const std::string& name){    auto res = mNames.find(name);    if(res != end(mNames)){        res -> second++;        return true;    }    return false;}

void NameDB::addNewName(const std::string& name){    mNames[name] = 1;}

// main.cpp#include<iostream>#include"namedb.h"

using namespace std;

int main(){    NameDB names("names.txt");    cout << names.getNameRank("Daniel") << endl;    cout << names.getNameRank("Jacob") << endl;    cout << names.getNameRank("William") << endl;    return 0;}
