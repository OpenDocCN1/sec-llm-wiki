void intt(){  srand(time(NULL));  for(int i=0;i<maxn+5</maxn+;i++)    for(int j=0;j<maxm+5</maxm+;j++)      square[i][j]=9;  memset(flag,true,sizeof(flag));  int xx=rand() % maxn+1;  int yy=rand() % maxm+1;  int num=1;  while(num<=99)  {    while(square[xx][yy]==-1||xx==xxx&&yy==yyy)    {      xx=rand()%maxn+1;      yy=rand()%maxm+1;    }    square[xx][yy]=-1;    num++;  }  for(int i=1;i<=maxn;i++)    for(int j=1;j<=maxm;j++)      if(square[i][j]!=-1) square[i][j]=count_num(i,j);}
