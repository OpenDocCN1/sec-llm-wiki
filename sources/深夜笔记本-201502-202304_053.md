```
from qqwry import QQwryimport argparseparser = argparse.ArgumentParser(description='帮助信息')parser.add_argument('-f','--file',help="输入一个ip文件",default='')args = parser.parse_args()def ipquery(file):    q = QQwry()    q.load_file('qqwry.dat') #载入纯真数据库    f = open(file,'r',encoding='utf-8')    outfile = open('result.xlsx','w+',encoding='utf-8')    for i in f.readlines():        i = i.strip()        res = q.lookup(i) #调用lookup查询ip归属，返回元组类型        outfile.write(i+"        "+res[0]+"\n") if __name__ == '__main__':    if args.file != "":        file1 = args.file        ipquery(file1)    else:        print("You must input a file!")
```
