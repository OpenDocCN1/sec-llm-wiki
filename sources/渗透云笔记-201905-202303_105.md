```
def get\_base64\_diff\_value(s1, s2):
    base64chars = 'ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/'
    res = 0
    for i in xrange(len(s2)):
        if s1\[i\] != s2\[i\]:
            return abs(base64chars.index(s1\[i\]) - base64chars.index(s2\[i\]))
    return res

def solve\_stego():
    with open('code.txt', 'rb') as f:
        file\_lines = f.readlines()
    bin\_str = ''
    for line in file\_lines:
        steg\_line = line.replace('\\n', '')
        norm\_line = line.replace('\\n', '').decode('base64').encode('base64').replace('\\n', '')
        diff = get\_base64\_diff\_value(steg\_line, norm\_line)
        print diff
        pads\_num = steg\_line.count('=')
        if diff:
            bin\_str += bin(diff)\[2:\].zfill(pads\_num \* 2)
        else:
            bin\_str += '0' \* pads\_num \* 2
        print goflag(bin\_str)

def goflag(bin\_str):
    res\_str = ''
    for i in xrange(0, len(bin\_str), 8):
        res\_str += chr(int(bin\_str\[i:i + 8\], 2))
    return res\_str

solve\_stego()
```
