    #将起始IP和结束IP转换为整数    start_ip = ip_to_int(start_ip)    end_ip = ip_to_int(end_ip)

def ip_to_int(ip):    # 把IP字符串转换为整数    octets = list(map(int, ip.split('.')))    return (octets[0] << 24) + (octets[1] << 16) + (octets[2] << 8) + octets[3]
