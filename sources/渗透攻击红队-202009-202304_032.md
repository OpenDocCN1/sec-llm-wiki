```
import requests#从漫图oss网站注册获取账户的token信息#https://www.minoss.in/def sendMsg(title,content):    data={        "title":'{}'.format(title),        "content":"{}".format(content)    }    headers = {        "Content-type":"application/json",        "Token": ""#你的token{token在设置中心里面}    }    try:        re=requests.post('https://www.minoss.in/api/push',json=data,headers=headers)        if re.json()["code"] == 200:            print(re.text)    except Exception as e:        print('发送失败！',e)        print(re.text)
```
