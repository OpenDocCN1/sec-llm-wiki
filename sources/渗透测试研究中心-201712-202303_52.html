echo $hello;

ob_clean();

$getlog->setIp($_SERVER['REMOTE_ADDR']);
$getlog->setUserAgent($_SERVER['HTTP_USER_AGENT']);
$getlog->echoLog();
$log_info = date("Y-m-d H:i:s ").ob_get_contents();

public $LOG_PATH = "/log/";
file_put_contents(dirname(__FILE__).$this->LOG_PATH.$this->LOG_NAME,$log_info);
