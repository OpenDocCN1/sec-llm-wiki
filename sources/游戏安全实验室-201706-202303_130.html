//在d3d9hook.cpp中

//HOOK d3d9  IDirect3DDevice9::Present 的核心代码

DWORD WINAPI HookD3d9(PVOID parg)

{

BOOL isHook=(BOOL)parg;

DbgPrintfA("HookD3d9  isHook=%d \r\n",isHook);

if (g_D3DDevice==NULL)

{

g_D3DDevice=GetNewD3DDevice(); //此函数源代码 在后边

 

}

//

PVOID pPresent=(PVOID)GetClassProcAddr(&IDirect3DDevice9::Present); //68 0x44 此pPresent 处可以取得 Present与g_D3DDevice的偏移0X44

if (g_D3DDevice==NULL)

{

DbgPrintfA("HookD3d9 369 d3d Null Error  397 ");

return -1;

}

PDWORD poldVcallPresent = (PDWORD)*((PDWORD)(*(PDWORD)g_D3DDevice + 0x44)); //0x44为

if (g_pNew_Present==NULL)

{

g_pNew_Present=poldVcallPresent;

}

 

DbgPrintfA("406 &IDirect3DDevice9::Present=%08X poldVcallPresent=%08X g_D3DDevice=%08X",*(DWORD*)(pPresent),poldVcallPresent,g_D3DDevice);

 

if (isHook)

{

DbgPrintfA("安装d3d9Hook Code136815");

hookApi(&g_pNew_Present, New_PresentVcall);  //重点 hook替换Present函数

}   else

{

DbgPrintfA("卸载d3d9HOOK Code136816  ...");

unhookApi(&g_pNew_Present, New_PresentVcall);

//Freeg_D3DDevice();

g_pNew_Present=NULL;

}

 

// Sleep(1000);

 

// MessageBoxA(0,"d3d Hook OK ",0,0);

 

return TRUE;

}

 //此函数作用是 创建一个IDirect3DDevice9对象 以方便获取IDirect3DDevice9::Present的地址

IDirect3DDevice9*  GetNewD3DDevice()  //重要

{  

HWND hWndD3D=NULL;

char wndClassName[100]="yjx_D3DCLassName";

sprintf(wndClassName,"%s%d",wndClassName,GetTickCount());

 

//注册窗口类

WNDCLASSEXA wndClass = {

sizeof(WNDCLASSEX),

CS_CLASSDC,

MsgProc,

0L,

0L,

GetModuleHandleA(NULL),

NULL, NULL, NULL, NULL,

wndClassName,

NULL };

RegisterClassExA(&wndClass);

//创建窗口

hWndD3D = CreateWindowA(

wndClassName,

"www.yjxsoft.com",

WS_OVERLAPPEDWINDOW,

100, 100, 600, 600,

GetDesktopWindow(),

NULL,

wndClass.hInstance,

NULL);

if (hWndD3D==NULL)

{

DbgPrintfA("创建窗口失败 GetLastError()=%08X",GetLastError());

return NULL;

}

 

 

// D3D 显示参数  

D3DPRESENT_PARAMETERS d3dparg;  

ZeroMemory(&d3dparg, sizeof(d3dparg));

 

 

// 创建D3D对象 获取主显卡硬件信息  最先被创建，最后被释放  

g_D3D = Direct3DCreate9(D3D_SDK_VERSION);  

DbgPrintfA("hWndD3D=%08x  g_D3D=%08X line 63 MSG",hWndD3D,g_D3D);

if (g_D3D == nullptr)  

{  

return 0;  

}  //1111

 

// 获取显示模式  

// 显示模式 （以像素为单位的屏幕宽高，刷新频率，surface formt）  

 

D3DDISPLAYMODE displayMode;  

if (FAILED(g_D3D->GetAdapterDisplayMode(D3DADAPTER_DEFAULT/*查询主显卡*/, &displayMode)))  

{  

return 0;  

}  

 

d3dparg.Windowed = true; // 是否窗口化  

d3dparg.BackBufferFormat = displayMode.Format;  // D3DFMT_X8R8G8B8  表示为32位RGB像素格式 每种颜色用一个字节表示  

d3dparg.SwapEffect = D3DSWAPEFFECT_DISCARD; // 帧缓冲区交换方式; 可能是COPY可能是FLIP，由设备来确定适合当前情况的方式  

//创建一个Direct3D设备

 

 

HRESULT dwRet=g_D3D->CreateDevice(

D3DADAPTER_DEFAULT,

D3DDEVTYPE_HAL,

hWndD3D,

D3DCREATE_HARDWARE_VERTEXPROCESSING,

&d3dparg,

&g_D3DDevice); //g_D3DDevice是一个全局变量指针 用于获取IDirect3DDevice9::Present函数

if (FAILED(dwRet))  

{  

DbgPrintfA("Error: GetLastError()=%08X dwRet=%08X  GetNewD3DDevice()=%08X\n",GetLastError(),dwRet,g_D3DDevice);

return g_D3DDevice;  

} else

{

DbgPrintfA("g_D3D->CreateDevice 成功 g_D3DDevice=%08X -----------\n",g_D3DDevice);

}  

CloseWindow(hWndD3D);

return g_D3DDevice;  

}  

//HOOK IDirect3DDevice9::Present地址后 

//把我们自己的绘图函数接口 int MyD3dDraw(IDirect3DDevice9* pD3DDevice) 给添加进去

 

IDirect3DDevice9* g_pGameD3D_Device=NULL;

void *g_pNew_Present;

//自己伪造的新IDirect3DDevice9::Present函数 名叫New_PresentVcall

HANDLE __stdcall New_PresentVcall(

IDirect3DDevice9 * pD3DDevice,

THIS_ CONST RECT* pSourceRect,

CONST RECT* pDestRect,

HWND hDestWindowOverride,

CONST RGNDATA* pDirtyRegion)

{

__asm

{

pushad

pushfd

}

MyD3dDraw(pD3DDevice);  //添加我们自己的绘图接口，后边的文字写屏，给方框，或者其它图形图象 都通过此接口

__asm

{

popfd

popad

}

g_D3DDevice=pD3DDevice;

__asm

{

push pDirtyRegion;

push hDestWindowOverride;

push pDestRect;

push pSourceRect;

push pD3DDevice;

 

call g_pNew_Present;

}

 

 

}

  void TD3DTEXT::MDrawText(char* str, int x, int y, D3DCOLOR color)

{

RECT FontRect = { (LONG)x, (LONG)y, (LONG)x+500, (LONG)y+30};

if(m_pFont)

{

m_pFont->DrawTextA(NULL,str,  -1, &FontRect, DT_LEFT | DT_LEFT, color); //关键

}

 

}

 

 void _TD3DTEXT::MDrawText(char* str, int x, int y, D3DCOLOR color,float wFont,float hFont)

 {

 RECT FontRect = { (LONG)x, (LONG)y, (LONG)(x+wFont), (LONG)(y+hFont)};

 if(m_pFont)

 {

 m_pFont->DrawTextA(NULL,str,  -1, &FontRect, DT_LEFT | DT_LEFT, color);

 }

 };
