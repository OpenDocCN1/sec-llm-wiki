//HOOK d3d9  IDirect3DDevice9::Present 的核心代码

DWORD WINAPI HookD3d9(PVOID parg)

{

BOOL isHook=(BOOL)parg;

DbgPrintfA("HookD3d9  isHook=%d \\r\\n",isHook);

if (g\_D3DDevice==NULL)

{

g\_D3DDevice=GetNewD3DDevice(); //此函数源代码 在后边

 

}

//

PVOID pPresent=(PVOID)GetClassProcAddr(&IDirect3DDevice9::Present); //68 0x44 此pPresent 处可以取得 Present与g\_D3DDevice的偏移0X44

if (g\_D3DDevice==NULL)

{

DbgPrintfA("HookD3d9 369 d3d Null Error  397 ");

return -1;

}

PDWORD poldVcallPresent = (PDWORD)\*((PDWORD)(\*(PDWORD)g\_D3DDevice + 0x44)); //0x44为

if (g\_pNew\_Present==NULL)

{

g\_pNew\_Present=poldVcallPresent;

}

 

DbgPrintfA("406 &IDirect3DDevice9::Present=%08X poldVcallPresent=%08X g\_D3DDevice=%08X",\*(DWORD\*)(pPresent),poldVcallPresent,g\_D3DDevice);

 

if (isHook)

{

DbgPrintfA("安装d3d9Hook Code136815");

hookApi(&g\_pNew\_Present, New\_PresentVcall);  //重点 hook替换Present函数

}   else

{

DbgPrintfA("卸载d3d9HOOK Code136816  ...");

unhookApi(&g\_pNew\_Present, New\_PresentVcall);

//Freeg\_D3DDevice();

g\_pNew\_Present=NULL;

}

 

// Sleep(1000);

 

// MessageBoxA(0,"d3d Hook OK ",0,0);

 

return TRUE;

}

 //此函数作用是 创建一个IDirect3DDevice9对象 以方便获取IDirect3DDevice9::Present的地址

IDirect3DDevice9\*  GetNewD3DDevice()  //重要

{  

HWND hWndD3D=NULL;

char wndClassName\[100\]="yjx\_D3DCLassName";

sprintf(wndClassName,"%s%d",wndClassName,GetTickCount());

 

//注册窗口类

WNDCLASSEXA wndClass = {

sizeof(WNDCLASSEX),

CS\_CLASSDC,

MsgProc,

0L,

0L,

GetModuleHandleA(NULL),

NULL, NULL, NULL, NULL,

wndClassName,

NULL };

RegisterClassExA(&wndClass);

//创建窗口

hWndD3D = CreateWindowA(

wndClassName,

"www.yjxsoft.com",

WS\_OVERLAPPEDWINDOW,

100, 100, 600, 600,

GetDesktopWindow(),

NULL,

wndClass.hInstance,

NULL);

if (hWndD3D==NULL)

{

DbgPrintfA("创建窗口失败 GetLastError()=%08X",GetLastError());

return NULL;

}

 

 

// D3D 显示参数  

D3DPRESENT\_PARAMETERS d3dparg;  

ZeroMemory(&d3dparg, sizeof(d3dparg));

 

 

// 创建D3D对象 获取主显卡硬件信息  最先被创建，最后被释放  

g\_D3D = Direct3DCreate9(D3D\_SDK\_VERSION);  

DbgPrintfA("hWndD3D=%08x  g\_D3D=%08X line 63 MSG",hWndD3D,g\_D3D);

if (g\_D3D == nullptr)  

{  

return 0;  

}  //1111

 

// 获取显示模式  

// 显示模式 （以像素为单位的屏幕宽高，刷新频率，surface formt）  

 

D3DDISPLAYMODE displayMode;  

if (FAILED(g\_D3D->GetAdapterDisplayMode(D3DADAPTER\_DEFAULT/\*查询主显卡\*/, &displayMode)))  

{  

return 0;  

}  

 

d3dparg.Windowed = true; // 是否窗口化  

d3dparg.BackBufferFormat = displayMode.Format;  // D3DFMT\_X8R8G8B8  表示为32位RGB像素格式 每种颜色用一个字节表示  

d3dparg.SwapEffect = D3DSWAPEFFECT\_DISCARD; // 帧缓冲区交换方式; 可能是COPY可能是FLIP，由设备来确定适合当前情况的方式  

//创建一个Direct3D设备

 

 

HRESULT dwRet=g\_D3D->CreateDevice(

D3DADAPTER\_DEFAULT,

D3DDEVTYPE\_HAL,

hWndD3D,

D3DCREATE\_HARDWARE\_VERTEXPROCESSING,

&d3dparg,

&g\_D3DDevice); //g\_D3DDevice是一个全局变量指针 用于获取IDirect3DDevice9::Present函数

if (FAILED(dwRet))  

{  

DbgPrintfA("Error: GetLastError()=%08X dwRet=%08X  GetNewD3DDevice()=%08X\\n",GetLastError(),dwRet,g\_D3DDevice);

return g\_D3DDevice;  

} else

{

DbgPrintfA("g\_D3D->CreateDevice 成功 g\_D3DDevice=%08X -----------\\n",g\_D3DDevice);

}  

CloseWindow(hWndD3D);

return g\_D3DDevice;  

}

IDirect3DDevice9\* g\_pGameD3D\_Device=NULL;

void \*g\_pNew\_Present;

//自己伪造的新IDirect3DDevice9::Present函数 名叫New\_PresentVcall

HANDLE \_\_stdcall New\_PresentVcall(

IDirect3DDevice9 \* pD3DDevice,

THIS\_ CONST RECT\* pSourceRect,

CONST RECT\* pDestRect,

HWND hDestWindowOverride,

CONST RGNDATA\* pDirtyRegion)

{

\_\_asm

{

}

}
