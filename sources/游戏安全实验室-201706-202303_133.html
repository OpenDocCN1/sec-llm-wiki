很多游戏挂机辅助，是需要大量上号的

而大量上号又受到机器硬件的影响，导致无法开足够数量的账号

其中一个制约条件就是显卡，那么对于显卡最好的优化方式就是黑屏

以下是通用型黑屏优化的效果以及实现原理和代码

 

通用型黑屏优化 下边效果图：

 

 

 

原理:hook Direct3DDevice9::Present  在伪造的 Present  里添加黑屏代码 详情见源代码

下边是核心代码

Dll 入口函数

// dllmain.cpp : 定义 DLL 应用程序的入口点。

#include "stdafx.h"

IDirect3DDevice9*  GetNewD3DDevice(HWND hWnd)  ;

BOOL APIENTRY DllMain( HMODULE hModule,

                       DWORD  ul_reason_for_call,

                       LPVOID lpReserved

 )

{

switch (ul_reason_for_call)

{

case DLL_PROCESS_ATTACH:

{

//GetNewD3DDevice(NULL);

//HookD3d9((PVOID)TRUE); //必须放在单独线程中

CreateThread(0,0,(LPTHREAD_START_ROUTINE)HookD3d9,(PVOID)TRUE,0,0);

break;

}

case DLL_THREAD_ATTACH:

{

break;

}

case DLL_THREAD_DETACH:

{

break;

}

case DLL_PROCESS_DETACH:

{

HookD3d9(FALSE);

break;

}

break;

}

return TRUE;

}

 

 

DWORD WINAPI HookD3d9(PVOID parg)

{

BOOL isHook=(BOOL)parg;

DbgPrintfA("HookD3d9  isHook=%d \r\n",isHook);

if (g_D3DDevice==NULL)

{

g_D3DDevice=GetNewD3DDevice(); //此函数必须在单独线程执行 不能在入口函数执行

 

}

//

PVOID pPresent=(PVOID)GetClassProcAddr(&IDirect3DDevice9::Present); //68 0x44 此pPresent 处可以取得 Present与g_D3DDevice的偏移0X44

if (g_D3DDevice==NULL)

{

DbgPrintfA("HookD3d9 369 d3d Null Error  397 ");

return -1;

}

PDWORD poldVcallPresent = (PDWORD)*((PDWORD)(*(PDWORD)g_D3DDevice + 0x44)); //0x44为

if (g_pNew_Present==NULL)

{

g_pNew_Present=poldVcallPresent;

}

 

DbgPrintfA("406 &IDirect3DDevice9::Present=%08X poldVcallPresent=%08X g_D3DDevice=%08X",*(DWORD*)(pPresent),poldVcallPresent,g_D3DDevice);

 

if (isHook)

{

DbgPrintfA("安装d3d9Hook Code136815");

hookApi(&g_pNew_Present, New_PresentVcall);

}   else

{

DbgPrintfA("卸载d3d9HOOK Code136816  ...");

unhookApi(&g_pNew_Present, New_PresentVcall);

//Freeg_D3DDevice();

g_pNew_Present=NULL;

}

 

// Sleep(1000);

 

// MessageBoxA(0,"d3d Hook OK ",0,0);

 

return TRUE;

}

 

 

HANDLE __stdcall New_PresentVcall(

IDirect3DDevice9 * pD3DDevice,

THIS_ CONST RECT* pSourceRect,

CONST RECT* pDestRect,

HWND hDestWindowOverride,

CONST RGNDATA* pDirtyRegion)

{

D3DRECT r1={0,0,1024,768};

 pD3DDevice->Clear( 1, &r1,D3DCLEAR_TARGET, 0xFF000000,1.0f,0); //0xFF000000是黑屏 ，0xFFFF0000就是红色 红屏了呗

//pD3DDevice->Clear( 1, &r1, D3DCLEAR_TARGET, 0xFFFFFFFF, 0, 0 );

//return NULL;

__asm

{

pushad

pushfd

}

//MyD3dDraw(pD3DDevice);

__asm

{

popfd

popad

}

g_D3DDevice=pD3DDevice;

__asm

{

push pDirtyRegion;

push hDestWindowOverride;

push pDestRect;

push pSourceRect;

push pD3DDevice;

 

call g_pNew_Present;

}
