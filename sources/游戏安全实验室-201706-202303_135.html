原理:hook Direct3DDevice9::Present  在伪造的 Present  里添加黑屏代码 详情见源代码

下边是核心代码

// dllmain.cpp : 定义 DLL 应用程序的入口点。

                       DWORD  ul_reason_for_call,

switch (ul_reason_for_call)

case DLL_PROCESS_ATTACH:

//GetNewD3DDevice(NULL);

case DLL_THREAD_ATTACH:

DWORD WINAPI HookD3d9(PVOID parg)

{

BOOL isHook=(BOOL)parg;

DbgPrintfA("HookD3d9  isHook=%d \r\n",isHook);

if (g_D3DDevice==NULL)

{

g_D3DDevice=GetNewD3DDevice(); //此函数必须在单独线程执行 不能在入口函数执行

 

}

//

PVOID pPresent=(PVOID)GetClassProcAddr(&IDirect3DDevice9::Present); //68 0x44 此pPresent 处可以取得 Present与g_D3DDevice的偏移0X44

if (g_D3DDevice==NULL)

{

DbgPrintfA("HookD3d9 369 d3d Null Error  397 ");

return -1;

}

PDWORD poldVcallPresent = (PDWORD)*((PDWORD)(*(PDWORD)g_D3DDevice + 0x44)); //0x44为

if (g_pNew_Present==NULL)

{

g_pNew_Present=poldVcallPresent;

}

 

DbgPrintfA("406 &IDirect3DDevice9::Present=%08X poldVcallPresent=%08X g_D3DDevice=%08X",*(DWORD*)(pPresent),poldVcallPresent,g_D3DDevice);

 

if (isHook)

{

DbgPrintfA("安装d3d9Hook Code136815");

hookApi(&g_pNew_Present, New_PresentVcall);

}   else

{

DbgPrintfA("卸载d3d9HOOK Code136816  ...");

unhookApi(&g_pNew_Present, New_PresentVcall);

//Freeg_D3DDevice();

g_pNew_Present=NULL;

}

 

// Sleep(1000);

 

// MessageBoxA(0,"d3d Hook OK ",0,0);

 

return TRUE;

}

//pD3DDevice->Clear( 1, &r1, D3DCLEAR_TARGET, 0xFFFFFFFF, 0, 0 ); //此行表示黑屏，但得调用原Present

Sleep(100); //降帧 如果想效果明显的话 可以增加延迟时间，但画面也会越卡

return NULL; //如果直接返回 就表示不更新画面，一直停留在之前的画面

__asm

{

pushad

}

//MyD3dDraw(pD3DDevice);

__asm

{

popfd

}

g_D3DDevice=pD3DDevice;

__asm

{

push pDirtyRegion;

push hDestWindowOverride;

call g_pNew_Present;

}
