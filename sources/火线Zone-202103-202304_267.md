xdebug.remote_enable = on　　;是否开启远程调试
xdebug.remote_handler = dbgp　　;指定远程调试的处理协议
xdebug.remote_host= localhost　　;指定远程调试的主机名
xdebug.remote_port = 9000　　;指定远程调试的端口号
