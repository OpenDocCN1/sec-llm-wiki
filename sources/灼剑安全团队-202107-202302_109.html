0x03 邮件发送脚本

一般用在粘合工具最后，执行完脚本后发个邮件提醒下

代码如下

from email.mime.text import MIMETextimport smtplibclass mails():    def config(self,data,bodys):        mail=MIMEText('''{0}'''.format(bodys))        mail["Subject"]=data        mail['From']='Beard的贴心大老铁@163.com'        mail['To'] = 'xxxx@qq.com'        return mail
    def fa(self,data,bodys):        smtp=smtplib.SMTP()        smtp.connect('smtp.163.com',25)        smtp.login('xxx@163.com','xxxx')        mail=self.config(data,bodys)        smtp.sendmail('xxx@163.com',' xxxx@qq.com',mail.as_string())a=mails()a.fa("我是内容","我是头")
