```
import uuidimport base64from Crypto.Cipher import AES
def encode_rememberme(data):    BS = AES.block_size    pad = lambda s: s + ((BS - len(s) % BS) * chr(BS - len(s) % BS)).encode()    key = base64.b64decode("kPH+bIxk5D2deZiIxcaaaA==")    iv = uuid.uuid4().bytes    encryptor = AES.new(key, AES.MODE_CBC, iv)    base64_ciphertext = base64.b64encode(iv + encryptor.encrypt(pad(data)))    return base64_ciphertext
def decode_rememberme(data):    BS = AES.block_size    key = base64.b64decode("kPH+bIxk5D2deZiIxcaaaA==")    unpad = lambda s: s[0:-s[-1]]    iv = base64.b64decode(data)[0:16]    cipher = AES.new(key, AES.MODE_CBC,iv)    decrypted  = unpad(cipher.decrypt(base64.b64decode(data)[16:]))    return decrypted
```

rememberMe的构成方式大概为先获得一个序列化字节流，然后以密钥和偏移量对序列化字节流进行加密，再把偏移量和字节流合并进行base64编码。  
所以这里偏移量可以随机也可以使用固定的iv = b'qqqqqqqqqqqqqqqq'
