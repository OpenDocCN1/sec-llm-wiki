no_can_no_bb

这道题随机生成了一个key，并对flag进行aes加密，解读后发现，生成的key是16位，其中前3位为随机生成后转为字节流，后13位为固定补位，生成密钥的可能性只有100万种，所以我们很自然就想到暴力破解，下面是解题脚本：

import randomfrom Crypto.Util.number import long_to_bytesfrom Crypto.Cipher import AES
def pad(m):   tmp = 16-(len(m)%16)   return m + bytes([tmp for _ in range(tmp)])
def decrypt(m,key):   aes = AES.new(key,AES.MODE_ECB)   return aes.decrypt(m)
if __name__ == "__main__":   while True:       cipher = b'\x9d\x18K\x84n\xb8b|\x18\xad4\xc6\xfc\xec\xfe\x14\x0b_T\xe3\x1b\x03Q\x96e\x9e\xb8MQ\xd5\xc3\x1c'       key = pad(long_to_bytes(random.randrange(1,1<<20)))       c = decrypt(cipher,key)       if c[:5] == 'cazy{':           print(c)           break

no_cry_no_can

这道题简单来说就是将key循环作为密钥流，和原文异或后得到密文，且我们知道了原文的前五位和key长度不超过5位这两个条件，所以将前五位和密文异或就可以得到key：_\x112\xffa

from Crypto.Util.number import*
def can_decrypt(flag,key):   block_len = len(flag) // len(key) + 1   new_key = key * block_len   return bytes([i^j for i,j in zip(flag,new_key)])
c = can_decrypt(b'<pH\x86\x1a&"m\xce\x12\x00pm\x97U1uA\xcf\x0c:NP\xcf\x18~l', b'_\x112\xffa')print(c)
