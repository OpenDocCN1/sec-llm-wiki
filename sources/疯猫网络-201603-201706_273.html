sub=""

fileAnswer() {
MYRAND=$[$[$RANDOM%${LINE}]+1]

answer0=$(echo $sub | cut -d"@" -f1)
answer1=$(echo $sub | cut -d"@" -f2)
answer2=$(echo $sub | cut -d"@" -f3)
answer3=$(echo $sub | cut -d"@" -f4)
}
