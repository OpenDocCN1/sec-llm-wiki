这里直接把代码贴上了，还有已经把python编译成exe了，有需要可以直接下载附件。

  41. def b64():#编码
  42.     txt = scr3.get('0.0', 'end')
  43.     strip = txt.strip('\n')
  44.     e = base64.b64encode(strip.encode('utf-8')).decode('ascii') 
  45.     scr4.insert(END,e)

  47. def b643():#解码
  48.     txt = scr3.get('0.0', 'end')
  49.     strip = txt.strip('\n')
  50.     e = base64.b64decode(strip.encode('utf-8'))
  51.     b = e.decode()
  52.     scr4.insert(END,b)

  54. def b32():
  55.     txt = scr3.get('0.0', 'end')
  56.     strip = txt.strip('\n')

  62.     strip = txt.strip('\n')
  63.     e = base64.b32decode(strip.encode('utf-8'))
  64.     b = e.decode()
  65.     scr4.insert(END,b)

  67. def b642():#清除
