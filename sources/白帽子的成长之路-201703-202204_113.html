defget_links(filename):#从filename中获取url

   file_object = open(filename,"r+")

   links_list = file_object.readlines()

   file_object.close()

return links_list

 

defsave_link(vlunfilename , vlu_urls):#将存在漏洞的url存入vlunfilename文件中

   file_object = open(vlunfilename,"w+")

   file_object.writelines(vlu_urls)

   file_object.close()
