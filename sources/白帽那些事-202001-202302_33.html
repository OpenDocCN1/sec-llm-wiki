SSR开机自启

cd /etc/init.dtouch ssrchmod +x ssrvim ssr


其中ssr的内容为


#!/bin/sh



### BEGIN INIT INFO

# Provides: ssr

# Required-Start: $network $remote_fs $local_fs

# Required-Stop: $network $remote_fs $local_fs

# Default-Start: 2 3 4 5

# Default-Stop: 0 1 6

# Short-Description: ssr

# Description: shadowsocks

### END INIT INFO



sudo ssr start   



exit 0
