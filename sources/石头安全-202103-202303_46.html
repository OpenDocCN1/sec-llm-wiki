# 想法： xray 从txt中读取url， 逐个扫描。
import re
from urllib.parse import urlparse
import time
import os


def xrayscan(url,outfilename):
    scan_command = "./xray_linux_amd64 webscan --basic-crawler {} --html-output {}.html".format(url,outfilename)
    print("\033[33m"+scan_command+"\033[0m")
    os.system(scan_command)

def get_url():
    with open('xray_url.txt','r') as f:
        urls = f.readlines()
        pattern = re.compile(r'^(https|http)://')
        try:
            for url in urls:
                if (len(url) ==1):
                    continue
                if not pattern.match(url.strip()):
                    targeturl = "http://"+url
                else:
                    targeturl = url
                print("\033[32m[info]-------START-------"+targeturl.strip()+"---------START--------\033[0m")
                outputfilename = urlparse(targeturl).netloc

                print("\033[34m"+outputfilename+"\033[0m")
                xrayscan(targeturl.strip(),outputfilename.strip())
                print("\033[32m[info]------END--------" + targeturl.strip() + "---------END--------\033[0m")
                print("\033[34m"+url.strip()+"的结果输出到"+outputfilename+"\033[0m")
        except Exception as e:
            print(e)




def get_info():
    with open('xray_url.txt', 'r') as f:
        lines = f.readlines()
        num_lines = len([l for l in lines if l.strip(' \n') != ''])
    print("\033[36m==========================统计===================================== \033[0m")
    print("\033[36m==================从文件中提取出"+str(num_lines)+"个域名============================   \033[0m")


    time.sleep(1)
if __name__ == '__main__':
    start_time = time.time()
    get_info()
    get_url()
    end_time = time.time()
    print("共计用时",end_time - start_time)
