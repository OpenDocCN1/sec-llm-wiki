class Deduction {    String name;    int amount;
    public Deduction(String name, int amount) {        this.name = name;        this.amount = amount;    }}
class TaxCalculator {    List<TaxBracket> taxBrackets;    List<Deduction> deductions;    int income;
    public TaxCalculator(int income, List<TaxBracket> taxBrackets, List<Deduction> deductions) {        this.income = income;        this.taxBrackets = taxBrackets;        this.deductions = deductions;    }
    public double calculateTax() {        int taxableIncome = income;        for (Deduction deduction : deductions) {            taxableIncome -= deduction.amount;        }
        double tax = 0;        for (TaxBracket bracket : taxBrackets) {            if (taxableIncome <= bracket.upperBound) {                tax += (taxableIncome - bracket.lowerBound ) * bracket.rate;                break;            } else {                tax += (bracket.upperBound - bracket.lowerBound ) * bracket.rate;            }        }        return tax;    }}
