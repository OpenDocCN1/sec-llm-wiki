import java.util.ArrayList;import java.util.Arrays;import java.util.List;
class TaxBracket {    int id;    int lowerBound;    int upperBound;    double rate;
    public TaxBracket(int id, int lowerBound, int upperBound, double rate) {        this.id = id;        this.lowerBound = lowerBound;        this.upperBound = upperBound;        this.rate = rate;    }}
class Deduction {    String name;    int amount;
    public Deduction(String name, int amount) {        this.name = name;        this.amount = amount;    }}
class TaxCalculator {    List<TaxBracket> taxBrackets;    List<Deduction> deductions;    int income;
    public TaxCalculator(int income, List<TaxBracket> taxBrackets, List<Deduction> deductions) {        this.income = income;        this.taxBrackets = taxBrackets;        this.deductions = deductions;    }
    public double calculateTax() {        int taxableIncome = income;        for (Deduction deduction : deductions) {            taxableIncome -= deduction.amount;        }
        double tax = 0;        for (TaxBracket bracket : taxBrackets) {            if (taxableIncome <= bracket.upperBound) {                tax += (taxableIncome - bracket.lowerBound ) * bracket.rate;                break;            } else {                tax += (bracket.upperBound - bracket.lowerBound ) * bracket.rate;            }        }        return tax;    }}
public class Main {    public static void main(String[] args) {        List<TaxBracket> taxBrackets = new ArrayList<>(Arrays.asList(            new TaxBracket(1, 0, 15000, 0.05),            new TaxBracket(2, 15001, 45000, 0.10),            new TaxBracket(3, 45001, 120000, 0.20),            new TaxBracket(4, 120001, 200000, 0.30),            new TaxBracket(5, 200001, Integer.MAX_VALUE, 0.40)        ));
        List<Deduction> deductions = new ArrayList<>(Arrays.asList(            new Deduction("Standard", 5000),            new Deduction("Health_Insurance", 2000),            new Deduction("Retirement_Savings", 3000)        ));
