#-- coding:UTF-8 --
import re
import random
import os
import requests
import sys
from urllib.parse import urlsplit
headers = {
    "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/91.0.4472.101 Safari/537.36 Edg/91.0.864.48",
    "Cookie": "security=low; PHPSESSID=rvs97lr92tqoomm05jm6e9lah5"
}
def main():
    #1.输入URL
    URL = input("目标网址: ")
    print("\n你输入的网址是", URL)

    #2.对目标发起请求
    r = requests.get(URL)
    #print(r.status_code)
    #3.判断是否连接正常
    r = requests.get(URL, timeout=10)
    if r.status_code == 200 or r.status_code == 301:
        print("\n可以正常访问")
    else:
        print("\n连接失败")

    #4.加载xss payload
    print("\n开始加载payload。请慢慢等待")
    payloads = []
    #filename = os.path.join(sys.path[0], "xss", "xsspayload.txt")
    f = open('xsspayload.txt')
    for i in f:
        payloads.append(i.strip())

    #5.拆分URL寻找参数点
    s = urlsplit(URL).query
    kkk = re.findall(r".*?name=(.*)", s)

    #print(result)
    #6.把payload替换query
    for _payload in payloads:
        #print(urls.replace(kkk[0],_payload))
        _url = URL.replace(kkk[0], _payload)
        print("[xss 测试]:", _url)
        #7.重新发送
        _str = requests.get(_url,headers=headers,timeout=10)
        if _str.status_code !=200:
            print("访问不正常重来吧")
            break
        #8.开始检测页面是否有包含payload信息
        result = _str.text
        if (result.find(_payload) != -1):
            #print("xss 检测成功payload为:%s" % _url)
            print("\nxss 检测成功payload为:%s" % _payload)
            print("建议直接访问:%s" % _url)
            break
    return False
if __name__=='__main__':
    try:
        main()
    except KeyboardInterrupt:
        print("killed by user.")
