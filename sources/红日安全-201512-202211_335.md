function getip()

{

   if (getenv('HTTP\_CLIENT\_IP'))

    {

       $ip = getenv('HTTP\_CLIENT\_IP');

    }

   

   elseif (getenv('HTTP\_X\_FORWARDED\_FOR'))

    {//获取客户端用代理服务器访问时的真实ip 地址

       $ip = getenv('HTTP\_X\_FORWARDED\_FOR');

    }

   elseif (getenv('HTTP\_X\_FORWARDED'))

    {

       $ip = getenv('HTTP\_X\_FORWARDED');

    }

   elseif (getenv('HTTP\_FORWARDED\_FOR'))

    {

       $ip = getenv('HTTP\_FORWARDED\_FOR');

    }

   elseif (getenv('HTTP\_FORWARDED'))

    {

       $ip = getenv('HTTP\_FORWARDED');

    }

}
