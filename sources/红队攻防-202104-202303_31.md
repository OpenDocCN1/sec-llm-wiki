**编写js加密代码**

调用这个CryptoJS库js文件，网上都能找

*

```
function jiami(str,o,i) {    //str 是我们要加密的字符，o是iv偏移量  i是密钥   var str = CryptoJS.enc.Utf8.parse(str);    //加密的字符  var iv = CryptoJS.enc.Utf8.parse(o)  var key = CryptoJS.enc.Utf8.parse(i)  return CryptoJS.AES.encrypt(str, key, {    iv: iv,    mode: CryptoJS.mode.CBC,    padding: CryptoJS.pad.Pkcs7  }).ciphertext.toString().toUpperCase()}
```

# AES CBC 128 PKSC7def get_aes_psswd(data, key, po):      # js传入的参数    jsstr = get_js()    ctx = execjs.compile(jsstr) #加载JS文件    return (ctx.call('jiami', data, key, po))  #调用js方法  第一个参数是JS的方法名，后面的data和key，和po是js方法的参数def get_js():    f = open("1.js", 'r', encoding='utf-8') # 打开JS文件    line = f.readline()       htmlstr = ''    while line:             htmlstr = htmlstr+line                  line = f.readline()                 return htmlstr    
if __name__ == '__main__':    fileurl = r'E:\1.txt'  #字典的绝对路径        with open(fileurl, 'r', encoding='utf-8') as zidian: #获取加密字典内容        with open('encrypt.txt', 'w', encoding='utf-8') as file: #创建encrypt.txt            print('正在加密')            for password in zidian:                file.write(get_aes_psswd(password, '这里填你的iv偏移量', '填你的密钥')+'\n')  #这是我们要传入的密钥和iv偏移量    print("加密完毕")
