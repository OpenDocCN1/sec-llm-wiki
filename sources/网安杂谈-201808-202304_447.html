import win32com.client
def fuzz(path,password):
try:
xlsx = win32com.client.Dispatch("Excel.Application")
xlwb = xlsx.Workbooks.Open(path, False, True, None, Password=password)
if xlwb:
print('[yes]:' + str(password))
except Exception as e:
print("[%s]" % password, repr(e))

if __name__ == '__main__':
file = r"C:\Users\Myself\Desktop\xlsfuzz\sx.xls"
