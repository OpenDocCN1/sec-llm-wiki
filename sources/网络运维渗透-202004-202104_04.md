```
#!/usr/bin/python# Env: python2
import binascii, struct
for i in range(0, 65535):  height = struct.pack('>i', i)  #CRC: CBD6DF8A  data = '\x49\x48\x44\x52\x00\x00\x01\xF4' + height + '\x08\x06\x00\x00\x00'  crc32result = binascii.crc32(data) & 0xffffffff  if crc32result == 0xCBD6DF8A:    print ''.join(map(lambda c: "%02X" % ord(c), height))
```

参考文章：https://www.freebuf.com/column/205244.html                   隐写术总结.note![](../Images/be8088cf181454c7468c632baf08f900.png)爆破成功，真实的高对应的十六进制是：000001F4。修改回来保存，重新打开图片就能看到flag。

```
#!/usr/bin/python# Env: python3# Author: afei_0and1# -*- coding: utf8 -*-
import binascii, struct, sys
def burst_png_width_height():    if (len(sys.argv) < 2):        print("-----------------------------------------------")        print(" ")        print("Useg: python %s <path/img.png>" % sys.argv[0])        print("eg: python steg_Bugku-burst_png_width_height.py c:\1.png")        print(" ")        print("-----------------------------------------------")        return    png_path = sys.argv[1]
    #pngdata = open(r'C:\Users\Administrator\Desktop\file\dabai.png', 'rb').read()    pngdata = open(png_path, 'rb').read()        #爆破宽    for i in range(1024):        # pngdata[12:16]表示IHDR标识，pngdata[20:24]表示图片的高        data = pngdata[12:16] + struct.pack('>i',i)+ pngdata[20:29]        crc32result = binascii.crc32(data) & 0xffffffff        #CRC: 6D7C7135        if crc32result == 0x6D7C7135:            print('width: ' +str(i))            print("width Hex: " +hex(i))            print("")
    #爆破高    for i in range(1024):        # pngdata[16:20]表示图片的高        data = pngdata[12:20] + struct.pack('>i', i) + pngdata[24:29]        crc32result = binascii.crc32(data) & 0xffffffff        #CRC: 6D7C7135        if crc32result == 0x6D7C7135:            print('height: ' +str(i))            print("height Hex: " +hex(i))
if (__name__ == '__main__'):    burst_png_width_height()
```
