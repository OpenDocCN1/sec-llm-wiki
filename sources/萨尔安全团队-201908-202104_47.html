headers = {
    'Accept': 'text/html,application/xhtml+xml,application/xml;q=0.9,*/*;q=0.8',
    'Accept-Encoding': 'gzip, deflate, compress',

def getfromBaidu(word,page):
    start = time.clock()
    process = int(input('请输入您的进程数目：'))
    print('请稍后...正在分配进程中...')
    url = 'http://www.baidu.com.cn/s?wd=' + urllib.parse.quote(word) + '&pn='  # word为关键词，pn是百度用来分页的..
    # pool = multiprocessing.Pool(multiprocessing.cpu_count())
    pool = multiprocessing.Pool(process)
    #默认30页 k=5
    for k in range(1, int(page)):
        result = pool.apply_async(geturl, (url, k))# 多进程
    pool.close()
    pool.join()
    end = time.clock()
    print('爬行用时：' + str(end-start))

def geturl(url, k):
    path = url + str((k - 1) * 10)
    response = request.urlopen(path)
    page = response.read()
    soup = BeautifulSoup(page, 'lxml')
    tagh3 = soup.find_all('h3')
    for h3 in tagh3:
        href = h3.find('a').get('href')
        # print(href)
        baidu_url = requests.get(url=href, headers=headers, allow_redirects=False,timeout = 3)
        real_url = baidu_url.headers['Location']  #得到网页原始地址
        if real_url.startswith('http'):
            # urls.insert({"url": real_url})
            print('爬行获取到的URL地址是：' + real_url)
            print('请勿关闭，等待爬行结束！否则没有urls会保存！')
            all.write(real_url + '\n')

import time
import requests
import re
#------------------------------//---------------
print ('软件名称：批量URL_Fuzz检测工具')
print ('Time_sleep：3 S')
time.sleep(3)

#------------------------------//注意事项
print ('请将网址导入目录下urls.txt，要求一个网址一个回车，网址格式：http://www.baidu.com')
print ('fuzz语法集合在目录下fuzz.txt，可以手工补充')
print ('Time_sleep：3 S')
time.sleep(3)
#------------------------------//注意事项
pattern = '\w+.\w+.\w+'
#------------------------------//URL请求模块
for url in open('urls.txt','r'):
    match = re.search(pattern,url.strip())
    url = match.group()
    print ('[info]TargetUrl:'+ url)
    for fuzz_test in open('fuzz.txt','r',encoding='UTF-8'):
        try:
            print ('[info]Payload:'+ fuzz_test)
            Finally_url = 'http://' + url.strip() + '/' + fuzz_test.strip()
            headers = {
                'User - Agent': 'Mozilla / 5.0(Windows NT 10.0;Win64 x64;rv: 66.0) Gecko / 20100101 Firefox / 66.0',
            }
            # requests.packages.urllib3.disable_warnings()
            #防止警告
            req = requests.get(Finally_url,headers = headers ,verify=False)
            if req.status_code == 200:
                print ('返回状态码:'+ str(req.status_code) + '|' + ' 返回包长度是：' + str(len(req.content)))
                print ('[info]TargetUrl:'+ url + '[info]Payload:'+ fuzz_test + '返回状态码:'+ str(req.status_code) + '|' + ' 返回包长度是：' + str(len(req.content)) ,file=open('result.txt','a+'))
            else:
                print ('返回状态码:'+ str(req.status_code))
        except Exception as e:
            print (e)
            continue
