```
#include<stdio.h>//flag{3688BAF108A4B42CB42E66EAE1091ECB}int main(){  char stdscan[39];  char stds[39]={"v}qvk\"&)(SQW!!(P$S$#SS$#U'&TQT!!) URRl"};  printf("请输入要验证的flag:");  scanf("%s",&stdscan);  int j=strlen(stds);  int i;  for(i=1;i<j+1;i++){    if(i%2==0){      stdscan[i-1]=stdscan[i-1]^0x11;    }else{      stdscan[i-1]=stdscan[i-1]^0x10;    }  }
  if(strcmp(stdscan,stds)==0){    printf("恭喜，flag验证正确！");  }else{    printf("flag验证错误！");  }
}
```

  

简单梳理一下代码的逻辑

1、定义一个stdscan长度39的char变量接收输入数据，定义stds长度39的char加密后的字符串（说明一下，我懒得再去截图改代码了，为什么明明38个字节的长度，要定义39呢？输入接收的变量可以定义是38，但是我们定义的比对字符串是39的长度，因为里面有一个双引号，我们需要用到转义符\\，所以是39个字节长度）

2、接收输入的flag

3、直接for循环，循环次数是输入字符串的长度，这里我没有像原题还做了一个输入字符串长度的判断，如果你想做可以加一个 if(strlen(stdscan)==38){继续运算对比}。然后取余如果=0则是偶数就亦或0x11，否则就是奇数，则亦或0x10。

4、最终比对stdscan和stds是否一模一样，如果是则证明flag验证成功，反之失败！
