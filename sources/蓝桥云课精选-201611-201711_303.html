整理课程列表

本题目考察的是 C 语言语法基础。其中的 sort 方法是对字符串的冒泡排序，而字符串顺序由字典序决定。我们需要注意的是 sort 方法的两个参数：char *p[] 这是一个指针数组，我们需要传入的是指向多个字符串起始位置的指针集合，所以在参数传递时需要写作*p[]。而参数 n 是集合的个数。strlen 函数只能获取单个字符串的长度个数，而对于一个字符串集合，我们需要用总量/每个指针开辟的空间量，所以使用 sizeof(courses) / sizeof(courses[0]) 可以获取到数组中精确的个数。而在冒泡排序的判断条件上，我们可以使用 string.h 中的 strcmp 方法对于两个字符串进行字典序大小的判断。

参考解答可以见下列代码：

#include <stdio.h>
#include <string.h>

void sort(char *p[], int n) { 
       char *tmp;
       int i, j, k;
       for (i = 0; i < n; ++ i) {
            k = i;
            for (j = i; j < n; ++ j) { 
                  if (strcmp(p[j], p[k]) < 0) {
                       k = j;
                   }
            } 
            if (k != j) {
          ‍      tmp = p[k];
                p[k] = p[i];
                p[i] = tmp;‍
        }
    }
}int main() { 
    char *courses[] = {"C","C++","PHP","ASP","ASP.NET","C#","JAVA","BASIC","PASCAL","COBOL"};   
    int n = sizeof(courses) / sizeof(courses[0]);
    sort(courses, n);    
    int i;    
    for (i = 0; i < n; ++ i) {        
        printf("%s\n", courses[i]);
    }    
    return 0;
}

远程 Shell

这道题实际上考察的是 Linux 上的 C 语言网络编程。并且只给出了客户端的源代码和服务端的二进制文件。需要根据客户端代码编译和运行的状态来找 bug，必要的时候需要进行调试跟踪。

在客户端代码文件中包含了多个 socket 编程中容易犯的错误：

错误1：TCP 连接

服务器端使用 TCP 连接，代码中创建 socket 的时候用的是 UDP 连接方式，需要使用：

int sock = socket(AF_INET, SOCK_STREAM, 0);

错误2：端口号使用网络字节序

字节序使用错误也是常见的 socket 错误，需要使用 htons 函数将主机字节序的端口号改为网络字节序：

connection.sin_port = htons(PORT);

错误3：强制类型转化

connect 函数中第二个参数需要使用 struct sockaddr 类型指针，但 connection 的类型不匹配，这个错误在编译的提示中可以很容易看出来。修改的代码为：

if (connect(sock, (struct sockaddr *)&connection, sizeof(connection)) != 0)

错误4：if 中的 == 误用

pid 变量没有初始化直接在 if 中用作 == 比较，此处应该是 =，pid 为 fork 创建后返回的进程 ID：

if(pid = fork())

最终修改后的代码：

#include "share.h"
void send_cmd(int sock, int pid) {

混乱的打字机

这道题实际上考察的是在多线程下的竞争问题。由于两个线程同时对 printer 方法进行调用，所以其输出结果交错在一起。我们可以使用信号量对访问的线程数做出限制，从而避免竞争问题。

参考解答可以见下列代码：

#include <stdio.h>#include <pthread.h>#include <unistd.h> #include <semaphore.h>sem_t sem; //信号量  void init() {
    sem_init(&sem, 0, 1); //初始化信号量，初始值为 1 }void printer(char *str) {  
    sem_wait(&sem); // 减一  
    while(*str) {  
        putchar(*str);    
        fflush(stdout);  
        str++;  
        sleep(1);  
    }  
    sem_post(&sem); // 加一  }  void *thread_fun1(void *arg) {  
    char *str1 = "hello";  
    printer(str1);  
}  void *thread_fun2(void *arg) {  
    char *str2 = "world";  
    printer(str2);  
}  int main(void) {  
    pthread_t tid1, tid2;   
    init()    // 创建 2 个线程  
    pthread_create(&tid1, NULL, thread_fun1, NULL);  
    pthread_create(&tid2, NULL, thread_fun2, NULL);  

    // 等待线程结束，回收其资源  
    pthread_join(tid1, NULL);  
    pthread_join(tid2, NULL);   
    return 0;  
}

逃出迷宫

对于一个迷宫数组，我们的思路可以有两种：

  • DFS（深度优先搜索）

  • BFS（广度优先搜索）

在深度优先搜索中，最简单的方法是我们可以对每一个可以到达的点进行染色，直到染色到不能再继续位置。而染色的点都是我们可以走到的地方。

这里有一个坑，就是即使是系统对数组进行初始化为 -1 的操作，但是在越界区域我们无法判断其初值为多少。所以在 x < 0 和 y < 0 的地方，我们需要进行剪枝操作，即排除掉该非法条件下的所有情况。

参考解答可以见下列代码：

void dfs(int x, int y) {    if (x < 0 || y < 0) return ;    if (maze[x][y] == 0) {
        dfs(x + 1, y);
        dfs(x - 1, y);
        dfs(x, y + 1);
        dfs(x, y - 1);
    }
}int haveExit() {

实现队列

该题目为数据结构基础题。其考察的内容就是对队列先进先出特点的掌握以及对于 C 语言中创建链式数据结构的熟练程度。

参考代码如下：

#include <stdio.h>
#include <stdlib.h>
struct node {
    int val;
    struct node* next;
};
static struct node *phead = NULL;
static struct node *create_node(int val) {
    struct node *pnode = NULL;
    pnode = (struct node*)malloc(sizeof(struct node));
    if (!pnode) {       
          return NULL;
    }
    pnode -> val = val;
    pnode -> next = NULL;    
    return pnode;
}
static void destory_single_link() {
    struct node *pnode = NULL;    
    while (phead != NULL) {
        pnode = phead;
        phead = phead -> next;        
        free(pnode);
    }
}
static void add(int val) {    
    if (!phead) {
        phead = create_node(val);        
        return ;
    }    
    struct node *pnode = create_node(val);    
    struct node *pend = phead;    
    while (pend -> next) {
        pend = pend -> next;
    }
    pend -> next = pnode;
}
static int front() {    
    return phead -> val;
}
static int pop() {    
    int ret = phead -> val;    
    struct node *pnode = phead;
    phead = phead -> next;   
    free(pnode);    return ret;
}
static int count() {
    int count = 0;    
    struct node *pend = phead;    
    while (pend) {
        pend = pend -> next;
        count ++;
    }    
    return count ++;
}
bool is_empty() {    
    if (count() == 0) {
            return true;
    } else {
            return false;
    }
}

常见 Q&A

为什么测试样例可以通过，在提交后无法通过？

A：在代码提交后，后端会对数据进行加强测试。如果您的代码不具备所有数据的通用性，则无法通过该挑战。

为什么在非 TODO 注释位置不允许修改代码？

A：实验楼题目多为运维性质。后端数据对拍系统以及评判系统还需改善。在目前的实现方式是使用脚本嵌入代码进行数据测试。在后期我们会考虑到用户提交代码的自由性，对后端功能进行加强。请大家见谅！

在由于系统问题造成的罚时，是不是应该由评判系统进行 rejudge 操作？

A：确实这样。但是实验楼的评测与一般的 Online Judge 评判不尽相同。由于实验楼提交的是一个虚拟环境，其提交的信息较为庞大。我们的攻城狮正在努力攻克这一难题，在未来会上线 rejudge 系统以确保楼赛公平性。敬请期待。

我对我的程序进行了各种大数据检测，但是在实验楼环境中也无法 pass？

A：也许是你所使用的编译器与楼赛中的编译器不尽相同，或者是代码存储的路径有所偏差。最好的检测方法是在虚拟环境中对其代码进行数据检测。
