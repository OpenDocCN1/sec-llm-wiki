实验原理

在开始开发之前，我们先来设计一下游戏算法。

扫雷游戏的规则很简单：

游戏面板上有一些格子，每个格子中有一个数字（空白表示数字为 0）或是地雷，格子中的数字表示格子周围格子中地雷的数量。玩家要做的就是把数字格子找出来，时间花的越少越好。

除边界上的格子外，每个格子周围有 8 个格子：上、下、左、右、4 个斜角。所以数字范围是 0~8。

所以我们的算法如下:

根据用户选择的难易程度（有初、中、高三个级别，级别越高地雷和格子数量越多），随机产生一定个数的地雷并随机放在格子中。然后遍历格子，计算每个格子中的数字，标记在格子上。玩家左键点击格子时显示格子内容（如果遇到地雷则挑战失败，游戏结束），右键点击格子时标记格子为地雷，真到正确标记所有地雷并打开所有非地雷格子，挑战成功，游戏结束。

小技巧：由于格子中数字范围是 0~8，所以为了便于计算，我们可以把地雷所在的格子中的数字记为 9。

绘制格子

完成上面的步骤后，下面就要画格子了，为了让代码更清晰，我们把游戏实现部分和调用部分分开，游戏实现部分放在跟 index.html 同目录下的 jms.js 中，游戏调用部分放在同目录下的 index.js 中。

画格子需要传入一些参数，如放格子的表格的 id，格子的数量（用行数和列数表示）。另外，游戏的其他数据也要进行初始化。

  1. //在jms.js中
  2. (function () {
  3.     // 初始化扫雷对象，初始化数据
  4.     var JMS = function (id,rowCount,colCount, minLandMineCount, maxLandMineCount) {
  5.         if (!(this instanceof JMS))
  6.             return new JMS(id, rowCount, colCount, minLandMineCount, maxLandMineCount);
  7.         this.doc = document;
  8.         this.table = this.doc.getElementById(id);//画格子的表格
  9.         this.cells = this.table.getElementsByTagName("td");//小格子
  10.         this.rowCount = rowCount || 10;//格子行数
  11.         this.colCount = colCount || 10;//格子列数
  12.         this.landMineCount = 0;//地雷个数
  13.         this.markLandMineCount = 0;//标记的地雷个数
  14.         this.minLandMineCount = minLandMineCount || 10;//地雷最少个数
  15.         this.maxLandMineCount = maxLandMineCount || 20;//地雷最多个数
  16.         this.arrs = [];//格子对应的数组
  17.         this.beginTime = null;//游戏开始时间
  18.         this.endTime = null;//游戏结束时间
  19.         this.currentSetpCount = 0;//当前走的步数
  20.         this.endCallBack = null;//游戏结束时的回调函数
  21.         this.landMineCallBack = null;//标记为地雷时更新剩余地雷个数的回调函数
  22.         this.doc.oncontextmenu = function () {//禁用右键菜单
  23.             return false;
  24.         };
  25.         this.drawMap();
  26.     };

  28.     // 在 JMS 的原型中创建格子
  29.     JMS.prototype = {
  30.         //画格子
  31.         drawMap: function () {
  32.             var tds = [];
  33.             // 为了兼容浏览器
  34.             if (window.ActiveXObject && parseInt(navigator.userAgent.match(/msie ([\d.]+)/i)[1]) < 8) {
  35.                 // 创建引入新的 css 样式文件
  36.                 var css = '#JMS_main table td{background-color:#888;}',
  37.                     // 获取 head 标签
  38.                     head = this.doc.getElementsByTagName("head")[0],
  39.                     // 创建 style 标签
  40.                     style = this.doc.createElement("style");
  41.                     style.type = "text/css";
  42.                 if (style.styleSheet) {
  43.                     // 将 css 样式赋给 style 标签
  44.                     style.styleSheet.cssText = css;
  45.                 } else {
  46.                     // 在 style 标签中创建节点
  47.                     style.appendChild(this.doc.createTextNode(css));
  48.                 }
  49.                 // 再将 style 标签创建为 head 标签的子标签
  50.                 head.appendChild(style);
  51.             }
  52.             // 循环创建表格
  53.             for (var i = 0; i < this.rowCount; i++) {
  54.                 tds.push("<tr>");
  55.                 for (var j = 0; j < this.colCount; j++) {
  56.                     tds.push("<td id='m_" + i + "_" + j + "'></td>");
  57.                 }
  58.                 tds.push("</tr>");
  59.             }
  60.             this.setTableInnerHTML(this.table, tds.join(""));
  61.         },
  62.         //添加HTML到Table
  63.         setTableInnerHTML: function (table, html) {
  64.             if (navigator && navigator.userAgent.match(/msie/i)) {
  65.                 // 在 table 的 owner document 内创建 div
  66.                 var temp = table.ownerDocument.createElement('div');
  67.                 // 创建表格的 tbody 内容
  68.                 temp.innerHTML = '<table><tbody>' + html + '</tbody></table>';
  69.                 if (table.tBodies.length == 0) {
  70.                     var tbody = document.createElement("tbody");
  71.                     table.appendChild(tbody);
  72.                 }
  73.                 table.replaceChild(temp.firstChild.firstChild, table.tBodies[0]);
  74.             } else {
  75.                 table.innerHTML = html;
  76.             }
  77.         }
  78.     };

  80.     window.JMS = JMS;
  81. })();

游戏初始化

现在，我们开始对游戏初始化，主要分三步：

  1. 把所有格子（代码中用一个数组表示）初始化为 0

  2. 随机生成地雷个数，把地雷随机放到数组中，数组项值设置为 9

  3. 计算其他格子中的数字，值放入数组中

在 jms.js 中 JMS.prototype 内加入如下代码：
  1. //在jms.js中JMS.prototype内加入
  2. //初始化，一是设置数组默认值为0，二是确定地雷个数
  3. init: function () {
  4.     for (var i = 0; i < this.rowCount; i++) {
  5.         this.arrs[i] = [];
  6.         for (var j = 0; j < this.colCount; j++) {
  7.             this.arrs[i][j] = 0;
  8.         }
  9.     }
  10.     this.landMineCount = this.selectFrom(this.minLandMineCount, this.maxLandMineCount);
  11.     this.markLandMineCount = 0;
  12.     this.beginTime = null;
  13.     this.endTime = null;
  14.     this.currentSetpCount = 0;
  15. },
  16. //把是地雷的数组项的值设置为9
  17. landMine: function () {
  18.     var allCount = this.rowCount * this.colCount - 1,
  19.         tempArr = {};
  20.     for (var i = 0; i < this.landMineCount; i++) {
  21.         var randomNum = this.selectFrom(0, allCount),
  22.             rowCol = this.getRowCol(randomNum);
  23.         if (randomNum in tempArr) {
  24.             i--;
  25.             continue;
  26.         }
  27.         this.arrs[rowCol.row][rowCol.col] = 9;
  28.         tempArr[randomNum] = randomNum;
  29.     }
  30. },
  31. //计算其他格子中的数字
  32. calculateNoLandMineCount: function () {
  33.     for (var i = 0; i < this.rowCount; i++) {
  34.         for (var j = 0; j < this.colCount; j++) {
  35.             if (this.arrs[i][j] == 9)
  36.                 continue;
  37.             if (i > 0 && j > 0) {
  38.                 if (this.arrs[i - 1][j - 1] == 9)
  39.                     this.arrs[i][j]++;
  40.             }
  41.             if (i > 0) {
  42.                 if (this.arrs[i - 1][j] == 9)
  43.                     this.arrs[i][j]++;
  44.             }
  45.             if (i > 0 && j < this.colCount - 1) {
  46.                 if (this.arrs[i - 1][j + 1] == 9)
  47.                     this.arrs[i][j]++;
  48.             }
  49.             if (j > 0) {
  50.                 if (this.arrs[i][j - 1] == 9)
  51.                     this.arrs[i][j]++;
  52.             }
  53.             if (j < this.colCount - 1) {
  54.                 if (this.arrs[i][j + 1] == 9)
  55.                     this.arrs[i][j]++;
  56.             }
  57.             if (i < this.rowCount - 1 && j > 0) {
  58.                 if (this.arrs[i + 1][j - 1] == 9)
  59.                     this.arrs[i][j]++;
  60.             }
  61.             if (i < this.rowCount - 1) {
  62.                 if (this.arrs[i + 1][j] == 9)
  63.                     this.arrs[i][j]++;
  64.             }
  65.             if (i < this.rowCount - 1 && j < this.colCount - 1) {
  66.                 if (this.arrs[i + 1][j + 1] == 9)
  67.                     this.arrs[i][j]++;
  68.             }
  69.         }
  70.     }
  71. },
  72. //获取一个随机数
  73. selectFrom: function (iFirstValue, iLastValue) {
  74.     var iChoices = iLastValue - iFirstValue + 1;
  75.     return Math.floor(Math.random() * iChoices + iFirstValue);
  76. },
  77. //通过数值找到行数和列数
  78. getRowCol: function (val) {
  79.     return {
  80.         row: parseInt(val / this.colCount),
  81.         col: val % this.colCount
  82.     };
  83. },
