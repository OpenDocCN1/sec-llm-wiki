2. 统计当前根目录代码行数。

  1. # coding=utf-8
  2. import os
  3. import time
  4. # 设定根目录
  5. basedir = './'
  6. filelists = []
  7. # 指定想要统计的文件类型
  8. whitelist = ['cpp', 'h']
  9. #遍历文件, 递归遍历文件夹中的所有
  10. def getFile(basedir):
  11.     global filelists
  12.     for parent,dirnames,filenames in os.walk(basedir):
  13.         for filename in filenames:
  14.             ext = filename.split('.')[-1]
  15.             #只统计指定的文件类型，略过一些log和cache文件
  16.             if ext in whitelist:
  17.                 filelists.append(os.path.join(parent,filename))
  18. #统计一个的行数
  19. def countLine(fname):
  20.     count = 0
  21.     # 把文件做二进制看待,read.
  22.     for file_line in open(fname, 'rb').readlines():
  23.         if file_line != '' and file_line != '\n': #过滤掉空行
  24.             count += 1
  25.     print (fname + '----' , count)
  26.     return count
  27. if __name__ == '__main__' :
  28.     startTime = time.clock()
  29.     getFile(basedir)
  30.     totalline = 0
  31.     for filelist in filelists:
  32.         totalline = totalline + countLine(filelist)
  33.     print ('total lines:',totalline)
  34.     print ('Done! Cost Time: %0.2f second' % (time.clock() - startTime))

  12. args_parser.add_argument('-dst', '--dst', type=str, nargs='?', required=True, help='(REQUIRED) Directory where files will be moved to.')
  13. args_parser.add_argument('-days', '--days', type=int, nargs='?', default=240, help='(OPTIONAL) Days value specifies the minimum age of files to be moved. Default is 240.')
  14. args = args_parser.parse_args()

  16. if args.days < 0:
  17.     args.days = 0

  19. src = args.src  # 设置源目录
  20. dst = args.dst  # 设置目标目录
  21. days = args.days # 设置天数
  22. now = time.time()  # 获得当前时间

  24. if not os.path.exists(dst):
  25.     os.mkdir(dst)

  27. for f in os.listdir(src):  # 遍历源目录所有文件
  28.     if os.stat(f).st_mtime < now - days * 86400:  # 判断是否超过240天
  29.         if os.path.isfile(f):  # 检查是否是文件
  30.             shutil.move(f, dst)  # 移动文件

6. 下载 Leetcode 的算法题。

  1. import sys
  2. import re
  3. import os
  4. import argparse
  5. import requests
  6. from lxml import html as lxml_html

  8. try:
  9.     import html
  10. except ImportError:
  11.     import HTMLParser
  12.     html = HTMLParser.HTMLParser()

  14. try:
  15.     import cPickle as pk
  16. except ImportError:
  17.     import pickle as pk

  19. class LeetcodeProblems(object):
  20.     def get_problems_info(self):
  21.         leetcode_url = 'https://leetcode.com/problemset/algorithms'
  22.         res = requests.get(leetcode_url)
  23.         if not res.ok:
  24.             print('request error')
  25.             sys.exit()
  26.         cm = res.text
  27.         cmt = cm.split('tbody>')[-2]
  28.         indexs = re.findall(r'<td>(\d+)</td>', cmt)
  29.         problem_urls = ['https://leetcode.com' + url \
  30.                         for url in re.findall(
  31.                             r'<a href="(/problems/.+?)"', cmt)]
  32.         levels = re.findall(r"<td value='\d*'>(.+?)</td>", cmt)
  33.         tinfos = zip(indexs, levels, problem_urls)
  34.         assert (len(indexs) == len(problem_urls) == len(levels))
  35.         infos = []
  36.         for info in tinfos:
  37.             res = requests.get(info[-1])
  38.             if not res.ok:
  39.                 print('request error')
  40.                 sys.exit()
  41.             tree = lxml_html.fromstring(res.text)
  42.             title = tree.xpath('//meta[@property="og:title"]/@content')[0]
  43.             description = tree.xpath('//meta[@property="description"]/@content')
  44.             if not description:
  45.                 description = tree.xpath('//meta[@property="og:description"]/@content')[0]
  46.             else:
  47.                 description = description[0]
  48.             description = html.unescape(description.strip())
  49.             tags = tree.xpath('//div[@id="tags"]/following::a[@class="btn btn-xs btn-primary"]/text()')
  50.             infos.append(
  51.                 {
  52.                     'title': title,
  53.                     'level': info[1],
  54.                     'index': int(info[0]),
  55.                     'description': description,
  56.                     'tags': tags
  57.                 }
  58.             )

  60.         with open('leecode_problems.pk', 'wb') as g:
  61.             pk.dump(infos, g)
  62.         return infos

  64.     def to_text(self, pm_infos):
  65.         if self.args.index:
  66.             key = 'index'
  67.         elif self.args.title:
  68.             key = 'title'
  69.         elif self.args.tag:
  70.             key = 'tags'
  71.         elif self.args.level:
  72.             key = 'level'
  73.         else:
  74.             key = 'index'

  76.         infos = sorted(pm_infos, key=lambda i: i[key])

  78.         text_template = '## {index} - {title}\n' \
  79.             '~{level}~  {tags}\n' \
  80.             '{description}\n' + '\n' * self.args.line
  81.         text = ''
  82.         for info in infos:
  83.             if self.args.rm_blank:
  84.                 info['description'] = re.sub(r'[\n\r]+', r'\n', info['description'])
  85.             text += text_template.format(**info)

  87.         with open('leecode problems.txt', 'w') as g:
  88.             g.write(text)

  90.     def run(self):
  91.         if os.path.exists('leecode_problems.pk') and not self.args.redownload:
  92.             with open('leecode_problems.pk', 'rb') as f:
  93.                 pm_infos = pk.load(f)
  94.         else:
  95.             pm_infos = self.get_problems_info()

  97.         print('find %s problems.' % len(pm_infos))
  98.         self.to_text(pm_infos)

  100. def handle_args(argv):
  101.     p = argparse.ArgumentParser(description='extract all leecode problems to location')
  102.     p.add_argument('--index', action='store_true', help='sort by index')
  103.     p.add_argument('--level', action='store_true', help='sort by level')
  104.     p.add_argument('--tag', action='store_true', help='sort by tag')
  105.     p.add_argument('--title', action='store_true', help='sort by title')
  106.     p.add_argument('--rm_blank', action='store_true', help='remove blank')
  107.     p.add_argument('--line', action='store', type=int, default=10, help='blank of two problems')
  108.     p.add_argument('-r', '--redownload', action='store_true', help='redownload data')
  109.     args = p.parse_args(argv[1:])
  110.     return args

  112. def main(argv):
  113.     args = handle_args(argv)
  114.     x = LeetcodeProblems()
  115.     x.args = args
  116.     x.run()

  118. if __name__ == '__main__':
  119.     argv = sys.argv
  120.     main(argv)

7. 将 Markdown 转换为 HTML。

  1. import sys
  2. import os

  4. from bs4 import BeautifulSoup
  5. import markdown


  8. class MarkdownToHtml:

  10.     headTag = '<head><meta charset="utf-8" /></head>'

  12.     def __init__(self,cssFilePath = None):
  13.         if cssFilePath != None:
  14.             self.genStyle(cssFilePath)

  16.     def genStyle(self,cssFilePath):
  17.         with open(cssFilePath,'r') as f:
  18.             cssString = f.read()
  19.         self.headTag = self.headTag[:-7] + '<style type="text/css">{}</style>'.format(cssString) + self.headTag[-7:]

  21.     def markdownToHtml(self, sourceFilePath, destinationDirectory = None, outputFileName = None):
  22.         if not destinationDirectory:
  23.             # 未定义输出目录则将源文件目录(注意要转换为绝对路径)作为输出目录
  24.             destinationDirectory = os.path.dirname(os.path.abspath(sourceFilePath))
  25.         if not outputFileName:
  26.             # 未定义输出文件名则沿用输入文件名
  27.             outputFileName = os.path.splitext(os.path.basename(sourceFilePath))[0] + '.html'
  28.         if destinationDirectory[-1] != '/':
  29.             destinationDirectory += '/'
  30.         with open(sourceFilePath,'r', encoding='utf8') as f:
  31.             markdownText = f.read()
  32.         # 编译出原始 HTML 文本
  33.         rawHtml = self.headTag + markdown.markdown(markdownText,output_format='html5')
  34.         # 格式化 HTML 文本为可读性更强的格式
  35.         beautifyHtml = BeautifulSoup(rawHtml,'html5lib').prettify()
  36.         with open(destinationDirectory + outputFileName, 'w', encoding='utf8') as f:
  37.             f.write(beautifyHtml)


  40. if __name__ == "__main__":
  41.     mth = MarkdownToHtml()
  42.     # 做一个命令行参数列表的浅拷贝，不包含脚本文件名
  43.     argv = sys.argv[1:]
  44.     # 目前列表 argv 可能包含源文件路径之外的元素（即选项信息）
  45.     # 程序最后遍历列表 argv 进行编译 markdown 时，列表中的元素必须全部是源文件路径
  46.     outputDirectory = None
  47.     if '-s' in argv:
  48.         cssArgIndex = argv.index('-s') +1
  49.         cssFilePath = argv[cssArgIndex]
  50.         # 检测样式表文件路径是否有效
  51.         if not os.path.isfile(cssFilePath):
  52.             print('Invalid Path: '+cssFilePath)
  53.             sys.exit()
  54.         mth.genStyle(cssFilePath)
  55.         # pop 顺序不能随意变化
  56.         argv.pop(cssArgIndex)
  57.         argv.pop(cssArgIndex-1)
  58.     if '-o' in argv:
  59.         dirArgIndex = argv.index('-o') +1
  60.         outputDirectory = argv[dirArgIndex]
  61.         # 检测输出目录是否有效
  62.         if not os.path.isdir(outputDirectory):
  63.             print('Invalid Directory: ' + outputDirectory)
  64.             sys.exit()
  65.         # pop 顺序不能随意变化
  66.         argv.pop(dirArgIndex)
  67.         argv.pop(dirArgIndex-1)
  68.     # 至此，列表 argv 中的元素均是源文件路径
  69.     # 遍历所有源文件路径
  70.     for filePath in argv:
  71.         # 判断文件路径是否有效
  72.         if os.path.isfile(filePath):
  73.             mth.markdownToHtml(filePath, outputDirectory)
  74.         else:
  75.             print('Invalid Path: ' + filePath)

8. 文本文件编码检测与转换。

  1. import sys
  2. import os
  3. import argparse
  4. from chardet.universaldetector import UniversalDetector

  6. parser = argparse.ArgumentParser(description = '文本文件编码检测与转换')
  7. parser.add_argument('filePaths', nargs = '+',
  8.                    help = '检测或转换的文件路径')

  10.                    help = '''

  12. ASCII, (Default) UTF-8 (with or without a BOM), UTF-16 (with a BOM),

  14. ISO-2022-KR, KOI8-R, MacCyrillic, IBM855, IBM866, ISO-8859-5, windows-1251, ISO-8859-2, windows-1250, EUC-KR,
  15. ISO-8859-5, windows-1251, ISO-8859-1, windows-1252, ISO-8859-7, windows-1253, ISO-8859-8, windows-1255, TIS-620
  16. ''')
  17. parser.add_argument('-o', '--output',
  18.                    help = '输出目录')
  19. # 解析参数，得到一个 Namespace 对象
  20. args = parser.parse_args()
  21. # 输出目录不为空即视为开启转换, 若未指定转换编码，则默认为 UTF-8
  22. if args.output != None:
  23.     if not args.encoding:
  24.         # 默认使用编码 UTF-8
  25.         args.encoding = 'UTF-8'
  26.     # 检测用户提供的输出目录是否有效
  27.     if not os.path.isdir(args.output):
  28.         print('Invalid Directory: ' + args.output)
  29.         sys.exit()
  30.     else:
  31.         if args.output[-1] != '/':
  32.             args.output += '/'
  33. # 实例化一个通用检测器
  34. detector = UniversalDetector()
  35. print()
  36. print('Encoding (Confidence)',':','File path')
  37. for filePath in args.filePaths:
  38.     # 检测文件路径是否有效，无效则跳过
  39.     if not os.path.isfile(filePath):
  40.         print('Invalid Path: ' + filePath)
  41.         continue
  42.     # 重置检测器
  43.     detector.reset()
  44.     # 以二进制模式读取文件
  45.     for each in open(filePath, 'rb'):
  46.         # 检测器读取数据
  47.         detector.feed(each)
  48.         # 若检测完成则跳出循环
  49.         if detector.done:
  50.             break
  51.     # 关闭检测器
  52.     detector.close()
  53.     # 读取结果
  54.     charEncoding = detector.result['encoding']
  55.     confidence = detector.result['confidence']
  56.     # 打印信息
  57.     if charEncoding is None:
  58.         charEncoding = 'Unknown'
  59.         confidence = 0.99
  60.     print('{} {:>12} : {}'.format(charEncoding.rjust(8),
  61.         '('+str(confidence*100)+'%)', filePath))
  62.     if args.encoding and charEncoding != 'Unknown' and confidence > 0.6:
  63.         # 若未设置输出目录则覆盖源文件
  64.         outputPath = args.output + os.path.basename(filePath) if args.output else filePath
  65.         with open(filePath, 'r', encoding = charEncoding, errors = 'replace') as f:
  66.             temp = f.read()
  67.         with open(outputPath, 'w', encoding = args.encoding, errors = 'replace') as f:
  68.             f.write(temp)
