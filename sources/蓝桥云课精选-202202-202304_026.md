public class Main {    private static long[][] dp = new long[1 << 22][22];    private static int[][] g = new int[22][22];
    public static int gcd(int a, int b) {        if (b == 0) return a;        return gcd(b, a % b);    }
    public static void main(String[] args) {        int n = 1 << 21;        for (int i = 1; i <= 21; i++) {            for (int j = 1; j <= 21; j++) {                if (gcd(i, j) == 1) {                    g[i - 1][j - 1] = g[j - 1][i - 1] = 1;                }            }        }        dp[1][0] = 1;        for (int i = 1; i < n; i++) {            for (int j = 0; j < 21; j++) {                if ((i >> j & 1) == 0) {                    continue;                }                for (int k = 0; k < 21; k++) {                    if (g[j][k] == 0 || (i >> k & 1) != 0) {                        continue;                    }                    dp[i + (1 << k)][k] += dp[i][j];                }            }        }        long res = 0;        for (int i = 0; i < 21; i++) {            res += dp[(n - 1)][i];        }        System.out.println(res);    }}
