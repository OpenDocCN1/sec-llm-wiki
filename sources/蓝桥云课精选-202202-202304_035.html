1、Java组

import java.util.*;
public class Main {    public static void main(String[] args) {        Scanner scan = new Scanner(System.in);
        int n = scan.nextInt();        int[] arr=new int[n];        for (int i = 0; i < n; i++) {            arr[i]=scan.nextInt();        }        Arrays.sort(arr);        int max = arr[n-1];        int min = arr[0];        int d = Integer.MAX_VALUE;        for (int i = 1; i < n; i++) {            d=Math.min(d,arr[i]-arr[i-1]);        }        if(d == 0){  // 如果差数为0，则直接返回n即可          System.out.println(n);          return;        }
        int count = (max-min)/d +1;        System.out.println(count);        scan.close();    }}

2、C/C++组

#include <iostream>#include <algorithm>using namespace std;int gcd(int a, int b) {//求最大公约数    return b == 0 ? a : gcd(b, a % b);}int a[100005], n, ans;int main() {    cin >> n;    for(int i = 0; i < n; i++)        cin >> a[i];    sort(a, a + n);//排序    ans = a[1] - a[0];//最大与最小项的差肯定是d的倍数    for(int i = 2; i < n; i++)        ans = gcd(ans, a[i] - a[i - 1]);    if(a[n - 1] - a[0] == 0)//若每一项都相同，直接输出n        cout << n;    else        cout << (a[n - 1] - a[0]) / ans + 1;    return 0;}
