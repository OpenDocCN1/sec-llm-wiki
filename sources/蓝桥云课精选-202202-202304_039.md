public class Main {    public static void main(String[] args) {        Scanner in = new Scanner(System.in);        int n = in.nextInt();        long[] a = new long[n + 10];        long[] b = new long[n + 10];                for(int i = 1; i <= n; i++)        {            a[i] = in.nextLong();            //预处理前缀和            b[i] = b[i - 1] + a[i];        }        long S = 0;        for(int i = 1; i <= n; i++)            S += a[i] * (b[n] - b[i]);        System.out.println(S);    }}
