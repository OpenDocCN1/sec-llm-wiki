```
#include <iostream>
using namespace std;
int main() {    int m = 2019, n = 324;
    int cnt = 0;
    int temp;
    while(true) {        cnt += m/n;        temp = m%n; if(temp==0) break;        m = n;        n = temp;    }
    cout << cnt << endl;
    return 0;}
```
