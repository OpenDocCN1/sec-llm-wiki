1、C/C++

#include <iostream>using namespace std;
int n, m;int a[1002][1002];//保存图片
int main(){
  cin >> n >> m;  for(int i = 0; i < n; i++) {    for(int j = 0; j < m; j++) {      cin >> a[i][j];    }  }  for(int i = 0; i < m; i++) {    for(int j = n - 1; j >= 0; j--) {      cout << a[j][i] << " ";    }    cout << endl;  }  return 0;}

2、Java

import java.util.*;
public class Main {
    public static void main(String[] args) {        Scanner scanner = new Scanner(System.in);                int n = scanner.nextInt();        int m = scanner.nextInt();                int[][] a = new int[n][m];                for (int i = 0; i < a.length; i++) {            for (int j = 0; j < a[i].length; j++) {                a[i][j] = scanner.nextInt();            }        }
        for (int i = 0; i < m; i++) {            for (int j = n - 1; j >= 0; j--) {                System.out.print(a[j][i] + " ");            }            System.out.println();        }        scanner.close();    }    }
