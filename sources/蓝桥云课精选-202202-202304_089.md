#include<bits/stdc++.h>using namespace std;const int N = 2e5 + 10;int a[N];signed main(){    int T = 1;    cin >> T;    while(T --){        int n , sum = 0 , ans = 0;        cin >> n;        for(int i = 1 ; i <= n ; i ++){            cin >> a[i];            sum ^= a[i];        }        if(!sum) {            cout << 0 << '\n';            continue ;        }        for(int j = 20 ; j >= 0 ; j --){            int one = 0 , zero = 0;            for(int i = 1 ; i <= n ; i ++){                if(a[i] >> j & 1) one ++ ;                else zero ++ ;            }            if(one & 1){                if(zero % 2 && one != 1) ans = -1;                else ans = 1;                break ;            }        }        cout << ans << '\n';    }    return 0;}

public class Main {    private static int[] a = new int[200010];
    public static void main(String[] args) {        InputReader cin = new InputReader(System.in);        int T = cin.nextInt();        while ((T--) != 0) {            int n = cin.nextInt();            int sum = 0;            int ans = 0;            for (int i = 1; i <= n; i++) {                a[i] = cin.nextInt();                sum ^= a[i];            }            if (sum == 0) {                System.out.println(0);                continue;            }            for (int j = 20; j >= 0; j--) {                int one = 0 , zero = 0;                for (int i = 1; i <= n; i++) {                   if ((a[i] >> j & 1) != 0) {                        one++;                    } else {                        zero++;                    }                }                if (one % 2 != 0) {                    if ((zero % 2) != 0 && one != 1) {                        ans = -1;                    } else {                        ans = 1;                    }                    break;                }            }            System.out.println(ans);        }    }}

T = int(input())for Y in range(1, T + 1):    a = []    sum = 0    ans = int(0)    mp = list(map(int, input().split()))    n = mp[0]    ma = int(0)    for j in range(1, len(mp)):        a.append(mp[j])        sum ^= mp[j]        ma = max(ma, mp[j])    x = 1    if sum == 0:        print(0)        continue    while x < ma:        x = x * 2    x = x * 2    while x > 0:        one = 0        zero = 0        for i in range(0, n):            if int(int(a[i] / x) % 2) == 1:                one += 1            else:                zero += 1        if (int(one % 2)) == 1:            if (int(zero & 1)) == 1 and one > 1:                ans = -1            else:                ans = 1            break        x = int(x / 2)    print(ans)
