```
#include<bits/stdc++.h>using namespace std;signed main(){    int r = 1 , c = 1 , ans = 1;    while(r != 20 || c != 20){        if(r == 1){            if(c & 1) c ++ ;            else r ++ , c --;        }        else if(c == 1){            if(r % 2 == 0) r ++;            else r -- , c ++;        }        else if((r + c) % 2) r ++ , c -- ;        else r -- , c ++ ;        ans ++ ;    }    cout << ans << '\n';    return 0;}
```
