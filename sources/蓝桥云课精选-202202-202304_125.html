1、C/C++ A组

#include <cstdio>#include <iostream>using namespace std;#define MAXN 100005#define MAXA 1100005int a, N, fa[MAXA], vis[MAXA];int get_father(int x) { // 查找祖先+合并，路径压缩    if (x == fa[x]) return x;    return fa[x] = get_father(fa[x]);}int main() {    cin >> N;    for (int i = 1; i < MAXA; i++)        fa[i] = i; // 并查集初始化    for (int i = 0; i < N; i++) {        cin >> a;        int ans = vis[a] ? get_father(a) + 1 : a;        cout << ans << ' ';        vis[ans] = 1;
        if (ans != 1 && vis[ans - 1])//是否和前后连续            fa[ans - 1] = ans;        if (vis[ans + 1])            fa[ans] = ans + 1;    }    return 0;}

2、Java A组

import java.util.*;import java.io.*;
class InputReader {    public BufferedReader Reader;    public StringTokenizer tokenizer;
    public InputReader(InputStream stream) {        Reader = new BufferedReader(new InputStreamReader(System.in), 32768);        tokenizer = null;    }
    public String next() {        while (tokenizer == null || !tokenizer.hasMoreTokens()) {            try {                tokenizer = new StringTokenizer(Reader.readLine());            } catch (IOException e) {                throw new RuntimeException(e);            }        }        return tokenizer.nextToken();    }
    public int nextInt() {        return Integer.parseInt(next());    }
    public long nextLong() {        return Long.parseLong(next());    }
    public float nextFloat() {        return Float.parseFloat(next());    }
    public double nextDouble() {        return Double.parseDouble(next());    }}
public class Main {
    //father[i]表示,数据i(即data[i]的值)应该修改为father[i]    static int[] father = new int[100005];    static int[] data;    public static void main(String[] args) {
        InputReader sc = new InputReader(System.in);
        int num = sc.nextInt();
        data = new int[num];
        for(int i=1;i<100005;i++) {            //初始化，显然开始时每个数据都不需要修改            father[i] = i;        }
        for(int i=0;i<num;i++) {
            data[i] = sc.nextInt();            //将修改的值赋值给data[i]            int x =find(data[i]);
            data[i] = x;            //修改data[i]对应的father值，显然当下次还有数据为x时，需要修改为x+1对应的father            father[data[i]] = find(x+1);
        }
        for(int i:data) {            System.out.print(i+" ");
        }
    }
    //查找父节点操作    private static int find(int x) {
        if(x==father[x])return x;        else {            //路径压缩            return father[x] = find(father[x]);        }
    }
}
