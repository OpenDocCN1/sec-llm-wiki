1、C/C++ A组

#include<bits/stdc++.h>using namespace std;const int N = 1e5 + 10;char s[N];int pre[N] , suf[N] , last[30];signed main(){    cin >> s + 1;    int n = strlen(s + 1) , ans = 0;    for(int i = 1 ; i <= n ; i ++){        pre[i] = last[s[i] - 'a'];        last[s[i] - 'a'] = i;    }    for(int j = 0 ; j <= 25 ; j ++) last[j] = n + 1;    for(int i = n ; i >= 1 ; i --){        suf[i] = last[s[i] - 'a'];        last[s[i] - 'a'] = i;    }    for(int i = 1 ; i <= n ; i ++){        ans += (i - pre[i]) * (suf[i] - i);    }    cout << ans << '\n';    return 0;}

2、Java A组

import java.util.*;public class Main{  public static void main(String[] args){    Scanner cin = new Scanner(System.in);    String s = cin.nextLine();    int[] pre = new int[s.length() + 1];    int[] suf = new int[s.length() + 1];    int[] last = new int[30];    int ans = 0;    Arrays.fill(last , -1);    for(int i = 0 ; i < s.length() ; i ++){      int x = s.charAt(i) - 'a';      pre[i] = last[x];      last[x] = i;    }    Arrays.fill(last , s.length());    for(int i = s.length() - 1 ; i >= 0 ; i --){      int x = s.charAt(i) - 'a';      suf[i] = last[x];      last[x] = i;    }    for(int i = 0 ; i < s.length() ; i ++){      ans += (i - pre[i]) * (suf[i] - i);    }    System.out.println(ans);  }}
