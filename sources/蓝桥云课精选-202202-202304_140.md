    private static void bfs() {        int up = 0;        for (int i = 0; i <= up; i++) {            for (int j : vec[i]) {                if (vis[j]) {                    continue;                }                vis[j] = true;                que.offer(j);                while (!que.isEmpty()) {                    int u = que.peek();                    que.poll();                    for (Pair<Integer, Integer> k : G[u]) {                        int y = k.first;                        int z = k.second;                        if (!ok[y]) {                            continue;                        }                        int now = Math.max(dp[u], dp[y]) + Math.max(t[u], t[y]);                        if (now < dp[z]) {                            dp[z] = now;                            vec[dp[z]].add(z);                        }                        ok[z] = true;                        up = Math.max(up, now);                    }                }            }        }    }    public static void main(String[] args) {        InputReader cin = new InputReader(System.in);        n = cin.nextInt();        m = cin.nextInt();        K = cin.nextInt();        T = cin.nextInt();        java.util.Arrays.fill(dp , 1000000000);        for (int i = 1; i <= n; i++) {            t[i] = cin.nextInt();        }        for (int i = 1; i <= m; i++) {            x = cin.nextInt();            ok[x] = true;            dp[x] = 0;            vec[0].add(x);        }        for (int i = 1; i <= K; i++) {            x = cin.nextInt();            y = cin.nextInt();            z = cin.nextInt();            G[x].add(new Pair<Integer, Integer>(y, z));            G[y].add(new Pair<Integer, Integer>(x, z));        }        bfs();        System.out.println(dp[T]);    }}
