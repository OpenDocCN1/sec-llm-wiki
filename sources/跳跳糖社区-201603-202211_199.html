def get_order(g: int, _p: int, factors: list) -> tuple([int, list]):    order = _p - 1    ord_factors = []    for f in factors:        if pow(g, order // f, _p) == 1:            order //= f        else:            ord_factors.append(f)    return order, [(i, ord_factors.count(i)) for i in set(ord_factors)]
