import javax.swing.*;import java.awt.*;import java.awt.event.WindowAdapter;import java.awt.event.WindowEvent;
public class DemoPaint {    public static void main(String[] args) {        // 运行主类        new MyPaint();    }}class MyPaint extends Frame{
    // 图形大小    public MyPaint(){        // 设置title        setTitle("冰墩墩");        // 窗口大小 窗口大小 要大于图片大小        setBounds(200,200,600,500);        // 设置可见性        setVisible(true);        // 监听按钮 点击x 退出程序        addWindowListener(new WindowAdapter() {            @Override            public void windowClosing(WindowEvent e) {                System.exit(0);            }        });    }    // 画笔方法    @Override    public void paint(Graphics g) {        // 设置颜色        // 加载图片资源        Image image = new ImageIcon("D:\\Program Files\\jichu\\src\\javaGui\\1.jpeg").getImage();        // 面板引入图片        g.drawImage(image,0,0,this);    }
}
