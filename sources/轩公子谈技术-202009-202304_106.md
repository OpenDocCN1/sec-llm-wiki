# 导入docx模块 调用Document函数# 安装方法 pip install python-docxfrom docx import Document# 格式转换 pip install pypiwin32from win32com import client as wcimport os
# 创建doc文件def CreateDoc(file):    # 初始化文档函数    document = Document()    # 保存文档 path为全局路径 file传参    document.save(path + file)
# 创建docx文件def Saveas(file):    # 开启word进程（个人理解）    word = wc.Dispatch("Word.Application")    # 打开源文件doc     doc = word.Documents.Open(path + file)    # 另存docx    12 表示docx的格式    doc.SaveAs(path + "{}x".format(file), 12)    # 关闭文档    doc.Close()    # 退出进程    word.Quit()
# 写文件def Write(file):    # 打开修改的文件      wenjian = Document(path + file)    # 添加标签 level为一级标题    wenjian.add_heading("第一条", level=1)    # 添加字段 一行代表一段    wenjian.add_paragraph("hello，world")    wenjian.add_paragraph("hello，word")    wenjian.add_paragraph("hello，author")    # 保存文件    wenjian.save(path + file)
# 读文件def read(file):    # 选择读取的文件    red_file = Document(path + file)    # 读取文档里的段落 课切片 后续深入讲解     neirong = red_file.paragraphs    # 遍历内容      for txt in neirong:        print(txt.text)
if __name__ == '__main__':    # 定义全局路径    path = "C:\\Users\\Administrator\\Desktop\\"
    # 自定义文档名称    file = input("请输入文档名称：")    # 对文档后缀进行格式化    doc = "{}.doc".format(file)    docx = "{}.docx".format(file)
    # 创建doc    CreateDoc(doc)    # 创建docx    Saveas(doc)    # 删除doc文档    os.remove(path + doc)    # 写文件    Write(docx)    # 读文件    read(docx)
