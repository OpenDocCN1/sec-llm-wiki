安装expect工具

expect是自动化脚本解释型的工具
yum install expect -y

主要命令    spawn 新建一个进程  这个进程是由expect控制    send  发送字符串给expect控制的进程    set   设置变量   set user root    exp_continue 重新执行expect分支    set timeout -1 设置超时时间 永远等待    set timeout 10 设置超时时间 10秒    interact 将脚本的控制权交给用户,用户继续使用shell执行命令

#!/bin/sh#1.获取在线的主机 能ping通在线>ip.txtfor i in 7 8do        ping -c1 -W1 10.0.0.$i &>/dev/null        if [ $? -eq 0 ];then           echo 10.0.0.$i >>ip.txt        fidone#2.判断公钥是否生成if [ ! -f ~/.ssh/id_rsa ];then   ssh-keygen -P "" -f ~/.ssh/id_rsafi #3.批量分发秘钥pass=1while read linedo      /usr/bin/expect<<EOF         set timeout 2         spawn ssh-copy-id root@$line         expect {                "yes/no" { send "yes\r"; exp_continue }                "password:" { send "$pass\r" };         }         expect eofEOFdone<ip.txt
