一些经验，希望对你有用

  

java我安装到了/opt/java，可以换成你自己的路径

KEYTOOL=/opt/java/bin/keytool

CACERTS=/opt/java/jre/lib/security/cacerts

  

DOMAIN=yunweibang.com（修改为你的二级域名）

DOMAIN\_ALIAS=\`echo $DOMAIN|sed 's/\\.//g'\`

$KEYTOOL -delete -alias $DOMAIN\_ALIAS -keystore $CACERTS -storepass changeit

$KEYTOOL -genkey -v -alias $DOMAIN\_ALIAS -keyalg RSA -storetype pkcs12 -keystore keystore -validity 36500 \\

\-dname "CN=\*.$DOMAIN,OU=$DOMAIN\_ALIAS,O=$DOMAIN\_ALIAS,L=BJ,ST=BJ,C=CN" -storepass changeit -keypass changeit

$KEYTOOL -export -alias $DOMAIN\_ALIAS -storetype pkcs12 -keystore keystore -file $DOMAIN\_ALIAS.cer -storepass changeit

$KEYTOOL -import -trustcacerts -alias $DOMAIN\_ALIAS -keystore $CACERTS -file $DOMAIN\_ALIAS.cer -storepass changeit
