package main

import "fmt"

func main() {
	if 7%2 == 0 {
		fmt.Println(" even")
	} else {
		fmt.Println("odd")
	}

	if num := 9; num < 0 {
		fmt.Println(num, "is negative")
	} else if num < 10 {
		fmt.Println(num, "has 1 digit")
	} else {
		fmt.Println(num, "has multiple digits")
	}
}
