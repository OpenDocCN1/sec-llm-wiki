package main

import "fmt"

func main() {
	var a = "hello"
	fmt.Println(a)

	var b, c int = 1, 2
	fmt.Println(b, c)

	var d = true
	fmt.Println(d)

	var e int
	fmt.Println(e)

	//:= 语法是声明并初始化变量的简写，
	//例如 var f string = "xxx" 可以简写为下边这样。
	f := "world"
	fmt.Println(f)
}

运行:

$go run 变量.go
hello
1 2
true
0
world
