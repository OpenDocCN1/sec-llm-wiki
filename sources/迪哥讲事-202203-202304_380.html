// JSON.parse()-->将JSON字符串转变为JSON对象

  var obj = {
    name : "Lee",
    age: 18
  }

  var str = JSON.stringify(obj);//JSON对象-->JSON字符串
  console.log(str)//{"name":"Lee","age":18}
  obj1 = JSON.parse(str)
  console.log(obj1)//{name: 'Lee', age: 18}
