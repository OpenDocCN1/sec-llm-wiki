function setSystemStatus(status) {
  div.setAttribute('class', status);
  switch (status) {
    case 'running':
      div.innerHTML = '系统正常运行中';
      break;
    case 'warning':
      div.innerHTML = '系统有警告';
      break;
    case 'danger':
      div.innerHTML = '系统存在危险';
      break;
    default:
      div.innerHTML = '系统处于为知状态';

  }

}
