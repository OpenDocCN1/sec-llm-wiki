定义

一个函数和它的周围状态的引用捆绑在一起的组合.
注意:这里的函数的周围状态指的是这个函数定义的附近而不是执行的附近
模块化编程闭包非常有用

函数作为返回值 


//函数作为返回值
function test() {
  const a = 1;
  return function () {//这里的匿名函数和上面的a捆绑在一起构成了一个闭包
    console.log('a:', a);
  }
}

const fn = test();
const a = 2;
fn()//a: 1

函数作为参数 


//函数作为参数
function test1(fn1) {
  const b = 1;
  fn1();
}

const b = 2;

function fn1() {//这里的fn1和上面的b构成了一个闭包
  console.log('b:', b);
}

test1(fn1)//b: 2
