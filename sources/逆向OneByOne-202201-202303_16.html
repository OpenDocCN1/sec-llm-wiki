(6)、通过油猴写一个hook脚本来定位 window.gdxidpyhxde这个生成位置，油猴脚本如下，点击链接查看油猴使用方法

(function() {
    'use strict';
    var _fp =window.gdxidpyhxde;
    Object.defineProperty(window,'gdxidpyhxde',{
        get:function(){
            console.log("调用fp",_fp)
            return _fp;
        },
        set:function(val){
            console.log("生成fp",val)
            debugger;
            this._value = val;
            return val;
        }
    })
})();
