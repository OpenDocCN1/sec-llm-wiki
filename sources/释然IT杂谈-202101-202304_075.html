7、logstash日志收集配置

cd /etc/logstash/conf.d
vim network.conf
------写入以下内容--------
input {
    udp {
        port => 3500
        type => "network"
    }
}
filter {
 if [type] == "network" {
        grok {
            match => {
                "message" => "<%{BASE10NUM:syslog_pri}>(?<switchtime>.*) %{DATA:hostname} %{DATA:ddModuleName}/%{POSINT:severity}/%{DATA:Brief}:%{GREEDYDATA:message}"
            }
            remove_field => [ "timestamp" ]
            add_field => {
                "severity_code" => "%{severity}"
            }
            overwrite => ["message"]
        }
  }
  mutate {
        gsub => [
            "severity", "0", "Emergency",
            "severity", "1", "Alert",
            "severity", "2", "Critical",
            "severity", "3", "Error",
            "severity", "4", "Warning",
            "severity", "5", "Notice",
            "severity", "6", "Informational",
            "severity", "7", "Debug"
        ]
    }
}
output {
  #stdout { codec => rubydebug }
  if [type] == "network" {
        elasticsearch {
                hosts => ["http://127.0.0.1:9200"]
                index => "filebeat-syslog-%{+YYYY.MM.dd}"
#                ssl => true
#               cacert => "/etc/logstash/certs/ca.crt"
                user => "elastic"
                password => "xxxxx"

        }
  }
}
/usr/share/logstash/bin/logstash -f /etc/logstash/conf.d/network.conf     ###运行
