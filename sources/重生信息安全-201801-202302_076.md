    int t = 0, x = 0, y = 0, i = 0;    unsigned char c = 0;    int g = 3;
    while (indata[x] != 0) {        // 需要解码的数据对应的ASCII值对应base64_suffix_map的值        c = base64_suffix_map[indata[x++]];        if (c == 255) return -1;// 对应的值不在转码表中        if (c == 253) continue;// 对应的值是换行或者回车        if (c == 254) { c = 0; g--; }// 对应的值是'='        t = (t << 6) | c; // 将其依次放入一个int型中占3字节        if (++y == 4) {            outdata[i++] = (unsigned char)((t >> 16) & 0xff);            if (g > 1) outdata[i++] = (unsigned char)((t >> 8) & 0xff);            if (g > 2) outdata[i++] = (unsigned char)(t & 0xff);            y = t = 0;        }    }
