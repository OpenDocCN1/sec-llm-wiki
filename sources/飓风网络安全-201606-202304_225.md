#ip转换方法

def ip2num(ip):

    ip=\[int(x) for x in ip.split('.')\]

    return ip\[0\] <<24 | ip\[1\]<<16 | ip\[2\]<<8 |ip\[3\]

def num2ip(num):

    return '%s.%s.%s.%s' %( (num & 0xff000000) >>24,

                     (num & 0x00ff0000) >>16,

                (num & 0x0000ff00) >>8,

                 num & 0x000000ff )

 

#把输入的ip范围组合成一个list形式

def get\_ip(ip):

    start,end = \[ip2num(x) for x in ip.split('-') \]

    return \[ num2ip(num) for num in range(start,end+1) if num & 0xff \]

 

#确认指定的ip是否连通的方法

def check\_port(ip):

    #调用系统的telnetlib，telnet指定ip的指定端口，如果成功，则

    try:

        tn = telnetlib.Telnet(ip, port=445, timeout=10)

 

        #给ip后面加一个换行符，用来在txt里面换行显示

        new\_ip = ip + '\\n'

 

        #以追加的形式将端口开放的ip列出来

        file\_object = open('thefile.txt', 'a')

        file\_object.write(new\_ip)

        file\_object.close()

        print ip+True

    except:

        print('{}ip 的端口不通'.format(ip))

  

iplist = get\_ip('192.168.1.1-192.168.2.1')
