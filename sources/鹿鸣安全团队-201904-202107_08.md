# -*- coding: utf-8 -*-#Author: Konmu#功能：实现爬取美文网的文章内容
import reimport urllib.requestimport requestsimport pprintimport html
def download_novel_html():    url = "http://www.748219.com/meiwen/13094.html"  # 因为编码原因,先获取二进制内容再进行解码    txt = requests.get(url).content.decode('utf-8')     # 转成该网站的格式，额，之前网站源码download下来有乱码，应该就是没转码的原因    new_content=[]    reg1=r'<h1>(.*?)</h1>'    title=re.findall(reg1,txt)    pprint.pprint(title)    reg2=r'<p>(.*?)</p>'    content=re.findall(reg2,txt)    for html_encoded_content in content:        html_encoded_content = html.unescape(html_encoded_content)        new_content.append(html_encoded_content)    pprint.pprint(new_content)    with open('美文.txt','w',encoding='utf-8') as f:        for i in new_content:            f.write(i+'\n')            '''测试：content=download_novel_html()
'''
