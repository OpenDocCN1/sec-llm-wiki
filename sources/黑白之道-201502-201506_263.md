function sign( $array , $do ){  
// 签名算法  
$request\_str = '';  
$salt = '';  
$sign = '';  
// 对应apk中的 Arrays.sort 数组排序，测试php需用 ksort   
ksort( $array );  
foreach ($array as $key => $value) {  
$request\_str .= $value;  
}  
if ( $do == 'querypwd') {  
$sign = md5( $request\_str . SALT );  
} elseif ( $do == 'querydhid' ) {  
$sign = md5( $request\_str . SALT2 );  
} else {  
NULL;  
}  
return $sign;  
}
