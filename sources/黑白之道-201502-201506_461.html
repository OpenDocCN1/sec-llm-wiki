// 本代码是"自动抢红包"的一个简单实现

// 针对最近TMALL推出的“抢红包”活动

// 请使用Visual Studio在Windows环境下编译运行

// 已经编译好的可执行文件下载地址：

// http://code.google.com/p/javadsm/downloads/list

// 祝你好运！

// Designed By Leyuan Pan

// To my dear Yongyu Dai

#include <Windows.h>

#include <stdlib.h>

#include <stdio.h>

int main(){

int n;

int N;

POINT pt1, pt2;

int x1, y1;

int x2, y2;

int flag;

n = 0;

N = 100;

flag = 0;

printf("本代码是\"自动抢红包\"的一个简单实现\n");

printf("针对最近TMALL推出的“抢红包”活动\n");

printf("请使用Visual Studio在Windows环境下编译运行\n");

printf("已经编译好的可执行文件下载地址：\n");

printf("http://code.google.com/p/javadsm/downloads/list\n");

printf("祝你好运！\n");

printf("Designed By Leyuan Pan\n");

printf("To my dear Yongyu Dai\n");

printf("\n\n");

printf("请输入连续抢红包的总次数（默认为100次）: ");

scanf("%d", &N);

printf("程序将自动运行%d秒，运行过程中，随时按鼠标右键终止程序！\n", N);

printf("请用鼠标左键点击“抢”!\n");

while(1){

if(flag == 0){

if(GetAsyncKeyState(VK_LBUTTON)) {

GetCursorPos(&pt1);

flag = 1;

printf("请用鼠标左键点击“叉”!\n");

}

}else{

if(GetAsyncKeyState(VK_LBUTTON)) {

GetCursorPos(&pt2);

break;

}

}

if(GetAsyncKeyState(VK_RBUTTON)) {

return 0;

}

Sleep(100);

}

x1 = pt1.x;

y1 = pt1.y;

x2 = pt2.x;

y2 = pt2.y;

printf("Ready?\n");

Sleep(1000);

printf("GO!\n");

while( n < N){

// 抢

SetCursorPos(x1, y1);

mouse_event(MOUSEEVENTF_LEFTDOWN, 0, 0, 0, NULL);

mouse_event(MOUSEEVENTF_LEFTUP, 0, 0, 0, NULL);

Sleep(100);

// 叉掉

SetCursorPos(x2, y2);

mouse_event(MOUSEEVENTF_LEFTDOWN, 0, 0, 0, NULL);

mouse_event(MOUSEEVENTF_LEFTUP, 0, 0, 0, NULL);

Sleep(100);

n++;

printf("Done / Total = %d / %d\r", n, N);

// Stop

if(GetAsyncKeyState(VK_RBUTTON)) {

printf("\nStop!\n");

break;

}

}

system("pause");

return 0;

}
