import requests
import json
def ipToLoc(ip,key):
    serverloc='http://restapi.amap.com/v3/ip?'
    ipapi=serverloc+'ip='+ip+'&key='+key
    output=requests.get(ipapi)
    info=json.loads(output.text)
    print info
    if info['status']:
        print '省份：'+info['province'].encode('utf-8')
        print '城市：'+info['city'].encode('utf-8')
    return info['rectangle']
def locToAdd(key,location,radius,extensions='all',roadlevel=1,homeorcorp=2):
    serverloc='http://restapi.amap.com/v3/geocode/regeo?'
    ipapi=serverloc+'key='+key+'&location='+location+'&radius='+str(radius)+'&extensions='+extensions+'&roadlevel='+str(roadlevel)+'&homeorcorp='+str(homeorcorp)
    output=requests.get(ipapi)
    info=json.loads(output.text)
    print info['regeocode']['formatted_address']
    for key in info['regeocode']['addressComponent']:
        print info['regeocode']['addressComponent'][key]

if __name__=='__main__':
