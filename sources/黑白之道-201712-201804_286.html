********************redo log ********************

archive log list;（查询归档是否开启等信息）

 


alter system archive log start;--启动自动存档

 


alter system switch logfile;--强行进行一次日志switch


 


alter system checkpoint;--强制进行一次checkpoint


 


/*加入一个日志组*/


alter database add logfile group 4('/u01/app/oracle/oradata/ora11g/redo04.log' )  size 10M;


 


/*加入日志组的一个成员*/


alter database add logfile member '/u01/app/oracle/oradata/ora11g/redo05.log'to group 4;

/*删除日志组:当前日志组不能删；活动的日志组不能删；非归档的日志组不能删*/


alter database drop logfile group 3;



/*删除日志组中的某个成员，但每个组的最后一个成员不能被删除*/


alter database drop logfile member '/u01/app/oracle/oradata/ora11g/redo04.log';



/*清除在线日志*/


alter database clear logfile '/u01/app/oracle/oradata/ora11g/redo01.log';


 

alter database clear logfile group 4;



/*清除非归档日志*/


alter database clear unarchived logfile group 3;



/*数据库归档模式到非归档模式的互换,要启动到mount状态下才能改变;startup mount;然后再打开数据库.*/


 

alter database noarchivelog/archivelog; （打开/关闭oracle归档日志）


 


achive log start;---启动自动归档



select * from v$archived_log; （归档日志一些信息）
