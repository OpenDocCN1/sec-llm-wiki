def scan(ip, port):
    server = telnetlib.Telnet()      # 创建一个Telnet对象
    try:
        title = "开放"
        server.open(ip,port)         # 利用Telnet对象的open方法来创建TCP连接测试端口是否开放
        a='{0}:{1}端口'.format(ip, port)
        write([a,title])
        print(a,title)
    except Exception as e:
         title = "关闭"
         b = '{0}:{1}端口'.format(ip,port)
         write([b, title])
         print(b,title)
    finally:
         server.close()
