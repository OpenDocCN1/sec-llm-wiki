1、通过sed替换

[root@node01 wolf]# find / -type f -name wolf.log

/tmp/wolf.log

[root@node01 wolf]# sed -i 's#wolf#yujing#g' find / -type f -name wolf.log
[root@node01 wolf]# find / -type f -name "wolf.log" | xargs cat

yujing
