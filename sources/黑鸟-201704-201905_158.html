def ip2Octal(ip):

      return '.'.join(format(int(x), '04o') for x in ip.split('.'))


def ip_as_urlencoded(ip):

      ip = ip.split("/")[0]

      en=""

      for i in ip :

            if i.isdigit() :

                  en += "%3{}".format(i)

            elif i == "." :

                  en += "%2E"

            elif i == ":" :

                  en += "%3A"

      return en
